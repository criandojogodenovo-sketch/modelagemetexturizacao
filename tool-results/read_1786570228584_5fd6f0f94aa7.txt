    34→// Helper: amostrar altura do heightmap numa posição (x,z) do mundo
    35→function sampleTerrainHeight(hm, seg, worldX, worldZ, terrainWidth, terrainDepth, heightScale) {
    36→  const nx = (worldX / terrainWidth + 0.5) * seg
    37→  const nz = (worldZ / terrainDepth + 0.5) * seg
    38→  const ix = Math.max(0, Math.min(seg, Math.round(nx)))
    39→  const iz = Math.max(0, Math.min(seg, Math.round(nz)))
    40→  const idx = iz * (seg + 1) + ix
    41→  const h = hm[idx] || 0
    42→  return h * heightScale
    43→}

    75→function autoSplat(hm, seg, maxLayers = 4) {
    76→  const sm = new Float32Array((seg + 1) * (seg + 1) * maxLayers)
    77→  for (let i = 0; i < hm.length; i++) {
    78→    const h = (hm[i] + 1) * 0.5
    79→    if (h < 0.25) { sm[i*4]=1; sm[i*4+1]=0; sm[i*4+2]=0; sm[i*4+3]=0 }
    80→    else if (h < 0.5) { sm[i*4]=0.4; sm[i*4+1]=0.6; sm[i*4+2]=0; sm[i*4+3]=0 }
    81→    else if (h < 0.75) { sm[i*4]=0; sm[i*4+1]=0.5; sm[i*4+2]=0.5; sm[i*4+3]=0 }
    82→    else { sm[i*4]=0; sm[i*4+1]=0; sm[i*4+2]=0.3; sm[i*4+3]=0.7 }
    83→  }
    84→  return sm
    85→}
