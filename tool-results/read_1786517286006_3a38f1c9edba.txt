     1→/**
     2→ * TerrainEditor — Editor de Terrenos profissional (Unity-aligned).
     3→ *
     4→ * **Reescrito em P7** para alinhar com o padrão Unity Terrain:
     5→ *
     6→ *  Tabs separadas (estilo Unity Inspector):
     7→ *   1. Sculpt — pincéis de escultura (Elevar, Rebaixar, Suavizar, Achatar,
     8→ *               Definir Altura, Ruído, Rampa) + falloff types (Smooth/Linear/
     9→ *               Constant/Sharp) + spacing para drag contínuo
    10→ *   2. Paint Texture — 4 camadas com blending suave (Float32Array splatmap),
    11→ *               pintura manual + auto-splat por altura/inclinação, camadas
    12→ *               customizáveis (cor + nome)
    13→ *   3. Details — dispersão de objetos (foliage) com regras de altura/inclinação
    14→ *   4. Settings — dimensões, resolução, escala, import/export PNG heightmap,
    15→ *               seed, Perlin params, exportar para a cena
    16→ *
    17→ *  Pré-visualização 2D com brush cursor + drag painting contínuo.
    18→ *  Exporta para a cena como TerrainObject com heightmap + splatmap + layers.
    19→ */
    20→import { useState, useRef, useEffect, useCallback } from 'react'
    21→import { useStore } from '../../../store/useStore'
    22→import { IconClose } from '../../ui/Icons'
    23→import HeightmapPreview from './HeightmapPreview'
    24→import {
    25→  SCULPT_BRUSHES,
    26→  DEFAULT_TEXTURE_LAYERS,
    27→  DEFAULT_TERRAIN_CONFIG,
    28→  DEFAULT_BRUSH,
    29→  DEFAULT_SCATTER,
    30→  MAX_LAYERS,
    31→} from '../../../utils/terrain/terrainPresets'
    32→import {
    33→  generateHeightmap,
    34→  applyBrush as applyBrushOp,
    35→  applyRamp,
    36→  falloff as falloffFn,
    37→  createSplatmap,
    38→  paintSplat,
    39→  autoSplatByHeight,
    40→  heightmapStats,
    41→  heightmapToPNG,
    42→  pngToHeightmap,
    43→  hexToRgb,
    44→} from '../../../utils/terrain/terrainMath'
    45→
    46→const FALLOFFS = [
    47→  { id: 'smooth',   label: 'Smooth'   },
    48→  { id: 'linear',   label: 'Linear'   },
    49→  { id: 'constant', label: 'Constant' },
    50→  { id: 'sharp',    label: 'Sharp'    },
    51→]
    52→
    53→const TABS = [
    54→  { id: 'sculpt',   label: 'Escultura',   icon: '⛏️' },
    55→  { id: 'paint',    label: 'Textura',     icon: 'palette' },
    56→  { id: 'details',  label: 'Detalhes',    icon: 'tree' },
    57→  { id: 'settings', label: 'Definições',  icon: 'settings' },
    58→]
    59→
    60→export default function TerrainEditor({ onClose }) {
    61→  const addConectToScene = useStore((s) => s.addConectToScene)
    62→  const updateConect = useStore((s) => s.updateConect)
    63→  const activeSceneId = useStore((s) => s.activeSceneId)
    64→  const objects = useStore((s) => s.objects)
    65→  const toast = useStore((s) => s.toast)
    66→
    67→  const [activeTab, setActiveTab] = useState('sculpt')
    68→  const [config, setConfig] = useState(DEFAULT_TERRAIN_CONFIG)
    69→  const [brush, setBrush] = useState(DEFAULT_BRUSH)
    70→  const [rampPoints, setRampPoints] = useState([])
    71→  const [heightmap, setHeightmap] = useState(null)
    72→  const [splatmap, setSplatmap] = useState(null)
    73→  const [textureLayers, setTextureLayers] = useState(DEFAULT_TEXTURE_LAYERS)
    74→  const [activeLayerIdx, setActiveLayerIdx] = useState(0)
    75→  const [scatter, setScatter] = useState(DEFAULT_SCATTER)
    76→  const [scatteredPoints, setScatteredPoints] = useState([])
    77→
    78→  // Gerar heightmap + splatmap iniciais
    79→  useEffect(() => {
    80→    if (!heightmap) {
    81→      const hm = generateHeightmap(config.segments, config)
    82→      setHeightmap(hm)
    83→      const sm = autoSplatByHeight(hm, config.segments, textureLayers, MAX_LAYERS)
    84→      setSplatmap(sm)
    85→    }
    86→    // eslint-disable-next-line react-hooks/exhaustive-deps
    87→  }, [])
    88→
    89→  // ===== Operações =====
    90→
    91→  const regenerate = useCallback(() => {
    92→    const hm = generateHeightmap(config.segments, config)
    93→    setHeightmap(hm)
    94→    const sm = autoSplatByHeight(hm, config.segments, textureLayers, MAX_LAYERS)
    95→    setSplatmap(sm)
    96→    toast('Heightmap gerado', 'success', 1200)
    97→  }, [config, textureLayers, toast])
    98→
    99→  // Aplicar pincel — chamado a cada stamp (do HeightmapPreview drag)
   100→  const handlePaint = useCallback((x, z, isStart) => {
   101→    if (!heightmap) return
   102→    const seg = config.segments
   103→
   104→    if (brush.mode === 'ramp') {
   105→      // Modo rampa: 2 cliques
   106→      if (!isStart) return
   107→      const newPoints = [...rampPoints, [x, z]]
   108→      if (newPoints.length === 2) {
   109→        const hm = new Float32Array(heightmap)
   110→        applyRamp(hm, seg, newPoints[0], newPoints[1], brush)
   111→        setHeightmap(hm)
   112→        // Auto-splat update
   113→        const sm = autoSplatByHeight(hm, seg, textureLayers, MAX_LAYERS)
   114→        setSplatmap(sm)
   115→        setRampPoints([])
   116→        toast('Rampa aplicada', 'success', 1000)
   117→      } else {
   118→        setRampPoints(newPoints)
   119→      }
   120→      return
   121→    }
   122→
   123→    // Sculpt mode (raise/lower/smooth/flatten/setHeight/noise)
   124→    if (activeTab === 'sculpt') {
   125→      const hm = new Float32Array(heightmap)
   126→      applyBrushOp(hm, seg, x, z, { ...brush, deltaTime: isStart ? 1 : 0.4 })
   127→      setHeightmap(hm)
   128→      // Atualizar splatmap automaticamente? Em Unity, a pintura de textura
   129→      // é separada da escultura — NÃO recriar splatmap aqui.
   130→      // Mas se o utilizador quiser auto-update, pode clicar "Auto-textura".
   131→    } else if (activeTab === 'paint') {
   132→      // Paint texture mode
   133→      if (!splatmap) return
   134→      const sm = new Float32Array(splatmap)
   135→      paintSplat(sm, seg, x, z, activeLayerIdx, { ...brush, deltaTime: isStart ? 1 : 0.4, maxLayers: MAX_LAYERS })
   136→      setSplatmap(sm)
   137→    }
   138→  }, [heightmap, splatmap, brush, config.segments, textureLayers, rampPoints, activeTab, activeLayerIdx, toast])
   139→
   140→  // Auto-splat update
   141→  const autoTexture = useCallback(() => {
   142→    if (!heightmap) return
   143→    const sm = autoSplatByHeight(heightmap, config.segments, textureLayers, MAX_LAYERS)
   144→    setSplatmap(sm)
   145→    toast('Texturas aplicadas por altura/inclinação', 'success', 1200)
   146→  }, [heightmap, config.segments, textureLayers, toast])
   147→
   148→  // Limpar textura (só camada 0)
   149→  const clearTexture = useCallback(() => {
   150→    if (!heightmap) return
   151→    const cellCount = (config.segments + 1) * (config.segments + 1)
   152→    setSplatmap(createSplatmap(cellCount, MAX_LAYERS))
   153→    toast('Textura limpa (camada 0 dominante)', 'info', 1000)
   154→  }, [heightmap, config.segments, toast])
   155→
   156→  // Adicionar camada
   157→  const addLayer = () => {
   158→    if (textureLayers.length >= MAX_LAYERS) {
   159→      toast(`Máximo de ${MAX_LAYERS} camadas`, 'error')
   160→      return
   161→    }
   162→    const colors = ['#a0522d', '#cd853f', '#778899', '#bdb76b', '#8fbc8f']
   163→    const newLayer = {
   164→      id: `layer${Date.now()}`,
   165→      label: `Camada ${textureLayers.length + 1}`,
   166→      color: colors[textureLayers.length % colors.length],
   167→      textureURL: null,
   168→    }
   169→    setTextureLayers([...textureLayers, newLayer])
   170→  }
   171→
   172→  // Remover camada
   173→  const removeLayer = (idx) => {
   174→    if (textureLayers.length <= 1) {
   175→      toast('Precisas de pelo menos 1 camada', 'error')
   176→      return
   177→    }
   178→    const next = textureLayers.filter((_, i) => i !== idx)
   179→    setTextureLayers(next)
   180→    if (activeLayerIdx >= next.length) setActiveLayerIdx(0)
   181→    // Recriar splatmap com novo nº de camadas — redistribuir pesos
   182→    if (heightmap && splatmap) {
   183→      const sm = autoSplatByHeight(heightmap, config.segments, next, MAX_LAYERS)
   184→      setSplatmap(sm)
   185→    }
   186→  }
   187→
   188→  // Editar camada
   189→  const updateLayer = (idx, patch) => {
   190→    setTextureLayers(textureLayers.map((l, i) => (i === idx ? { ...l, ...patch } : l)))
   191→  }
   192→
   193→  // Dispersar objetos
   194→  const scatterObjects = () => {
   195→    if (!heightmap || !activeSceneId) {
   196→      toast('Gera um heightmap primeiro', 'error')
   197→      return
   198→    }
   199→    if (!scatter.objectName) {
   200→      toast('Seleciona um objeto para dispersar', 'error')
   201→      return
   202→    }
   203→    const obj = objects.find((o) => o.name === scatter.objectName)
   204→    if (!obj) {
   205→      toast('Objeto não encontrado', 'error')
   206→      return
   207→    }
   208→    const seg = config.segments
   209→    const count = scatter.density
   210→    let placed = 0
   211→    const points = []
   212→    const rng = mulberry32(Date.now() & 0x7fffffff)
   213→    let min = Infinity, max = -Infinity
   214→    for (let i = 0; i < heightmap.length; i++) {
   215→      if (heightmap[i] < min) min = heightmap[i]
   216→      if (heightmap[i] > max) max = heightmap[i]
   217→    }
   218→    const range = max - min || 1
   219→    for (let i = 0; i < count * 5 && placed < count; i++) {
   220→      const x = Math.floor(rng() * seg)
   221→      const z = Math.floor(rng() * seg)
   222→      const h = heightmap[z * (seg + 1) + x]
   223→      const normalized = (h - min) / range
   224→      if (normalized < scatter.minHeight || normalized > scatter.maxHeight) continue
   225→      const hRight = heightmap[z * (seg + 1) + Math.min(seg, x + 1)]
   226→      const hDown = heightmap[Math.min(seg, z + 1) * (seg + 1) + x]
   227→      const slope = (Math.abs(hRight - h) + Math.abs(hDown - h)) / range
   228→      if (slope > scatter.maxSlope) continue
   229→      const worldX = (x / seg - 0.5) * config.width
   230→      const worldZ = (z / seg - 0.5) * config.depth
   231→      const worldY = h * config.heightScale
   232→      const rotY = scatter.randomRotation ? rng() * Math.PI * 2 : 0
   233→      const scale = 1 + (rng() - 0.5) * 2 * scatter.randomScale
   234→      useStore.getState().addObjectToScene(obj.id, [worldX, worldY, worldZ], [0, rotY, 0], [scale, scale, scale])
   235→      points.push([x, z])
   236→      placed++
   237→    }
   238→    setScatteredPoints(points)
   239→    toast(`${placed} objetos dispersos`, 'success')
   240→  }
   241→
   242→  // Exportar para a cena
   243→  const exportToScene = () => {
   244→    if (!activeSceneId) {
   245→      toast('Cria uma cena primeiro', 'error')
   246→      return
   247→    }
   248→    const conect = addConectToScene('TerrainObject', [0, 0, 0])
   249→    if (conect) {
   250→      updateConect(conect.instanceId, {
   251→        width: config.width,
   252→        depth: config.depth,
   253→        segments: config.segments,
   254→        heightScale: config.heightScale,
   255→        heightmapSeed: config.seed,
   256→        heightmap: heightmap ? Array.from(heightmap) : null,
   257→        splatmap: splatmap ? Array.from(splatmap) : null,
   258→        textureLayers,
   259→        maxLayers: MAX_LAYERS,
   260→      })
   261→      toast('Terreno exportado para a cena!', 'success')
   262→      if (onClose) onClose()
   263→    }
   264→  }
   265→
   266→  // Import heightmap PNG
   267→  const importInputRef = useRef(null)
   268→  const handleImportPNG = async (e) => {
   269→    const file = e.target.files?.[0]
   270→    if (!file) return
   271→    try {
   272→      const hm = await pngToHeightmap(file, config.segments)
   273→      setHeightmap(hm)
   274→      const sm = autoSplatByHeight(hm, config.segments, textureLayers, MAX_LAYERS)
   275→      setSplatmap(sm)
   276→      toast('Heightmap importado de PNG', 'success')
   277→    } catch (err) {
   278→      toast('Erro ao importar PNG: ' + err.message, 'error')
   279→    }
   280→    e.target.value = ''
   281→  }
   282→
   283→  // Export heightmap PNG
   284→  const handleExportPNG = () => {
   285→    if (!heightmap) {
   286→      toast('Sem heightmap para exportar', 'error')
   287→      return
   288→    }
   289→    const dataURL = heightmapToPNG(heightmap, config.segments)
   290→    const a = document.createElement('a')
   291→    a.href = dataURL
   292→    a.download = `heightmap-${config.segments}x${config.segments}.png`
   293→    a.click()
   294→    toast('Heightmap exportado como PNG', 'success')
   295→  }
   296→
   297→  // Stats do heightmap atual
   298→  const stats = heightmap ? heightmapStats(heightmap) : null
   299→
   300→  return (
   301→    <>
   302→      {onClose && <div className="drawer-backdrop show" onClick={onClose} />}
   303→      <aside className={`terrain-editor ${onClose ? 'open' : ''}`}>
   304→        <div className="panel-header">
   305→          <span>Editor de Terrenos</span>
   306→          {onClose && (
   307→            <button className="icon" onClick={onClose} title="Fechar">
   308→              <IconClose width={14} height={14} />
   309→            </button>
   310→          )}
   311→        </div>
   312→
   313→        {/* Tabs principais — estilo Unity Inspector */}
   314→        <div className="terrain-tabs">
   315→          {TABS.map((t) => (
   316→            <button
   317→              key={t.id}
   318→              className={`terrain-tab ${activeTab === t.id ? 'active' : ''}`}
   319→              onClick={() => setActiveTab(t.id)}
   320→              title={t.label}
   321→            >
   322→              <span style={{ fontSize: 14 }}>{t.icon}</span>
   323→              <span>{t.label}</span>
   324→            </button>
   325→          ))}
   326→        </div>
   327→
   328→        <div className="terrain-editor-body">
   329→          {/* ============ TAB: SCULPT ============ */}
   330→          {activeTab === 'sculpt' && (
   331→            <>
   332→              <div className="panel-section">
   333→                <h4>⛏️ Pincel de Escultura</h4>
   334→                <div className="terrain-brush-grid">
   335→                  {SCULPT_BRUSHES.map((b) => (
   336→                    <button
   337→                      key={b.id}
   338→                      className={`terrain-brush-btn ${brush.mode === b.id ? 'active' : ''}`}
   339→                      onClick={() => {
   340→                        setBrush({ ...brush, mode: b.id })
   341→                        if (b.id === 'ramp') setRampPoints([])
   342→                      }}
   343→                      title={`${b.label}: ${b.desc}`}
   344→                    >
   345→                      <span style={{ fontSize: 16 }}>{b.icon}</span>
   346→                      <span>{b.label}</span>
   347→                    </button>
   348→                  ))}
   349→                </div>
   350→                {brush.mode === 'ramp' && (
   351→                  <div className="small" style={{ color: 'var(--accent)', marginBottom: 8, marginTop: 4 }}>
   352→                    {rampPoints.length === 0 ? '👆 Clica no ponto 1 no preview' : '👆 Clica no ponto 2 no preview'}
   353→                  </div>
   354→                )}
   355→              </div>
   356→
   357→              <BrushControls brush={brush} setBrush={setBrush} />
   358→            </>
   359→          )}
   360→
   361→          {/* ============ TAB: PAINT TEXTURE ============ */}
   362→          {activeTab === 'paint' && (
   363→            <>
   364→              <div className="panel-section">
   365→                <h4>Camadas de Textura</h4>
   366→                <div className="terrain-layers-list">
   367→                  {textureLayers.map((layer, idx) => (
   368→                    <div
   369→                      key={layer.id}
   370→                      className={`terrain-layer-row ${activeLayerIdx === idx ? 'active' : ''}`}
   371→                      onClick={() => setActiveLayerIdx(idx)}
   372→                    >
   373→                      <input
   374→                        type="color"
   375→                        value={layer.color}
   376→                        onClick={(e) => e.stopPropagation()}
   377→                        onChange={(e) => updateLayer(idx, { color: e.target.value })}
   378→                        style={{ width: 28, height: 28, padding: 0, cursor: 'pointer', flexShrink: 0 }}
   379→                      />
   380→                      <input
   381→                        type="text"
   382→                        value={layer.label}
   383→                        onClick={(e) => e.stopPropagation()}
   384→                        onChange={(e) => updateLayer(idx, { label: e.target.value })}
   385→                        style={{ flex: 1, minWidth: 0 }}
   386→                      />
   387→                      <button
   388→                        className="icon danger"
   389→                        onClick={(e) => { e.stopPropagation(); removeLayer(idx) }}
   390→                        title="Remover camada"
   391→                        style={{ padding: '4px 6px', minWidth: 'auto' }}
   392→                      >
   393→                        ✕
   394→                      </button>
   395→                    </div>
   396→                  ))}
   397→                </div>
   398→                <button onClick={addLayer} style={{ width: '100%', marginTop: 6 }} disabled={textureLayers.length >= MAX_LAYERS}>
   399→                  + Adicionar Camada
   400→                </button>
   401→                <div className="small muted mt-2">
   402→                  Camada ativa: <strong style={{ color: textureLayers[activeLayerIdx]?.color }}>
   403→                    {textureLayers[activeLayerIdx]?.label}
   404→                  </strong>
   405→                </div>
   406→              </div>
   407→
   408→              <BrushControls brush={brush} setBrush={setBrush} hideMode />
   409→
   410→              <div className="panel-section">
   411→                <h4>Auto-Textura</h4>
   412→                <div className="small muted mb-2">
   413→                  Distribui automaticamente as camadas por altura e inclinação:
   414→                  baixa altitude → relva, alta inclinação → pedra, topo → neve.
   415→                </div>
   416→                <button onClick={autoTexture} style={{ width: '100%', marginBottom: 4 }}>Aplicar Auto-Textura
   417→                </button>
   418→                <button onClick={clearTexture} style={{ width: '100%' }}>
   419→                  🧹 Limpar Textura
   420→                </button>
   421→              </div>
   422→            </>
   423→          )}
   424→
   425→          {/* ============ TAB: DETAILS (foliage) ============ */}
   426→          {activeTab === 'details' && (
   427→            <>
   428→              <div className="panel-section">
   429→                <h4>Dispersão de Objetos</h4>
   430→                <div className="prop-row">
   431→                  <label>Objeto a dispersar</label>
   432→                  <select
   433→                    value={scatter.objectName}
   434→                    onChange={(e) => setScatter({ ...scatter, objectName: e.target.value })}
   435→                  >
   436→                    <option value="">— Selecionar —</option>
   437→                    {objects.map((o) => (
   438→                      <option key={o.id} value={o.name}>{o.name}</option>
   439→                    ))}
   440→                  </select>
   441→                </div>
   442→                <div className="prop-row">
   443→                  <label>Densidade: {scatter.density}</label>
   444→                  <input type="range" min="1" max="200" step="1" value={scatter.density}
   445→                    onChange={(e) => setScatter({ ...scatter, density: Number(e.target.value) })} />
   446→                </div>
   447→                <div className="prop-row">
   448→                  <label>Altura mín: {scatter.minHeight.toFixed(2)}</label>
   449→                  <input type="range" min="0" max="1" step="0.05" value={scatter.minHeight}
   450→                    onChange={(e) => setScatter({ ...scatter, minHeight: Number(e.target.value) })} />
   451→                </div>
   452→                <div className="prop-row">
   453→                  <label>Altura máx: {scatter.maxHeight.toFixed(2)}</label>
   454→                  <input type="range" min="0" max="1" step="0.05" value={scatter.maxHeight}
   455→                    onChange={(e) => setScatter({ ...scatter, maxHeight: Number(e.target.value) })} />
   456→                </div>
   457→                <div className="prop-row">
   458→                  <label>Inclinação máx: {scatter.maxSlope.toFixed(2)}</label>
   459→                  <input type="range" min="0.05" max="1" step="0.05" value={scatter.maxSlope}
   460→                    onChange={(e) => setScatter({ ...scatter, maxSlope: Number(e.target.value) })} />
   461→                </div>
   462→                <div className="prop-row">
   463→                  <label>
   464→                    <input
   465→                      type="checkbox"
   466→                      checked={scatter.randomRotation}
   467→                      onChange={(e) => setScatter({ ...scatter, randomRotation: e.target.checked })}
   468→                      style={{ width: 'auto', display: 'inline-block', marginRight: 6 }}
   469→                    />
   470→                    Rotação aleatória
   471→                  </label>
   472→                </div>
   473→                <div className="prop-row">
   474→                  <label>Variação de escala: ±{Math.round(scatter.randomScale * 100)}%</label>
   475→                  <input type="range" min="0" max="0.5" step="0.05" value={scatter.randomScale}
   476→                    onChange={(e) => setScatter({ ...scatter, randomScale: Number(e.target.value) })} />
   477→                </div>
   478→                <button onClick={scatterObjects} className="primary" style={{ width: '100%', marginTop: 8 }}>Dispersar no Terreno
   479→                </button>
   480→                {scatteredPoints.length > 0 && (
   481→                  <button
   482→                    onClick={() => setScatteredPoints([])}
   483→                    style={{ width: '100%', marginTop: 4 }}
   484→                  >
   485→                    🧹 Limpar markers (não remove objetos)
   486→                  </button>
   487→                )}
   488→              </div>
   489→            </>
   490→          )}
   491→
   492→          {/* ============ TAB: SETTINGS ============ */}
   493→          {activeTab === 'settings' && (
   494→            <>
   495→              <div className="panel-section">
   496→                <h4>📏 Dimensões</h4>
   497→                <div className="prop-row">
   498→                  <label>Largura: {config.width}m</label>
   499→                  <input type="range" min="10" max="200" step="5" value={config.width}
   500→                    onChange={(e) => setConfig({ ...config, width: Number(e.target.value) })} />
   501→                </div>
   502→                <div className="prop-row">
   503→                  <label>Profundidade: {config.depth}m</label>
   504→                  <input type="range" min="10" max="200" step="5" value={config.depth}
   505→                    onChange={(e) => setConfig({ ...config, depth: Number(e.target.value) })} />

   671→// ============================================================
   672→//  Sub-componente: FalloffPreview (mini-gráfico)
   673→// ============================================================
   674→function FalloffPreview({ brush }) {
   675→  const canvasRef = useRef(null)
   676→  useEffect(() => {
   677→    const canvas = canvasRef.current
   678→    if (!canvas) return
   679→    const ctx = canvas.getContext('2d')
   680→    const w = canvas.width = canvas.clientWidth * 2
   681→    const h = canvas.height = 40 * 2
   682→    ctx.clearRect(0, 0, w, h)
   683→    // Desenhar perfil de falloff
   684→    ctx.strokeStyle = '#2f81f7'
   685→    ctx.lineWidth = 2
   686→    ctx.beginPath()
   687→    const radius = brush.size
   688→    for (let x = 0; x <= w; x++) {
   689→      const dist = (x / w) * radius * 2 - radius
   690→      const f = falloffFn(Math.abs(dist), radius, brush.falloff)
   691→      const y = h - f * h * 0.85 - h * 0.05
   692→      if (x === 0) ctx.moveTo(x, y)
   693→      else ctx.lineTo(x, y)
   694→    }
   695→    ctx.stroke()
   696→    // Linha central
   697→    ctx.strokeStyle = 'rgba(255,255,255,0.2)'
   698→    ctx.beginPath()
   699→    ctx.moveTo(w / 2, 0)
   700→    ctx.lineTo(w / 2, h)
   701→    ctx.stroke()
   702→  }, [brush.falloff, brush.size])
   703→  return (
   704→    <div style={{ marginTop: 8 }}>
   705→      <div className="small muted mb-1">Perfil do falloff:</div>
   706→      <canvas ref={canvasRef} style={{ width: '100%', height: 40, background: 'var(--bg-app)', borderRadius: 4, border: '1px solid var(--border-soft)' }} />
   707→    </div>
   708→  )
   709→}

   711→// PRNG local — não quero importar do terrainMath para evitar bundle duplo
   712→function mulberry32(seed) {
   713→  let s = seed >>> 0
   714→  return function () {
   715→    s = (s + 0x6D2B79F5) >>> 0
   716→    let t = s
   717→    t = Math.imul(t ^ (t >>> 15), t | 1)
   718→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
   719→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
   720→  }
   721→}
