     1→/**
     2→ * ConectRenderer — renderiza um Conect na cena 3D.
     3→ *
     4→ * Dependendo do tipo de Conect:
     5→ *  - VisualObject: renderiza o modelo do catálogo (sourceObjectId)
     6→ *  - RigidObject/StaticObject/StopObject: renderiza um cubo placeholder + física
     7→ *  - PersonalObject: renderiza um capsule/cubo como jogador
     8→ *  - TriggerObject: renderiza um wireframe transparente (não visível no jogo)
     9→ *  - LuminousObject: adiciona uma luz à cena
    10→ *  - ParticleObject: sistema de partículas
    11→ *  - TerrainObject: terreno com heightmap
    12→ *  - WaterObject: plano de água
    13→ *  - PathObject: renderiza waypoints
    14→ *  - CheckpointObject: renderiza uma bandeira/marca
    15→ *  - Outros (UI, Sound, Sky, Fog, View, etc.): sem visual 3D (só lógica)
    16→ */
    17→import { forwardRef, useMemo, useEffect, useRef } from 'react'
    18→import { useThree, useFrame } from '@react-three/fiber'
    19→import * as THREE from 'three'
    20→import { useStore } from '../../store/useStore'
    21→import { findConectDefinition } from '../../utils/conects/taxonomy'
    22→import { createWaterProMaterial } from '../../utils/waterShaderPro'
    23→import { createSkyProMaterial, calculateSunDirection } from '../../utils/skyShaderPro'
    24→import SceneObject from '../3d/SceneObject'
    25→
    26→const ConectRenderer = forwardRef(function ConectRenderer({ conect, objects, setMeshRef }, meshRef) {
    27→  const def = findConectDefinition(conect.type)
    28→
    29→  // Se não tem visual, não renderiza mesh — mas pode adicionar luzes, etc.
    30→  if (conect.type === 'LuminousObject') {
    31→    return <LuminousMesh conect={conect} setMeshRef={setMeshRef} />
    32→  }
    33→
    34→  if (conect.type === 'TerrainObject') {
    35→    return <TerrainMesh conect={conect} setMeshRef={setMeshRef} />
    36→  }
    37→
    38→  if (conect.type === 'WaterObject') {
    39→    return <WaterMesh conect={conect} setMeshRef={setMeshRef} />
    40→  }
    41→
    42→  if (conect.type === 'PathObject') {
    43→    return <PathMesh conect={conect} setMeshRef={setMeshRef} />
    44→  }
    45→
    46→  if (conect.type === 'CheckpointObject') {
    47→    return <CheckpointMesh conect={conect} setMeshRef={setMeshRef} />
    48→  }
    49→
    50→  if (conect.type === 'ParticleObject') {
    51→    return <ParticleMesh conect={conect} setMeshRef={setMeshRef} />
    52→  }
    53→
    54→  // ViewObject: renderiza um gizmo de câmara visível no editor (selecionável)
    55→  if (conect.type === 'ViewObject') {
    56→    return <ViewObjectMesh conect={conect} setMeshRef={setMeshRef} />
    57→  }
    58→
    59→  // TrailObject: rasto visual atrás de um objeto em movimento
    60→  if (conect.type === 'TrailObject') {
    61→    return <TrailMesh conect={conect} setMeshRef={setMeshRef} />
    62→  }
    63→
    64→  // ReflectObject: sonda de reflexo (renderiza uma esfera espelhada)
    65→  if (conect.type === 'ReflectObject') {
    66→    return <ReflectMesh conect={conect} setMeshRef={setMeshRef} />
    67→  }
    68→
    69→  // SkyObject: renderiza esfera de céu com shader procedural
    70→  if (conect.type === 'SkyObject') {
    71→    return <SkyMesh conect={conect} setMeshRef={setMeshRef} />
    72→  }
    73→
    74→  // SunObject: luz direcional (sol)
    75→  if (conect.type === 'SunObject') {
    76→    return <SunLightMesh conect={conect} setMeshRef={setMeshRef} />
    77→  }
    78→
    79→  // PointObject: luz pontual
    80→  if (conect.type === 'PointObject') {
    81→    return <PointLightMesh conect={conect} setMeshRef={setMeshRef} />
    82→  }
    83→
    84→  // SpotObject: luz spot
    85→  if (conect.type === 'SpotObject') {
    86→    return <SpotLightMesh conect={conect} setMeshRef={setMeshRef} />
    87→  }
    88→
    89→  // AmbientObject: luz ambiente (sem gizmo, só aplica à cena)
    90→  if (conect.type === 'AmbientObject') {
    91→    return <AmbientLightMesh conect={conect} setMeshRef={setMeshRef} />
    92→  }
    93→
    94→  // ReferenceObject: renderiza conteúdo de outra cena
    95→  if (conect.type === 'ReferenceObject') {
    96→    return <ReferenceMesh conect={conect} setMeshRef={setMeshRef} />
    97→  }
    98→
    99→  if (!def?.hasVisual && conect.type !== 'VisualObject') {
   100→    // Sem visual — ligar o meshRef a null para que physics não tente usar
   101→    useEffect(() => { setMeshRef?.(null) }, [])
   102→    return null
   103→  }
   104→
   105→  // VisualObject: usa modelo do catálogo
   106→  if (conect.type === 'VisualObject') {
   107→    const obj = objects.find((o) => o.id === conect.sourceObjectId)
   108→    if (!obj) return null
   109→    const sceneObj = {
   110→      ...obj,
   111→      id: conect.instanceId,
   112→      position: conect.position,
   113→      rotation: conect.rotation,
   114→      scale: conect.scale,
   115→    }
   116→    return (
   117→      <SceneObject
   118→        ref={(node) => setMeshRef?.(node)}
   119→        obj={sceneObj}
   120→        isSelected={false}
   121→        onSelect={() => {}}
   122→      />
   123→    )
   124→  }
   125→
   126→  // Rigid/Static/Stop/Personal: placeholder com geometria simples
   127→  return (
   128→    <PlaceholderMesh
   129→      ref={(node) => {
   130→        if (typeof meshRef === 'function') meshRef(node)
   131→        else if (meshRef) meshRef.current = node
   132→        setMeshRef?.(node)
   133→      }}
   134→      conect={conect}
   135→    />
   136→  )
   137→})
   138→
   139→export default ConectRenderer

   158→      receiveShadow
   159→      userData={{ conectInstanceId: conect.instanceId }}

   167→// ===== LuminousObject =====
   168→// Luz real + gizmo visual (esfera amarela) para seleção no editor
   169→function LuminousMesh({ conect, setMeshRef }) {
   170→  const lightProps = {
   171→    intensity: conect.intensity,
   172→    color: conect.color,
   173→    castShadow: conect.castShadow,
   174→  }
   175→  let lightElement
   176→  if (conect.lightType === 'directional') {
   177→    lightElement = <directionalLight {...lightProps} position={[0, 0, 0]} />
   178→  } else if (conect.lightType === 'spot') {
   179→    lightElement = <spotLight {...lightProps} distance={conect.distance} angle={Math.PI / 4} penumbra={0.3} position={[0, 0, 0]} />
   180→  } else {
   181→    lightElement = <pointLight {...lightProps} distance={conect.distance} position={[0, 0, 0]} />
   182→  }
   183→  return (
   184→    <group
   185→      ref={setMeshRef}
   186→      position={conect.position}
   187→      rotation={conect.rotation}
   188→      userData={{ conectInstanceId: conect.instanceId }}
   189→    >
   190→      {lightElement}
   191→      {/* Gizmo visual: esfera amarela para seleção */}
   192→      <mesh>
   193→        <sphereGeometry args={[0.2, 12, 12]} />
   194→        <meshBasicMaterial color="#ffd700" />
   195→      </mesh>
   196→      {/* Halo ao redor */}
   197→      <mesh>
   198→        <sphereGeometry args={[0.35, 12, 12]} />
   199→        <meshBasicMaterial color="#ffd700" transparent opacity={0.2} />
   200→      </mesh>
   201→    </group>
   202→  )
   203→}

   205→// ===== TerrainObject =====
   206→function TerrainMesh({ conect, setMeshRef }) {
   207→  const geometry = useMemo(() => {
   208→    const seg = conect.segments || 64
   209→    const g = new THREE.PlaneGeometry(
   210→      conect.width || 50, conect.depth || 50, seg, seg
   211→    )
   212→    g.rotateX(-Math.PI / 2)
   213→    const positions = g.attributes.position
   214→    const heightScale = conect.heightScale || 5
   215→
   216→    // Se há heightmap exportado do TerrainEditor, usá-lo
   217→    if (conect.heightmap && conect.heightmap.length > 0) {
   218→      const hm = conect.heightmap
   219→      for (let i = 0; i < positions.count && i < hm.length; i++) {
   220→        positions.setY(i, hm[i] * heightScale)
   221→      }
   222→    } else {
   223→      // Fallback: gerar procedural
   224→      const seed = conect.heightmapSeed || 1
   225→      for (let i = 0; i < positions.count; i++) {
   226→        const x = positions.getX(i)
   227→        const z = positions.getZ(i)
   228→        const h =
   229→          Math.sin(x * 0.3 + seed) * Math.cos(z * 0.3 + seed) * 0.5 +
   230→          Math.sin(x * 0.1 + z * 0.1) * 0.3
   231→        positions.setY(i, h * heightScale)
   232→      }
   233→    }
   234→    g.computeVertexNormals()
   235→
   236→    // Vertex colors from splatmap (multi-layer blending) — P7
   237→    if (conect.splatmap && conect.textureLayers && conect.splatmap.length > 0) {
   238→      const maxLayers = conect.maxLayers || 4
   239→      const layers = conect.textureLayers
   240→      const colors = new Float32Array(positions.count * 3)
   241→      const sm = conect.splatmap
   242→      const hexToRgb = (hex) => {
   243→        const r = parseInt(hex.slice(1, 3), 16) / 255
   244→        const g = parseInt(hex.slice(3, 5), 16) / 255
   245→        const b = parseInt(hex.slice(5, 7), 16) / 255
   246→        return [r, g, b]
   247→      }
   248→      const layerRgb = layers.map((l) => hexToRgb(l.color || '#5a7d3a'))
   249→      for (let i = 0; i < positions.count; i++) {
   250→        let r = 0, g = 0, b = 0
   251→        for (let l = 0; l < maxLayers && l < layers.length; l++) {
   252→          const w = sm[i * maxLayers + l] || 0
   253→          const c = layerRgb[l] || [0.35, 0.49, 0.23]
   254→          r += c[0] * w
   255→          g += c[1] * w
   256→          b += c[2] * w
   257→        }
   258→        colors[i * 3] = r
   259→        colors[i * 3 + 1] = g
   260→        colors[i * 3 + 2] = b
   261→      }
   262→      g.setAttribute('color', new THREE.BufferAttribute(colors, 3))
   263→    }
   264→
   265→    return g
   266→  }, [conect.width, conect.depth, conect.segments, conect.heightScale, conect.heightmapSeed, conect.heightmap, conect.splatmap, conect.textureLayers, conect.maxLayers])
   267→
   268→  const hasVertexColors = !!(conect.splatmap && conect.textureLayers && conect.splatmap.length > 0)
   269→
   270→  return (
   271→    <mesh
   272→      ref={setMeshRef}
   273→      geometry={geometry}
   274→      position={conect.position}
   275→      receiveShadow
   276→      castShadow
   277→    >
   278→      <meshStandardMaterial
   279→        color={hasVertexColors ? '#ffffff' : (conect.textureColor || '#5a7d3a')}
   280→        vertexColors={hasVertexColors}
   281→        roughness={0.9}
   282→        metalness={0}
   283→      />
   284→    </mesh>
   285→  )
   286→}

   288→// ===== WaterObject =====
   289→function WaterMesh({ conect, setMeshRef }) {
   290→  const meshRef = useRef()
   291→  const time = useRef(0)
   292→  const waterQuality = useStore((s) => s.renderSettings?.waterQuality || 'basic')
   293→
   294→  // Material Pro (Gerstner + caustics + IOR + Fresnel) quando quality = professional
   295→  const proMaterial = useMemo(() => {
   296→    if (waterQuality !== 'professional') return null
   297→    return createWaterProMaterial({
   298→      color: conect.color || '#2f81f7',
   299→      deepColor: conect.deepColor || '#0a3d5c',
   300→      opacity: conect.opacity ?? 0.85,
   301→      waveHeight: conect.waveHeight ?? 0.2,
   302→      waveSpeed: conect.waveSpeed ?? 0.5,
   303→      waterMode: conect.waterMode || 'lake',
   304→      flowDirection: conect.flowDirection || 0,
   305→      foamEnabled: conect.foamEnabled !== false,
   306→      foamThreshold: conect.foamThreshold ?? 0.7,
   307→      depthGradient: conect.depthGradient !== false,
   308→      skyColor: '#88aacc',
   309→    })
   310→  }, [waterQuality, conect.color, conect.deepColor, conect.opacity, conect.waveHeight,
   311→      conect.waveSpeed, conect.waterMode, conect.flowDirection, conect.foamEnabled,
   312→      conect.foamThreshold, conect.depthGradient])
   313→
   314→  const geometry = useMemo(() => {
   315→    const [w, h] = conect.size || [20, 20]
   316→    const segs = waterQuality === 'professional' ? 48 : 32
   317→    const g = new THREE.PlaneGeometry(w, h, segs, segs)
   318→    g.rotateX(-Math.PI / 2)
   319→    return g
   320→  }, [conect.size, waterQuality])
   321→
   322→  // Animar ondas no useFrame (apenas para water básico; Pro usa shader)
   323→  useFrame((state, delta) => {
   324→    if (!meshRef.current) return
   325→    if (proMaterial) {
   326→      // Pro: actualizar uniforms do shader
   327→      if (proMaterial.uniforms) {
   328→        proMaterial.uniforms.uTime.value += delta
   329→        proMaterial.uniforms.uCameraPos.value.copy(state.camera.position)
   330→      }
   331→      return
   332→    }
   333→    // Básico: animar vértices
   334→    time.current += delta * (conect.waveSpeed || 0.5)
   335→    const pos = meshRef.current.geometry.attributes.position
   336→    const waveHeight = conect.waveHeight || 0.1
   337→    for (let i = 0; i < pos.count; i++) {
   338→      const x = pos.getX(i)
   339→      const z = pos.getZ(i)
   340→      const y = Math.sin(x * 0.5 + time.current) * waveHeight
   341→        + Math.cos(z * 0.5 + time.current * 0.7) * waveHeight * 0.5
   342→      pos.setY(i, y)
   343→    }
   344→    pos.needsUpdate = true
   345→    meshRef.current.geometry.computeVertexNormals()
   346→  })
   347→
   348→  return (
   349→    <mesh
   350→      ref={(node) => {
   351→        meshRef.current = node
   352→        if (typeof setMeshRef === 'function') setMeshRef(node)
   353→      }}
   354→      position={conect.position}
   355→      geometry={geometry}
   356→      material={proMaterial || undefined}
   357→    >
   358→      {!proMaterial && (
   359→        <meshStandardMaterial
   360→          color={conect.color || '#2f81f7'}
   361→          transparent
   362→          opacity={conect.opacity ?? 0.6}
   363→          roughness={0.1}
   364→          metalness={0.3}
   365→        />
   366→      )}
   367→    </mesh>
   368→  )
   369→}

   371→// ===== PathObject =====
   372→function PathMesh({ conect, setMeshRef }) {
   373→  if (!conect.points || conect.points.length === 0) return null
   374→  return (
   375→    <group ref={setMeshRef} position={conect.position}>
   376→      {conect.points.map((p, i) => (
   377→        <mesh key={i} position={p}>
   378→          <sphereGeometry args={[0.15, 12, 12]} />
   379→          <meshBasicMaterial color="#2f81f7" />
   380→        </mesh>
   381→      ))}
   382→      {/* Linhas entre pontos */}
   383→      {conect.points.length > 1 && (
   384→        <Line points={conect.points} loop={conect.loop} />
   385→      )}
   386→    </group>
   387→  )
   388→}
   389→
   390→function Line({ points, loop }) {
   391→  const geometry = useMemo(() => {
   392→    const pts = loop ? [...points, points[0]] : points
   393→    return new THREE.BufferGeometry().setFromPoints(pts.map((p) => new THREE.Vector3(...p)))
   394→  }, [points, loop])
   395→  return (
   396→    <line geometry={geometry}>
   397→      <lineBasicMaterial color="#2f81f7" transparent opacity={0.5} />
   398→    </line>
   399→  )
   400→}

   418→// ===== ParticleObject =====
   419→function ParticleMesh({ conect, setMeshRef }) {
   420→  const groupRef = useRef()
   421→  const timeRef = useRef(0)
   422→
   423→  // Usar BufferGeometry + Points para performance (1 draw call em vez de N)
   424→  const count = Math.min(conect.maxParticles || 100, 300)
   425→  const positionsRef = useRef(new Float32Array(count * 3))
   426→  const velocitiesRef = useRef(new Float32Array(count * 3))
   427→  const lifeRef = useRef(new Float32Array(count))
   428→
   429→  // Inicializar partículas
   430→  useMemo(() => {
   431→    const spread = conect.spread || 1
   432→    const speed = conect.particleSpeed || 0.5
   433→    for (let i = 0; i < count; i++) {
   434→      positionsRef.current[i * 3] = (Math.random() - 0.5) * spread
   435→      positionsRef.current[i * 3 + 1] = 0
   436→      positionsRef.current[i * 3 + 2] = (Math.random() - 0.5) * spread
   437→      velocitiesRef.current[i * 3] = (Math.random() - 0.5) * speed
   438→      velocitiesRef.current[i * 3 + 1] = Math.random() * speed + 0.2
   439→      velocitiesRef.current[i * 3 + 2] = (Math.random() - 0.5) * speed
   440→      lifeRef.current[i] = Math.random() * (conect.particleLife || 2)
   441→    }
   442→  }, [count, conect.spread, conect.particleSpeed, conect.particleLife])
   443→
   444→  const geometry = useMemo(() => {
   445→    const geo = new THREE.BufferGeometry()
   446→    geo.setAttribute('position', new THREE.BufferAttribute(positionsRef.current, 3))
   447→    return geo
   448→  }, [])
   449→
   450→  const material = useMemo(() => {
   451→    return new THREE.PointsMaterial({
   452→      color: conect.color || '#ffffff',
   453→      size: conect.particleSize || 0.1,
   454→      transparent: true,
   455→      opacity: 0.7,
   456→      sizeAttenuation: true,
   457→      depthWrite: false,
   458→    })
   459→  }, [conect.color, conect.particleSize])
   460→
   461→  // Animar partículas no useFrame
   462→  useFrame((_, delta) => {
   463→    timeRef.current += delta
   464→    const positions = positionsRef.current
   465→    const velocities = velocitiesRef.current
   466→    const life = lifeRef.current
   467→    const gravity = conect.gravity ?? -0.5
   468→    const maxLife = conect.particleLife || 2
   469→    const spread = conect.spread || 1
   470→
   471→    for (let i = 0; i < count; i++) {
   472→      // Atualizar vida
   473→      life[i] += delta
   474→      if (life[i] >= maxLife) {
   475→        // Reciclar partícula
   476→        positions[i * 3] = (Math.random() - 0.5) * spread
   477→        positions[i * 3 + 1] = 0
   478→        positions[i * 3 + 2] = (Math.random() - 0.5) * spread
   479→        velocities[i * 3] = (Math.random() - 0.5) * (conect.particleSpeed || 0.5)
   480→        velocities[i * 3 + 1] = Math.random() * (conect.particleSpeed || 0.5) + 0.2
   481→        velocities[i * 3 + 2] = (Math.random() - 0.5) * (conect.particleSpeed || 0.5)
   482→        life[i] = 0
   483→      } else {
   484→        // Mover partícula
   485→        positions[i * 3] += velocities[i * 3] * delta
   486→        positions[i * 3 + 1] += velocities[i * 3 + 1] * delta
   487→        positions[i * 3 + 2] += velocities[i * 3 + 2] * delta
   488→        // Aplicar gravidade
   489→        velocities[i * 3 + 1] += gravity * delta
   490→      }
   491→    }
   492→    geometry.attributes.position.needsUpdate = true
   493→  })
   494→
   495→  // Cleanup
   496→  useEffect(() => {
   497→    return () => {
   498→      geometry.dispose()
   499→      material.dispose()
   500→    }
   501→  }, [geometry, material])
   502→
   503→  return (
   504→    <points
   505→      ref={(node) => {
   506→        groupRef.current = node
   507→        if (typeof setMeshRef === 'function') setMeshRef(node)
   508→      }}
   509→      position={conect.position}
   510→      geometry={geometry}
   511→      material={material}
   512→    />
   513→  )
   514→}

   516→// ===== ViewObject (câmara visível no editor) =====
   517→function ViewObjectMesh({ conect, setMeshRef }) {
   518→  return (
   519→    <group
   520→      ref={setMeshRef}
   521→      position={conect.position}
   522→      rotation={conect.rotation}
   523→      scale={conect.scale}
   524→      userData={{ conectInstanceId: conect.instanceId, isViewObject: true }}
   525→    >
   526→      <mesh castShadow>
   527→        <boxGeometry args={[0.5, 0.35, 0.6]} />
   528→        <meshStandardMaterial color="#f4a261" emissive="#f4a261" emissiveIntensity={0.2} />
   529→      </mesh>
   530→      <mesh position={[0, 0, -0.45]} rotation={[Math.PI / 2, 0, 0]}>
   531→        <cylinderGeometry args={[0.18, 0.22, 0.3, 16]} />
   532→        <meshStandardMaterial color="#2a9d8f" emissive="#2a9d8f" emissiveIntensity={0.1} />
   533→      </mesh>
   534→      <mesh position={[0, 0, -1.5]}>

   537→      </mesh>
   538→      {/* Indicador do papel da câmara */}
   539→      {conect.cameraRole && (
   540→        <mesh position={[0, 0.4, 0]}>
   541→          <sphereGeometry args={[0.1, 8, 8]} />
   542→          <meshBasicMaterial color={
   543→            conect.cameraRole === 'player' ? '#2f81f7'
   544→            : conect.cameraRole === 'primary' ? '#3fb950'
   545→            : '#d29922' // secondary

   564→// ===== TrailObject (rasto visual) =====
   565→// Renderiza uma linha que segue a posição do objeto pai.
   566→// Em tempo de execução, o GameRunner atualiza as posições do trail.
   567→function TrailMesh({ conect, setMeshRef }) {
   568→  const groupRef = useRef()
   569→  const lineRef = useRef()
   570→  const maxPoints = conect.length || 30
   571→  const positionsRef = useRef(new Float32Array(maxPoints * 3))
   572→  const currentIndexRef = useRef(0)
   573→  const frameCountRef = useRef(0)
   574→
   575→  // Inicializar todas as posições na posição inicial do trail
   576→  useMemo(() => {
   577→    const px = conect.position?.[0] || 0
   578→    const py = conect.position?.[1] || 0
   579→    const pz = conect.position?.[2] || 0
   580→    for (let i = 0; i < maxPoints; i++) {
   581→      positionsRef.current[i * 3] = px
   582→      positionsRef.current[i * 3 + 1] = py
   583→      positionsRef.current[i * 3 + 2] = pz
   584→    }
   585→  }, [maxPoints, conect.position])
   586→
   587→  const geometry = useMemo(() => {
   588→    const geo = new THREE.BufferGeometry()
   589→    geo.setAttribute('position', new THREE.BufferAttribute(positionsRef.current, 3))
   590→    return geo
   591→  }, [])
   592→
   593→  // Seguir o followTarget e atualizar o trail
   594→  useFrame(() => {
   595→    if (!conect.followTarget) return
   596→    // Procurar o mesh do followTarget nos conectMeshRefs
   597→    const targetMesh = window._flirConectMeshRefs?.get(conect.followTarget)
   598→      || window._flirMeshRefs?.get(conect.followTarget)
   599→    if (!targetMesh) return
   600→
   601→    // Atualizar a cada N frames (não todos os frames — performance)
   602→    const updateInterval = Math.max(1, Math.floor(60 / (conect.updateRate || 30)))
   603→    frameCountRef.current++
   604→    if (frameCountRef.current < updateInterval) return
   605→    frameCountRef.current = 0
   606→
   607→    // Shift: mover todos os pontos uma posição para trás
   608→    const positions = positionsRef.current
   609→    for (let i = maxPoints - 1; i > 0; i--) {
   610→      positions[i * 3] = positions[(i - 1) * 3]
   611→      positions[i * 3 + 1] = positions[(i - 1) * 3 + 1]
   612→      positions[i * 3 + 2] = positions[(i - 1) * 3 + 2]
   613→    }
   614→    // Novo ponto na posição do target
   615→    positions[0] = targetMesh.position.x
   616→    positions[1] = targetMesh.position.y
   617→    positions[2] = targetMesh.position.z
   618→
   619→    geometry.attributes.position.needsUpdate = true
   620→    geometry.setDrawRange(0, maxPoints)
   621→  })
   622→
   623→  // Cleanup
   624→  useEffect(() => {
   625→    return () => { geometry.dispose() }
   626→  }, [geometry])
   627→
   628→  return (
   629→    <group ref={setMeshRef} position={[0, 0, 0]}>
   630→      <line ref={lineRef} geometry={geometry}>
   631→        <lineBasicMaterial
   632→          color={conect.color || '#2f81f7'}
   633→          transparent
   634→          opacity={conect.fade !== false ? 0.6 : 1}
   635→          linewidth={conect.width || 2}
   636→        />
   637→      </line>
   638→    </group>
   639→  )
   640→}

   659→// ===== SkyObject — esfera de céu com shader procedural =====
   660→function SkyMesh({ conect, setMeshRef }) {
   661→  const { scene } = useThree()
   662→  const meshRef = useRef()
   663→
   664→  // Ler propriedades com os nomes correctos da taxonomy
   665→  const skyType = conect.skyType || 'gradient'
   666→  const topColor = conect.topColor || conect.gradientTop || '#1a4d8f'
   667→  const bottomColor = conect.bottomColor || conect.gradientBottom || '#aac4e8'
   668→  const solidColor = conect.solidColor || conect.color || '#87ceeb'
   669→
   670→  // Material Pro para procedural (Rayleigh + Mie scattering)
   671→  const proMaterial = useMemo(() => {
   672→    if (skyType !== 'procedural') return null
   673→    // Calcular direção do sol a partir de elevation/azimuth
   674→    const sunDir = calculateSunDirection(
   675→      conect.sunElevation || 25,   // graus
   676→      172,                          // dia do ano (verão)
   677→      0                             // latitude equador
   678→    )
   679→    return createSkyProMaterial({
   680→      sunDirection: sunDir.toArray(),
   681→      sunIntensity: 15,
   682→      rayleigh: conect.rayleigh ?? 2.5,
   683→      mie: 0.5,
   684→      turbidity: conect.turbidity ?? 10,
   685→      starsEnabled: conect.starsEnabled || false,
   686→    })
   687→  }, [skyType, conect.sunElevation, conect.rayleigh, conect.turbidity, conect.starsEnabled])
   688→
   689→  // Animar uTime do sky shader
   690→  useFrame((_, delta) => {
   691→    if (proMaterial && proMaterial.uniforms?.uTime) {
   692→      proMaterial.uniforms.uTime.value += delta
   693→    }
   694→  })
   695→
   696→  // Aplicar background à cena
   697→  useEffect(() => {
   698→    if (!scene) return
   699→    if (skyType === 'procedural' && proMaterial) {
   700→      // Procedural usa esfera com shader — limpar background para não tapar
   701→      scene.background = null
   702→    } else if (skyType === 'gradient') {
   703→      // Gradiente via canvas
   704→      const canvas = document.createElement('canvas')
   705→      canvas.width = 2; canvas.height = 256
   706→      const ctx = canvas.getContext('2d')
   707→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
   708→      grad.addColorStop(0, topColor)
   709→      grad.addColorStop(1, bottomColor)
   710→      ctx.fillStyle = grad
   711→      ctx.fillRect(0, 0, 2, 256)
   712→      const tex = new THREE.CanvasTexture(canvas)
   713→      tex.colorSpace = THREE.SRGBColorSpace
   714→      scene.background = tex
   715→      return () => { tex.dispose() }
   716→    } else if (skyType === 'hdri' && conect.hdriUrl) {
   717→      import('three/examples/jsm/loaders/RGBELoader.js').then(({ RGBELoader }) => {
   718→        const loader = new RGBELoader()
   719→        loader.load(conect.hdriUrl, (texture) => {
   720→          texture.mapping = THREE.EquirectangularReflectionMapping
   721→          scene.background = texture
   722→          scene.environment = texture
   723→        })
   724→      }).catch(() => {
   725→        scene.background = new THREE.Color(solidColor)
   726→      })
   727→    } else {
   728→      // solid
   729→      scene.background = new THREE.Color(solidColor)
   730→    }
   731→  }, [scene, skyType, topColor, bottomColor, solidColor, conect.hdriUrl, proMaterial])
   732→
   733→  // Esfera grande com material de céu
   734→  if (skyType === 'procedural' && proMaterial) {
   735→    // Esfera com shader procedural (Rayleigh + Mie)
   736→    return (
   737→      <mesh
   738→        ref={(node) => {
   739→          meshRef.current = node
   740→          if (typeof setMeshRef === 'function') setMeshRef(node)
   741→        }}
   742→        scale={[100, 100, 100]}
   743→        material={proMaterial}
   744→      >
   745→        <sphereGeometry args={[1, 32, 16]} />
   746→      </mesh>
   747→    )
   748→  }
   749→  if (skyType === 'solid' || skyType === 'hdri') {
   750→    return (
   751→      <mesh ref={setMeshRef} scale={[100, 100, 100]}>
   752→        <sphereGeometry args={[1, 32, 16]} />
   753→        <meshBasicMaterial
   754→          color={solidColor}
   755→          side={THREE.BackSide}
   756→          fog={false}
   757→        />
   758→      </mesh>
   759→    )
   760→  }
   761→  // gradient: background canvas, mesh invisível
   762→  return (
   763→    <mesh ref={setMeshRef} visible={false} scale={[0.001, 0.001, 0.001]}>
   764→      <sphereGeometry args={[1, 8, 8]} />
   765→      <meshBasicMaterial />
   766→    </mesh>
   767→  )
   768→}

   845→// ===== ReferenceObject — renderiza conteúdo de outra cena =====
   846→function ReferenceMesh({ conect, setMeshRef }) {
   847→  const scenes = useStore((s) => s.scenes)
   848→  const targetScene = scenes.find(s => s.id === conect.targetSceneId)
   849→
   850→  if (!targetScene) {
   851→    // Sem cena de destino — mostrar gizmo de placeholder
   852→    return (
   853→      <mesh ref={setMeshRef} position={conect.position} rotation={conect.rotation} scale={conect.scale}>
   854→        <boxGeometry args={[0.5, 0.5, 0.5]} />
   855→        <meshBasicMaterial color="#8957e5" wireframe />
   856→      </mesh>
   857→    )
   858→  }
   859→
   860→  // Renderizar objetos do catálogo da cena referenciada
   861→  const refObjects = (targetScene.objects || []).map((instance) => {
   862→    const obj = objects.find((o) => o.id === instance.objectId)
   863→    if (!obj) return null
   864→    const sceneObj = {
   865→      ...obj,
   866→      id: `${conect.instanceId}_ref_${instance.instanceId}`,
   867→      position: [
   868→        (conect.position?.[0] || 0) + (instance.position?.[0] || 0),
   869→        (conect.position?.[1] || 0) + (instance.position?.[1] || 0),
   870→        (conect.position?.[2] || 0) + (instance.position?.[2] || 0),
   871→      ],
   872→      rotation: instance.rotation || [0, 0, 0],
   873→      scale: instance.scale || [1, 1, 1],
   874→    }
   875→    return <SceneObject key={instance.instanceId} obj={sceneObj} isSelected={false} onSelect={() => {}} />
   876→  }).filter(Boolean)
   877→
   878→  return (
   879→    <group ref={setMeshRef} position={conect.position} rotation={conect.rotation} scale={conect.scale}>
   880→      {refObjects}
   881→    </group>
   882→  )
   883→}
