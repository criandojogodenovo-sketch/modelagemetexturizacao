    57→// Componente interno que aplica o fundo da cena
    58→function SceneBackgroundSolid({ background }) {
    59→  const { scene } = useThree()
    60→  useEffect(() => {
    61→    if (background.type === 'gradient') {
    62→      const canvas = document.createElement('canvas')
    63→      canvas.width = 2; canvas.height = 256
    64→      const ctx = canvas.getContext('2d')
    65→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
    66→      grad.addColorStop(0, background.gradientTop)
    67→      grad.addColorStop(1, background.gradientBottom)
    68→      ctx.fillStyle = grad; ctx.fillRect(0, 0, 2, 256)
    69→      const tex = new THREE.CanvasTexture(canvas)
    70→      tex.colorSpace = THREE.SRGBColorSpace
    71→      scene.background = tex
    72→      return () => tex.dispose()
    73→    } else {
    74→      scene.background = new THREE.Color(background.color)
    75→    }
    76→  }, [background, scene])
    77→  return null
    78→}

    80→// ===== FogApplier — aplica FogObject à cena 3D =====
    81→function FogApplier({ conects }) {
    82→  const { scene } = useThree()
    83→  const fogConect = (conects || []).find(c => c.type === 'FogObject')
    84→  useEffect(() => {
    85→    if (!fogConect) {
    86→      scene.fog = null
    87→      return
    88→    }
    89→    if (fogConect.fogType === 'exponential') {
    90→      scene.fog = new THREE.FogExp2(fogConect.color || '#a0a0a0', fogConect.density || 0.02)
    91→    } else {
    92→      scene.fog = new THREE.Fog(fogConect.color || '#a0a0a0', fogConect.near || 5, fogConect.far || 50)
    93→    }
    94→  }, [fogConect?.fogType, fogConect?.color, fogConect?.near, fogConect?.far, fogConect?.density, scene])
    95→  return null
    96→}

    98→// ===== InstancingRenderer — adiciona HardwareInstancingSystems à cena e atualiza por frame =====
    99→function InstancingRenderer() {
   100→  const { scene, camera } = useThree()
   101→  const addedRef = useRef(new Set())
   102→
   103→  useFrame(() => {
   104→    const systems = window._flirInstancingSystems
   105→    if (!systems || systems.length === 0) return
   106→    for (const sys of systems) {
   107→      // Adicionar à cena se ainda não foi adicionado
   108→      if (!addedRef.current.has(sys)) {
   109→        try {
   110→          sys.addToScene(scene)
   111→          addedRef.current.add(sys)
   112→        } catch (e) {
   113→          // já foi adicionado ou sistema dispuesto
   114→        }
   115→      }
   116→      // Atualizar frustum culling + LOD + matrizes por instância
   117→      try { sys.update(camera) } catch {}
   118→    }
   119→  })
   120→
   121→  // Limpar ao desmontar
   122→  useEffect(() => {
   123→    return () => {
   124→      for (const sys of addedRef.current) {
   125→        try {
   126→          for (const mesh of sys.instanceMeshes || []) {
   127→            scene.remove(mesh)
   128→          }
   129→        } catch {}
   130→      }
   131→      addedRef.current.clear()
   132→    }
   133→  }, [scene])
   134→
   135→  return null
   136→}

   138→// ===== TerrainSculptBridge — ativa escultura 3D no viewport quando um TerrainObject está selecionado =====
   139→function TerrainSculptBridge({ activeScene, conectMeshRefs, isGameMode, onDragStateChange }) {
   140→  // Procurar se há um TerrainObject selecionado e um modo de escultura ativo
   141→  const terrainSculptActive = useStore((s) => s.terrainSculptActive)
   142→  const terrainBrush = useStore((s) => s.terrainBrush)
   143→  const selectedConectId = useStore((s) => s.selectedConectId)
   144→  const updateConect = useStore((s) => s.updateConect)
   145→
   146→  // Encontrar o TerrainObject atualmente selecionado
   147→  const terrainConect = useMemo(() => {
   148→    if (!activeScene?.conects || !selectedConectId) return null
   149→    const c = activeScene.conects.find(c => c.instanceId === selectedConectId && c.type === 'TerrainObject')
   150→    return c || null
   151→  }, [activeScene, selectedConectId])
   152→
   153→  const terrainMesh = useMemo(() => {
   154→    if (!terrainConect) return null
   155→    return conectMeshRefs.current.get(terrainConect.instanceId) || null
   156→  }, [terrainConect, conectMeshRefs, activeScene])
   157→
   158→  // Callback quando heightmap é alterado pela escultura 3D
   159→  const handleHeightmapChange = useCallback((newHm) => {
   160→    if (!terrainConect) return
   161→    // Persistir alteração no conect
   162→    updateConect(terrainConect.instanceId, {
   163→      heightmap: Array.from(newHm),
   164→    })
   165→  }, [terrainConect, updateConect])
   166→
   167→  if (!terrainSculptActive || !terrainConect || !terrainMesh || isGameMode) return null
   168→
   169→  return (
   170→    <TerrainSculpt3D
   171→      terrainMesh={terrainMesh}
   172→      heightmap={terrainConect.heightmap ? new Float32Array(terrainConect.heightmap) : null}
   173→      seg={terrainConect.segments || 64}
   174→      brushMode={terrainBrush?.mode || 'raise'}
   175→      brushSize={terrainBrush?.size || 8}
   176→      brushStrength={terrainBrush?.strength || 0.5}
   177→      falloffType={terrainBrush?.falloff || 'smooth'}
   178→      onHeightmapChange={handleHeightmapChange}
   179→      onDragStateChange={onDragStateChange}
   180→      isActive={terrainSculptActive}
   181→    />
   182→  )
   183→}

   186→function GameMode({ activeScene, objects, meshRefs, conectMeshRefs, isGameMode }) {
   187→  const { camera } = useThree()
   188→  const physicsRef = useRef(null)
   189→  const runtimesRef = useRef(new Map())
   190→  const animPlayersRef = useRef(new Map())
   191→  const npcAIsRef = useRef(new Map())
   192→  const timerStatesRef = useRef(new Map())
   193→  const joystickRef = useRef({ x: 0, z: 0, active: false })

   195→  const weaponStateRef = useRef({ equipped: false, ammo: 0, maxAmmo: 0, damage: 0, fireRate: 0.3, range: 50, reloadTime: 2, lastShot: 0 })
   196→  const collisionEventsRef = useRef(new Map()) // instanceId → Set de otherIds em contacto
   197→  const checkpointRef = useRef(null) // último checkpoint registado
   198→  const skyRef = useRef(null) // referência ao SkyObject ativo
   199→
   200→  // Expor meshRefs globalmente para TrailObject poder seguir objetos
   201→  useEffect(() => {
   202→    window._flirConectMeshRefs = conectMeshRefs
   203→    window._flirMeshRefs = meshRefs
   204→    return () => {
   205→      window._flirConectMeshRefs = null
   206→      window._flirMeshRefs = null
   207→    }
   208→  }, [conectMeshRefs, meshRefs])
   209→
   210→  // P2.5 fix: armazenar a cena ativa num ref para que modificações feitas pelo
   211→  // FlirCode (createObject, etc.) NÃO reiniciem o jogo (evita loop infinito).
   212→  // O useEffect só deve correr quando isGameMode muda (true→false ou false→true),
   213→  // não quando activeScene é modificado por spawnObject.
   214→  const activeSceneRef = useRef(activeScene)
   215→  const objectsRef = useRef(objects)
   216→  const gameStartedRef = useRef(false)
   217→
   218→  // Expor joystickRef globalmente para o GameUIOverlay poder atualizá-lo
   219→  useEffect(() => {
   220→    window._flirJoystick = joystickRef.current
   221→    return () => { window._flirJoystick = null }
   222→  }, [])
   223→
   224→  // Use o ref para o setup
   225→  const setupScene = activeSceneRef.current
   226→
   227→  // Fase 5: Expor multiplayer globalmente e configurar sync de estado do PersonalObject
   228→  useEffect(() => {
   229→    if (!isGameMode) return
   230→    let mp
   231→    import('../../utils/multiplayer/multiplayerManager').then(({ multiplayer }) => {
   232→      mp = multiplayer
   233→      window._multiplayer = mp
   234→      // Configurar getter de estado local (posição do PersonalObject)
   235→      mp.setLocalStateGetter(() => {
   236→        for (const conect of setupScene?.conects || []) {
   237→          if (conect.type === 'PersonalObject') {
   238→            const mesh = conectMeshRefs.current.get(conect.instanceId)
   239→            if (mesh) {
   240→              return {
   241→                position: [mesh.position.x, mesh.position.y, mesh.position.z],
   242→                rotation: [mesh.rotation.x, mesh.rotation.y, mesh.rotation.z],
   243→                playerId: mp.playerId,
   244→              }
   245→            }
   246→          }
   247→        }
   248→        return null
   249→      })
   250→      // Registar callbacks para eventos multiplayer → FlirCode
   251→      mp.on('playerJoin', (playerId) => {
   252→        for (const rt of runtimesRef.current.values()) {
   253→          rt.triggerEvent('onPlayerJoin', { playerId })
   254→        }
   255→      })
   256→      mp.on('playerLeave', (playerId) => {
   257→        for (const rt of runtimesRef.current.values()) {
   258→          rt.triggerEvent('onPlayerLeave', { playerId })
   259→        }
   260→      })
   261→      mp.on('message', (playerId, data) => {
   262→        for (const rt of runtimesRef.current.values()) {
   263→          rt.triggerEvent('onMessage', { playerId, data })
   264→        }
   265→      })
   266→    })
   267→    return () => {
   268→      window._multiplayer = null
   269→    }
   270→  }, [isGameMode, setupScene])
   271→
   272→  // Atualizar refs sem re-disparar o useEffect
   273→  useEffect(() => {
   274→    // Só atualizar o ref se o jogo ainda não começou OU se a cena mudou de verdade
   275→    // (não por spawnObject, mas por mudança de cena manual via changeScene)
   276→    if (!gameStartedRef.current || (activeScene && activeScene.id !== activeSceneRef.current?.id)) {
   277→      activeSceneRef.current = activeScene
   278→      objectsRef.current = objects
   279→    } else {
   280→      // Jogo em curso: manter o snapshot antigo, mas atualizar objects (para spawn)
   281→      objectsRef.current = objects
   282→    }
   283→  }, [activeScene, objects])
   284→
   285→  // Setup quando o modo jogo é activado
   286→  useEffect(() => {
   287→    if (!isGameMode || !setupScene) return
   288→    gameStartedRef.current = true
   289→
   290→    debugLog('Jogo iniciado', 'log', 'Game')
   291→
   292→    // Criar gameContext
   293→    const gameContext = {
   294→      globalVars: { _score: 0 },
   295→      // P2.5: setVar/getVar — expostos no gameContext para o FlirCode
   296→      setVar: (name, value) => { gameContext.globalVars[name] = value },
   297→      getVar: (name) => gameContext.globalVars[name] ?? 0,
   298→      moveObject: (instanceId, direction, speed) => {
   299→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   300→        if (mesh) {
   301→          mesh.position.x += direction[0] * speed * 0.016
   302→          mesh.position.y += direction[1] * speed * 0.016
   303→          mesh.position.z += direction[2] * speed * 0.016
   304→        }
   305→      },
   306→      rotateObject: (instanceId, rotation) => {
   307→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   308→        if (mesh) {
   309→          mesh.rotation.x += THREE.MathUtils.degToRad(rotation[0]) * 0.016
   310→          mesh.rotation.y += THREE.MathUtils.degToRad(rotation[1]) * 0.016
   311→          mesh.rotation.z += THREE.MathUtils.degToRad(rotation[2]) * 0.016
   312→        }
   313→      },
   314→      playAnimation: (instanceId, clip) => {
   315→        const player = animPlayersRef.current.get(instanceId)
   316→        if (player) {
   317→          // Usar blendTime do AnimationBoostObject se existir
   318→          const boost = (setupScene?.conects || []).find((c) => c.type === 'AnimationBoostObject')
   319→          player.play(clip, { loop: true, blendTime: boost?.blendTime ?? 0.3 })
   320→        }
   321→      },
   322→      playSound: (url) => { try { new Audio(url).play() } catch {} },
   323→      playSoundByName: (name) => {
   324→        const sc = (setupScene.conects || []).find((c) => c.type === 'SoundObject' && c.name === name)
   325→        if (sc && sc.url) { try { const a = new Audio(sc.url); a.volume = sc.volume ?? 1; a.loop = sc.loop || false; a.play() } catch {} }
   326→      },
   327→      destroyObject: (instanceId) => {
   328→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   329→        if (mesh) mesh.visible = false
   330→      },
   331→      spawnObject: (objectName, position) => {
   332→        const obj = objectsRef.current.find((o) => o.name === objectName)
   333→        if (obj) useStore.getState().addObjectToScene(obj.id, position)
   334→      },
   335→      changeScene: (sceneName) => {
   336→        const scenes = useStore.getState().scenes
   337→        const target = scenes.find((s) => s.name === sceneName)
   338→        if (target) useStore.getState().setActiveScene(target.id)
   339→      },
   340→      setVisible: (instanceId, visible) => {
   341→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   342→        if (mesh) mesh.visible = visible
   343→      },
   344→      applyForce: (instanceId, force) => physicsRef.current?.applyForce(instanceId, force),
   345→      jumpPlayer: (instanceId) => physicsRef.current?.jumpPersonal(instanceId),
   346→      showUIScreen: (name) => {
   347→        const s = useStore.getState().uiScreens.find((s) => s.name === name)
   348→        if (s) useStore.getState().setUIScreenVisible(s.id, true)
   349→      },
   350→      hideUIScreen: (name) => {
   351→        const s = useStore.getState().uiScreens.find((s) => s.name === name)
   352→        if (s) useStore.getState().setUIScreenVisible(s.id, false)
   353→      },
   354→      getUIValue: (name) => {
   355→        // Procurar em uiScreens
   356→        for (const s of useStore.getState().uiScreens) {
   357→          const el = s.elements.find((e) => e.name === name)
   358→          if (el) return el.value ?? el.text ?? ''
   359→        }
   360→        // Procurar em Conects de UI (TextObject, ButtonObject)
   361→        const scene = useStore.getState().scenes.find((s) => s.id === useStore.getState().activeSceneId)
   362→        const conect = (scene?.conects || []).find((c) => c.name === name && (c.type === 'TextObject' || c.type === 'ButtonObject'))
   363→        if (conect) return conect.text ?? conect.label ?? conect.value ?? ''
   364→        return ''
   365→      },
   366→      setUIValue: (name, val) => {
   367→        // Procurar em uiScreens
   368→        for (const s of useStore.getState().uiScreens) {
   369→          const el = s.elements.find((e) => e.name === name)
   370→          if (el) { useStore.getState().updateUIElement(el.id, { value: val, text: val, label: val }); return }
   371→        }
   372→        // Procurar em Conects de UI (TextObject, ButtonObject)
   373→        const state = useStore.getState()
   374→        const scene = state.scenes.find((s) => s.id === state.activeSceneId)
   375→        if (scene) {
   376→          const conect = (scene.conects || []).find((c) => c.name === name && (c.type === 'TextObject' || c.type === 'ButtonObject'))
   377→          if (conect) {
   378→            state.updateConect(conect.instanceId, { text: String(val), label: String(val) })
   379→            return
   380→          }
   381→        }
   382→      },
   383→      triggerUIEvent: (eventName, payload) => {
   384→        for (const rt of runtimesRef.current.values()) rt.triggerEvent(eventName, payload)
   385→      },
   386→      collidingWith: (instanceId, type) => {
   387→        if (!physicsRef.current) return false
   388→        // Usar contactos reais do cannon-es em vez de distância
   389→        const contacts = physicsRef.current.getContacts?.(instanceId)
   390→        if (contacts && contacts.length > 0) {
   391→          for (const otherId of contacts) {
   392→            const otherEntry = physicsRef.current.bodies.get(otherId)
   393→            if (!otherEntry) continue
   394→            if (otherEntry.conect.type === type || otherEntry.conect.name === type) {
   395→              return true
   396→            }
   397→          }
   398→          return false
   399→        }
   400→        // Fallback: usar eventos de colisão registados
   401→        const collisionSet = collisionEventsRef.current.get(instanceId)
   402→        if (collisionSet && collisionSet.size > 0) {
   403→          for (const otherId of collisionSet) {
   404→            const otherEntry = physicsRef.current.bodies.get(otherId)
   405→            if (!otherEntry) continue
   406→            if (otherEntry.conect.type === type || otherEntry.conect.name === type) {
   407→              return true
   408→            }
   409→          }
   410→        }
   411→        return false
   412→      },
   413→      distanceTo: (instanceId, targetName) => {

   421→          }
   422→        }
   423→        for (const conect of setupScene.conects || []) {
   424→          if (conect.name === targetName) {
   425→            const target = conectMeshRefs.current.get(conect.instanceId)
   426→            const source = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   427→            if (source && target) return source.position.distanceTo(target.position)
   428→          }
   429→        }
   430→        return 0
   431→      },
   432→      isTouching: () => joystickRef.current.active,
   433→      // Fase 5: Multiplayer functions
   434→      sendMessage: (data) => {
   435→        try {
   436→          const mp = window._multiplayer
   437→          if (mp && mp.connected) mp.sendMessage(data)
   438→        } catch {}
   439→      },

   443→      },
   444→      getPlayerState: (playerId) => {

   446→        return mp && mp.connected ? mp.getPlayerState(playerId) : null
   447→      },
   448→
   449→      // ===== Inventário (ItemObject auto-pickup) =====
   450→      addToInventory: (itemName, qty = 1) => {
   451→        inventoryRef.current[itemName] = (inventoryRef.current[itemName] || 0) + qty
   452→        window._flirInventory = inventoryRef.current
   453→        debugLog(`Item "${itemName}" adicionado (total: ${inventoryRef.current[itemName]})`, 'log', 'Inventory')
   454→      },
   455→      removeFromInventory: (itemName, qty = 1) => {
   456→        if (!inventoryRef.current[itemName]) return
   457→        inventoryRef.current[itemName] = Math.max(0, inventoryRef.current[itemName] - qty)
   458→        if (inventoryRef.current[itemName] === 0) delete inventoryRef.current[itemName]
   459→        window._flirInventory = inventoryRef.current
   460→      },
   461→      getInventoryCount: (itemName) => inventoryRef.current[itemName] || 0,
   462→      hasItem: (itemName) => (inventoryRef.current[itemName] || 0) > 0,
   463→
   464→      // ===== Combate (WeaponObject) =====
   465→      equipWeapon: (damage, ammo, fireRate, range, reloadTime) => {
   466→        weaponStateRef.current = {
   467→          equipped: true,
   468→          ammo, maxAmmo: ammo,
   469→          damage: damage || 10,
   470→          fireRate: fireRate || 0.3,
   471→          range: range || 50,
   472→          reloadTime: reloadTime || 2,
   473→          lastShot: 0,
   474→        }
   475→        window._flirCrosshair = true
   476→        debugLog(`Arma equipada (dano: ${damage}, munições: ${ammo})`, 'log', 'Weapon')
   477→      },
   478→      shoot: () => {
   479→        const w = weaponStateRef.current
   480→        if (!w.equipped || w.ammo <= 0) return false
   481→        const now = performance.now() / 1000
   482→        if (now - w.lastShot < w.fireRate) return false
   483→        w.lastShot = now
   484→        w.ammo--

   938→    window.addEventListener('keyup', onKeyUp)
   939→    window._flirKeys = keys
   940→
   941→    return () => {
   942→      for (const rt of runtimesRef.current.values()) rt.dispose()
   943→      runtimesRef.current.clear()
   944→      for (const ai of npcAIsRef.current.values()) ai.dispose()
   945→      npcAIsRef.current.clear()
   946→      for (const [, s] of timerStatesRef.current) { if (s.interval) clearInterval(s.interval); if (s.audio) s.audio.pause() }
   947→      timerStatesRef.current.clear()
   948→      animPlayersRef.current.clear()
   949→      physicsRef.current?.dispose()
   950→      physicsRef.current = null
   951→      window.removeEventListener('keydown', onKeyDown)
   952→      window.removeEventListener('keyup', onKeyUp)
   953→      window._flirGameContext = null
   954→      gameStartedRef.current = false
   955→    }
   956→  }, [isGameMode, setupScene])
   957→
   958→  // Loop do jogo
   959→  useFrame((_, delta) => {
   960→    if (!isGameMode) return
   961→
   962→    // Física
   963→    if (physicsRef.current) {
   964→      physicsRef.current.update(delta)
   965→      // Sincronizar meshes com bodies
   966→      for (const [id, entry] of physicsRef.current.bodies) {
   967→        const mesh = meshRefs.current.get(id) || conectMeshRefs.current.get(id)
   968→        if (mesh) {
   969→          mesh.position.copy(entry.body.position)
   970→          mesh.quaternion.copy(entry.body.quaternion)
   971→        }
   972→      }
   973→    }
   974→
   975→    // Actualizar variáveis de posição do jogador para FlirCode
   976→    const playerConect = (setupScene?.conects || []).find((c) => c.type === 'PersonalObject')
   977→    if (playerConect) {
   978→      const playerMesh = conectMeshRefs.current.get(playerConect.instanceId)
   979→      if (playerMesh) {
   980→        gameContext.setVar('_player_x', playerMesh.position.x)
   981→        gameContext.setVar('_player_y', playerMesh.position.y)
   982→        gameContext.setVar('_player_z', playerMesh.position.z)
   983→        gameContext.setVar('_y_pos', playerMesh.position.y)
   984→      }
   985→    }
   986→
   987→    // FlirCode onTick
   988→    for (const rt of runtimesRef.current.values()) {
   989→      rt.update(delta)
   990→      rt.triggerEvent('tick', { deltaTime: delta })
   991→    }
   992→
   993→    // Animation players
   994→    for (const player of animPlayersRef.current.values()) player.update(delta)
   995→
   996→    // NPC AI
   997→    for (const ai of npcAIsRef.current.values()) ai.update(delta)
   998→
   999→    // Timers
  1000→    for (const [id, state] of timerStatesRef.current) {
  1001→      if (id.startsWith('_')) continue
  1002→      state.remaining -= delta
  1003→      if (state.remaining <= 0) {
  1004→        const rt = runtimesRef.current.get(id)
  1005→        if (rt) rt.triggerEvent('onTimer')
  1006→        if (state.loop) state.remaining = state.duration
  1007→        else timerStatesRef.current.delete(id)
  1008→      }
  1009→    }
  1010→
  1011→    // Joystick/WASD → PersonalObject (camera-relative movement, like Godot's Input.get_vector + transform.basis)
  1012→    const keys = window._flirKeys || {}
  1013→    if (joystickRef.current.active || keys['w'] || keys['a'] || keys['s'] || keys['d']) {
  1014→      // Read camera yaw so movement is relative to where the player is looking (FPS standard).
  1015→      // camera.rotation was set earlier via camera.rotation.set(pitch, yaw, 0, 'YXZ') OR will be set below.
  1016→      // We read from window._flirCameraRotation (the source of truth updated by CameraTouchZoneControl).
  1017→      const yaw = window._flirCameraRotation?.yaw || 0
  1018→      const cosY = Math.cos(yaw)
  1019→      const sinY = Math.sin(yaw)
  1020→      for (const conect of activeScene.conects || []) {
  1021→        if (conect.type === 'PersonalObject') {
  1022→          const speed = conect.moveSpeed || conect.speed || 5
  1023→          let mx = 0, mz = 0 // mx = right axis, mz = forward axis (negative = forward)
  1024→          if (joystickRef.current.active) { mx = joystickRef.current.x * speed; mz = joystickRef.current.z * speed }
  1025→          if (keys['w']) mz = -speed
  1026→          if (keys['s']) mz = speed
  1027→          if (keys['a']) mx = -speed
  1028→          if (keys['d']) mx = speed
  1029→          // Rotate (mx, mz) by camera yaw on the XZ plane:
  1030→          //   forwardVec = (-sin(yaw), -cos(yaw))  [camera looks down -Z at yaw=0]
  1031→          //   rightVec   = ( cos(yaw), -sin(yaw))
  1032→          //   worldVel   = mx * rightVec + (-mz) * forwardVec
  1033→          // Simplified:
  1034→          //   vx =  mx * cos(yaw) + mz * sin(yaw)
  1035→          //   vz = -mx * sin(yaw) + mz * cos(yaw)
  1036→          const vx =  mx * cosY + mz * sinY
  1037→          const vz = -mx * sinY + mz * cosY
  1038→          physicsRef.current?.movePersonal(conect.instanceId, [vx, 0, vz], 1)
  1039→          if ((keys[' '] || keys['space']) && conect.canJump) {
  1040→            physicsRef.current?.jumpPersonal(conect.instanceId)
  1041→          }
  1042→        }
  1043→      }
  1044→    }
  1045→
  1046→    // Câmara: ViewObject ativa
  1047→    const viewConects = (activeScene.conects || []).filter((c) => c.type === 'ViewObject')
  1048→    const activeView = viewConects.find((c) => c.cameraRole === 'player') || viewConects.find((c) => c.cameraRole === 'primary') || viewConects[0]
  1049→    // Aplicar FOV/Near/Far da ViewObject (ou gameCamera) à câmara do Canvas
  1050→    if (activeView) {
  1051→      const targetFov = activeView.fov || activeScene.gameCamera?.fov || 60
  1052→      const targetNear = activeView.near || activeScene.gameCamera?.near || 0.1
  1053→      const targetFar = activeView.far || activeScene.gameCamera?.far || 200
  1054→      if (camera.fov !== targetFov) {
  1055→        camera.fov = targetFov
  1056→        camera.near = targetNear
  1057→        camera.far = targetFar
  1058→        camera.updateProjectionMatrix()
  1059→      }
  1060→    } else if (activeScene.gameCamera) {
  1061→      // Sem ViewObject — usar gameCamera
  1062→      const gc = activeScene.gameCamera
  1063→      const targetFov = gc.fov || 60
  1064→      const targetNear = gc.near || 0.1
  1065→      const targetFar = gc.far || 200
  1066→      if (camera.fov !== targetFov) {
  1067→        camera.fov = targetFov
  1068→        camera.near = targetNear
  1069→        camera.far = targetFar
  1070→        camera.updateProjectionMatrix()
  1071→      }
  1072→    }
  1073→
  1074→    // Ler rotação da CameraTouchZone (FPS/BR-style)
  1075→    const camRotation = window._flirCameraRotation || { yaw: 0, pitch: 0, enabled: false }
  1076→
  1077→    if (activeView) {
  1078→      // Se cameraRole='player' e não tem followTarget, seguir PersonalObject
  1079→      let targetId = activeView.followTarget
  1080→      if (!targetId && activeView.cameraRole === 'player') {
  1081→        const player = (activeScene.conects || []).find((c) => c.type === 'PersonalObject')
  1082→        if (player) targetId = player.instanceId
  1083→      }
  1084→      const mode = activeView.followMode || 'none'
  1085→      if (targetId && mode !== 'none') {
  1086→        const targetMesh = meshRefs.current.get(targetId) || conectMeshRefs.current.get(targetId)
  1087→        if (targetMesh) {
  1088→          const dist = activeView.followDistance || 6
  1089→          const height = activeView.followHeight || 3
  1090→          if (mode === 'first') {
  1091→            // First-person: câmara na posição dos olhos do jogador, rotação pelo toque
  1092→            const eyeHeight = activeView.eyeHeight || 1.6
  1093→            camera.position.set(
  1094→              targetMesh.position.x,
  1095→              targetMesh.position.y + eyeHeight,
  1096→              targetMesh.position.z
  1097→            )
  1098→            // Aplicar rotação da CameraTouchZone (FPS)
  1099→            if (camRotation.enabled) {
  1100→              // YXZ order = yaw (Y) depois pitch (X) — padrão FPS
  1101→              camera.rotation.set(camRotation.pitch, camRotation.yaw, 0, 'YXZ')
  1102→            } else {
  1103→              // Sem touch input — olhar para frente
  1104→              const rot = activeView.rotation || [0, 0, 0]
  1105→              camera.rotation.set(rot[0], rot[1], rot[2], 'YXZ')
  1106→            }
  1107→          } else if (mode === 'third') {
  1108→            // Third-person: se há touch input, orbita à volta do jogador
  1109→            if (camRotation.enabled) {
  1110→              const orbitDist = dist
  1111→              const offsetY = Math.sin(camRotation.pitch) * orbitDist
  1112→              const offsetX = Math.sin(camRotation.yaw) * Math.cos(camRotation.pitch) * orbitDist
  1113→              const offsetZ = Math.cos(camRotation.yaw) * Math.cos(camRotation.pitch) * orbitDist
  1114→              camera.position.set(
  1115→                targetMesh.position.x + offsetX,
  1116→                targetMesh.position.y + height + offsetY,
  1117→                targetMesh.position.z + offsetZ
  1118→              )
  1119→              camera.lookAt(targetMesh.position.x, targetMesh.position.y + 1, targetMesh.position.z)
  1120→            } else {
  1121→              // Third-person clássico
  1122→              camera.position.lerp(new THREE.Vector3(
  1123→                targetMesh.position.x,
  1124→                targetMesh.position.y + height,
  1125→                targetMesh.position.z + dist
  1126→              ), 0.1)
  1127→              camera.lookAt(targetMesh.position)
  1128→            }
  1129→          } else if (mode === 'top') {
  1130→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x, targetMesh.position.y + dist, targetMesh.position.z), 0.1)
  1131→            camera.lookAt(targetMesh.position)
  1132→          } else if (mode === 'side') {
  1133→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x + dist, targetMesh.position.y + height / 2, targetMesh.position.z), 0.1)
  1134→            camera.lookAt(targetMesh.position)
  1135→          }
  1136→        }
  1137→      } else {
  1138→        // Câmara estática na posição da ViewObject
  1139→        camera.position.set(...(activeView.position || [5, 4, 6]))
  1140→        if (camRotation.enabled) {
  1141→          // Mesmo em câmara estática, permite olhar à volta com touch
  1142→          camera.rotation.set(camRotation.pitch, camRotation.yaw, 0, 'YXZ')
  1143→        } else if (activeView.rotation) {
  1144→          camera.rotation.set(...activeView.rotation)
  1145→        } else {
  1146→          camera.lookAt(0, 0, 0)
  1147→        }
  1148→      }
  1149→    } else if (activeScene.gameCamera) {
  1150→      // Sem ViewObject — usar gameCamera estática
  1151→      const gc = activeScene.gameCamera
  1152→      camera.position.set(...(gc.position || [5, 4, 6]))
  1153→      if (camRotation.enabled) {
  1154→        camera.rotation.set(camRotation.pitch, camRotation.yaw, 0, 'YXZ')
  1155→      } else if (gc.rotation) {
  1156→        camera.rotation.set(...gc.rotation)
  1157→      } else {
  1158→        camera.lookAt(0, 0, 0)
  1159→      }
  1160→    }
  1161→
  1162→    // CheckpointObject — registar checkpoint quando jogador toca
  1163→    const player3 = (setupScene?.conects || []).find((c) => c.type === 'PersonalObject')
  1164→    if (player3) {
  1165→      const playerMesh = conectMeshRefs.current.get(player3.instanceId)
  1166→      if (playerMesh) {
  1167→        for (const cp of setupScene?.conects || []) {
  1168→          if (cp.type === 'CheckpointObject') {
  1169→            const cpMesh = conectMeshRefs.current.get(cp.instanceId)
  1170→            if (cpMesh && cpMesh.visible !== false) {
  1171→              const dist = playerMesh.position.distanceTo(cpMesh.position)
  1172→              if (dist <= (cp.triggerRadius || 2)) {
  1173→                // Registar checkpoint
  1174→                checkpointRef.current = {
  1175→                  position: [cpMesh.position.x, cpMesh.position.y, cpMesh.position.z],
  1176→                  sceneId: setupScene.id,
  1177→                }
  1178→                // Guardar no localStorage
  1179→                try {
  1180→                  localStorage.setItem('flir_checkpoint', JSON.stringify(checkpointRef.current))
  1181→                } catch (e) {}
  1182→                // Disparar evento
  1183→                const rt = runtimesRef.current.get(cp.instanceId)
  1184→                if (rt) rt.triggerEvent('onCheckpoint', { position: checkpointRef.current.position })
  1185→                debugLog('Checkpoint atingido!', 'log', 'Checkpoint')
  1186→              }
  1187→            }
  1188→          }
  1189→        }
  1190→      }
  1191→    }
  1192→    if (player3) {
  1193→      const playerMesh = conectMeshRefs.current.get(player3.instanceId)
  1194→      if (playerMesh) {
  1195→        for (const nav of setupScene?.conects || []) {
  1196→          if (nav.type === 'NavigatorObject' && nav.targetSceneId) {
  1197→            const navMesh = conectMeshRefs.current.get(nav.instanceId)
  1198→            if (navMesh && navMesh.visible !== false) {
  1199→              const dist = playerMesh.position.distanceTo(navMesh.position)
  1200→              if (dist <= (nav.triggerRadius || 2)) {
  1201→                // Transportar para a cena de destino
  1202→                debugLog(`Portal ativado! A mudar para cena ${nav.targetSceneId}`, 'log', 'Navigator')
  1203→                useStore.getState().closeScenePreview()
  1204→                setTimeout(() => {
  1205→                  useStore.getState().setActiveScene(nav.targetSceneId)
  1206→                  useStore.getState().openScenePreview()

  1208→                break
  1209→              }
  1210→            }
  1211→          }
  1212→        }
  1213→      }
  1214→    }
  1215→
  1216→    // ItemObject — auto-pickup quando jogador próximo
  1217→    if (player3) {
  1218→      const playerMesh = conectMeshRefs.current.get(player3.instanceId)
  1219→      if (playerMesh) {
  1220→        for (const item of setupScene?.conects || []) {
  1221→          if (item.type === 'ItemObject' && item.autoPickup !== false) {
  1222→            const itemMesh = conectMeshRefs.current.get(item.instanceId)
  1223→            if (itemMesh && itemMesh.visible !== false) {

  1232→              }
  1233→            }
  1234→          }
  1235→        }
  1236→      }
  1237→    }
  1238→
  1239→    // SpawnObject — spawning temporal de objetos
  1240→    for (const spawn of setupScene?.conects || []) {
  1241→      if (spawn.type === 'SpawnObject' && spawn.autoStart !== false && spawn.objectToSpawn) {
  1242→        if (!timerStatesRef.current.has(spawn.instanceId + '_spawn')) {
  1243→          timerStatesRef.current.set(spawn.instanceId + '_spawn', {
  1244→            remaining: spawn.interval || 3,
  1245→            duration: spawn.interval || 3,
  1246→            loop: true,
  1247→          })
  1248→        }
  1249→        const st = timerStatesRef.current.get(spawn.instanceId + '_spawn')
  1250→        st.remaining -= delta
  1251→        if (st.remaining <= 0) {
  1252→          st.remaining = st.duration
  1253→          // Spawn do objeto
  1254→          const spawnMesh = conectMeshRefs.current.get(spawn.instanceId)
  1255→          const spawnPos = spawnMesh
  1256→            ? [spawnMesh.position.x, spawnMesh.position.y, spawnMesh.position.z]
  1257→            : spawn.position || [0, 1, 0]
  1258→          gameContext.spawnObject(spawn.objectToSpawn, spawnPos)
  1259→          debugLog(`Spawn: "${spawn.objectToSpawn}" em ${spawnPos}`, 'log', 'Spawn')
  1260→        }
  1261→      }
  1262→    }
  1263→
  1264→    // GroupObject — aplicar parenting
  1265→    for (const grp of setupScene?.conects || []) {
  1266→      if (grp.type === 'GroupObject' && grp.children && grp.children.length > 0) {
  1267→        const groupMesh = conectMeshRefs.current.get(grp.instanceId)
  1268→        if (groupMesh && !groupMesh.userData._grouped) {
  1269→          for (const childId of grp.children) {
  1270→            const childMesh = conectMeshRefs.current.get(childId) || meshRefs.current.get(childId)
  1271→            if (childMesh && childMesh.parent !== groupMesh) {
  1272→              groupMesh.attach(childMesh)
  1273→            }
  1274→          }
  1275→          groupMesh.userData._grouped = true
  1276→        }
  1277→      }
  1278→    }
  1279→  })
  1280→
  1281→  return null
  1282→}

  1397→          <FogApplier conects={activeScene?.conects} />

  1414→          {/* Grelha — só no editor */}

  1438→          ))}

  1457→              touches={{ ONE: THREE.TOUCH.ROTATE, TWO: THREE.TOUCH.DOLLY_PAN }}

  1463→          <TerrainSculptBridge

  1471→          <InstancingRenderer />
  1472→

  1475→        </Suspense>
  1476→      </Canvas>
  1477→    </div>
  1478→  )
