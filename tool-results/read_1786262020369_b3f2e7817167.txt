    58→// Componente interno que aplica o fundo da cena
    59→function SceneBackgroundSolid({ background }) {
    60→  const { scene } = useThree()
    61→  useEffect(() => {
    62→    if (background.type === 'gradient') {
    63→      const canvas = document.createElement('canvas')
    64→      canvas.width = 2; canvas.height = 256
    65→      const ctx = canvas.getContext('2d')
    66→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
    67→      grad.addColorStop(0, background.gradientTop)
    68→      grad.addColorStop(1, background.gradientBottom)
    69→      ctx.fillStyle = grad; ctx.fillRect(0, 0, 2, 256)
    70→      const tex = new THREE.CanvasTexture(canvas)
    71→      tex.colorSpace = THREE.SRGBColorSpace
    72→      scene.background = tex
    73→      return () => tex.dispose()
    74→    } else {
    75→      scene.background = new THREE.Color(background.color)
    76→    }
    77→  }, [background, scene])
    78→  return null
    79→}

    82→function GameMode({ activeScene, objects, meshRefs, conectMeshRefs, isGameMode }) {
    83→  const { camera } = useThree()
    84→  const physicsRef = useRef(null)
    85→  const runtimesRef = useRef(new Map())
    86→  const animPlayersRef = useRef(new Map())
    87→  const npcAIsRef = useRef(new Map())
    88→  const timerStatesRef = useRef(new Map())
    89→  const joystickRef = useRef({ x: 0, z: 0, active: false })
    90→  const weaponStateRef = useRef({ equipped: false, ammo: 0, maxAmmo: 0, damage: 0, fireRate: 0.3, range: 50, reloadTime: 2, lastShot: 0 })
    91→  const inventoryRef = useRef({})
    92→
    93→  // P2.5 fix: armazenar a cena ativa num ref para que modificações feitas pelo
    94→  // FlirCode (createObject, etc.) NÃO reiniciem o jogo (evita loop infinito).
    95→  // O useEffect só deve correr quando isGameMode muda (true→false ou false→true),
    96→  // não quando activeScene é modificado por spawnObject.
    97→  const activeSceneRef = useRef(activeScene)
    98→  const objectsRef = useRef(objects)
    99→  const gameStartedRef = useRef(false)
   100→
   101→  // Expor joystickRef globalmente para o GameUIOverlay poder atualizá-lo
   102→  useEffect(() => {
   103→    window._flirJoystick = joystickRef.current
   104→    window._flirInventory = inventoryRef.current
   105→    return () => { window._flirJoystick = null; window._flirInventory = null }
   106→  }, [])
   107→
   108→  // Use o ref para o setup
   109→  const setupScene = activeSceneRef.current
   110→
   111→  // Fase 5: Expor multiplayer globalmente e configurar sync de estado do PersonalObject
   112→  useEffect(() => {
   113→    if (!isGameMode) return
   114→    let mp
   115→    import('../../utils/multiplayer/multiplayerManager').then(({ multiplayer }) => {
   116→      mp = multiplayer
   117→      window._multiplayer = mp
   118→      // Configurar getter de estado local (posição do PersonalObject)
   119→      mp.setLocalStateGetter(() => {
   120→        for (const conect of setupScene?.conects || []) {
   121→          if (conect.type === 'PersonalObject') {
   122→            const mesh = conectMeshRefs.current.get(conect.instanceId)
   123→            if (mesh) {
   124→              return {
   125→                position: [mesh.position.x, mesh.position.y, mesh.position.z],
   126→                rotation: [mesh.rotation.x, mesh.rotation.y, mesh.rotation.z],
   127→                playerId: mp.playerId,
   128→              }
   129→            }
   130→          }
   131→        }
   132→        return null
   133→      })
   134→      // Registar callbacks para eventos multiplayer → FlirCode
   135→      mp.on('playerJoin', (playerId) => {
   136→        for (const rt of runtimesRef.current.values()) {
   137→          rt.triggerEvent('onPlayerJoin', { playerId })
   138→        }
   139→      })
   140→      mp.on('playerLeave', (playerId) => {
   141→        for (const rt of runtimesRef.current.values()) {
   142→          rt.triggerEvent('onPlayerLeave', { playerId })
   143→        }
   144→      })
   145→      mp.on('message', (playerId, data) => {
   146→        for (const rt of runtimesRef.current.values()) {
   147→          rt.triggerEvent('onMessage', { playerId, data })
   148→        }
   149→      })
   150→    })
   151→    return () => {
   152→      window._multiplayer = null
   153→    }
   154→  }, [isGameMode, setupScene])
   155→
   156→  // Atualizar refs sem re-disparar o useEffect
   157→  useEffect(() => {
   158→    // Só atualizar o ref se o jogo ainda não começou OU se a cena mudou de verdade
   159→    // (não por spawnObject, mas por mudança de cena manual via changeScene)
   160→    if (!gameStartedRef.current || (activeScene && activeScene.id !== activeSceneRef.current?.id)) {
   161→      activeSceneRef.current = activeScene
   162→      objectsRef.current = objects
   163→    } else {
   164→      // Jogo em curso: manter o snapshot antigo, mas atualizar objects (para spawn)
   165→      objectsRef.current = objects
   166→    }
   167→  }, [activeScene, objects])
   168→
   169→  // Setup quando o modo jogo é activado
   170→  useEffect(() => {
   171→    if (!isGameMode || !setupScene) return
   172→    gameStartedRef.current = true
   173→
   174→    debugLog('Jogo iniciado', 'log', 'Game')
   175→
   176→    // Criar gameContext
   177→    const gameContext = {
   178→      globalVars: { _score: 0 },
   179→      // P2.5: setVar/getVar — expostos no gameContext para o FlirCode
   180→      setVar: (name, value) => { gameContext.globalVars[name] = value },
   181→      getVar: (name) => gameContext.globalVars[name] ?? 0,
   182→      moveObject: (instanceId, direction, speed) => {
   183→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   184→        if (mesh) {
   185→          mesh.position.x += direction[0] * speed * 0.016
   186→          mesh.position.y += direction[1] * speed * 0.016
   187→          mesh.position.z += direction[2] * speed * 0.016
   188→        }
   189→      },
   190→      rotateObject: (instanceId, rotation) => {
   191→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   192→        if (mesh) {
   193→          mesh.rotation.x += THREE.MathUtils.degToRad(rotation[0]) * 0.016
   194→          mesh.rotation.y += THREE.MathUtils.degToRad(rotation[1]) * 0.016
   195→          mesh.rotation.z += THREE.MathUtils.degToRad(rotation[2]) * 0.016
   196→        }
   197→      },
   198→      playAnimation: (instanceId, clip) => {
   199→        const player = animPlayersRef.current.get(instanceId)
   200→        if (player) player.play(clip, { loop: true })
   201→      },
   202→      playSound: (url) => { try { new Audio(url).play() } catch {} },
   203→      playSoundByName: (name) => {
   204→        const sc = (setupScene.conects || []).find((c) => c.type === 'SoundObject' && c.name === name)
   205→        if (sc && sc.url) { try { const a = new Audio(sc.url); a.volume = sc.volume ?? 1; a.loop = sc.loop || false; a.play() } catch {} }
   206→      },
   207→      destroyObject: (instanceId) => {
   208→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   209→        if (mesh) mesh.visible = false
   210→      },
   211→      spawnObject: (objectName, position) => {
   212→        const obj = objectsRef.current.find((o) => o.name === objectName)
   213→        if (obj) useStore.getState().addObjectToScene(obj.id, position)
   214→      },
   215→      changeScene: (sceneName) => {
   216→        const scenes = useStore.getState().scenes
   217→        const target = scenes.find((s) => s.name === sceneName)
   218→        if (target) useStore.getState().setActiveScene(target.id)
   219→      },
   220→      setVisible: (instanceId, visible) => {
   221→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   222→        if (mesh) mesh.visible = visible
   223→      },
   224→      applyForce: (instanceId, force) => physicsRef.current?.applyForce(instanceId, force),
   225→      jumpPlayer: (instanceId) => physicsRef.current?.jumpPersonal(instanceId),
   226→      showUIScreen: (name) => {
   227→        const s = useStore.getState().uiScreens.find((s) => s.name === name)
   228→        if (s) useStore.getState().setUIScreenVisible(s.id, true)
   229→      },
   230→      hideUIScreen: (name) => {
   231→        const s = useStore.getState().uiScreens.find((s) => s.name === name)
   232→        if (s) useStore.getState().setUIScreenVisible(s.id, false)
   233→      },
   234→      getUIValue: (name) => {
   235→        for (const s of useStore.getState().uiScreens) {
   236→          const el = s.elements.find((e) => e.name === name)
   237→          if (el) return el.value ?? el.text ?? ''
   238→        }
   239→        return ''
   240→      },
   241→      setUIValue: (name, val) => {
   242→        for (const s of useStore.getState().uiScreens) {
   243→          const el = s.elements.find((e) => e.name === name)
   244→          if (el) { useStore.getState().updateUIElement(el.id, { value: val, text: val, label: val }); return }
   245→        }
   246→      },
   247→      triggerUIEvent: (eventName, payload) => {
   248→        for (const rt of runtimesRef.current.values()) rt.triggerEvent(eventName, payload)
   249→      },
   250→      collidingWith: (instanceId, type) => {
   251→        if (!physicsRef.current) return false
   252→        const entry = physicsRef.current.bodies.get(instanceId)
   253→        if (!entry) return false
   254→        for (const [otherId, otherEntry] of physicsRef.current.bodies) {
   255→          if (otherId === instanceId) continue
   256→          if (otherEntry.conect.type === type || otherEntry.conect.name === type) {
   257→            if (entry.body.position.distanceTo(otherEntry.body.position) < 1.5) return true
   258→          }
   259→        }
   260→        return false
   261→      },
   262→      distanceTo: (instanceId, targetName) => {
   263→        // P2.5: procurar em objects E conects da cena ativa
   264→        for (const inst of setupScene.objects || []) {
   265→          const obj = objectsRef.current.find((o) => o.id === inst.objectId)
   266→          if (obj && (obj.name === targetName || inst.name === targetName)) {
   267→            const target = meshRefs.current.get(inst.instanceId)
   268→            const source = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   269→            if (source && target) return source.position.distanceTo(target.position)
   270→          }
   271→        }
   272→        for (const conect of setupScene.conects || []) {
   273→          if (conect.name === targetName) {
   274→            const target = conectMeshRefs.current.get(conect.instanceId)
   275→            const source = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   276→            if (source && target) return source.position.distanceTo(target.position)
   277→          }
   278→        }
   279→        return 0
   280→      },
   281→      isTouching: () => joystickRef.current.active,
   282→      // Fase 5: Multiplayer functions
   283→      sendMessage: (data) => {
   284→        try {
   285→          const mp = window._multiplayer
   286→          if (mp && mp.connected) mp.sendMessage(data)
   287→        } catch {}
   288→      },
   289→      getPlayers: () => {
   290→        const mp = window._multiplayer
   291→        return mp && mp.connected ? mp.getOtherPlayers().length + 1 : 1
   292→      },
   293→      getPlayerState: (playerId) => {
   294→        const mp = window._multiplayer
   295→        return mp && mp.connected ? mp.getPlayerState(playerId) : null
   296→      },
   297→      // Sistema 3: Sinais — emite para todos os runtimes da cena
   298→      emitSignal: (name, data) => {
   299→        for (const rt of runtimesRef.current.values()) {
   300→          rt.triggerEvent('onSignal', { name, data })
   301→        }
   302→        debugLog(`Signal emitido: ${name}`, 'log', 'Signals')
   303→      },
   304→      // Sistema 2: Armas e combate
   305→      shoot: () => {
   306→        const weaponState = weaponStateRef.current
   307→        if (!weaponState || !weaponState.equipped) {
   308→          debugLog('shoot() sem arma equipada', 'warning', 'Weapon')
   309→          return
   310→        }
   311→        const now = performance.now() / 1000
   312→        if (now - weaponState.lastShot < weaponState.fireRate) return
   313→        if (weaponState.ammo <= 0) {
   314→          debugLog('Sem munição — pressiona reload()', 'warning', 'Weapon')
   315→          return
   316→        }
   317→        weaponState.lastShot = now
   318→        weaponState.ammo--
   319→        debugLog(`Disparo! Munição restante: ${weaponState.ammo}`, 'log', 'Weapon')
   320→        // Raycast do centro do ecrã (câmara) ou da posição do jogador
   321→        const player = (setupScene?.conects || []).find((c) => c.type === 'PersonalObject')
   322→        if (player) {
   323→          const playerMesh = conectMeshRefs.current.get(player.instanceId)
   324→          if (playerMesh) {
   325→            const origin = new THREE.Vector3(playerMesh.position.x, playerMesh.position.y + 1, playerMesh.position.z)
   326→            const dir = new THREE.Vector3(0, 0, -1)
   327→            dir.applyQuaternion(playerMesh.quaternion)
   328→            const ray = new THREE.Raycaster(origin, dir, 0, weaponState.range)
   329→            // Verificar colisão com outros conects
   330→            let hit = false
   331→            for (const [id, entry] of physicsRef.current?.bodies || []) {
   332→              if (id === player.instanceId) continue
   333→              const targetMesh = meshRefs.current.get(id) || conectMeshRefs.current.get(id)
   334→              if (targetMesh) {
   335→                const intersects = ray.intersectObject(targetMesh, true)
   336→                if (intersects.length > 0) {
   337→                  // Aplicar dano
   338→                  gameContext.takeDamage(id, weaponState.damage)
   339→                  debugLog(`Alvo atingido: ${entry.conect.name} (-${weaponState.damage} HP)`, 'log', 'Weapon')
   340→                  hit = true
   341→                  break
   342→                }
   343→              }
   344→            }
   345→            if (!hit) debugLog('Disparo falhou (sem alvo)', 'log', 'Weapon')
   346→          }
   347→        }
   348→      },
   349→      reload: () => {
   350→        const ws = weaponStateRef.current
   351→        if (!ws || !ws.equipped) return
   352→        if (ws.ammo >= ws.maxAmmo) return
   353→        debugLog(`A recarregar... (${ws.reloadTime}s)`, 'log', 'Weapon')
   354→        setTimeout(() => {
   355→          ws.ammo = ws.maxAmmo
   356→          debugLog(`Recarregado! Munição: ${ws.ammo}`, 'log', 'Weapon')
   357→        }, ws.reloadTime * 1000)
   358→      },
   359→      equipWeapon: (weaponName) => {
   360→        const weapon = (setupScene?.conects || []).find((c) => c.type === 'WeaponObject' && (c.name === weaponName || c.instanceId === weaponName))
   361→        if (!weapon) {
   362→          debugLog(`Arma "${weaponName}" não encontrada`, 'warning', 'Weapon')
   363→          return
   364→        }
   365→        weaponStateRef.current = {
   366→          equipped: true,
   367→          damage: weapon.damage || 25,
   368→          fireRate: weapon.fireRate || 0.3,
   369→          range: weapon.range || 50,
   370→          maxAmmo: weapon.maxAmmo || 30,
   371→          ammo: weapon.maxAmmo || 30,
   372→          reloadTime: weapon.reloadTime || 2,
   373→          lastShot: 0,
   374→          weaponId: weapon.instanceId,
   375→        }
   376→        // Mostrar mira se necessário
   377→        if (weapon.showCrosshair) {
   378→          window._flirCrosshair = true
   379→        }
   380→        debugLog(`Arma "${weapon.name}" equipada (${weaponStateRef.current.ammo} balas)`, 'log', 'Weapon')
   381→      },
   382→      getAmmo: () => weaponStateRef.current?.ammo ?? 0,
   383→      takeDamage: (instanceId, amount) => {
   384→        // Reduzir vida do alvo
   385→        for (const conect of setupScene?.conects || []) {
   386→          if (conect.instanceId === instanceId) {
   387→            const currentHealth = conect.health ?? 100
   388→            const newHealth = Math.max(0, currentHealth - amount)
   389→            useStore.getState().updateConect(instanceId, { health: newHealth })
   390→            debugLog(`${conect.name} recebeu ${amount} de dano (vida: ${newHealth})`, 'log', 'Combat')
   391→            // Disparar onDamage no runtime do alvo
   392→            const rt = runtimesRef.current.get(instanceId)
   393→            if (rt) rt.triggerEvent('onDamage', { amount, source: 'weapon' })
   394→            if (newHealth <= 0) {
   395→              debugLog(`${conect.name} foi destruído`, 'log', 'Combat')
   396→              const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   397→              if (mesh) mesh.visible = false
   398→            }
   399→            break
   400→          }
   401→        }
   402→      },
   403→      getHealth: (instanceId) => {
   404→        for (const conect of setupScene?.conects || []) {
   405→          if (conect.instanceId === instanceId) return conect.health ?? 100
   406→        }
   407→        return 100
   408→      },
   409→      // Sistema 3: Inventário
   410→      addToInventory: (itemName, quantity = 1) => {
   411→        const inv = inventoryRef.current
   412→        inv[itemName] = (inv[itemName] || 0) + quantity
   413→        debugLog(`Item "${itemName}" adicionado (${quantity}). Total: ${inv[itemName]}`, 'log', 'Inventory')
   414→        // Trigger onPickup em todos os runtimes
   415→        for (const rt of runtimesRef.current.values()) {
   416→          rt.triggerEvent('onPickup', { itemName, quantity })
   417→        }
   418→      },
   419→      removeFromInventory: (itemName, quantity = 1) => {
   420→        const inv = inventoryRef.current
   421→        if (!inv[itemName]) return
   422→        inv[itemName] = Math.max(0, inv[itemName] - quantity)
   423→        if (inv[itemName] === 0) delete inv[itemName]
   424→        debugLog(`Item "${itemName}" removido (${quantity}). Restante: ${inv[itemName] || 0}`, 'log', 'Inventory')
   425→      },
   426→      getInventoryCount: (itemName) => inventoryRef.current[itemName] || 0,
   427→      hasItem: (itemName) => (inventoryRef.current[itemName] || 0) > 0,
   428→      // Sistema: Links — navegar para cena ou tela de UI
   429→      linkTo: (target, subTarget) => {
   430→        if (target === 'scene') {
   431→          // Mudar de cena
   432→          const scenes = useStore.getState().scenes
   433→          const targetScene = scenes.find((s) => s.name === subTarget || s.id === subTarget)
   434→          if (targetScene) {
   435→            useStore.getState().setActiveScene(targetScene.id)
   436→            debugLog(`Link: mudou para cena "${targetScene.name}"`, 'log', 'Links')
   437→          } else {
   438→            debugLog(`Link: cena "${subTarget}" não encontrada`, 'warning', 'Links')
   439→          }
   440→        } else if (target === 'screen') {
   441→          // Mostrar tela de UI e esconder as outras
   442→          const screens = useStore.getState().uiScreens
   443→          const targetScreen = screens.find((s) => s.name === subTarget || s.id === subTarget)
   444→          if (targetScreen) {
   445→            screens.forEach((s) => useStore.getState().setUIScreenVisible(s.id, s.id === targetScreen.id))
   446→            debugLog(`Link: mudou para tela "${targetScreen.name}"`, 'log', 'Links')
   447→          } else {
   448→            debugLog(`Link: tela "${subTarget}" não encontrada`, 'warning', 'Links')
   449→          }
   450→        } else if (target === 'url') {
   451→          // Abrir URL externa
   452→          window.open(subTarget, '_blank')
   453→          debugLog(`Link: abriu URL "${subTarget}"`, 'log', 'Links')
   454→        }
   455→      },
   456→    }
   457→    window._flirGameContext = gameContext
   458→
   459→    // Física
   460→    const gravity = setupScene.physics?.gravity || [0, -9.82, 0]
   461→    physicsRef.current = createPhysicsSystem({ gravity: gravity[1] })
   462→
   463→    // Registar conects com física
   464→    setTimeout(() => {
   465→      if (!physicsRef.current) return
   466→      for (const conect of setupScene.conects || []) {
   467→        const mesh = conectMeshRefs.current.get(conect.instanceId)
   468→        if (mesh) physicsRef.current.addConect(conect, mesh)
   469→      }
   470→      for (const conect of setupScene.conects || []) {
   471→        if (conect.type === 'JointObject' && conect.targetA && conect.targetB) {
   472→          physicsRef.current.addJoint(conect)
   473→        }
   474→      }
   475→    }, 50)
   476→
   477→    // Eventos de física
   478→    physicsRef.current.on('onCollision', ({ instanceId, otherInstanceId }) => {
   479→      const rt = runtimesRef.current.get(instanceId)
   480→      if (rt) rt.triggerEvent('onCollision', { other: otherInstanceId })
   481→    })
   482→    physicsRef.current.on('onTriggerEnter', ({ instanceId, otherInstanceId }) => {
   483→      const rt = runtimesRef.current.get(instanceId)
   484→      if (rt) rt.triggerEvent('onEnterZone', { other: otherInstanceId })
   485→    })
   486→    physicsRef.current.on('onTriggerExit', ({ instanceId, otherInstanceId }) => {
   487→      const rt = runtimesRef.current.get(instanceId)
   488→      if (rt) rt.triggerEvent('onExitZone', { other: otherInstanceId })
   489→    })
   490→
   491→    // FlirCode/FlirScript runtimes
   492→    // Sistema 2: obter classes FlirCode do store
   493→    const flirCodeClasses = useStore.getState().flirCodeClasses || []
   494→
   495→    const setupRuntime = (instance, scriptData) => {
   496→      try {
   497→        // Sistema 2: se o conect tem flirCodeClass definido, usar o código da classe
   498→        let source = null
   499→        let className = null
   500→
   501→        if (instance.flirCodeClass) {
   502→          const cls = flirCodeClasses.find((c) => c.name === instance.flirCodeClass)
   503→          if (cls) {
   504→            source = cls.source
   505→            className = cls.name
   506→          }
   507→        }
   508→
   509→        // Se há script próprio, concatenar com a classe (script próprio por cima)
   510→        if (scriptData && typeof scriptData === 'string' && scriptData.startsWith('FLIRCODE:')) {
   511→          const ownScript = scriptData.slice(9)
   512→          source = source ? source + '\n' + ownScript : ownScript
   513→        }
   514→
   515→        if (!source) return
   516→
   517→        const rt = createFlirCodeRuntime(source, {
   518→          ...gameContext,
   519→          _instanceId: instance.instanceId,
   520→          className, // Sistema 2: passar nome da classe para o runtime
   521→          mesh: conectMeshRefs.current.get(instance.instanceId) || meshRefs.current.get(instance.instanceId),
   522→        })
   523→        if (!rt.hasErrors) {
   524→          runtimesRef.current.set(instance.instanceId, rt)
   525→          rt.triggerEvent('beginPlay')
   526→        }
   527→      } catch (err) {
   528→        debugLog('Erro ao inicializar script: ' + err.message, 'error', 'Script')
   529→      }
   530→    }
   531→    for (const inst of setupScene.objects || []) setupRuntime(inst, inst.flirScript)
   532→    for (const conect of setupScene.conects || []) setupRuntime(conect, conect.flirScript)
   533→
   534→    // Animation players
   535→    // NOTA: as animações vivem no objeto do catálogo (em objects), não na instância.
   536→    // Para instâncias de objetos (setupScene.objects), precisamos de ir buscar o
   537→    // objeto original via inst.objectId. Para conects (PersonalObject/NpcObject com
   538→    // sourceObjectId), fazemos o mesmo via conect.sourceObjectId.
   539→    const setupAnimationPlayer = (inst, sourceObjectId) => {
   540→      if (!sourceObjectId) return
   541→      const catalogObj = objectsRef.current.find((o) => o.id === sourceObjectId)
   542→      const animations = catalogObj?.animations || inst.animations
   543→      if (!animations || Object.keys(animations).length === 0) return
   544→      const getMesh = () => meshRefs.current.get(inst.instanceId) || conectMeshRefs.current.get(inst.instanceId)
   545→      // getBones: extrai THREE.Bone[] do SkinnedMesh (se o objeto tem esqueleto)
   546→      // Isto é CRÍTICO — sem isto, applyPose nunca modifica os bones e o
   547→      // SkinnedMesh não deforma.
   548→      const getBones = () => {
   549→        const mesh = getMesh()
   550→        if (mesh && mesh.isSkinnedMesh && mesh.skeleton) return mesh.skeleton.bones
   551→        return null
   552→      }
   553→      const player = createAnimationPlayer(animations, getMesh, getBones)
   554→      animPlayersRef.current.set(inst.instanceId, player)
   555→      if (animations.idle) player.play('idle', { loop: true })
   556→      else { const f = Object.keys(animations)[0]; if (f) player.play(f, { loop: true }) }
   557→    }
   558→    for (const inst of setupScene.objects || []) setupAnimationPlayer(inst, inst.objectId)
   559→    for (const conect of setupScene.conects || []) setupAnimationPlayer(conect, conect.sourceObjectId)
   560→
   561→    // NPC AI
   562→    for (const conect of setupScene.conects || []) {
   563→      if (conect.type === 'NpcObject') {
   564→        const ai = createNPCAI(conect, {
   565→          getPlayerPos: () => {
   566→            const player = (setupScene.conects || []).find((c) => c.type === 'PersonalObject')
   567→            if (!player) return null
   568→            const pm = conectMeshRefs.current.get(player.instanceId)
   569→            return pm ? [pm.position.x, pm.position.y, pm.position.z] : null
   570→          },
   571→          getPathPoints: (pathId) => {
   572→            const path = (setupScene.conects || []).find((c) => c.instanceId === pathId)
   573→            return path?.points || null
   574→          },
   575→          physicsMove: (id, dir, speed) => physicsRef.current?.movePersonal(id, dir, speed),
   576→          physicsJump: (id) => physicsRef.current?.jumpPersonal(id),
   577→          emitEvent: (en, payload) => {
   578→            const rt = runtimesRef.current.get(conect.instanceId)
   579→            if (rt) {
   580→              if (en === 'OnSeePlayer') rt.triggerEvent('onSeePlayer', payload)
   581→              else if (en === 'OnLoseSight') rt.triggerEvent('onLoseSight', payload)
   582→            }
   583→          },
   584→        })
   585→        npcAIsRef.current.set(conect.instanceId, ai)
   586→      }
   587→    }
   588→
   589→    // Timers
   590→    for (const conect of setupScene.conects || []) {
   591→      if (conect.type === 'TimerObject' && conect.autoStart) {
   592→        timerStatesRef.current.set(conect.instanceId, { remaining: conect.duration || 5, loop: conect.loop || false, duration: conect.duration || 5 })
   593→      }
   594→    }
   595→
   596→    // SkyObject / FogObject
   597→    const skyConect = (setupScene.conects || []).find((c) => c.type === 'SkyObject')
   598→    if (skyConect) {
   599→      // Aplicado via SceneBackgroundSolid se mudar o background
   600→    }
   601→    const fogConect = (setupScene.conects || []).find((c) => c.type === 'FogObject')
   602→    // Fog aplicado no useFrame
   603→
   604→    // Joystick touch
   605→    const onTouchStart = (e) => { if (e.touches.length === 1) { joystickRef.current.active = true } }
   606→    const onTouchEnd = () => { joystickRef.current.active = false; joystickRef.current.x = 0; joystickRef.current.z = 0 }
   607→    window.addEventListener('touchstart', onTouchStart)
   608→    window.addEventListener('touchend', onTouchEnd)
   609→
   610→    // Teclado
   611→    const keys = {}
   612→    const onKeyDown = (e) => { keys[e.key.toLowerCase()] = true }
   613→    const onKeyUp = (e) => { keys[e.key.toLowerCase()] = false }
   614→    window.addEventListener('keydown', onKeyDown)
   615→    window.addEventListener('keyup', onKeyUp)
   616→    window._flirKeys = keys
   617→
   618→    return () => {
   619→      for (const rt of runtimesRef.current.values()) rt.dispose()
   620→      runtimesRef.current.clear()
   621→      for (const ai of npcAIsRef.current.values()) ai.dispose()
   622→      npcAIsRef.current.clear()
   623→      for (const [, s] of timerStatesRef.current) { if (s.interval) clearInterval(s.interval); if (s.audio) s.audio.pause() }
   624→      timerStatesRef.current.clear()
   625→      animPlayersRef.current.clear()
   626→      physicsRef.current?.dispose()
   627→      physicsRef.current = null
   628→      window.removeEventListener('touchstart', onTouchStart)
   629→      window.removeEventListener('touchend', onTouchEnd)
   630→      window.removeEventListener('keydown', onKeyDown)
   631→      window.removeEventListener('keyup', onKeyUp)
   632→      window._flirGameContext = null
   633→      gameStartedRef.current = false
   634→    }
   635→  }, [isGameMode, setupScene])
   636→
   637→  // Loop do jogo
   638→  useFrame((_, delta) => {
   639→    if (!isGameMode) return
   640→
   641→    // OTIMIZAÇÃO: limpar cache de poses no início de cada frame
   642→    // (permite que 200 NPCs partilhem a mesma pose calculada)
   643→    clearPoseCache()
   644→
   645→    // Física
   646→    if (physicsRef.current) {
   647→      physicsRef.current.update(delta)
   648→      // Sincronizar meshes com bodies
   649→      for (const [id, entry] of physicsRef.current.bodies) {
   650→        const mesh = meshRefs.current.get(id) || conectMeshRefs.current.get(id)
   651→        if (mesh) {
   652→          mesh.position.copy(entry.body.position)
   653→          mesh.quaternion.copy(entry.body.quaternion)
   654→        }
   655→      }
   656→    }
   657→
   658→    // FlirCode onTick
   659→    for (const rt of runtimesRef.current.values()) {
   660→      rt.update(delta)
   661→      rt.triggerEvent('tick', { deltaTime: delta })
   662→    }
   663→
   664→    // Animation players
   665→    for (const player of animPlayersRef.current.values()) player.update(delta)
   666→
   667→    // NPC AI
   668→    for (const ai of npcAIsRef.current.values()) ai.update(delta)
   669→
   670→    // Timers
   671→    for (const [id, state] of timerStatesRef.current) {
   672→      if (id.startsWith('_')) continue
   673→      state.remaining -= delta
   674→      if (state.remaining <= 0) {
   675→        const rt = runtimesRef.current.get(id)
   676→        if (rt) rt.triggerEvent('onTimer')
   677→        if (state.loop) state.remaining = state.duration
   678→        else timerStatesRef.current.delete(id)
   679→      }
   680→    }
   681→
   682→    // Joystick → PersonalObject — usar setupScene (ref)
   683→    const keys = window._flirKeys || {}
   684→    if (joystickRef.current.active || keys['w'] || keys['a'] || keys['s'] || keys['d']) {
   685→      for (const conect of setupScene?.conects || []) {
   686→        if (conect.type === 'PersonalObject') {
   687→          const speed = conect.moveSpeed || 5
   688→          let mx = 0, mz = 0
   689→          if (joystickRef.current.active) { mx = joystickRef.current.x * speed; mz = joystickRef.current.z * speed }
   690→          if (keys['w']) mz = -speed
   691→          if (keys['s']) mz = speed
   692→          if (keys['a']) mx = -speed
   693→          if (keys['d']) mx = speed
   694→          physicsRef.current?.movePersonal(conect.instanceId, [mx, 0, mz], 1)
   695→          if ((keys[' '] || keys['space']) && conect.canJump) {
   696→            physicsRef.current?.jumpPersonal(conect.instanceId)
   697→          }
   698→        }
   699→      }
   700→    }
   701→
   702→    // Sistema 3: Auto-pickup de ItemObjects
   703→    const player2 = (setupScene?.conects || []).find((c) => c.type === 'PersonalObject')
   704→    if (player2) {
   705→      const playerMesh = conectMeshRefs.current.get(player2.instanceId)
   706→      if (playerMesh) {
   707→        for (const item of setupScene?.conects || []) {
   708→          if (item.type === 'ItemObject' && item.autoPickup !== false) {
   709→            const itemMesh = conectMeshRefs.current.get(item.instanceId)
   710→            if (itemMesh && itemMesh.visible !== false) {
   711→              const dist = playerMesh.position.distanceTo(itemMesh.position)
   712→              if (dist <= (item.pickupRadius || 2)) {
   713→                // Apanhar item
   714→                gameContext.addToInventory(item.itemName || 'Item', item.quantity || 1)
   715→                itemMesh.visible = false
   716→                debugLog(`Item "${item.itemName}" apanhado!`, 'log', 'Inventory')
   717→              }
   718→            }
   719→          }
   720→        }
   721→      }
   722→    }
   723→
   724→    // Câmara: ViewObject ativa — usar setupScene (ref) para consistência
   725→    const viewConects = (setupScene?.conects || []).filter((c) => c.type === 'ViewObject')
   726→    const activeView = viewConects.find((c) => c.cameraRole === 'player') || viewConects.find((c) => c.cameraRole === 'primary') || viewConects[0]
   727→    if (activeView) {
   728→      // Se cameraRole='player' e não tem followTarget, seguir PersonalObject
   729→      let targetId = activeView.followTarget
   730→      if (!targetId && activeView.cameraRole === 'player') {
   731→        const player = (setupScene?.conects || []).find((c) => c.type === 'PersonalObject')
   732→        if (player) targetId = player.instanceId
   733→      }
   734→      if (targetId && activeView.followMode && activeView.followMode !== 'none') {
   735→        const targetMesh = meshRefs.current.get(targetId) || conectMeshRefs.current.get(targetId)
   736→        if (targetMesh) {
   737→          const mode = activeView.followMode
   738→          const dist = activeView.followDistance || 6
   739→          const height = activeView.followHeight || 3
   740→          if (mode === 'third') {
   741→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x, targetMesh.position.y + height, targetMesh.position.z + dist), 0.1)
   742→            camera.lookAt(targetMesh.position)
   743→          } else if (mode === 'top') {
   744→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x, targetMesh.position.y + dist, targetMesh.position.z), 0.1)
   745→            camera.lookAt(targetMesh.position)
   746→          } else if (mode === 'side') {
   747→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x + dist, targetMesh.position.y + height / 2, targetMesh.position.z), 0.1)
   748→            camera.lookAt(targetMesh.position)
   749→          }
   750→        }
   751→      } else {
   752→        // Câmara estática na posição da ViewObject
   753→        camera.position.set(...(activeView.position || [5, 4, 6]))
   754→        // Sempre olhar para a origem (ou para o centro dos objetos da cena)
   755→        // a menos que followMode esteja definido
   756→        camera.lookAt(0, 0, 0)
   757→      }
   758→    }
   759→  })
   760→
   761→  return null
   762→}
