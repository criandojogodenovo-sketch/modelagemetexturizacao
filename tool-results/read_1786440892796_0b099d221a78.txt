     1→/**
     2→ * flircode.js — interpretador da linguagem FlirCode.
     3→ *
     4→ * Linguagem de scripting por texto próprio (não é GDScript nem JavaScript).
     5→ * Compila para o mesmo gameContext que o executor.js usa, garantindo
     6→ * compatibilidade total com Conects, eventos e físicas existentes.
     7→ *
     8→ * Sintaxe:
     9→ *   begincode ... endcode          — blocos de código
    10→ *   fun nome(params) begincode ... endcode  — declarar função
    11→ *   var nome = valor               — declarar variável
    12→ *   $$ comentário                  — linha de comentário
    13→ *   if (cond), else if (cond), else — condicionais
    14→ *   repeat in number(n, i) begincode ... endcode — ciclo por quantidade
    15→ *   repeat +n until m begincode ... endcode — ciclo por incremento
    16→ *   repeat -n until m begincode ... endcode — ciclo por decremento
    17→ *   switch (var) begincode case v begincode ... endcode default begincode ... endcode endcode
    18→ *
    19→ * Eventos (funções especiais chamadas automaticamente):
    20→ *   fun onStart()        — BeginPlay
    21→ *   fun onTick()         — Tick (a cada frame)
    22→ *   fun onCollide(outro) — OnCollision
    23→ *   fun onTouch()        — OnTouch
    24→ *   fun onSeePlayer()    — OnSeePlayer (NPC)
    25→ *   fun onLoseSight()    — OnLoseSight (NPC)
    26→ *   fun onTimer()        — OnTimer
    27→ *
    28→ * Funções embutidas:
    29→ *   playAnim("name"), playSound("name"), move(x,y,z), rotate(x,y,z), scale(x,y,z)
    30→ *   destroy(object), createObject("name",x,y,z), changeScene("name"), wait(seconds)
    31→ *   setVar("name",value), getVar("name"), showUI("name"), hideUI("name")
    32→ *   print("msg"), warn("msg"), error("msg")
    33→ *   collidingWith("type"), distanceTo(object), isTouching()
    34→ */
    35→import { debugLog } from '../debug/debugStore'
    36→
    37→// ===== Parser =====
    38→// Converte texto FlirCode num AST (Abstract Syntax Tree) que o runtime executa.
    39→// O parser é um tokenizer + recursive descent parser simples.
    40→
    41→export function parseFlirCode(source) {
    42→  const errors = []
    43→  const lines = source.split('\n')
    44→  const functions = {}
    45→  const classes = {} // Sistema 2: suporte a classes
    46→
    47→  // Pré-processar: remover comentários e linhas vazias
    48→  const cleanLines = []
    49→  for (let i = 0; i < lines.length; i++) {
    50→    const trimmed = lines[i].trim()
    51→    if (trimmed.startsWith('$$') || trimmed === '') continue
    52→    cleanLines.push({ text: trimmed, line: i + 1 })
    53→  }
    54→
    55→  // Parser de funções e classes (top-level)
    56→  let idx = 0
    57→  while (idx < cleanLines.length) {
    58→    const line = cleanLines[idx]
    59→
    60→    // Sistema 2: "class Nome begincode" ou "class Nome extends Base begincode"
    61→    const classMatch = line.text.match(/^class\s+(\w+)(?:\s+extends\s+(\w+))?\s*begincode$/)
    62→    if (classMatch) {
    63→      const className = classMatch[1]
    64→      const extendsName = classMatch[2] || null
    65→      const body = parseBlock(cleanLines, idx + 1, errors)
    66→      if (body.error) {
    67→        errors.push({ line: line.line, message: `Classe "${className}": ${body.error}` })
    68→        idx = body.nextIdx
    69→        continue
    70→      }
    71→      // Extrair variáveis e funções da classe
    72→      const classVars = {}
    73→      const classFuncs = {}
    74→      for (const stmt of body.statements) {
    75→        if (stmt.type === 'var' || stmt.type === 'assign') {
    76→          classVars[stmt.name] = stmt.value
    77→        } else if (stmt.type === 'function') {
    78→          classFuncs[stmt.name] = stmt
    79→        }
    80→      }
    81→      classes[className] = {
    82→        name: className,
    83→        extends: extendsName,
    84→        vars: classVars,
    85→        functions: classFuncs,
    86→        rawStatements: body.statements,
    87→        line: line.line,
    88→      }
    89→      idx = body.nextIdx
    90→      continue
    91→    }
    92→
    93→    // Procurar "fun nome(params) begincode"
    94→    const funMatch = line.text.match(/^fun\s+(\w+)\s*\(([^)]*)\)\s*begincode$/)
    95→    if (funMatch) {
    96→      const funName = funMatch[1]
    97→      const params = funMatch[2].split(',').map((p) => p.trim()).filter((p) => p)
    98→      const body = parseBlock(cleanLines, idx + 1, errors)
    99→      if (body.error) {
   100→        errors.push({ line: line.line, message: `Função "${funName}": ${body.error}` })
   101→        idx = body.nextIdx
   102→        continue
   103→      }
   104→      functions[funName] = { name: funName, params, body: body.statements, line: line.line }
   105→      idx = body.nextIdx
   106→    } else {
   107→      // Pode ser uma variável global ou instrução solta
   108→      // Por agora, ignorar (só suportamos funções no top-level)
   109→      idx++
   110→    }
   111→  }
   112→
   113→  return { functions, classes, errors }
   114→}
   115→
   116→// Parser de bloco begincode...endcode
   117→function parseBlock(lines, startIdx, errors) {
   118→  const statements = []
   119→  let idx = startIdx
   120→  let depth = 1 // já entramos num begincode
   121→
   122→  while (idx < lines.length && depth > 0) {
   123→    const line = lines[idx]
   124→    const text = line.text
   125→
   126→    if (text === 'endcode') {
   127→      depth--
   128→      if (depth === 0) {
   129→        return { statements, nextIdx: idx + 1 }
   130→      }
   131→      statements.push({ type: 'endcode', line: line.line })
   132→      idx++
   133→      continue
   134→    }
   135→
   136→    // Sistema 2: reconhecer "fun nome(params) begincode" dentro de classes
   137→    const funMatch = text.match(/^fun\s+(\w+)\s*\(([^)]*)\)\s*begincode$/)
   138→    if (funMatch) {
   139→      const funName = funMatch[1]
   140→      const params = funMatch[2].split(',').map((p) => p.trim()).filter((p) => p)
   141→      const body = parseBlock(lines, idx + 1, errors)
   142→      if (body.error) {
   143→        errors.push({ line: line.line, message: `Função "${funName}": ${body.error}` })
   144→        idx = body.nextIdx
   145→        continue
   146→      }
   147→      statements.push({
   148→        type: 'function',
   149→        name: funName,
   150→        params,
   151→        body: body.statements,
   152→        line: line.line,
   153→        nextIdx: body.nextIdx,
   154→      })
   155→      idx = body.nextIdx
   156→      continue
   157→    }
   158→
   159→    if (text.endsWith('begincode')) {
   160→      // Sub-bloco (if, else, repeat, etc.)
   161→      const stmt = parseStatement(lines, idx, errors)
   162→      if (stmt) {
   163→        statements.push(stmt)
   164→        idx = stmt.nextIdx
   165→      } else {
   166→        idx++
   167→      }
   168→    } else {
   169→      // Instrução simples
   170→      const stmt = parseSimpleStatement(text, line.line, errors)
   171→      if (stmt) statements.push(stmt)
   172→      idx++
   173→    }
   174→  }
   175→
   176→  if (depth > 0) {
   177→    return { error: 'begincode sem endcode correspondente', nextIdx: idx }
   178→  }
   179→  return { statements, nextIdx: idx }
   180→}
   181→
   182→// Parser de uma instrução individual
   183→function parseSimpleStatement(text, lineNum, errors) {
   184→  // var nome = valor
   185→  let m = text.match(/^var\s+(\w+)\s*=\s*(.+)$/)
   186→  if (m) return { type: 'var', name: m[1], value: parseValue(m[2]), line: lineNum }
   187→
   188→  // if (cond) begincode
   189→  m = text.match(/^if\s*\((.+)\)\s*begincode$/)
   190→  if (m) return { type: 'if', condition: m[1], line: lineNum }
   191→
   192→  // else if (cond) begincode
   193→  m = text.match(/^else\s+if\s*\((.+)\)\s*begincode$/)
   194→  if (m) return { type: 'elseif', condition: m[1], line: lineNum }
   195→
   196→  // else begincode
   197→  m = text.match(/^else\s*begincode$/)
   198→  if (m) return { type: 'else', line: lineNum }
   199→
   200→  // repeat in number(quantidade, variavel) begincode
   201→  m = text.match(/^repeat\s+in\s+number\s*\((\d+),\s*(\w+)\)\s*begincode$/)
   202→  if (m) return { type: 'repeat_n', count: parseInt(m[1]), varName: m[2], line: lineNum }
   203→
   204→  // repeat +numero until numero begincode
   205→  m = text.match(/^repeat\s+\+(\d+)\s+until\s+(\d+)\s*begincode$/)
   206→  if (m) return { type: 'repeat_inc', step: parseInt(m[1]), until: parseInt(m[2]), line: lineNum }
   207→
   208→  // repeat -numero until numero begincode
   209→  m = text.match(/^repeat\s+-(\d+)\s+until\s+(\d+)\s*begincode$/)
   210→  if (m) return { type: 'repeat_dec', step: parseInt(m[1]), until: parseInt(m[2]), line: lineNum }
   211→
   212→  // switch (var) begincode
   213→  m = text.match(/^switch\s*\((.+)\)\s*begincode$/)
   214→  if (m) return { type: 'switch', varName: m[1], line: lineNum }
   215→
   216→  // case valor begincode
   217→  m = text.match(/^case\s+(.+)\s*begincode$/)
   218→  if (m) return { type: 'case', value: m[1].trim(), line: lineNum }
   219→
   220→  // default begincode
   221→  m = text.match(/^default\s*begincode$/)
   222→  if (m) return { type: 'default', line: lineNum }
   223→
   224→  // Chamada de função embutida: nome(args)
   225→  m = text.match(/^(\w+)\s*\(([^)]*)\)$/)
   226→  if (m) {
   227→    const funcName = m[1]
   228→    const args = m[2].split(',').map((a) => a.trim()).filter((a) => a)
   229→    return { type: 'call', funcName, args: args.map(parseValue), line: lineNum }
   230→  }
   231→
   232→  // Atribuição: nome = valor
   233→  m = text.match(/^(\w+)\s*=\s*(.+)$/)
   234→  if (m) return { type: 'assign', name: m[1], value: parseValue(m[2]), line: lineNum }
   235→
   236→  // Se não reconhecer, ignorar (não travar)
   237→  return { type: 'unknown', text, line: lineNum }
   238→}
   239→
   240→// Parser de statement (para sub-blocos)
   241→function parseStatement(lines, idx, errors) {
   242→  const stmt = parseSimpleStatement(lines[idx].text, lines[idx].line, errors)
   243→  if (!stmt) return null
   244→  // Se o statement abre um bloco (if, repeat, etc.), parsear o corpo
   245→  if (['if', 'elseif', 'else', 'repeat_n', 'repeat_inc', 'repeat_dec', 'switch', 'case', 'default'].includes(stmt.type)) {
   246→    const body = parseBlock(lines, idx + 1, errors)
   247→    if (body.error) {
   248→      errors.push({ line: lines[idx].line, message: body.error })
   249→      return { ...stmt, body: [], nextIdx: body.nextIdx }
   250→    }
   251→    return { ...stmt, body: body.statements, nextIdx: body.nextIdx }
   252→  }
   253→  return { ...stmt, nextIdx: idx + 1 }
   254→}

   256→// Parser de valores (números, strings, variáveis, expressões simples)
   257→function parseValue(text) {
   258→  text = text.trim()
   259→  // String
   260→  if (text.startsWith('"') && text.endsWith('"')) {
   261→    return { type: 'string', value: text.slice(1, -1) }
   262→  }
   263→  // Expressão com concatenação (+) — suporta "string" + var + "string" + ...
   264→  // Não divide dentro de strings (respeita aspas)
   265→  if (text.includes('+') && !/^-?\d/.test(text)) {
   266→    const parts = splitPlus(text)
   267→    if (parts.length > 1) {
   268→      return { type: 'concat', parts: parts.map(parseValue) }
   269→    }
   270→  }
   271→  // Número
   272→  const num = parseFloat(text)
   273→  if (!isNaN(num)) return { type: 'number', value: num }
   274→  // Booleano
   275→  if (text === 'true') return { type: 'boolean', value: true }
   276→  if (text === 'false') return { type: 'boolean', value: false }
   277→  // Sistema 2: 'this' refere-se ao próprio objeto (instanceId)
   278→  if (text === 'this') return { type: 'this' }
   279→  // Chamada de função embutida como valor: identifier(args)
   280→  // Ex: getVar("teste"), distanceTo("Cubo"), collidingWith("tipo"), isTouching(), getUIValue("nome")
   281→  const callMatch = text.match(/^(\w+)\s*\(([^)]*)\)$/)
   282→  if (callMatch) {
   283→    const funcName = callMatch[1]
   284→    const args = callMatch[2].split(',').map((a) => a.trim()).filter((a) => a)
   285→    return { type: 'call_value', funcName, args: args.map(parseValue) }
   286→  }
   287→  // Variável
   288→  return { type: 'var', name: text }
   289→}
   290→
   291→// Divide uma expressão por + respeitando aspas (não divide dentro de strings)
   292→function splitPlus(text) {
   293→  const parts = []
   294→  let current = ''
   295→  let inString = false
   296→  for (let i = 0; i < text.length; i++) {
   297→    const c = text[i]
   298→    if (c === '"') inString = !inString
   299→    if (c === '+' && !inString) {
   300→      parts.push(current.trim())
   301→      current = ''
   302→    } else {
   303→      current += c
   304→    }
   305→  }
   306→  parts.push(current.trim())
   307→  return parts.filter((p) => p)
   308→}

   310→// ===== Runtime =====
   311→// Executa o AST compilado, chamando as funções embutidas do gameContext.
   312→
   313→export function createFlirCodeRuntime(source, gameContext) {
   314→  const { functions, classes, errors } = parseFlirCode(source)
   315→
   316→  // Se há erros de sintaxe, reportar e não executar
   317→  for (const err of errors) {
   318→    debugLog(`Erro de sintaxe FlirCode (linha ${err.line}): ${err.message}`, 'error', 'FlirCode')
   319→  }
   320→
   321→  // Variáveis locais do script
   322→  const localVars = {}
   323→
   324→  // Sistema 2: resolver hierarquia de classes e fundir funções/variáveis
   325→  // Se gameContext.className estiver definido, usar essa classe
   326→  const className = gameContext.className || null
   327→  if (className && classes[className]) {
   328→    // Construir cadeia de herança: [Base, ..., Sub]
   329→    const chain = []
   330→    let curr = classes[className]
   331→    while (curr) {
   332→      chain.unshift(curr) // adicionar no início para Base ficar primeiro
   333→      curr = curr.extends ? classes[curr.extends] : null
   334→    }
   335→    // Inicializar variáveis da classe (da base para a sub)
   336→    for (const cls of chain) {
   337→      for (const [varName, varValue] of Object.entries(cls.vars || {})) {
   338→        localVars[varName] = varValue ? (varValue.value !== undefined ? varValue.value : varValue) : 0
   339→      }
   340→    }
   341→    // Fundir funções (sub override base)
   342→    for (const cls of chain) {
   343→      for (const [fnName, fnDef] of Object.entries(cls.functions || {})) {
   344→        // Converter function statement para o formato esperado pelo runtime
   345→        functions[fnName] = {
   346→          name: fnName,
   347→          params: fnDef.params || [],
   348→          body: fnDef.body || [],
   349→          line: fnDef.line,
   350→        }
   351→      }
   352→    }
   353→    debugLog(`Classe "${className}" aplicada: ${Object.keys(localVars).length} vars, ${Object.keys(functions).length} funções`, 'log', 'FlirCode')
   354→  }
   355→
   356→  // Mapear nomes de eventos PT → nome interno
   357→  const eventMap = {
   358→    onStart: 'beginPlay',
   359→    onTick: 'tick',
   360→    onCollide: 'onCollision',
   361→    onTouch: 'onTouch',
   362→    onSeePlayer: 'onSeePlayer',
   363→    onLoseSight: 'onLoseSight',
   364→    onTimer: 'onTimer',
   365→    onEnterZone: 'onEnterZone',
   366→    onExitZone: 'onExitZone',
   367→    onClick: 'onClick',
   368→    onChange: 'onChange',
   369→    onSubmit: 'onSubmit',
   370→    // Fase 5: Eventos de multiplayer
   371→    onPlayerJoin: 'onPlayerJoin',
   372→    onPlayerLeave: 'onPlayerLeave',
   373→    onMessage: 'onMessage',
   374→    // Sistema 3: Evento de sinais
   375→    onSignal: 'onSignal',
   376→    // Sistema 2: Evento de dano
   377→    onDamage: 'onDamage',
   378→    // Sistema 3: Evento de apanhar item
   379→    onPickup: 'onPickup',
   380→    // Sistema: Game State
   381→    onGameStateChange: 'onGameStateChange',
   382→  }
   383→
   384→  // Avaliar um valor (número, string, variável, etc.)
   385→  function evalValue(val) {
   386→    if (!val) return null
   387→    switch (val.type) {
   388→      case 'number': return val.value
   389→      case 'string': return val.value
   390→      case 'boolean': return val.value
   391→      case 'this': return gameContext._instanceId // Sistema 2: this = próprio objeto
   392→      case 'concat': return val.parts.map((p) => {
   393→        const v = evalValue(p)
   394→        return v === null || v === undefined ? '' : String(v)
   395→      }).join('')
   396→      case 'call_value': {
   397→        // Avaliar a função embutida e retornar o resultado
   398→        return execBuiltin(val.funcName, val.args, {})
   399→      }
   400→      case 'var': {
   401→        // Primeiro procura em localVars, depois em globalVars (via getVar)
   402→        if (val.name in localVars) return localVars[val.name]
   403→        const gv = gameContext.getVar?.(val.name)
   404→        return gv ?? 0
   405→      }
   406→      default: return null
   407→    }
   408→  }
   409→
   410→  // Avaliar condição (string como "x > 5", "a == b", etc.)
   411→  function evalCondition(cond) {
   412→    // Simplificado: suporta > < == != >= <=
   413→    const m = cond.match(/^(.+?)\s*(>=|<=|==|!=|>|<)\s*(.+)$/)
   414→    if (!m) return !!evalValue(parseValue(cond))
   415→    const left = evalValue(parseValue(m[1].trim()))
   416→    const op = m[2]
   417→    const right = evalValue(parseValue(m[3].trim()))
   418→    switch (op) {
   419→      case '>': return left > right
   420→      case '<': return left < right
   421→      case '>=': return left >= right
   422→      case '<=': return left <= right
   423→      case '==': return left == right
   424→      case '!=': return left != right
   425→    }
   426→    return false
   427→  }
   428→
   429→  // Executar uma lista de statements
   430→  function execStatements(statements, params = {}) {
   431→    for (const stmt of statements) {
   432→      try {
   433→        execStatement(stmt, params)
   434→      } catch (err) {
   435→        debugLog(`Erro ao executar (linha ${stmt.line}): ${err.message}`, 'error', 'FlirCode')
   436→        // Não travar — continuar
   437→      }
   438→    }
   439→  }
   440→
   441→  // Executar um statement
   442→  function execStatement(stmt, params) {
   443→    switch (stmt.type) {
   444→      case 'var':
   445→        localVars[stmt.name] = evalValue(stmt.value)
   446→        break
   447→      case 'assign':
   448→        localVars[stmt.name] = evalValue(stmt.value)
   449→        break
   450→      case 'if': {
   451→        // if / else if / else chain: o parser coloca elseif e else como statements
   452→        // subsequentes no mesmo nível. Processamos o if e, se falso, procuramos
   453→        // elseif/else nos statements seguintes do bloco pai.
   454→        // Mas como o parser coloca elseif/else como statements separados com body,
   455→        // precisamos de uma abordagem diferente: o if já tem o body, e elseif/else
   456→        // são statements irmãos. O execStatements percorre linearmente.
   457→        // Solução: o if executa o body se true. Se false, NÃO executa.
   458→        // O elseif/else são executados como statements independentes, mas só
   459→        // têm efeito se o if anterior foi false. Usamos uma flag.
   460→        if (evalCondition(stmt.condition)) {
   461→          execStatements(stmt.body || [], params)
   462→          params._ifChainMatched = true
   463→        } else {
   464→          params._ifChainMatched = false
   465→        }
   466→        break
   467→      }
   468→      case 'elseif': {
   469→        // Só executa se nenhum if/elseif anterior no chain foi true
   470→        if (!params._ifChainMatched && evalCondition(stmt.condition)) {
   471→          execStatements(stmt.body || [], params)
   472→          params._ifChainMatched = true
   473→        }
   474→        break
   475→      }
   476→      case 'else': {
   477→        // Só executa se nenhum if/elseif anterior no chain foi true
   478→        if (!params._ifChainMatched) {
   479→          execStatements(stmt.body || [], params)
   480→          params._ifChainMatched = true
   481→        }
   482→        break
   483→      }
   484→      case 'switch': {
   485→        // O switch contém cases e default no seu body
   486→        // Resolver o valor da variável do switch
   487→        let switchVal
   488→        if (localVars[stmt.varName] !== undefined) {
   489→          switchVal = String(localVars[stmt.varName])
   490→        } else if (gameContext.getVar) {
   491→          switchVal = String(gameContext.getVar(stmt.varName) ?? '')
   492→        } else {
   493→          switchVal = ''
   494→        }
   495→        let matched = false
   496→        for (const caseStmt of stmt.body || []) {
   497→          if (caseStmt.type === 'case') {
   498→            // Remover aspas do valor do case
   499→            let caseVal = caseStmt.value.trim()
   500→            if ((caseVal.startsWith('"') && caseVal.endsWith('"')) || (caseVal.startsWith("'") && caseVal.endsWith("'"))) {
   501→              caseVal = caseVal.slice(1, -1)
   502→            }
   503→            if (switchVal === caseVal) {
   504→              execStatements(caseStmt.body || [], params)
   505→              matched = true
   506→              break
   507→            }
   508→          }
   509→        }
   510→        if (!matched) {
   511→          for (const caseStmt of stmt.body || []) {
   512→            if (caseStmt.type === 'default') {
   513→              execStatements(caseStmt.body || [], params)
   514→              break
   515→            }
   516→          }
   517→        }
   518→        break
   519→      }
   520→      // case e default são processados dentro do switch — não executar standalone
   521→      case 'case':
   522→      case 'default':
   523→        break
   524→      case 'call': {
   525→        execBuiltin(stmt.funcName, stmt.args, params)
   526→        break
   527→      }
   528→      case 'repeat_n': {
   529→        localVars[stmt.varName] = 0
   530→        for (let i = 0; i < stmt.count; i++) {
   531→          localVars[stmt.varName] = i
   532→          execStatements(stmt.body || [], params)
   533→        }
   534→        break
   535→      }
   536→      case 'repeat_inc': {
   537→        let v = 0
   538→        while (v <= stmt.until) {
   539→          execStatements(stmt.body || [], { ...params, _repeat: v })
   540→          v += stmt.step
   541→        }
   542→        break
   543→      }
   544→      case 'repeat_dec': {
   545→        let v = stmt.until
   546→        while (v >= 0) {
   547→          execStatements(stmt.body || [], { ...params, _repeat: v })
   548→          v -= stmt.step
   549→        }
   550→        break
   551→      }
   552→      case 'endcode':
   553→      case 'unknown':
   554→        break
   555→    }
   556→  }
   557→
   558→  // Executar função embutida
   559→  function execBuiltin(name, args, params) {
   560→    const evaluatedArgs = args.map((a) => evalValue(a))
   561→    switch (name) {
   562→      case 'playAnim':
   563→        gameContext.playAnimation?.(gameContext._instanceId, evaluatedArgs[0])
   564→        break
   565→      case 'playSound':
   566→        // Procurar SoundObject pelo nome na cena ativa; se não encontrar, usar como URL
   567→        gameContext.playSoundByName?.(evaluatedArgs[0]) ?? gameContext.playSound?.(evaluatedArgs[0])
   568→        break
   569→      case 'move':
   570→        gameContext.moveObject?.(gameContext._instanceId, evaluatedArgs, 1)
   571→        break
   572→      case 'rotate':
   573→        gameContext.rotateObject?.(gameContext._instanceId, evaluatedArgs)
   574→        break
   575→      case 'scale':
   576→        if (gameContext.mesh) {
   577→          gameContext.mesh.scale.set(evaluatedArgs[0] || 1, evaluatedArgs[1] || 1, evaluatedArgs[2] || 1)
   578→        }
   579→        break
   580→      case 'destroy':
   581→        gameContext.destroyObject?.(gameContext._instanceId)
   582→        break
   583→      case 'createObject':
   584→        gameContext.spawnObject?.(evaluatedArgs[0], [evaluatedArgs[1], evaluatedArgs[2], evaluatedArgs[3]])
   585→        break
   586→      case 'changeScene':
   587→        gameContext.changeScene?.(evaluatedArgs[0])
   588→        break
   589→      case 'wait': {
   590→        // wait(seconds) — implementa um delay real via setTimeout
   591→        // Como FlirCode é síncrono, o wait só afeta a chamada atual:
   592→        // regista o delay e o runtime processa ações pendentes após o tempo
   593→        const delayMs = (evaluatedArgs[0] || 0) * 1000
   594→        debugLog('wait(' + evaluatedArgs[0] + 's) — aguardando', 'log', 'FlirCode')
   595→        // Usar Atomics.wait para um sleep síncrono não-bloqueante em worker
   596→        // Como não estamos em worker, usar setTimeout com flag
   597→        if (gameContext._waitQueue) {
   598→          gameContext._waitQueue.push(delayMs)
   599→        }
   600→        break
   601→      }
   602→      case 'setVar':
   603→        gameContext.globalVars = gameContext.globalVars || {}
   604→        gameContext.globalVars[evaluatedArgs[0]] = evaluatedArgs[1]
   605→        break
   606→      case 'getVar':
   607→        return gameContext.globalVars?.[evaluatedArgs[0]]
   608→      case 'showUI':
   609→        debugLog(`showUI: ${evaluatedArgs[0]}`, 'log', 'FlirCode')
   610→        gameContext.showUIScreen?.(evaluatedArgs[0])
   611→        break
   612→      case 'hideUI':
   613→        debugLog(`hideUI: ${evaluatedArgs[0]}`, 'log', 'FlirCode')
   614→        gameContext.hideUIScreen?.(evaluatedArgs[0])
   615→        break
   616→      case 'showUIScreen':
   617→        gameContext.showUIScreen?.(evaluatedArgs[0])
   618→        break
   619→      case 'hideUIScreen':
   620→        gameContext.hideUIScreen?.(evaluatedArgs[0])
   621→        break
   622→      case 'getUIValue':
   623→        return gameContext.getUIValue?.(evaluatedArgs[0]) ?? ''
   624→      case 'setUIValue':
   625→        gameContext.setUIValue?.(evaluatedArgs[0], evaluatedArgs[1])
   626→        break
   627→      case 'print':
   628→        debugLog(evaluatedArgs[0], 'log', 'FlirCode')
   629→        break
   630→      case 'warn':
   631→        debugLog(evaluatedArgs[0], 'warning', 'FlirCode')
   632→        break
   633→      case 'error':
   634→        debugLog(evaluatedArgs[0], 'error', 'FlirCode')
   635→        break
   636→      case 'collidingWith':
   637→        // Verificar se o objeto está a colidir com outro de um tipo específico
   638→        return gameContext.collidingWith?.(gameContext._instanceId, evaluatedArgs[0]) || false
   639→      case 'distanceTo':
   640→        // Calcular distância até outro objeto
   641→        return gameContext.distanceTo?.(gameContext._instanceId, evaluatedArgs[0]) || 0
   642→      case 'isTouching':
   643→        // Verificar se o utilizador está a tocar no ecrã
   644→        return gameContext.isTouching?.() || false
   645→      // Fase 5: Funções de multiplayer
   646→      case 'sendMessage':
   647→        // Envia dados customizados para outros jogadores
   648→        gameContext.sendMessage?.(evaluatedArgs[0])
   649→        break
   650→      case 'getPlayers':
   651→        // Retorna o número de jogadores ligados
   652→        return gameContext.getPlayers?.() || 1
   653→      case 'getPlayerState':
   654→        // Retorna o estado de um jogador específico
   655→        return gameContext.getPlayerState?.(evaluatedArgs[0]) || null
   656→      // Sistema 3: Sinais — emitSignal("nome", dados)
   657→      case 'emitSignal':
   658→        gameContext.emitSignal?.(evaluatedArgs[0], evaluatedArgs[1])
   659→        break
   660→      // Sistema 2: Armas e combate
   661→      case 'shoot':
   662→        gameContext.shoot?.()
   663→        break
   664→      case 'reload':
   665→        gameContext.reload?.()
   666→        break
   667→      case 'equipWeapon':
   668→        gameContext.equipWeapon?.(evaluatedArgs[0])
   669→        break
   670→      case 'getAmmo':
   671→        return gameContext.getAmmo?.() ?? 0
   672→      case 'takeDamage':
   673→        gameContext.takeDamage?.(gameContext._instanceId, evaluatedArgs[0])
   674→        break
   675→      case 'getHealth':
   676→        return gameContext.getHealth?.(gameContext._instanceId) ?? 100
   677→      // Sistema 3: Inventário
   678→      case 'addToInventory':
   679→        gameContext.addToInventory?.(evaluatedArgs[0], evaluatedArgs[1])
   680→        break
   681→      case 'removeFromInventory':
   682→        gameContext.removeFromInventory?.(evaluatedArgs[0], evaluatedArgs[1])
   683→        break
   684→      case 'getInventoryCount':
   685→        return gameContext.getInventoryCount?.(evaluatedArgs[0]) ?? 0
   686→      case 'hasItem':
   687→        return gameContext.hasItem?.(evaluatedArgs[0]) ?? false
   688→      // Sistema: Links — navegar para cena ou tela de UI
   689→      case 'linkTo':
   690→        gameContext.linkTo?.(evaluatedArgs[0], evaluatedArgs[1])
   691→        break
   692→      // Sistema: Game State
   693→      case 'setGameState':
   694→        gameContext.setGameState?.(evaluatedArgs[0])
   695→        break
   696→      case 'getGameState':
   697→        return gameContext.getGameState?.() || 'menu'
   698→      // Sistema: Save/Load Progress (localStorage do jogador)
   699→      case 'saveProgress':
   700→        gameContext.saveProgress?.(evaluatedArgs[0], evaluatedArgs[1])
   701→        break
   702→      case 'loadProgress':
   703→        return gameContext.loadProgress?.(evaluatedArgs[0])
   704→      // Sistema: Sequenciador
   705→      case 'playSequence':
   706→        gameContext.playSequence?.(evaluatedArgs[0])
   707→        break
   708→      // Sistema: Luzes
   709→      case 'setLightIntensity':
   710→        gameContext.setLightIntensity?.(evaluatedArgs[0], evaluatedArgs[1])
   711→        break
   712→      case 'setLightColor':
   713→        gameContext.setLightColor?.(evaluatedArgs[0], evaluatedArgs[1])
   714→        break
   715→      case 'setLightVisible':
   716→        gameContext.setLightVisible?.(evaluatedArgs[0], evaluatedArgs[1])
   717→        break
   718→      // Sistema: Data Assets (ScriptableObjects)
   719→      case 'getDataAsset':
   720→        return gameContext.getDataAsset?.(evaluatedArgs[0])
   721→      // Sistema: Autoloads (Singletons)
   722→      case 'getAutoload':
   723→        return gameContext.getAutoload?.(evaluatedArgs[0])
   724→      default:
   725→        debugLog(`Função desconhecida: ${name}`, 'warning', 'FlirCode')
   726→    }
   727→  }

   732→    hasErrors: errors.length > 0,

   734→    // Dispara um evento
   735→    triggerEvent(eventName, payload = {}) {
   736→      // Mapear nome interno → nome da função FlirCode
   737→      const funcName = Object.entries(eventMap).find(([_, en]) => en === eventName)?.[0]
   738→      if (!funcName) return
   739→      const fn = functions[funcName]
   740→      if (!fn) return
   741→      // Passar payload como parâmetros
   742→      const params = {}
   743→      if (fn.params.length > 0) {
   744→        // Sistema 3: para onSignal, passar name e data como parâmetros separados
   745→        if (funcName === 'onSignal' && fn.params.length >= 2) {
   746→          params[fn.params[0]] = payload.name
   747→          params[fn.params[1]] = payload.data
   748→        } else if (funcName === 'onMessage' && fn.params.length >= 2) {
   749→          params[fn.params[0]] = payload.playerId
   750→          params[fn.params[1]] = payload.data
   751→        } else {
   752→          params[fn.params[0]] = payload
   753→        }
   754→      }
   755→      gameContext._instanceId = gameContext._instanceId // já definido externamente
   756→      // Sistema 3: copiar params para localVars para ficarem acessíveis no script
   757→      for (const [k, v] of Object.entries(params)) {
   758→        localVars[k] = v
   759→      }
   760→      execStatements(fn.body, params)
   761→    },
   762→
   763→    update(delta) {
   764→      // Não usado — FlirCode é orientado a eventos
   765→    },
   766→
   767→    dispose() {
   768→      // limpar
   769→    },
   770→  }
