    57→    57→function SceneBackgroundSolid({ background }) {
    58→    58→  const { scene } = useThree()
    59→    59→  useEffect(() => {
    60→    60→    if (background.type === 'gradient') {
    61→    61→      const canvas = document.createElement('canvas')
    62→    62→      canvas.width = 2; canvas.height = 256
    63→    63→      const ctx = canvas.getContext('2d')
    64→    64→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
    65→    65→      grad.addColorStop(0, background.gradientTop)
    66→    66→      grad.addColorStop(1, background.gradientBottom)
    67→    67→      ctx.fillStyle = grad; ctx.fillRect(0, 0, 2, 256)
    68→    68→      const tex = new THREE.CanvasTexture(canvas)
    69→    69→      tex.colorSpace = THREE.SRGBColorSpace
    70→    70→      scene.background = tex
    71→    71→      return () => tex.dispose()
    72→    72→    } else {
    73→    73→      scene.background = new THREE.Color(background.color)
    74→    74→    }
    75→    75→  }, [background, scene])
    76→    76→  return null
    77→    77→}

   139→   139→      })

   141→   141→        for (const rt of runtimesRef.current.values()) {

   146→   146→    return () => {
   147→   147→      window._multiplayer = null

   152→   152→  useEffect(() => {

   154→   154→    // (não por spawnObject, mas por mudança de cena manual via changeScene)

   276→   276→      isTouching: () => joystickRef.current.active,
   277→   277→      // Fase 5: Multiplayer functions
   278→   278→      sendMessage: (data) => {

   292→   292→    }
   293→   293→    window._flirGameContext = gameContext
   294→   294→

   312→   312→

   383→   383→              else if (en === 'OnLoseSight') rt.triggerEvent('onLoseSight', payload)
   384→   384→            }
   385→   385→          },
   386→   386→        })
   387→   387→        npcAIsRef.current.set(conect.instanceId, ai)

   389→   389→    }
   390→   390→
   391→   391→    // Timers
   392→   392→    for (const conect of setupScene.conects || []) {
   393→   393→      if (conect.type === 'TimerObject' && conect.autoStart) {
   394→   394→        timerStatesRef.current.set(conect.instanceId, { remaining: conect.duration || 5, loop: conect.loop || false, duration: conect.duration || 5 })
   395→   395→      }
   396→   396→    }
   397→   397→
   398→   398→    // SkyObject / FogObject
   399→   399→    const skyConect = (setupScene.conects || []).find((c) => c.type === 'SkyObject')
   400→   400→    if (skyConect) {
   401→   401→      // Aplicado via SceneBackgroundSolid se mudar o background
   402→   402→    }
   403→   403→    const fogConect = (setupScene.conects || []).find((c) => c.type === 'FogObject')
   404→   404→    // Fog aplicado no useFrame

   406→   406→    // Joystick touch
   407→   407→    const onTouchStart = (e) => { if (e.touches.length === 1) { joystickRef.current.active = true } }
   408→   408→    const onTouchEnd = () => { joystickRef.current.active = false; joystickRef.current.x = 0; joystickRef.current.z = 0 }
   409→   409→    window.addEventListener('touchstart', onTouchStart)
   410→   410→    window.addEventListener('touchend', onTouchEnd)
   411→   411→
   412→   412→    // Teclado
   413→   413→    const keys = {}
   414→   414→    const onKeyDown = (e) => { keys[e.key.toLowerCase()] = true }
   415→   415→    const onKeyUp = (e) => { keys[e.key.toLowerCase()] = false }
   416→   416→    window.addEventListener('keydown', onKeyDown)
   417→   417→    window.addEventListener('keyup', onKeyUp)
   418→   418→    window._flirKeys = keys
   419→   419→
   420→   420→    return () => {
   421→   421→      for (const rt of runtimesRef.current.values()) rt.dispose()
   422→   422→      runtimesRef.current.clear()
   423→   423→      for (const ai of npcAIsRef.current.values()) ai.dispose()
   424→   424→      npcAIsRef.current.clear()
   425→   425→      for (const [, s] of timerStatesRef.current) { if (s.interval) clearInterval(s.interval); if (s.audio) s.audio.pause() }
   426→   426→      timerStatesRef.current.clear()
   427→   427→      animPlayersRef.current.clear()
   428→   428→      physicsRef.current?.dispose()
   429→   429→      physicsRef.current = null
   430→   430→      window.removeEventListener('touchstart', onTouchStart)
   431→   431→      window.removeEventListener('touchend', onTouchEnd)
   432→   432→      window.removeEventListener('keydown', onKeyDown)
   433→   433→      window.removeEventListener('keyup', onKeyUp)
   434→   434→      window._flirGameContext = null
   435→   435→      gameStartedRef.current = false
   436→   436→    }
   437→   437→  }, [isGameMode, setupScene])
   438→   438→
   439→   439→  // Loop do jogo
   440→   440→  useFrame((_, delta) => {
   441→   441→    if (!isGameMode) return
   442→   442→
   443→   443→    // Física
   444→   444→    if (physicsRef.current) {
   445→   445→      physicsRef.current.update(delta)
   446→   446→      // Sincronizar meshes com bodies
   447→   447→      for (const [id, entry] of physicsRef.current.bodies) {
   448→   448→        const mesh = meshRefs.current.get(id) || conectMeshRefs.current.get(id)
   449→   449→        if (mesh) {
   450→   450→          mesh.position.copy(entry.body.position)
   451→   451→          mesh.quaternion.copy(entry.body.quaternion)
   452→   452→        }
   453→   453→      }
   454→   454→    }
   455→   455→
   456→   456→    // FlirCode onTick
   457→   457→    for (const rt of runtimesRef.current.values()) {
   458→   458→      rt.update(delta)
   459→   459→      rt.triggerEvent('tick', { deltaTime: delta })
   460→   460→    }
   461→   461→
   462→   462→    // Animation players
   463→   463→    for (const player of animPlayersRef.current.values()) player.update(delta)
   464→   464→
   465→   465→    // NPC AI
   466→   466→    for (const ai of npcAIsRef.current.values()) ai.update(delta)
   467→   467→
   468→   468→    // Timers
   469→   469→    for (const [id, state] of timerStatesRef.current) {
   470→   470→      if (id.startsWith('_')) continue
   471→   471→      state.remaining -= delta
   472→   472→      if (state.remaining <= 0) {
   473→   473→        const rt = runtimesRef.current.get(id)
   474→   474→        if (rt) rt.triggerEvent('onTimer')
   475→   475→        if (state.loop) state.remaining = state.duration
   476→   476→        else timerStatesRef.current.delete(id)
   477→   477→      }
   478→   478→    }
   479→   479→
   480→   480→    // Joystick → PersonalObject
   481→   481→    const keys = window._flirKeys || {}
   482→   482→    if (joystickRef.current.active || keys['w'] || keys['a'] || keys['s'] || keys['d']) {
   483→   483→      for (const conect of activeScene.conects || []) {
   484→   484→        if (conect.type === 'PersonalObject') {
   485→   485→          const speed = conect.moveSpeed || 5
   486→   486→          let mx = 0, mz = 0
   487→   487→          if (joystickRef.current.active) { mx = joystickRef.current.x * speed; mz = joystickRef.current.z * speed }
   488→   488→          if (keys['w']) mz = -speed
   489→   489→          if (keys['s']) mz = speed
   490→   490→          if (keys['a']) mx = -speed
   491→   491→          if (keys['d']) mx = speed
   492→   492→          physicsRef.current?.movePersonal(conect.instanceId, [mx, 0, mz], 1)
   493→   493→          if ((keys[' '] || keys['space']) && conect.canJump) {
   494→   494→            physicsRef.current?.jumpPersonal(conect.instanceId)
   495→   495→          }
   496→   496→        }
   497→   497→      }
   498→   498→    }
   499→   499→
   500→   500→    // Câmara: ViewObject ativa
   501→   501→    const viewConects = (activeScene.conects || []).filter((c) => c.type === 'ViewObject')
   502→   502→    const activeView = viewConects.find((c) => c.cameraRole === 'player') || viewConects.find((c) => c.cameraRole === 'primary') || viewConects[0]
   503→   503→    if (activeView) {
   504→   504→      // Se cameraRole='player' e não tem followTarget, seguir PersonalObject
   505→   505→      let targetId = activeView.followTarget
   506→   506→      if (!targetId && activeView.cameraRole === 'player') {
   507→   507→        const player = (activeScene.conects || []).find((c) => c.type === 'PersonalObject')
   508→   508→        if (player) targetId = player.instanceId
   509→   509→      }
   510→   510→      if (targetId && activeView.followMode && activeView.followMode !== 'none') {
   511→   511→        const targetMesh = meshRefs.current.get(targetId) || conectMeshRefs.current.get(targetId)
   512→   512→        if (targetMesh) {
   513→   513→          const mode = activeView.followMode
   514→   514→          const dist = activeView.followDistance || 6
   515→   515→          const height = activeView.followHeight || 3
   516→   516→          if (mode === 'third') {
   517→   517→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x, targetMesh.position.y + height, targetMesh.position.z + dist), 0.1)
   518→   518→            camera.lookAt(targetMesh.position)
   519→   519→          } else if (mode === 'top') {
   520→   520→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x, targetMesh.position.y + dist, targetMesh.position.z), 0.1)
   521→   521→            camera.lookAt(targetMesh.position)
   522→   522→          } else if (mode === 'side') {
   523→   523→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x + dist, targetMesh.position.y + height / 2, targetMesh.position.z), 0.1)
   524→   524→            camera.lookAt(targetMesh.position)
   525→   525→          }
   526→   526→        }
   527→   527→      } else {
   528→   528→        // Câmara estática na posição da ViewObject
   529→   529→        camera.position.set(...(activeView.position || [5, 4, 6]))
   530→   530→        if (activeView.rotation) camera.rotation.set(...activeView.rotation)
   531→   531→        else camera.lookAt(0, 0, 0)
   532→   532→      }
   533→   533→    }
   534→   534→  })
   535→   535→
   536→   536→  return null
   537→   537→}

   540→   540→export default function SceneLevel3D() {
   541→   541→  const scenes = useStore((s) => s.scenes)
   542→   542→  const activeSceneId = useStore((s) => s.activeSceneId)
   543→   543→  const objects = useStore((s) => s.objects)
   544→   544→  const background = useStore((s) => s.background)
   545→   545→  const grid = useStore((s) => s.grid)
   546→   546→  const lights = useStore((s) => s.lights)
   547→   547→  const transformMode = useStore((s) => s.transformMode)
   548→   548→  const updateSceneInstance = useStore((s) => s.updateSceneInstance)
   549→   549→  const updateConect = useStore((s) => s.updateConect)
   550→   550→  const addObjectToScene = useStore((s) => s.addObjectToScene)
   551→   551→  const addConectToScene = useStore((s) => s.addConectToScene)
   552→   552→  const selectConect = useStore((s) => s.selectConect)
   553→   553→  const scenePreviewOpen = useStore((s) => s.scenePreviewOpen)
   554→   554→
   555→   555→  const [selectedInstanceId, setSelectedInstanceId] = useState(null)
   556→   556→  const [selectedType, setSelectedType] = useState(null)
   557→   557→  const orbitRef = useRef(null)
   558→   558→  const meshRefs = useRef(new Map())
   559→   559→  const conectMeshRefs = useRef(new Map())
   560→   560→  const [selectedMesh, setSelectedMesh] = useState(null)
   561→   561→
   562→   562→  const activeScene = scenes.find((s) => s.id === activeSceneId)
   563→   563→  const isGameMode = scenePreviewOpen
   564→   564→
   565→   565→  useEffect(() => {
   566→   566→    if (selectedInstanceId && meshRefs.current.has(selectedInstanceId)) {
   567→   567→      setSelectedMesh(meshRefs.current.get(selectedInstanceId))
   568→   568→    } else if (selectedInstanceId && conectMeshRefs.current.has(selectedInstanceId)) {
   569→   569→      setSelectedMesh(conectMeshRefs.current.get(selectedInstanceId))
   570→   570→    } else {
   571→   571→      setSelectedMesh(null)
   572→   572→    }
   573→   573→  }, [selectedInstanceId, activeScene])
   574→   574→
   575→   575→  const setMeshRef = useCallback((id, node) => {
   576→   576→    if (node) meshRefs.current.set(id, node)
   577→   577→    else meshRefs.current.delete(id)
   578→   578→  }, [])
   579→   579→
   580→   580→  const setConectMeshRef = useCallback((id, node) => {
   581→   581→    if (node) conectMeshRefs.current.set(id, node)
   582→   582→    else conectMeshRefs.current.delete(id)
   583→   583→  }, [])
   584→   584→
   585→   585→  const handleDrop = useCallback((e) => {
   586→   586→    if (isGameMode) return
   587→   587→    e.preventDefault()
   588→   588→    const objectId = e.dataTransfer.getData('text/objectId')
   589→   589→    const conectType = e.dataTransfer.getData('text/conectType')
   590→   590→    const rect = e.currentTarget.getBoundingClientRect()
   591→   591→    const x = ((e.clientX - rect.left) / rect.width) * 2 - 1
   592→   592→    const y = -((e.clientY - rect.top) / rect.height) * 2 + 1
   593→   593→    const position = [x * 3, 0.5, -y * 3]
   594→   594→    if (objectId) addObjectToScene(objectId, position)
   595→   595→    else if (conectType) addConectToScene(conectType, position)
   596→   596→  }, [addObjectToScene, addConectToScene, isGameMode])
   597→   597→
   598→   598→  const selectObject = useCallback((instanceId) => {
   599→   599→    if (isGameMode) return
   600→   600→    setSelectedInstanceId(instanceId)
   601→   601→    setSelectedType('object')
   602→   602→    selectConect(null)
   603→   603→  }, [selectConect, isGameMode])
   604→   604→
   605→   605→  const selectConectInstance = useCallback((instanceId) => {
   606→   606→    if (isGameMode) return
   607→   607→    setSelectedInstanceId(instanceId)
   608→   608→    setSelectedType('conect')
   609→   609→    selectConect(instanceId)
   610→   610→  }, [selectConect, isGameMode])
   611→   611→
   612→   612→  const handleTransformUpdate = useCallback(() => {
   613→   613→    if (!selectedMesh || !selectedInstanceId) return
   614→   614→    const transform = {
   615→   615→      position: [selectedMesh.position.x, selectedMesh.position.y, selectedMesh.position.z],
   616→   616→      rotation: [selectedMesh.rotation.x, selectedMesh.rotation.y, selectedMesh.rotation.z],
   617→   617→      scale: [selectedMesh.scale.x, selectedMesh.scale.y, selectedMesh.scale.z],
   618→   618→    }
   619→   619→    if (selectedType === 'conect') updateConect(selectedInstanceId, transform)
   620→   620→    else updateSceneInstance(selectedInstanceId, transform)
   621→   621→  }, [selectedMesh, selectedInstanceId, selectedType, updateConect, updateSceneInstance])
   622→   622→
   623→   623→  if (!activeScene) {
   624→   624→    return (
   625→   625→      <div className="viewport" style={{ display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
   626→   626→        <div className="empty-state">
   627→   627→          <div style={{ fontSize: 32, opacity: 0.4 }}></div>
   628→   628→          <div>Nenhuma cena ativa.</div>
   629→   629→        </div>
   630→   630→      </div>
   631→   631→    )
   632→   632→  }
   633→   633→
   634→   634→  return (
   635→   635→    <div className="viewport" onDragOver={(e) => !isGameMode && e.preventDefault()} onDrop={handleDrop}>
   636→   636→      <Canvas
   637→   637→        shadows
   638→   638→        dpr={[1, 2]}
   639→   639→        camera={{ position: [8, 6, 10], fov: 50, near: 0.1, far: 200 }}
   640→   640→        gl={{ antialias: true, preserveDrawingBuffer: true, alpha: false }}
   641→   641→        onPointerMissed={() => {
   642→   642→          if (!isGameMode) {
   643→   643→            setSelectedInstanceId(null)
   644→   644→            setSelectedType(null)
   645→   645→            selectConect(null)
   646→   646→          }
   647→   647→        }}
   648→   648→      >
   649→   649→        <Suspense fallback={null}>
   650→   650→          <SceneBackgroundSolid background={background} />
   651→   651→
   652→   652→          <ambientLight intensity={lights.ambient.intensity} color={lights.ambient.color} />
   653→   653→          <directionalLight
   654→   654→            intensity={lights.directional.intensity}
   655→   655→            color={lights.directional.color}
   656→   656→            position={lights.directional.position}
   657→   657→            castShadow
   658→   658→            shadow-mapSize-width={2048}
   659→   659→            shadow-mapSize-height={2048}
   660→   660→            shadow-camera-left={-20}
   661→   661→            shadow-camera-right={20}
   662→   662→            shadow-camera-top={20}
   663→   663→            shadow-camera-bottom={-20}
   664→   664→          />
   665→   665→          <hemisphereLight intensity={0.3} groundColor="#1a1a2e" color="#ffffff" />
   666→   666→
   667→   667→          {/* Grelha — só no editor */}
   668→   668→          {!isGameMode && grid.visible && (
   669→   669→            <Grid position={[0, 0, 0]} args={[grid.size, grid.divisions]} cellColor={grid.color} sectionColor={grid.color} sectionThickness={1.2} cellThickness={0.6} fadeDistance={30} fadeStrength={1} />
   670→   670→          )}
   671→   671→          {!isGameMode && <ContactShadows position={[0, 0.001, 0]} opacity={0.35} scale={40} blur={2.5} far={5} />}
   672→   672→
   673→   673→          {/* Objetos do catálogo */}
   674→   674→          {activeScene.objects.map((instance) => {
   675→   675→            const obj = objects.find((o) => o.id === instance.objectId)
   676→   676→            if (!obj) return null
   677→   677→            const sceneObj = { ...obj, id: instance.instanceId, position: instance.position, rotation: instance.rotation, scale: instance.scale }
   678→   678→            const isSelected = !isGameMode && instance.instanceId === selectedInstanceId
   679→   679→            const isPlayer = instance.instanceId === activeScene.playerObjectId
   680→   680→            return (
   681→   681→              <group key={instance.instanceId}>
   682→   682→                <SceneObject ref={(node) => setMeshRef(instance.instanceId, node)} obj={sceneObj} isSelected={isSelected} onSelect={() => selectObject(instance.instanceId)} />
   683→   683→                {isPlayer && !isGameMode && <PlayerMarker position={instance.position} />}
   684→   684→              </group>
   685→   685→            )
   686→   686→          })}
   687→   687→
   688→   688→          {/* Conects */}
   689→   689→          {(activeScene.conects || []).map((conect) => (
   690→   690→            <ConectSelectorWrapper key={conect.instanceId} conect={conect} objects={objects} isSelected={!isGameMode && conect.instanceId === selectedInstanceId} onSelect={() => selectConectInstance(conect.instanceId)} setMeshRef={(node) => setConectMeshRef(conect.instanceId, node)} isGameMode={isGameMode} />
   691→   691→          ))}
   692→   692→
   693→   693→          {/* GameCamera gizmo — só no editor */}
   694→   694→          {!isGameMode && !(activeScene.conects || []).some((c) => c.type === 'ViewObject') && (
   695→   695→            <GameCameraGizmo camera={activeScene.gameCamera} />
   696→   696→          )}
   697→   697→
   698→   698→          {/* TransformControls — só no editor */}
   699→   699→          {!isGameMode && selectedMesh && selectedInstanceId && (
   700→   700→            <TransformControls object={selectedMesh} mode={transformMode} size={1.2}
   701→   701→              onMouseDown={() => { if (orbitRef.current) orbitRef.current.enabled = false }}
   702→   702→              onMouseUp={() => { if (orbitRef.current) orbitRef.current.enabled = true; handleTransformUpdate() }}
   703→   703→              onObjectChange={handleTransformUpdate}
   704→   704→            />
   705→   705→          )}
   706→   706→
   707→   707→          {/* OrbitControls — só no editor */}
   708→   708→          {!isGameMode && (
   709→   709→            <OrbitControls ref={orbitRef} makeDefault enableDamping dampingFactor={0.08} minDistance={1} maxDistance={100} maxPolarAngle={Math.PI * 0.495}
   710→   710→              touches={{ ONE: THREE.TOUCH.ROTATE, TWO: THREE.TOUCH.DOLLY_PAN }}
   711→   711→            />
   712→   712→          )}
   713→   713→
   714→   714→          {/* GameMode — activado quando scenePreviewOpen */}
   715→   715→          <GameMode activeScene={activeScene} objects={objects} meshRefs={meshRefs} conectMeshRefs={conectMeshRefs} isGameMode={isGameMode} />
   716→   716→        </Suspense>
   717→   717→      </Canvas>
   718→   718→    </div>
   719→   719→  )
   720→   720→}
