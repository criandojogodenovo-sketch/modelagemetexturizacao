     1→/**
     2→ * ConectRenderer — renderiza um Conect na cena 3D.
     3→ *
     4→ * Dependendo do tipo de Conect:
     5→ *  - VisualObject: renderiza o modelo do catálogo (sourceObjectId)
     6→ *  - RigidObject/StaticObject/StopObject: renderiza um cubo placeholder + física
     7→ *  - PersonalObject: renderiza um capsule/cubo como jogador
     8→ *  - TriggerObject: renderiza um wireframe transparente (não visível no jogo)
     9→ *  - LuminousObject: adiciona uma luz à cena
    10→ *  - ParticleObject: sistema de partículas
    11→ *  - TerrainObject: terreno com heightmap
    12→ *  - WaterObject: plano de água
    13→ *  - PathObject: renderiza waypoints
    14→ *  - CheckpointObject: renderiza uma bandeira/marca
    15→ *  - Outros (UI, Sound, Sky, Fog, View, etc.): sem visual 3D (só lógica)
    16→ */
    17→import { forwardRef, useMemo, useEffect, useRef } from 'react'
    18→import { useThree, useFrame } from '@react-three/fiber'
    19→import * as THREE from 'three'
    20→import { useStore } from '../../store/useStore'
    21→import { findConectDefinition } from '../../utils/conects/taxonomy'
    22→import { createWaterProMaterial } from '../../utils/waterShaderPro'
    23→import { createSkyProMaterial, calculateSunDirection } from '../../utils/skyShaderPro'
    24→import { createRealWaterMaterial } from '../../utils/realWaterShader'
    25→import SceneObject from '../3d/SceneObject'

    27→const ConectRenderer = forwardRef(function ConectRenderer({ conect, objects, setMeshRef }, meshRef) {
    28→  const def = findConectDefinition(conect.type)
    29→
    30→  // Se não tem visual, não renderiza mesh — mas pode adicionar luzes, etc.
    31→  if (conect.type === 'LuminousObject') {
    32→    return <LuminousMesh conect={conect} setMeshRef={setMeshRef} />
    33→  }
    34→
    35→  if (conect.type === 'TerrainObject') {
    36→    return <TerrainMesh conect={conect} setMeshRef={setMeshRef} />
    37→  }
    38→
    39→  if (conect.type === 'WaterObject') {
    40→    return <WaterMesh conect={conect} setMeshRef={setMeshRef} />
    41→  }
    42→
    43→  if (conect.type === 'PathObject') {
    44→    return <PathMesh conect={conect} setMeshRef={setMeshRef} />
    45→  }
    46→
    47→  if (conect.type === 'CheckpointObject') {
    48→    return <CheckpointMesh conect={conect} setMeshRef={setMeshRef} />
    49→  }
    50→
    51→  if (conect.type === 'ParticleObject') {
    52→    return <ParticleMesh conect={conect} setMeshRef={setMeshRef} />
    53→  }
    54→
    55→  // ViewObject: renderiza um gizmo de câmara visível no editor (selecionável)
    56→  if (conect.type === 'ViewObject') {
    57→    return <ViewObjectMesh conect={conect} setMeshRef={setMeshRef} />
    58→  }
    59→
    60→  // TrailObject: rasto visual atrás de um objeto em movimento
    61→  if (conect.type === 'TrailObject') {
    62→    return <TrailMesh conect={conect} setMeshRef={setMeshRef} />
    63→  }
    64→
    65→  // ReflectObject: sonda de reflexo (renderiza uma esfera espelhada)
    66→  if (conect.type === 'ReflectObject') {
    67→    return <ReflectMesh conect={conect} setMeshRef={setMeshRef} />
    68→  }
    69→
    70→  // SkyObject: renderiza esfera de céu com shader procedural
    71→  if (conect.type === 'SkyObject') {
    72→    return <SkyMesh conect={conect} setMeshRef={setMeshRef} />
    73→  }
    74→
    75→  // SunObject: luz direcional (sol)
    76→  if (conect.type === 'SunObject') {
    77→    return <SunLightMesh conect={conect} setMeshRef={setMeshRef} />
    78→  }
    79→
    80→  // PointObject: luz pontual
    81→  if (conect.type === 'PointObject') {
    82→    return <PointLightMesh conect={conect} setMeshRef={setMeshRef} />
    83→  }
    84→
    85→  // SpotObject: luz spot
    86→  if (conect.type === 'SpotObject') {
    87→    return <SpotLightMesh conect={conect} setMeshRef={setMeshRef} />
    88→  }
    89→
    90→  // AmbientObject: luz ambiente (sem gizmo, só aplica à cena)
    91→  if (conect.type === 'AmbientObject') {
    92→    return <AmbientLightMesh conect={conect} setMeshRef={setMeshRef} />
    93→  }
    94→
    95→  // SpawnObject: ponto de spawn (gizmo em forma de estrela)
    96→  if (conect.type === 'SpawnObject') {
    97→    return <SpawnMarkerMesh conect={conect} setMeshRef={setMeshRef} />
    98→  }
    99→
   100→  // NavigatorObject: portal para outra cena
   101→  if (conect.type === 'NavigatorObject') {
   102→    return <NavigatorMesh conect={conect} setMeshRef={setMeshRef} />
   103→  }
   104→
   105→  // WeaponObject: arma visual (gizmo em forma de arma)
   106→  if (conect.type === 'WeaponObject') {
   107→    return <WeaponMesh conect={conect} setMeshRef={setMeshRef} />
   108→  }
   109→
   110→  // ItemObject: item coleccionável (gizmo em forma de diamante rotativo)
   111→  if (conect.type === 'ItemObject') {
   112→    return <ItemMesh conect={conect} setMeshRef={setMeshRef} />
   113→  }
   114→
   115→  // AreaObject: área de influência (gizmo wireframe)
   116→  if (conect.type === 'AreaObject') {
   117→    return <AreaMesh conect={conect} setMeshRef={setMeshRef} />
   118→  }
   119→
   120→  // Realism Conects — gizmos visuais no editor (efeitos reais aplicados no runtime pelo PostProcessingPanel)
   121→  if (conect.type === 'GIProbeObject') return <GIProbeMesh conect={conect} setMeshRef={setMeshRef} />
   122→  if (conect.type === 'SSRObject') return <SSRMesh conect={conect} setMeshRef={setMeshRef} />
   123→  if (conect.type === 'VolumetricFogObject') return <VolumetricFogMesh conect={conect} setMeshRef={setMeshRef} />
   124→  if (conect.type === 'SSSObject') return <SSSMesh conect={conect} setMeshRef={setMeshRef} />
   125→  if (conect.type === 'BloomObject') return <BloomMesh conect={conect} setMeshRef={setMeshRef} />
   126→  if (conect.type === 'DOFObject') return <DOFMesh conect={conect} setMeshRef={setMeshRef} />
   127→
   128→  // Markers
   129→  if (conect.type === 'EmptyObject') return <EmptyMesh conect={conect} setMeshRef={setMeshRef} />
   130→  if (conect.type === 'ArrowMarker') return <ArrowMarkerMesh conect={conect} setMeshRef={setMeshRef} />
   131→  if (conect.type === 'PointMarker') return <PointMarkerMesh conect={conect} setMeshRef={setMeshRef} />
   132→
   133→  // RealWater — água super-realista
   134→  if (conect.type === 'RealWaterObject') return <RealWaterMesh conect={conect} setMeshRef={setMeshRef} />
   135→
   136→  // ReferenceObject: renderiza conteúdo de outra cena
   137→  if (conect.type === 'ReferenceObject') {
   138→    return <ReferenceMesh conect={conect} setMeshRef={setMeshRef} />
   139→  }
   140→
   141→  if (!def?.hasVisual && conect.type !== 'VisualObject') {
   142→    // Sem visual — ligar o meshRef a null para que physics não tente usar
   143→    useEffect(() => { setMeshRef?.(null) }, [])
   144→    return null
   145→  }
   146→
   147→  // VisualObject: usa modelo do catálogo
   148→  if (conect.type === 'VisualObject') {
   149→    const obj = objects.find((o) => o.id === conect.sourceObjectId)
   150→    if (!obj) return null
   151→    const sceneObj = {
   152→      ...obj,
   153→      id: conect.instanceId,
   154→      position: conect.position,
   155→      rotation: conect.rotation,
   156→      scale: conect.scale,
   157→    }
   158→    return (
   159→      <SceneObject
   160→        ref={(node) => setMeshRef?.(node)}
   161→        obj={sceneObj}
   162→        isSelected={false}
   163→        onSelect={() => {}}
   164→      />
   165→    )
   166→  }
   167→
   168→  // Rigid/Static/Stop/Personal: placeholder com geometria simples
   169→  return (
   170→    <PlaceholderMesh
   171→      ref={(node) => {
   172→        if (typeof meshRef === 'function') meshRef(node)
   173→        else if (meshRef) meshRef.current = node
   174→        setMeshRef?.(node)
   175→      }}
   176→      conect={conect}
   177→    />
   178→  )
   179→})
   180→
   181→export default ConectRenderer

   183→// ===== Placeholder para conects com física =====
   184→const PlaceholderMesh = forwardRef(function PlaceholderMesh({ conect }, ref) {
   185→  // Em modo jogo com ViewObject FPS, esconder o PersonalObject para não tapar a câmara
   186→  const scenePreviewOpen = useStore((s) => s.scenePreviewOpen)
   187→  // Selector estreito: só lê os ViewObjects da cena ativa, não o array scenes inteiro
   188→  const hasFPSView = useStore((s) => {
   189→    if (!scenePreviewOpen || conect.type !== 'PersonalObject') return false
   190→    const activeScene = s.scenes.find((sc) => sc.id === s.activeSceneId)
   191→    if (!activeScene) return false
   192→    return (activeScene.conects || []).some((c) =>
   193→      c.type === 'ViewObject' && c.followMode === 'first' &&
   194→      (c.followTarget === conect.instanceId || c.cameraRole === 'player')
   195→    )
   196→  })
   197→
   198→  const color = conect.type === 'PersonalObject' ? '#3fb950'
   199→                : conect.type === 'StaticObject' ? '#6e7681'
   200→                : conect.type === 'StopObject' ? '#d29922'
   201→                : '#888888'
   202→  const geometry = conect.type === 'PersonalObject'
   203→    ? <capsuleGeometry args={[0.4, 1, 8, 16]} />
   204→    : <boxGeometry args={[1, 1, 1]} />
   205→  return (
   206→    <mesh
   207→      ref={ref}
   208→      position={conect.position}
   209→      rotation={conect.rotation}
   210→      scale={conect.scale}
   211→      visible={conect.visible !== false && !hasFPSView}
   212→      castShadow
   213→      receiveShadow
   214→      userData={{ conectInstanceId: conect.instanceId }}
   215→    >
   216→      {geometry}
   217→      <meshStandardMaterial color={color} roughness={0.6} metalness={0.1} />
   218→    </mesh>
   219→  )
   220→})

   222→// ===== LuminousObject =====
   223→// Luz real + gizmo visual (esfera amarela) para seleção no editor
   224→function LuminousMesh({ conect, setMeshRef }) {
   225→  const lightProps = {
   226→    intensity: conect.intensity,
   227→    color: conect.color,
   228→    castShadow: conect.castShadow,
   229→  }
   230→  let lightElement
   231→  if (conect.lightType === 'directional') {
   232→    lightElement = <directionalLight {...lightProps} position={[0, 0, 0]} />
   233→  } else if (conect.lightType === 'spot') {
   234→    lightElement = <spotLight {...lightProps} distance={conect.distance} angle={Math.PI / 4} penumbra={0.3} position={[0, 0, 0]} />
   235→  } else {
   236→    lightElement = <pointLight {...lightProps} distance={conect.distance} position={[0, 0, 0]} />
   237→  }
   238→  return (
   239→    <group
   240→      ref={setMeshRef}
   241→      position={conect.position}
   242→      rotation={conect.rotation}
   243→      userData={{ conectInstanceId: conect.instanceId }}
   244→    >
   245→      {lightElement}
   246→      {/* Gizmo visual: esfera amarela para seleção */}
   247→      <mesh>
   248→        <sphereGeometry args={[0.2, 12, 12]} />
   249→        <meshBasicMaterial color="#ffd700" />
   250→      </mesh>
   251→      {/* Halo ao redor */}
   252→      <mesh>
   253→        <sphereGeometry args={[0.35, 12, 12]} />
   254→        <meshBasicMaterial color="#ffd700" transparent opacity={0.2} />
   255→      </mesh>
   256→    </group>
   257→  )
   258→}

   260→// ===== TerrainObject =====
   261→function TerrainMesh({ conect, setMeshRef }) {
   262→  const geometry = useMemo(() => {
   263→    const seg = conect.segments || 64
   264→    const g = new THREE.PlaneGeometry(
   265→      conect.width || 50, conect.depth || 50, seg, seg
   266→    )
   267→    g.rotateX(-Math.PI / 2)
   268→    const positions = g.attributes.position
   269→    const heightScale = conect.heightScale || 5
   270→
   271→    // Se há heightmap exportado do TerrainEditor, usá-lo
   272→    if (conect.heightmap && conect.heightmap.length > 0) {
   273→      const hm = conect.heightmap
   274→      for (let i = 0; i < positions.count && i < hm.length; i++) {
   275→        positions.setY(i, hm[i] * heightScale)
   276→      }
   277→    } else {
   278→      // Fallback: gerar procedural
   279→      const seed = conect.heightmapSeed || 1
   280→      for (let i = 0; i < positions.count; i++) {
   281→        const x = positions.getX(i)
   282→        const z = positions.getZ(i)
   283→        const h =
   284→          Math.sin(x * 0.3 + seed) * Math.cos(z * 0.3 + seed) * 0.5 +
   285→          Math.sin(x * 0.1 + z * 0.1) * 0.3
   286→        positions.setY(i, h * heightScale)
   287→      }
   288→    }
   289→    g.computeVertexNormals()
   290→
   291→    // Vertex colors from splatmap (multi-layer blending) — P7
   292→    if (conect.splatmap && conect.textureLayers && conect.splatmap.length > 0) {
   293→      const maxLayers = conect.maxLayers || 4
   294→      const layers = conect.textureLayers
   295→      const colors = new Float32Array(positions.count * 3)
   296→      const sm = conect.splatmap
   297→      const hexToRgb = (hex) => {
   298→        const r = parseInt(hex.slice(1, 3), 16) / 255
   299→        const g = parseInt(hex.slice(3, 5), 16) / 255
   300→        const b = parseInt(hex.slice(5, 7), 16) / 255
   301→        return [r, g, b]
   302→      }
   303→      const layerRgb = layers.map((l) => hexToRgb(l.color || '#5a7d3a'))
   304→      for (let i = 0; i < positions.count; i++) {
   305→        let r = 0, g = 0, b = 0
   306→        for (let l = 0; l < maxLayers && l < layers.length; l++) {
   307→          const w = sm[i * maxLayers + l] || 0
   308→          const c = layerRgb[l] || [0.35, 0.49, 0.23]
   309→          r += c[0] * w
   310→          g += c[1] * w
   311→          b += c[2] * w
   312→        }
   313→        colors[i * 3] = r
   314→        colors[i * 3 + 1] = g
   315→        colors[i * 3 + 2] = b
   316→      }
   317→      g.setAttribute('color', new THREE.BufferAttribute(colors, 3))
   318→    }
   319→
   320→    return g
   321→  }, [conect.width, conect.depth, conect.segments, conect.heightScale, conect.heightmapSeed, conect.heightmap, conect.splatmap, conect.textureLayers, conect.maxLayers])
   322→
   323→  // Disposal: libertar geometry quando o componente desmonta
   324→  useEffect(() => () => { geometry.dispose() }, [geometry])
   325→
   326→  const hasVertexColors = !!(conect.splatmap && conect.textureLayers && conect.splatmap.length > 0)
   327→
   328→  return (
   329→    <mesh
   330→      ref={setMeshRef}
   331→      geometry={geometry}
   332→      position={conect.position}
   333→      receiveShadow
   334→      castShadow
   335→    >
   336→      <meshStandardMaterial
   337→        color={hasVertexColors ? '#ffffff' : (conect.textureColor || '#5a7d3a')}
   338→        vertexColors={hasVertexColors}
   339→        roughness={0.9}
   340→        metalness={0}
   341→        side={THREE.DoubleSide}
   342→      />
   343→    </mesh>
   344→  )
   345→}

   347→// ===== WaterObject =====
   348→function WaterMesh({ conect, setMeshRef }) {
   349→  const meshRef = useRef()
   350→  const time = useRef(0)
   351→  const waterQuality = useStore((s) => s.renderSettings?.waterQuality || 'basic')
   352→
   353→  // Material Pro (Gerstner + caustics + IOR + Fresnel) quando quality = professional
   354→  const proMaterial = useMemo(() => {
   355→    if (waterQuality !== 'professional') return null
   356→    return createWaterProMaterial({
   357→      color: conect.color || '#2f81f7',
   358→      deepColor: conect.deepColor || '#0a3d5c',
   359→      opacity: conect.opacity ?? 0.85,
   360→      waveHeight: conect.waveHeight ?? 0.2,
   361→      waveSpeed: conect.waveSpeed ?? 0.5,
   362→      waterMode: conect.waterMode || 'lake',
   363→      flowDirection: conect.flowDirection || 0,
   364→      foamEnabled: conect.foamEnabled !== false,
   365→      foamThreshold: conect.foamThreshold ?? 0.7,
   366→      depthGradient: conect.depthGradient !== false,
   367→      skyColor: '#88aacc',
   368→    })
   369→  }, [waterQuality, conect.color, conect.deepColor, conect.opacity, conect.waveHeight,
   370→      conect.waveSpeed, conect.waterMode, conect.flowDirection, conect.foamEnabled,
   371→      conect.foamThreshold, conect.depthGradient])
   372→
   373→  const geometry = useMemo(() => {
   374→    const [w, h] = conect.size || [20, 20]
   375→    const segs = waterQuality === 'professional' ? 48 : 32
   376→    const g = new THREE.PlaneGeometry(w, h, segs, segs)
   377→    g.rotateX(-Math.PI / 2)
   378→    return g
   379→  }, [conect.size, waterQuality])
   380→
   381→  // Disposal: libertar geometry e proMaterial
   382→  useEffect(() => () => { geometry.dispose(); proMaterial?.dispose() }, [geometry, proMaterial])
   383→
   384→  // Animar ondas no useFrame (apenas para water básico; Pro usa shader)
   385→  useFrame((state, delta) => {
   386→    if (!meshRef.current) return
   387→    if (proMaterial) {
   388→      // Pro: actualizar uniforms do shader
   389→      if (proMaterial.uniforms) {
   390→        proMaterial.uniforms.uTime.value += delta
   391→        proMaterial.uniforms.uCameraPos.value.copy(state.camera.position)
   392→      }
   393→      return
   394→    }
   395→    // Básico: animar vértices
   396→    time.current += delta * (conect.waveSpeed || 0.5)
   397→    const pos = meshRef.current.geometry.attributes.position
   398→    const waveHeight = conect.waveHeight || 0.1
   399→    for (let i = 0; i < pos.count; i++) {
   400→      const x = pos.getX(i)
   401→      const z = pos.getZ(i)
   402→      const y = Math.sin(x * 0.5 + time.current) * waveHeight
   403→        + Math.cos(z * 0.5 + time.current * 0.7) * waveHeight * 0.5
   404→      pos.setY(i, y)
   405→    }
   406→    pos.needsUpdate = true
   407→    meshRef.current.geometry.computeVertexNormals()
   408→  })
   409→
   410→  return (
   411→    <mesh
   412→      ref={(node) => {
   413→        meshRef.current = node
   414→        if (typeof setMeshRef === 'function') setMeshRef(node)
   415→      }}
   416→      position={conect.position}
   417→      geometry={geometry}
   418→      material={proMaterial || undefined}
   419→    >
   420→      {!proMaterial && (
   421→        <meshStandardMaterial
   422→          color={conect.color || '#2f81f7'}
   423→          transparent
   424→          opacity={conect.opacity ?? 0.6}
   425→          roughness={0.1}
   426→          metalness={0.3}
   427→        />
   428→      )}
   429→    </mesh>
   430→  )
   431→}

   433→// ===== PathObject =====
   434→function PathMesh({ conect, setMeshRef }) {
   435→  if (!conect.points || conect.points.length === 0) return null
   436→  return (
   437→    <group ref={setMeshRef} position={conect.position}>
   438→      {conect.points.map((p, i) => (
   439→        <mesh key={i} position={p}>
   440→          <sphereGeometry args={[0.15, 12, 12]} />
   441→          <meshBasicMaterial color="#2f81f7" />
   442→        </mesh>
   443→      ))}
   444→      {/* Linhas entre pontos */}
   445→      {conect.points.length > 1 && (
   446→        <Line points={conect.points} loop={conect.loop} />
   447→      )}
   448→    </group>
   449→  )
   450→}
   451→
   452→function Line({ points, loop }) {
   453→  const geometry = useMemo(() => {
   454→    const pts = loop ? [...points, points[0]] : points
   455→    return new THREE.BufferGeometry().setFromPoints(pts.map((p) => new THREE.Vector3(...p)))
   456→  }, [points, loop])
   457→  return (
   458→    <line geometry={geometry}>
   459→      <lineBasicMaterial color="#2f81f7" transparent opacity={0.5} />
   460→    </line>
   461→  )
   462→}

   480→// ===== ParticleObject =====
   481→function ParticleMesh({ conect, setMeshRef }) {
   482→  const groupRef = useRef()
   483→  const timeRef = useRef(0)
   484→
   485→  // Usar BufferGeometry + Points para performance (1 draw call em vez de N)
   486→  const count = Math.min(conect.maxParticles || 100, 300)
   487→  const positionsRef = useRef(new Float32Array(count * 3))
   488→  const velocitiesRef = useRef(new Float32Array(count * 3))
   489→  const lifeRef = useRef(new Float32Array(count))
   490→
   491→  // Inicializar partículas
   492→  useMemo(() => {
   493→    const spread = conect.spread || 1
   494→    const speed = conect.particleSpeed || 0.5
   495→    for (let i = 0; i < count; i++) {
   496→      positionsRef.current[i * 3] = (Math.random() - 0.5) * spread
   497→      positionsRef.current[i * 3 + 1] = 0
   498→      positionsRef.current[i * 3 + 2] = (Math.random() - 0.5) * spread
   499→      velocitiesRef.current[i * 3] = (Math.random() - 0.5) * speed
   500→      velocitiesRef.current[i * 3 + 1] = Math.random() * speed + 0.2
   501→      velocitiesRef.current[i * 3 + 2] = (Math.random() - 0.5) * speed
   502→      lifeRef.current[i] = Math.random() * (conect.particleLife || 2)
   503→    }
   504→  }, [count, conect.spread, conect.particleSpeed, conect.particleLife])
   505→
   506→  const geometry = useMemo(() => {
   507→    const geo = new THREE.BufferGeometry()
   508→    geo.setAttribute('position', new THREE.BufferAttribute(positionsRef.current, 3))
   509→    return geo
   510→  }, [])
   511→
   512→  const material = useMemo(() => {
   513→    return new THREE.PointsMaterial({
   514→      color: conect.color || '#ffffff',
   515→      size: conect.particleSize || 0.1,
   516→      transparent: true,
   517→      opacity: 0.7,
   518→      sizeAttenuation: true,
   519→      depthWrite: false,
   520→    })
   521→  }, [conect.color, conect.particleSize])
   522→
   523→  // Animar partículas no useFrame
   524→  useFrame((_, delta) => {
   525→    timeRef.current += delta
   526→    const positions = positionsRef.current
   527→    const velocities = velocitiesRef.current
   528→    const life = lifeRef.current
   529→    const gravity = conect.gravity ?? -0.5
   530→    const maxLife = conect.particleLife || 2
   531→    const spread = conect.spread || 1
   532→
   533→    for (let i = 0; i < count; i++) {
   534→      // Atualizar vida
   535→      life[i] += delta
   536→      if (life[i] >= maxLife) {
   537→        // Reciclar partícula
   538→        positions[i * 3] = (Math.random() - 0.5) * spread
   539→        positions[i * 3 + 1] = 0
   540→        positions[i * 3 + 2] = (Math.random() - 0.5) * spread
   541→        velocities[i * 3] = (Math.random() - 0.5) * (conect.particleSpeed || 0.5)
   542→        velocities[i * 3 + 1] = Math.random() * (conect.particleSpeed || 0.5) + 0.2
   543→        velocities[i * 3 + 2] = (Math.random() - 0.5) * (conect.particleSpeed || 0.5)
   544→        life[i] = 0
   545→      } else {
   546→        // Mover partícula
   547→        positions[i * 3] += velocities[i * 3] * delta
   548→        positions[i * 3 + 1] += velocities[i * 3 + 1] * delta
   549→        positions[i * 3 + 2] += velocities[i * 3 + 2] * delta
   550→        // Aplicar gravidade
   551→        velocities[i * 3 + 1] += gravity * delta
   552→      }
   553→    }
   554→    geometry.attributes.position.needsUpdate = true
   555→  })
   556→
   557→  // Cleanup
   558→  useEffect(() => {
   559→    return () => {
   560→      geometry.dispose()
   561→      material.dispose()
   562→    }
   563→  }, [geometry, material])
   564→
   565→  return (
   566→    <points
   567→      ref={(node) => {
   568→        groupRef.current = node
   569→        if (typeof setMeshRef === 'function') setMeshRef(node)
   570→      }}
   571→      position={conect.position}
   572→      geometry={geometry}
   573→      material={material}
   574→    />
   575→  )
   576→}

   626→// ===== TrailObject (rasto visual) =====
   627→// Renderiza uma linha que segue a posição do objeto pai.
   628→// Em tempo de execução, o GameRunner atualiza as posições do trail.
   629→function TrailMesh({ conect, setMeshRef }) {
   630→  const groupRef = useRef()
   631→  const lineRef = useRef()
   632→  const maxPoints = conect.length || 30
   633→  const positionsRef = useRef(new Float32Array(maxPoints * 3))
   634→  const currentIndexRef = useRef(0)
   635→  const frameCountRef = useRef(0)
   636→
   637→  // Inicializar todas as posições na posição inicial do trail
   638→  useMemo(() => {
   639→    const px = conect.position?.[0] || 0
   640→    const py = conect.position?.[1] || 0
   641→    const pz = conect.position?.[2] || 0
   642→    for (let i = 0; i < maxPoints; i++) {
   643→      positionsRef.current[i * 3] = px
   644→      positionsRef.current[i * 3 + 1] = py
   645→      positionsRef.current[i * 3 + 2] = pz
   646→    }
   647→  }, [maxPoints, conect.position])
   648→
   649→  const geometry = useMemo(() => {
   650→    const geo = new THREE.BufferGeometry()
   651→    geo.setAttribute('position', new THREE.BufferAttribute(positionsRef.current, 3))
   652→    return geo
   653→  }, [])
   654→
   655→  // Seguir o followTarget e atualizar o trail
   656→  useFrame(() => {
   657→    if (!conect.followTarget) return
   658→    // Procurar o mesh do followTarget nos conectMeshRefs
   659→    const targetMesh = window._flirConectMeshRefs?.get(conect.followTarget)
   660→      || window._flirMeshRefs?.get(conect.followTarget)
   661→    if (!targetMesh) return
   662→
   663→    // Atualizar a cada N frames (não todos os frames — performance)
   664→    const updateInterval = Math.max(1, Math.floor(60 / (conect.updateRate || 30)))
   665→    frameCountRef.current++
   666→    if (frameCountRef.current < updateInterval) return
   667→    frameCountRef.current = 0
   668→
   669→    // Shift: mover todos os pontos uma posição para trás
   670→    const positions = positionsRef.current
   671→    for (let i = maxPoints - 1; i > 0; i--) {
   672→      positions[i * 3] = positions[(i - 1) * 3]
   673→      positions[i * 3 + 1] = positions[(i - 1) * 3 + 1]
   674→      positions[i * 3 + 2] = positions[(i - 1) * 3 + 2]
   675→    }
   676→    // Novo ponto na posição do target
   677→    positions[0] = targetMesh.position.x
   678→    positions[1] = targetMesh.position.y
   679→    positions[2] = targetMesh.position.z
   680→
   681→    geometry.attributes.position.needsUpdate = true
   682→    geometry.setDrawRange(0, maxPoints)
   683→  })
   684→
   685→  // Cleanup
   686→  useEffect(() => {
   687→    return () => { geometry.dispose() }
   688→  }, [geometry])
   689→
   690→  return (
   691→    <group ref={setMeshRef} position={[0, 0, 0]}>
   692→      <line ref={lineRef} geometry={geometry}>
   693→        <lineBasicMaterial
   694→          color={conect.color || '#2f81f7'}
   695→          transparent
   696→          opacity={conect.fade !== false ? 0.6 : 1}
   697→          linewidth={conect.width || 2}
   698→        />
   699→      </line>
   700→    </group>
   701→  )
   702→}

   721→// ===== SkyObject — esfera de céu com shader procedural =====
   722→function SkyMesh({ conect, setMeshRef }) {
   723→  const { scene } = useThree()
   724→  const meshRef = useRef()
   725→
   726→  // Ler propriedades com os nomes correctos da taxonomy
   727→  const skyType = conect.skyType || 'gradient'
   728→  const topColor = conect.topColor || conect.gradientTop || '#1a4d8f'
   729→  const bottomColor = conect.bottomColor || conect.gradientBottom || '#aac4e8'
   730→  const solidColor = conect.solidColor || conect.color || '#87ceeb'
   731→
   732→  // Material Pro para procedural (Rayleigh + Mie scattering)
   733→  const proMaterial = useMemo(() => {
   734→    if (skyType !== 'procedural') return null
   735→    // Calcular direção do sol: priorizar sunPosition se definido (mais intuitivo)
   736→    // Senão, usar sunElevation (graus 0-90) convertido para hourOfDay
   737→    let sunDir
   738→    if (conect.sunPosition && Array.isArray(conect.sunPosition) && conect.sunPosition.length === 3) {
   739→      sunDir = new THREE.Vector3(conect.sunPosition[0], conect.sunPosition[1], conect.sunPosition[2]).normalize()
   740→    } else {
   741→      // sunElevation em graus (0=horizonte, 90=zénite) → converter para hourOfDay
   742→      // 0° = 6h/18h, 90° = 12h
   743→      const elevation = conect.sunElevation ?? 45
   744→      const hourOfDay = 12 - (90 - elevation) / 90 * 6  // mapeia 0°→6h, 90°→12h
   745→      sunDir = calculateSunDirection(hourOfDay, 172, 0)
   746→    }
   747→    return createSkyProMaterial({
   748→      sunDirection: [sunDir.x, sunDir.y, sunDir.z],
   749→      sunIntensity: 15,
   750→      rayleigh: conect.rayleigh ?? 2.5,
   751→      mie: 0.5,
   752→      turbidity: conect.turbidity ?? 10,
   753→      starsEnabled: conect.starsEnabled || false,
   754→    })
   755→  }, [skyType, conect.sunPosition, conect.sunElevation, conect.rayleigh, conect.turbidity, conect.starsEnabled])
   756→
   757→  // Disposal: libertar proMaterial quando desmonta
   758→  useEffect(() => () => { proMaterial?.dispose() }, [proMaterial])
   759→
   760→  // Animar uTime do sky shader
   761→  useFrame((_, delta) => {
   762→    if (proMaterial && proMaterial.uniforms?.uTime) {
   763→      proMaterial.uniforms.uTime.value += delta
   764→    }
   765→  })
   766→
   767→  // Aplicar background à cena
   768→  useEffect(() => {
   769→    if (!scene) return
   770→    if (skyType === 'procedural' && proMaterial) {
   771→      // Procedural usa esfera com shader — manter fallback azul céu caso a SkyMesh falhe
   772→      // (anteriormente era null, o que causava ecrã PRETO se a esfera não renderizasse)
   773→      scene.background = new THREE.Color(bottomColor || '#87ceeb')
   774→    } else if (skyType === 'gradient') {
   775→      // Gradiente via canvas
   776→      const canvas = document.createElement('canvas')
   777→      canvas.width = 2; canvas.height = 256
   778→      const ctx = canvas.getContext('2d')
   779→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
   780→      grad.addColorStop(0, topColor)
   781→      grad.addColorStop(1, bottomColor)
   782→      ctx.fillStyle = grad
   783→      ctx.fillRect(0, 0, 2, 256)
   784→      const tex = new THREE.CanvasTexture(canvas)
   785→      tex.colorSpace = THREE.SRGBColorSpace
   786→      scene.background = tex
   787→      return () => { tex.dispose() }
   788→    } else if (skyType === 'hdri' && conect.hdriUrl) {
   789→      import('three/examples/jsm/loaders/RGBELoader.js').then(({ RGBELoader }) => {
   790→        const loader = new RGBELoader()
   791→        loader.load(conect.hdriUrl, (texture) => {
   792→          texture.mapping = THREE.EquirectangularReflectionMapping
   793→          scene.background = texture
   794→          scene.environment = texture
   795→        })
   796→      }).catch(() => {
   797→        scene.background = new THREE.Color(solidColor)
   798→      })
   799→    } else {
   800→      // solid
   801→      scene.background = new THREE.Color(solidColor)
   802→    }
   803→  }, [scene, skyType, topColor, bottomColor, solidColor, conect.hdriUrl, proMaterial])
   804→
   805→  // Esfera grande com material de céu
   806→  if (skyType === 'procedural' && proMaterial) {
   807→    // Esfera com shader procedural (Rayleigh + Mie)
   808→    return (
   809→      <mesh
   810→        ref={(node) => {
   811→          meshRef.current = node
   812→          if (typeof setMeshRef === 'function') setMeshRef(node)
   813→        }}
   814→        scale={[100, 100, 100]}
   815→        material={proMaterial}
   816→      >
   817→        <sphereGeometry args={[1, 32, 16]} />
   818→      </mesh>
   819→    )
   820→  }
   821→  if (skyType === 'solid' || skyType === 'hdri') {
   822→    return (
   823→      <mesh ref={setMeshRef} scale={[100, 100, 100]}>
   824→        <sphereGeometry args={[1, 32, 16]} />
   825→        <meshBasicMaterial
   826→          color={solidColor}
   827→          side={THREE.BackSide}
   828→          fog={false}
   829→        />
   830→      </mesh>
   831→    )
   832→  }
   833→  // gradient: background canvas, mesh invisível
   834→  return (
   835→    <mesh ref={setMeshRef} visible={false} scale={[0.001, 0.001, 0.001]}>
   836→      <sphereGeometry args={[1, 8, 8]} />
   837→      <meshBasicMaterial />
   838→    </mesh>
   839→  )
   840→}

  1329→      flowSpeed: conect.flowSpeed,

  1334→      fresnelPower: conect.fresnelPower,
  1335→      ior: conect.ior,
  1336→      sunDirection: conect.sunDirection,
  1337→    })
  1338→  }, [
  1339→    conect.color, conect.deepColor, conect.clarity, conect.refraction,
  1340→    conect.reflection, conect.flowSpeed, conect.waveHeight, conect.waveFrequency,
  1341→    conect.foamThreshold, conect.foamColor, conect.fresnelPower, conect.ior, conect.sunDirection,

  1346→    const g = new THREE.PlaneGeometry(conect.size?.[0] || 50, conect.size?.[1] || 50, segs, segs)
  1347→    g.rotateX(-Math.PI / 2)
  1348→    return g
  1349→  }, [conect.size, conect.segments])
  1350→
  1351→  useFrame((state) => {
  1352→    if (material.userData.update) {

  1354→    }
  1355→  })
  1356→
  1357→  useEffect(() => {
  1358→    return () => {
  1359→      material.dispose()
  1360→      geometry.dispose()
  1361→    }
  1362→  }, [material, geometry])
  1363→
  1364→  return (
  1365→    <mesh
  1366→      ref={(node) => {
  1367→        meshRef.current = node
  1368→        setMeshRef?.(node)
  1369→      }}
  1370→      geometry={geometry}
  1371→      material={material}

  1374→      scale={conect.scale}

  1376→    />
  1377→  )
  1378→}
