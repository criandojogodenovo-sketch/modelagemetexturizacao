    34→    34→// Helper: amostrar altura do heightmap numa posição (x,z) do mundo
    35→    35→function sampleTerrainHeight(hm, seg, worldX, worldZ, terrainWidth, terrainDepth, heightScale) {
    36→    36→  const nx = (worldX / terrainWidth + 0.5) * seg
    37→    37→  const nz = (worldZ / terrainDepth + 0.5) * seg
    38→    38→  const ix = Math.max(0, Math.min(seg, Math.round(nx)))
    39→    39→  const iz = Math.max(0, Math.min(seg, Math.round(nz)))
    40→    40→  const idx = iz * (seg + 1) + ix
    41→    41→  const h = hm[idx] || 0
    42→    42→  return h * heightScale
    43→    43→}

    75→    75→function autoSplat(hm, seg, maxLayers = 4) {
    76→    76→  const sm = new Float32Array((seg + 1) * (seg + 1) * maxLayers)
    77→    77→  for (let i = 0; i < hm.length; i++) {
    78→    78→    const h = (hm[i] + 1) * 0.5
    79→    79→    if (h < 0.25) { sm[i*4]=1; sm[i*4+1]=0; sm[i*4+2]=0; sm[i*4+3]=0 }
    80→    80→    else if (h < 0.5) { sm[i*4]=0.4; sm[i*4+1]=0.6; sm[i*4+2]=0; sm[i*4+3]=0 }
    81→    81→    else if (h < 0.75) { sm[i*4]=0; sm[i*4+1]=0.5; sm[i*4+2]=0.5; sm[i*4+3]=0 }
    82→    82→    else { sm[i*4]=0; sm[i*4+1]=0; sm[i*4+2]=0.3; sm[i*4+3]=0.7 }
    83→    83→  }
    84→    84→  return sm
    85→    85→}
