     1→     1→/**
     2→     2→ * flircode.js — interpretador da linguagem FlirCode.
     3→     3→ *
     4→     4→ * Linguagem de scripting por texto próprio (não é GDScript nem JavaScript).
     5→     5→ * Compila para o mesmo gameContext que o executor.js usa, garantindo
     6→     6→ * compatibilidade total com Conects, eventos e físicas existentes.
     7→     7→ *
     8→     8→ * Sintaxe:
     9→     9→ *   begincode ... endcode          — blocos de código
    10→    10→ *   fun nome(params) begincode ... endcode  — declarar função
    11→    11→ *   var nome = valor               — declarar variável
    12→    12→ *   $$ comentário                  — linha de comentário
    13→    13→ *   if (cond), else if (cond), else — condicionais
    14→    14→ *   repeat in number(n, i) begincode ... endcode — ciclo por quantidade
    15→    15→ *   repeat +n until m begincode ... endcode — ciclo por incremento
    16→    16→ *   repeat -n until m begincode ... endcode — ciclo por decremento
    17→    17→ *   switch (var) begincode case v begincode ... endcode default begincode ... endcode endcode
    18→    18→ *
    19→    19→ * Eventos (funções especiais chamadas automaticamente):
    20→    20→ *   fun onStart()        — BeginPlay
    21→    21→ *   fun onTick()         — Tick (a cada frame)
    22→    22→ *   fun onCollide(outro) — OnCollision
    23→    23→ *   fun onTouch()        — OnTouch
    24→    24→ *   fun onSeePlayer()    — OnSeePlayer (NPC)
    25→    25→ *   fun onLoseSight()    — OnLoseSight (NPC)
    26→    26→ *   fun onTimer()        — OnTimer
    27→    27→ *
    28→    28→ * Funções embutidas:
    29→    29→ *   playAnim("name"), playSound("name"), move(x,y,z), rotate(x,y,z), scale(x,y,z)
    30→    30→ *   destroy(object), createObject("name",x,y,z), changeScene("name"), wait(seconds)
    31→    31→ *   setVar("name",value), getVar("name"), showUI("name"), hideUI("name")
    32→    32→ *   print("msg"), warn("msg"), error("msg")
    33→    33→ *   collidingWith("type"), distanceTo(object), isTouching()
    34→    34→ */
    35→    35→import { debugLog } from '../debug/debugStore'

    37→    37→// ===== Parser =====
    38→    38→// Converte texto FlirCode num AST (Abstract Syntax Tree) que o runtime executa.
    39→    39→// O parser é um tokenizer + recursive descent parser simples.
    40→    40→
    41→    41→export function parseFlirCode(source) {
    42→    42→  const errors = []
    43→    43→  const lines = source.split('\n')
    44→    44→  const functions = {}
    45→    45→  const classes = {} // Sistema 2: suporte a classes
    46→    46→
    47→    47→  // Pré-processar: remover comentários e linhas vazias
    48→    48→  const cleanLines = []
    49→    49→  for (let i = 0; i < lines.length; i++) {
    50→    50→    const trimmed = lines[i].trim()
    51→    51→    if (trimmed.startsWith('$$') || trimmed === '') continue
    52→    52→    cleanLines.push({ text: trimmed, line: i + 1 })
    53→    53→  }
    54→    54→
    55→    55→  // Parser de funções e classes (top-level)
    56→    56→  let idx = 0
    57→    57→  while (idx < cleanLines.length) {
    58→    58→    const line = cleanLines[idx]
    59→    59→
    60→    60→    // Sistema 2: "class Nome begincode" ou "class Nome extends Base begincode"
    61→    61→    const classMatch = line.text.match(/^class\s+(\w+)(?:\s+extends\s+(\w+))?\s*begincode$/)
    62→    62→    if (classMatch) {
    63→    63→      const className = classMatch[1]
    64→    64→      const extendsName = classMatch[2] || null
    65→    65→      const body = parseBlock(cleanLines, idx + 1, errors)
    66→    66→      if (body.error) {
    67→    67→        errors.push({ line: line.line, message: `Classe "${className}": ${body.error}` })
    68→    68→        idx = body.nextIdx
    69→    69→        continue
    70→    70→      }
    71→    71→      // Extrair variáveis e funções da classe
    72→    72→      const classVars = {}
    73→    73→      const classFuncs = {}
    74→    74→      for (const stmt of body.statements) {
    75→    75→        if (stmt.type === 'var' || stmt.type === 'assign') {
    76→    76→          classVars[stmt.name] = stmt.value
    77→    77→        } else if (stmt.type === 'function') {
    78→    78→          classFuncs[stmt.name] = stmt
    79→    79→        }
    80→    80→      }
    81→    81→      classes[className] = {
    82→    82→        name: className,
    83→    83→        extends: extendsName,
    84→    84→        vars: classVars,
    85→    85→        functions: classFuncs,
    86→    86→        rawStatements: body.statements,
    87→    87→        line: line.line,
    88→    88→      }
    89→    89→      idx = body.nextIdx
    90→    90→      continue
    91→    91→    }
    92→    92→
    93→    93→    // Procurar "fun nome(params) begincode"
    94→    94→    const funMatch = line.text.match(/^fun\s+(\w+)\s*\(([^)]*)\)\s*begincode$/)
    95→    95→    if (funMatch) {
    96→    96→      const funName = funMatch[1]
    97→    97→      const params = funMatch[2].split(',').map((p) => p.trim()).filter((p) => p)
    98→    98→      const body = parseBlock(cleanLines, idx + 1, errors)
    99→    99→      if (body.error) {
   100→   100→        errors.push({ line: line.line, message: `Função "${funName}": ${body.error}` })
   101→   101→        idx = body.nextIdx
   102→   102→        continue
   103→   103→      }
   104→   104→      functions[funName] = { name: funName, params, body: body.statements, line: line.line }
   105→   105→      idx = body.nextIdx
   106→   106→    } else {
   107→   107→      // Pode ser uma variável global ou instrução solta
   108→   108→      // Por agora, ignorar (só suportamos funções no top-level)
   109→   109→      idx++
   110→   110→    }
   111→   111→  }
   112→   112→
   113→   113→  return { functions, classes, errors }
   114→   114→}
   115→   115→
   116→   116→// Parser de bloco begincode...endcode
   117→   117→function parseBlock(lines, startIdx, errors) {
   118→   118→  const statements = []
   119→   119→  let idx = startIdx
   120→   120→  let depth = 1 // já entramos num begincode
   121→   121→
   122→   122→  while (idx < lines.length && depth > 0) {
   123→   123→    const line = lines[idx]
   124→   124→    const text = line.text
   125→   125→
   126→   126→    if (text === 'endcode') {
   127→   127→      depth--
   128→   128→      if (depth === 0) {
   129→   129→        return { statements, nextIdx: idx + 1 }
   130→   130→      }
   131→   131→      statements.push({ type: 'endcode', line: line.line })
   132→   132→      idx++
   133→   133→      continue
   134→   134→    }
   135→   135→
   136→   136→    // Sistema 2: reconhecer "fun nome(params) begincode" dentro de classes
   137→   137→    const funMatch = text.match(/^fun\s+(\w+)\s*\(([^)]*)\)\s*begincode$/)
   138→   138→    if (funMatch) {
   139→   139→      const funName = funMatch[1]
   140→   140→      const params = funMatch[2].split(',').map((p) => p.trim()).filter((p) => p)
   141→   141→      const body = parseBlock(lines, idx + 1, errors)
   142→   142→      if (body.error) {
   143→   143→        errors.push({ line: line.line, message: `Função "${funName}": ${body.error}` })
   144→   144→        idx = body.nextIdx
   145→   145→        continue
   146→   146→      }
   147→   147→      statements.push({
   148→   148→        type: 'function',
   149→   149→        name: funName,
   150→   150→        params,
   151→   151→        body: body.statements,
   152→   152→        line: line.line,
   153→   153→        nextIdx: body.nextIdx,
   154→   154→      })
   155→   155→      idx = body.nextIdx
   156→   156→      continue
   157→   157→    }
   158→   158→
   159→   159→    if (text.endsWith('begincode')) {
   160→   160→      // Sub-bloco (if, else, repeat, etc.)
   161→   161→      const stmt = parseStatement(lines, idx, errors)
   162→   162→      if (stmt) {
   163→   163→        statements.push(stmt)
   164→   164→        idx = stmt.nextIdx
   165→   165→      } else {
   166→   166→        idx++
   167→   167→      }
   168→   168→    } else {
   169→   169→      // Instrução simples
   170→   170→      const stmt = parseSimpleStatement(text, line.line, errors)
   171→   171→      if (stmt) statements.push(stmt)
   172→   172→      idx++
   173→   173→    }
   174→   174→  }
   175→   175→
   176→   176→  if (depth > 0) {
   177→   177→    return { error: 'begincode sem endcode correspondente', nextIdx: idx }
   178→   178→  }
   179→   179→  return { statements, nextIdx: idx }
   180→   180→}
   181→   181→
   182→   182→// Parser de uma instrução individual
   183→   183→function parseSimpleStatement(text, lineNum, errors) {
   184→   184→  // var nome = valor
   185→   185→  let m = text.match(/^var\s+(\w+)\s*=\s*(.+)$/)
   186→   186→  if (m) return { type: 'var', name: m[1], value: parseValue(m[2]), line: lineNum }
   187→   187→
   188→   188→  // if (cond) begincode
   189→   189→  m = text.match(/^if\s*\((.+)\)\s*begincode$/)
   190→   190→  if (m) return { type: 'if', condition: m[1], line: lineNum }
   191→   191→
   192→   192→  // else if (cond) begincode
   193→   193→  m = text.match(/^else\s+if\s*\((.+)\)\s*begincode$/)
   194→   194→  if (m) return { type: 'elseif', condition: m[1], line: lineNum }
   195→   195→
   196→   196→  // else begincode
   197→   197→  m = text.match(/^else\s*begincode$/)
   198→   198→  if (m) return { type: 'else', line: lineNum }
   199→   199→
   200→   200→  // repeat in number(quantidade, variavel) begincode
   201→   201→  m = text.match(/^repeat\s+in\s+number\s*\((\d+),\s*(\w+)\)\s*begincode$/)
   202→   202→  if (m) return { type: 'repeat_n', count: parseInt(m[1]), varName: m[2], line: lineNum }
   203→   203→
   204→   204→  // repeat +numero until numero begincode
   205→   205→  m = text.match(/^repeat\s+\+(\d+)\s+until\s+(\d+)\s*begincode$/)
   206→   206→  if (m) return { type: 'repeat_inc', step: parseInt(m[1]), until: parseInt(m[2]), line: lineNum }
   207→   207→
   208→   208→  // repeat -numero until numero begincode
   209→   209→  m = text.match(/^repeat\s+-(\d+)\s+until\s+(\d+)\s*begincode$/)
   210→   210→  if (m) return { type: 'repeat_dec', step: parseInt(m[1]), until: parseInt(m[2]), line: lineNum }
   211→   211→
   212→   212→  // switch (var) begincode
   213→   213→  m = text.match(/^switch\s*\((.+)\)\s*begincode$/)
   214→   214→  if (m) return { type: 'switch', varName: m[1], line: lineNum }
   215→   215→
   216→   216→  // case valor begincode
   217→   217→  m = text.match(/^case\s+(.+)\s*begincode$/)
   218→   218→  if (m) return { type: 'case', value: m[1].trim(), line: lineNum }
   219→   219→
   220→   220→  // default begincode
   221→   221→  m = text.match(/^default\s*begincode$/)
   222→   222→  if (m) return { type: 'default', line: lineNum }
   223→   223→
   224→   224→  // Chamada de função embutida: nome(args)
   225→   225→  m = text.match(/^(\w+)\s*\(([^)]*)\)$/)
   226→   226→  if (m) {
   227→   227→    const funcName = m[1]
   228→   228→    const args = m[2].split(',').map((a) => a.trim()).filter((a) => a)
   229→   229→    return { type: 'call', funcName, args: args.map(parseValue), line: lineNum }
   230→   230→  }
   231→   231→
   232→   232→  // Atribuição: nome = valor
   233→   233→  m = text.match(/^(\w+)\s*=\s*(.+)$/)
   234→   234→  if (m) return { type: 'assign', name: m[1], value: parseValue(m[2]), line: lineNum }
   235→   235→
   236→   236→  // Se não reconhecer, ignorar (não travar)
   237→   237→  return { type: 'unknown', text, line: lineNum }
   238→   238→}
   239→   239→
   240→   240→// Parser de statement (para sub-blocos)
   241→   241→function parseStatement(lines, idx, errors) {
   242→   242→  const stmt = parseSimpleStatement(lines[idx].text, lines[idx].line, errors)
   243→   243→  if (!stmt) return null
   244→   244→  // Se o statement abre um bloco (if, repeat, etc.), parsear o corpo
   245→   245→  if (['if', 'elseif', 'else', 'repeat_n', 'repeat_inc', 'repeat_dec', 'switch', 'case', 'default'].includes(stmt.type)) {
   246→   246→    const body = parseBlock(lines, idx + 1, errors)
   247→   247→    if (body.error) {
   248→   248→      errors.push({ line: lines[idx].line, message: body.error })
   249→   249→      return { ...stmt, body: [], nextIdx: body.nextIdx }
   250→   250→    }
   251→   251→    return { ...stmt, body: body.statements, nextIdx: body.nextIdx }
   252→   252→  }
   253→   253→  return { ...stmt, nextIdx: idx + 1 }
   254→   254→}

   256→   256→// Parser de valores (números, strings, variáveis, expressões simples)
   257→   257→function parseValue(text) {
   258→   258→  text = text.trim()
   259→   259→  // String
   260→   260→  if (text.startsWith('"') && text.endsWith('"')) {
   261→   261→    return { type: 'string', value: text.slice(1, -1) }
   262→   262→  }
   263→   263→  // Expressão com concatenação (+) — suporta "string" + var + "string" + ...
   264→   264→  // Não divide dentro de strings (respeita aspas)
   265→   265→  if (text.includes('+') && !/^-?\d/.test(text)) {
   266→   266→    const parts = splitPlus(text)
   267→   267→    if (parts.length > 1) {
   268→   268→      return { type: 'concat', parts: parts.map(parseValue) }
   269→   269→    }
   270→   270→  }
   271→   271→  // Número
   272→   272→  const num = parseFloat(text)
   273→   273→  if (!isNaN(num)) return { type: 'number', value: num }
   274→   274→  // Booleano
   275→   275→  if (text === 'true') return { type: 'boolean', value: true }
   276→   276→  if (text === 'false') return { type: 'boolean', value: false }
   277→   277→  // Sistema 2: 'this' refere-se ao próprio objeto (instanceId)
   278→   278→  if (text === 'this') return { type: 'this' }
   279→   279→  // Chamada de função embutida como valor: identifier(args)
   280→   280→  // Ex: getVar("teste"), distanceTo("Cubo"), collidingWith("tipo"), isTouching(), getUIValue("nome")
   281→   281→  const callMatch = text.match(/^(\w+)\s*\(([^)]*)\)$/)
   282→   282→  if (callMatch) {
   283→   283→    const funcName = callMatch[1]
   284→   284→    const args = callMatch[2].split(',').map((a) => a.trim()).filter((a) => a)
   285→   285→    return { type: 'call_value', funcName, args: args.map(parseValue) }
   286→   286→  }
   287→   287→  // Variável
   288→   288→  return { type: 'var', name: text }
   289→   289→}
   290→   290→
   291→   291→// Divide uma expressão por + respeitando aspas (não divide dentro de strings)
   292→   292→function splitPlus(text) {
   293→   293→  const parts = []
   294→   294→  let current = ''
   295→   295→  let inString = false
   296→   296→  for (let i = 0; i < text.length; i++) {
   297→   297→    const c = text[i]
   298→   298→    if (c === '"') inString = !inString
   299→   299→    if (c === '+' && !inString) {
   300→   300→      parts.push(current.trim())
   301→   301→      current = ''
   302→   302→    } else {
   303→   303→      current += c
   304→   304→    }
   305→   305→  }
   306→   306→  parts.push(current.trim())
   307→   307→  return parts.filter((p) => p)
   308→   308→}

   310→   310→// ===== Runtime =====
   311→   311→// Executa o AST compilado, chamando as funções embutidas do gameContext.
   312→   312→
   313→   313→export function createFlirCodeRuntime(source, gameContext) {
   314→   314→  const { functions, classes, errors } = parseFlirCode(source)
   315→   315→
   316→   316→  // Se há erros de sintaxe, reportar e não executar
   317→   317→  for (const err of errors) {
   318→   318→    debugLog(`Erro de sintaxe FlirCode (linha ${err.line}): ${err.message}`, 'error', 'FlirCode')
   319→   319→  }
   320→   320→
   321→   321→  // Variáveis locais do script
   322→   322→  const localVars = {}
   323→   323→
   324→   324→  // Sistema 2: resolver hierarquia de classes e fundir funções/variáveis
   325→   325→  // Se gameContext.className estiver definido, usar essa classe
   326→   326→  const className = gameContext.className || null
   327→   327→  if (className && classes[className]) {
   328→   328→    // Construir cadeia de herança: [Base, ..., Sub]
   329→   329→    const chain = []
   330→   330→    let curr = classes[className]
   331→   331→    while (curr) {
   332→   332→      chain.unshift(curr) // adicionar no início para Base ficar primeiro
   333→   333→      curr = curr.extends ? classes[curr.extends] : null
   334→   334→    }
   335→   335→    // Inicializar variáveis da classe (da base para a sub)
   336→   336→    for (const cls of chain) {
   337→   337→      for (const [varName, varValue] of Object.entries(cls.vars || {})) {
   338→   338→        localVars[varName] = varValue ? (varValue.value !== undefined ? varValue.value : varValue) : 0
   339→   339→      }
   340→   340→    }
   341→   341→    // Fundir funções (sub override base)
   342→   342→    for (const cls of chain) {
   343→   343→      for (const [fnName, fnDef] of Object.entries(cls.functions || {})) {
   344→   344→        // Converter function statement para o formato esperado pelo runtime
   345→   345→        functions[fnName] = {
   346→   346→          name: fnName,
   347→   347→          params: fnDef.params || [],
   348→   348→          body: fnDef.body || [],
   349→   349→          line: fnDef.line,
   350→   350→        }
   351→   351→      }
   352→   352→    }
   353→   353→    debugLog(`Classe "${className}" aplicada: ${Object.keys(localVars).length} vars, ${Object.keys(functions).length} funções`, 'log', 'FlirCode')
   354→   354→  }
   355→   355→
   356→   356→  // Mapear nomes de eventos PT → nome interno
   357→   357→  const eventMap = {
   358→   358→    onStart: 'beginPlay',
   359→   359→    onTick: 'tick',
   360→   360→    onCollide: 'onCollision',
   361→   361→    onTouch: 'onTouch',
   362→   362→    onSeePlayer: 'onSeePlayer',
   363→   363→    onLoseSight: 'onLoseSight',
   364→   364→    onTimer: 'onTimer',
   365→   365→    onEnterZone: 'onEnterZone',
   366→   366→    onExitZone: 'onExitZone',
   367→   367→    onClick: 'onClick',
   368→   368→    onChange: 'onChange',
   369→   369→    onSubmit: 'onSubmit',
   370→   370→    // Fase 5: Eventos de multiplayer
   371→   371→    onPlayerJoin: 'onPlayerJoin',
   372→   372→    onPlayerLeave: 'onPlayerLeave',
   373→   373→    onMessage: 'onMessage',
   374→   374→    // Sistema 3: Evento de sinais
   375→   375→    onSignal: 'onSignal',
   376→   376→    // Sistema 2: Evento de dano
   377→   377→    onDamage: 'onDamage',
   378→   378→    // Sistema 3: Evento de apanhar item
   379→   379→    onPickup: 'onPickup',
   380→   380→    // Sistema: Game State
   381→   381→    onGameStateChange: 'onGameStateChange',
   382→   382→  }
   383→   383→
   384→   384→  // Avaliar um valor (número, string, variável, etc.)
   385→   385→  function evalValue(val) {
   386→   386→    if (!val) return null
   387→   387→    switch (val.type) {
   388→   388→      case 'number': return val.value
   389→   389→      case 'string': return val.value
   390→   390→      case 'boolean': return val.value
   391→   391→      case 'this': return gameContext._instanceId // Sistema 2: this = próprio objeto
   392→   392→      case 'concat': return val.parts.map((p) => {
   393→   393→        const v = evalValue(p)
   394→   394→        return v === null || v === undefined ? '' : String(v)
   395→   395→      }).join('')
   396→   396→      case 'call_value': {
   397→   397→        // Avaliar a função embutida e retornar o resultado
   398→   398→        return execBuiltin(val.funcName, val.args, {})
   399→   399→      }
   400→   400→      case 'var': {
   401→   401→        // Primeiro procura em localVars, depois em globalVars (via getVar)
   402→   402→        if (val.name in localVars) return localVars[val.name]
   403→   403→        const gv = gameContext.getVar?.(val.name)
   404→   404→        return gv ?? 0
   405→   405→      }
   406→   406→      default: return null
   407→   407→    }
   408→   408→  }

   410→   410→  // Avaliar condição (string como "x > 5", "a == b", etc.)
   411→   411→  function evalCondition(cond) {
   412→   412→    // Simplificado: suporta > < == != >= <=
   413→   413→    const m = cond.match(/^(.+?)\s*(>=|<=|==|!=|>|<)\s*(.+)$/)
   414→   414→    if (!m) return !!evalValue(parseValue(cond))
   415→   415→    const left = evalValue(parseValue(m[1].trim()))
   416→   416→    const op = m[2]
   417→   417→    const right = evalValue(parseValue(m[3].trim()))
   418→   418→    switch (op) {
   419→   419→      case '>': return left > right
   420→   420→      case '<': return left < right
   421→   421→      case '>=': return left >= right
   422→   422→      case '<=': return left <= right
   423→   423→      case '==': return left == right
   424→   424→      case '!=': return left != right
   425→   425→    }
   426→   426→    return false
   427→   427→  }
   428→   428→
   429→   429→  // Executar uma lista de statements
   430→   430→  function execStatements(statements, params = {}) {
   431→   431→    for (const stmt of statements) {
   432→   432→      try {
   433→   433→        execStatement(stmt, params)
   434→   434→      } catch (err) {
   435→   435→        debugLog(`Erro ao executar (linha ${stmt.line}): ${err.message}`, 'error', 'FlirCode')
   436→   436→        // Não travar — continuar
   437→   437→      }
   438→   438→    }
   439→   439→  }

   441→   441→  // Executar um statement
   442→   442→  function execStatement(stmt, params) {
   443→   443→    switch (stmt.type) {
   444→   444→      case 'var':
   445→   445→        localVars[stmt.name] = evalValue(stmt.value)
   446→   446→        break
   447→   447→      case 'assign':
   448→   448→        localVars[stmt.name] = evalValue(stmt.value)
   449→   449→        break
   450→   450→      case 'if': {
   451→   451→        // if / else if / else chain: o parser coloca elseif e else como statements
   452→   452→        // subsequentes no mesmo nível. Processamos o if e, se falso, procuramos
   453→   453→        // elseif/else nos statements seguintes do bloco pai.
   454→   454→        // Mas como o parser coloca elseif/else como statements separados com body,
   455→   455→        // precisamos de uma abordagem diferente: o if já tem o body, e elseif/else
   456→   456→        // são statements irmãos. O execStatements percorre linearmente.
   457→   457→        // Solução: o if executa o body se true. Se false, NÃO executa.
   458→   458→        // O elseif/else são executados como statements independentes, mas só
   459→   459→        // têm efeito se o if anterior foi false. Usamos uma flag.
   460→   460→        if (evalCondition(stmt.condition)) {
   461→   461→          execStatements(stmt.body || [], params)
   462→   462→          params._ifChainMatched = true
   463→   463→        } else {
   464→   464→          params._ifChainMatched = false
   465→   465→        }
   466→   466→        break
   467→   467→      }
   468→   468→      case 'elseif': {
   469→   469→        // Só executa se nenhum if/elseif anterior no chain foi true
   470→   470→        if (!params._ifChainMatched && evalCondition(stmt.condition)) {
   471→   471→          execStatements(stmt.body || [], params)
   472→   472→          params._ifChainMatched = true
   473→   473→        }
   474→   474→        break
   475→   475→      }
   476→   476→      case 'else': {
   477→   477→        // Só executa se nenhum if/elseif anterior no chain foi true
   478→   478→        if (!params._ifChainMatched) {
   479→   479→          execStatements(stmt.body || [], params)
   480→   480→          params._ifChainMatched = true
   481→   481→        }
   482→   482→        break
   483→   483→      }
   484→   484→      case 'switch': {
   485→   485→        // O switch contém cases e default no seu body
   486→   486→        // Resolver o valor da variável do switch
   487→   487→        let switchVal
   488→   488→        if (localVars[stmt.varName] !== undefined) {
   489→   489→          switchVal = String(localVars[stmt.varName])
   490→   490→        } else if (gameContext.getVar) {
   491→   491→          switchVal = String(gameContext.getVar(stmt.varName) ?? '')
   492→   492→        } else {
   493→   493→          switchVal = ''
   494→   494→        }
   495→   495→        let matched = false
   496→   496→        for (const caseStmt of stmt.body || []) {
   497→   497→          if (caseStmt.type === 'case') {
   498→   498→            // Remover aspas do valor do case
   499→   499→            let caseVal = caseStmt.value.trim()
   500→   500→            if ((caseVal.startsWith('"') && caseVal.endsWith('"')) || (caseVal.startsWith("'") && caseVal.endsWith("'"))) {
   501→   501→              caseVal = caseVal.slice(1, -1)
   502→   502→            }
   503→   503→            if (switchVal === caseVal) {
   504→   504→              execStatements(caseStmt.body || [], params)
   505→   505→              matched = true
   506→   506→              break
   507→   507→            }
   508→   508→          }
   509→   509→        }
   510→   510→        if (!matched) {
   511→   511→          for (const caseStmt of stmt.body || []) {
   512→   512→            if (caseStmt.type === 'default') {
   513→   513→              execStatements(caseStmt.body || [], params)
   514→   514→              break
   515→   515→            }
   516→   516→          }
   517→   517→        }
   518→   518→        break
   519→   519→      }
   520→   520→      // case e default são processados dentro do switch — não executar standalone
   521→   521→      case 'case':
   522→   522→      case 'default':
   523→   523→        break
   524→   524→      case 'call': {
   525→   525→        execBuiltin(stmt.funcName, stmt.args, params)
   526→   526→        break
   527→   527→      }
   528→   528→      case 'repeat_n': {
   529→   529→        localVars[stmt.varName] = 0
   530→   530→        for (let i = 0; i < stmt.count; i++) {
   531→   531→          localVars[stmt.varName] = i
   532→   532→          execStatements(stmt.body || [], params)
   533→   533→        }
   534→   534→        break
   535→   535→      }
   536→   536→      case 'repeat_inc': {
   537→   537→        let v = 0
   538→   538→        while (v <= stmt.until) {
   539→   539→          execStatements(stmt.body || [], { ...params, _repeat: v })
   540→   540→          v += stmt.step
   541→   541→        }
   542→   542→        break
   543→   543→      }
   544→   544→      case 'repeat_dec': {
   545→   545→        let v = stmt.until
   546→   546→        while (v >= 0) {
   547→   547→          execStatements(stmt.body || [], { ...params, _repeat: v })
   548→   548→          v -= stmt.step
   549→   549→        }
   550→   550→        break
   551→   551→      }
   552→   552→      case 'endcode':
   553→   553→      case 'unknown':
   554→   554→        break
   555→   555→    }
   556→   556→  }

   558→   558→  // Executar função embutida
   559→   559→  function execBuiltin(name, args, params) {
   560→   560→    const evaluatedArgs = args.map((a) => evalValue(a))
   561→   561→    switch (name) {
   562→   562→      case 'playAnim':
   563→   563→        gameContext.playAnimation?.(gameContext._instanceId, evaluatedArgs[0])
   564→   564→        break
   565→   565→      case 'playSound':
   566→   566→        // Procurar SoundObject pelo nome na cena ativa; se não encontrar, usar como URL
   567→   567→        gameContext.playSoundByName?.(evaluatedArgs[0]) ?? gameContext.playSound?.(evaluatedArgs[0])
   568→   568→        break
   569→   569→      case 'move':
   570→   570→        gameContext.moveObject?.(gameContext._instanceId, evaluatedArgs, 1)
   571→   571→        break
   572→   572→      case 'rotate':
   573→   573→        gameContext.rotateObject?.(gameContext._instanceId, evaluatedArgs)
   574→   574→        break
   575→   575→      case 'scale':
   576→   576→        if (gameContext.mesh) {
   577→   577→          gameContext.mesh.scale.set(evaluatedArgs[0] || 1, evaluatedArgs[1] || 1, evaluatedArgs[2] || 1)
   578→   578→        }
   579→   579→        break
   580→   580→      case 'destroy':
   581→   581→        gameContext.destroyObject?.(gameContext._instanceId)
   582→   582→        break
   583→   583→      case 'createObject':
   584→   584→        gameContext.spawnObject?.(evaluatedArgs[0], [evaluatedArgs[1], evaluatedArgs[2], evaluatedArgs[3]])
   585→   585→        break
   586→   586→      case 'changeScene':
   587→   587→        gameContext.changeScene?.(evaluatedArgs[0])
   588→   588→        break
   589→   589→      case 'wait': {
   590→   590→        // wait(seconds) — implementa um delay real via setTimeout
   591→   591→        // Como FlirCode é síncrono, o wait só afeta a chamada atual:
   592→   592→        // regista o delay e o runtime processa ações pendentes após o tempo
   593→   593→        const delayMs = (evaluatedArgs[0] || 0) * 1000
   594→   594→        debugLog('wait(' + evaluatedArgs[0] + 's) — aguardando', 'log', 'FlirCode')
   595→   595→        // Usar Atomics.wait para um sleep síncrono não-bloqueante em worker
   596→   596→        // Como não estamos em worker, usar setTimeout com flag
   597→   597→        if (gameContext._waitQueue) {
   598→   598→          gameContext._waitQueue.push(delayMs)
   599→   599→        }
   600→   600→        break
   601→   601→      }
   602→   602→      case 'setVar':
   603→   603→        gameContext.globalVars = gameContext.globalVars || {}
   604→   604→        gameContext.globalVars[evaluatedArgs[0]] = evaluatedArgs[1]
   605→   605→        break
   606→   606→      case 'getVar':
   607→   607→        return gameContext.globalVars?.[evaluatedArgs[0]]
   608→   608→      case 'showUI':
   609→   609→        debugLog(`showUI: ${evaluatedArgs[0]}`, 'log', 'FlirCode')
   610→   610→        gameContext.showUIScreen?.(evaluatedArgs[0])
   611→   611→        break
   612→   612→      case 'hideUI':
   613→   613→        debugLog(`hideUI: ${evaluatedArgs[0]}`, 'log', 'FlirCode')
   614→   614→        gameContext.hideUIScreen?.(evaluatedArgs[0])
   615→   615→        break
   616→   616→      case 'showUIScreen':
   617→   617→        gameContext.showUIScreen?.(evaluatedArgs[0])
   618→   618→        break
   619→   619→      case 'hideUIScreen':
   620→   620→        gameContext.hideUIScreen?.(evaluatedArgs[0])
   621→   621→        break
   622→   622→      case 'getUIValue':
   623→   623→        return gameContext.getUIValue?.(evaluatedArgs[0]) ?? ''
   624→   624→      case 'setUIValue':
   625→   625→        gameContext.setUIValue?.(evaluatedArgs[0], evaluatedArgs[1])
   626→   626→        break
   627→   627→      case 'print':
   628→   628→        debugLog(evaluatedArgs[0], 'log', 'FlirCode')
   629→   629→        break
   630→   630→      case 'warn':
   631→   631→        debugLog(evaluatedArgs[0], 'warning', 'FlirCode')
   632→   632→        break
   633→   633→      case 'error':
   634→   634→        debugLog(evaluatedArgs[0], 'error', 'FlirCode')
   635→   635→        break
   636→   636→      case 'collidingWith':
   637→   637→        // Verificar se o objeto está a colidir com outro de um tipo específico
   638→   638→        return gameContext.collidingWith?.(gameContext._instanceId, evaluatedArgs[0]) || false
   639→   639→      case 'distanceTo':
   640→   640→        // Calcular distância até outro objeto
   641→   641→        return gameContext.distanceTo?.(gameContext._instanceId, evaluatedArgs[0]) || 0
   642→   642→      case 'isTouching':
   643→   643→        // Verificar se o utilizador está a tocar no ecrã
   644→   644→        return gameContext.isTouching?.() || false
   645→   645→      // Fase 5: Funções de multiplayer
   646→   646→      case 'sendMessage':
   647→   647→        // Envia dados customizados para outros jogadores
   648→   648→        gameContext.sendMessage?.(evaluatedArgs[0])
   649→   649→        break
   650→   650→      case 'getPlayers':
   651→   651→        // Retorna o número de jogadores ligados
   652→   652→        return gameContext.getPlayers?.() || 1
   653→   653→      case 'getPlayerState':
   654→   654→        // Retorna o estado de um jogador específico
   655→   655→        return gameContext.getPlayerState?.(evaluatedArgs[0]) || null
   656→   656→      // Sistema 3: Sinais — emitSignal("nome", dados)
   657→   657→      case 'emitSignal':
   658→   658→        gameContext.emitSignal?.(evaluatedArgs[0], evaluatedArgs[1])
   659→   659→        break
   660→   660→      // Sistema 2: Armas e combate
   661→   661→      case 'shoot':
   662→   662→        gameContext.shoot?.()
   663→   663→        break
   664→   664→      case 'reload':
   665→   665→        gameContext.reload?.()
   666→   666→        break
   667→   667→      case 'equipWeapon':
   668→   668→        gameContext.equipWeapon?.(evaluatedArgs[0])
   669→   669→        break
   670→   670→      case 'getAmmo':
   671→   671→        return gameContext.getAmmo?.() ?? 0
   672→   672→      case 'takeDamage':
   673→   673→        gameContext.takeDamage?.(gameContext._instanceId, evaluatedArgs[0])
   674→   674→        break
   675→   675→      case 'getHealth':
   676→   676→        return gameContext.getHealth?.(gameContext._instanceId) ?? 100
   677→   677→      // Sistema 3: Inventário
   678→   678→      case 'addToInventory':
   679→   679→        gameContext.addToInventory?.(evaluatedArgs[0], evaluatedArgs[1])
   680→   680→        break
   681→   681→      case 'removeFromInventory':
   682→   682→        gameContext.removeFromInventory?.(evaluatedArgs[0], evaluatedArgs[1])
   683→   683→        break
   684→   684→      case 'getInventoryCount':
   685→   685→        return gameContext.getInventoryCount?.(evaluatedArgs[0]) ?? 0
   686→   686→      case 'hasItem':
   687→   687→        return gameContext.hasItem?.(evaluatedArgs[0]) ?? false
   688→   688→      // Sistema: Links — navegar para cena ou tela de UI
   689→   689→      case 'linkTo':
   690→   690→        gameContext.linkTo?.(evaluatedArgs[0], evaluatedArgs[1])
   691→   691→        break
   692→   692→      // Sistema: Game State
   693→   693→      case 'setGameState':
   694→   694→        gameContext.setGameState?.(evaluatedArgs[0])
   695→   695→        break
   696→   696→      case 'getGameState':
   697→   697→        return gameContext.getGameState?.() || 'menu'
   698→   698→      // Sistema: Save/Load Progress (localStorage do jogador)
   699→   699→      case 'saveProgress':
   700→   700→        gameContext.saveProgress?.(evaluatedArgs[0], evaluatedArgs[1])
   701→   701→        break
   702→   702→      case 'loadProgress':
   703→   703→        return gameContext.loadProgress?.(evaluatedArgs[0])
   704→   704→      // Sistema: Sequenciador
   705→   705→      case 'playSequence':
   706→   706→        gameContext.playSequence?.(evaluatedArgs[0])
   707→   707→        break
   708→   708→      // Sistema: Luzes
   709→   709→      case 'setLightIntensity':
   710→   710→        gameContext.setLightIntensity?.(evaluatedArgs[0], evaluatedArgs[1])
   711→   711→        break
   712→   712→      case 'setLightColor':
   713→   713→        gameContext.setLightColor?.(evaluatedArgs[0], evaluatedArgs[1])
   714→   714→        break
   715→   715→      case 'setLightVisible':
   716→   716→        gameContext.setLightVisible?.(evaluatedArgs[0], evaluatedArgs[1])
   717→   717→        break
   718→   718→      // Sistema: Data Assets (ScriptableObjects)
   719→   719→      case 'getDataAsset':
   720→   720→        return gameContext.getDataAsset?.(evaluatedArgs[0])
   721→   721→      // Sistema: Autoloads (Singletons)
   722→   722→      case 'getAutoload':
   723→   723→        return gameContext.getAutoload?.(evaluatedArgs[0])
   724→   724→      default:
   725→   725→        debugLog(`Função desconhecida: ${name}`, 'warning', 'FlirCode')
   726→   726→    }
   727→   727→  }

   729→   729→  return {
   730→   730→    functions,
   731→   731→    errors,
   732→   732→    hasErrors: errors.length > 0,
   733→   733→
   734→   734→    // Dispara um evento
   735→   735→    triggerEvent(eventName, payload = {}) {
   736→   736→      // Mapear nome interno → nome da função FlirCode
   737→   737→      const funcName = Object.entries(eventMap).find(([_, en]) => en === eventName)?.[0]
   738→   738→      if (!funcName) return
   739→   739→      const fn = functions[funcName]
   740→   740→      if (!fn) return
   741→   741→      // Passar payload como parâmetros
   742→   742→      const params = {}
   743→   743→      if (fn.params.length > 0) {
   744→   744→        // Sistema 3: para onSignal, passar name e data como parâmetros separados
   745→   745→        if (funcName === 'onSignal' && fn.params.length >= 2) {
   746→   746→          params[fn.params[0]] = payload.name
   747→   747→          params[fn.params[1]] = payload.data
   748→   748→        } else if (funcName === 'onMessage' && fn.params.length >= 2) {
   749→   749→          params[fn.params[0]] = payload.playerId
   750→   750→          params[fn.params[1]] = payload.data
   751→   751→        } else {
   752→   752→          params[fn.params[0]] = payload
   753→   753→        }
   754→   754→      }
   755→   755→      gameContext._instanceId = gameContext._instanceId // já definido externamente
   756→   756→      // Sistema 3: copiar params para localVars para ficarem acessíveis no script
   757→   757→      for (const [k, v] of Object.entries(params)) {
   758→   758→        localVars[k] = v
   759→   759→      }
   760→   760→      execStatements(fn.body, params)
   761→   761→    },
   762→   762→
   763→   763→    update(delta) {
   764→   764→      // Não usado — FlirCode é orientado a eventos
   765→   765→    },
   766→   766→
   767→   767→    dispose() {
   768→   768→      // limpar
   769→   769→    },
   770→   770→  }
