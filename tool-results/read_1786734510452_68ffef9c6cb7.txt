    52→    return value
    53→  }))

   314→      // ---------- Helpers internos ----------
   315→      _pushHistory: () => {
   316→        const { objects, past } = get()
   317→        const next = [...past, snapshot(objects)]
   318→        if (next.length > MAX_HISTORY) next.shift()
   319→        set({ past: next, future: [] })
   320→      },
   321→
   322→      _pushHistoryCoalesced: (() => {
   323→        let lastKey = null
   324→        return (key) => {
   325→          const { objects, past } = get()
   326→          if (lastKey === key && past.length > 0) {
   327→            const next = [...past.slice(0, -1), snapshot(objects)]
   328→            set({ past: next })
   329→          } else {
   330→            const next = [...past, snapshot(objects)]
   331→            if (next.length > MAX_HISTORY) next.shift()
   332→            set({ past: next, future: [] })
   333→          }
   334→          lastKey = key
   335→        }
   336→      })(),
   337→
   338→      // ---------- Toasts ----------
   339→      toast: (message, type = 'info', duration = 2400) => {
   340→        const id = Math.random().toString(36).slice(2)

   343→          set((s) => ({ toasts: s.toasts.filter((t) => t.id !== id) }))
   344→        }, duration)
   345→      },

   422→      renameObject: (id, name) => {

   427→      },
   428→
   429→      updateObject: (id, patch) => {
   430→        set((s) => ({
   431→          objects: s.objects.map((o) => (o.id === id ? { ...o, ...patch } : o)),
   432→        }))
   433→      },
   434→
   435→      updateMaterial: (id, patch) => {
   436→        set((s) => ({
   437→          objects: s.objects.map((o) =>
   438→            o.id === id ? { ...o, material: { ...o.material, ...patch } } : o
   439→          ),
   440→        }))
   441→      },
   442→
   443→      commitMaterial: (id, patch) => {
   444→        get()._pushHistory()
   445→        set((s) => ({
   446→          objects: s.objects.map((o) =>
   447→            o.id === id ? { ...o, material: { ...o.material, ...patch } } : o
   448→          ),
   449→        }))
   450→      },
   451→
   452→      transformObject: (id, transform) => {
   453→        get()._pushHistoryCoalesced(`transform:${id}`)
   454→        set((s) => ({
   455→          objects: s.objects.map((o) => (o.id === id ? { ...o, ...transform } : o)),
   456→        }))
   457→      },
   458→
   459→      duplicateObject: (id) => {
   460→        const obj = get().objects.find((o) => o.id === id)
   461→        if (!obj) return
   462→        get()._pushHistory()
   463→        const copy = JSON.parse(JSON.stringify(obj))
   464→        copy.id = `obj_${Math.random().toString(36).slice(2, 10)}`
   465→        copy.name = `${obj.name} (cópia)`
   466→        copy.position = [obj.position[0] + 0.5, obj.position[1], obj.position[2] + 0.5]
   467→        // limpar parent para não herdar transformações
   468→        copy.parentId = null
   469→        set((s) => ({
   470→          objects: [...s.objects, copy],
   471→          selectedId: copy.id,
   472→        }))
   473→        get().toast('Objeto duplicado', 'success')
   474→      },
   475→
   476→      deleteObject: (id) => {
   477→        get()._pushHistory()
   478→        set((s) => ({
   479→          objects: s.objects
   480→            .filter((o) => o.id !== id)
   481→            .map((o) => (o.parentId === id ? { ...o, parentId: null } : o)),
   482→          selectedId: s.selectedId === id ? null : s.selectedId,
   483→        }))
   484→        get().toast('Objeto eliminado', 'info')
   485→      },
   486→
   487→      toggleVisibility: (id) => {
   488→        get()._pushHistory()
   489→        set((s) => ({
   490→          objects: s.objects.map((o) =>
   491→            o.id === id ? { ...o, visible: !o.visible } : o
   492→          ),
   493→        }))
   494→      },
   495→
   496→      extrudeObject: (id, amount = 0.5) => {
   497→        const obj = get().objects.find((o) => o.id === id)
   498→        if (!obj) return
   499→        get()._pushHistory()
   500→        const scaled = {
   501→          ...JSON.parse(JSON.stringify(obj)),
   502→          scale: [obj.scale[0], obj.scale[1] * (1 + amount), obj.scale[2]],
   503→        }
   504→        set((s) => ({
   505→          objects: s.objects.map((o) => (o.id === id ? scaled : o)),
   506→        }))
   507→        get().toast(`Extrude aplicado (${amount.toFixed(2)})`, 'success')
   508→      },
   509→
   510→      // ---------- Hierarquia (parent/child) ----------
   511→      setParent: (childId, parentId) => {
   512→        if (childId === parentId) return
   513→        // Evitar ciclos: o parentId não pode ser descendente do childId
   514→        const objects = get().objects
   515→        const wouldCycle = (id, target) => {
   516→          if (id === target) return true
   517→          const obj = objects.find((o) => o.id === id)
   518→          if (!obj || !obj.parentId) return false
   519→          return wouldCycle(obj.parentId, target)
   520→        }
   521→        if (wouldCycle(parentId, childId)) {
   522→          get().toast('Não é possível criar ciclo de parentesco', 'error')
   523→          return
   524→        }
   525→        get()._pushHistory()
   526→        set((s) => ({
   527→          objects: s.objects.map((o) =>
   528→            o.id === childId ? { ...o, parentId: parentId || null } : o
   529→          ),
   530→        }))
   531→        get().toast('Objetos agrupados', 'success')
   532→      },
   533→
   534→      clearParent: (childId) => {
   535→        get()._pushHistory()
   536→        set((s) => ({
   537→          objects: s.objects.map((o) =>
   538→            o.id === childId ? { ...o, parentId: null } : o
   539→          ),
   540→        }))
   541→      },
   542→
   543→      // ---------- Modificadores (não destrutivos) ----------
   544→      addModifier: (id, type) => {
   545→        const def = MODIFIER_TYPES[type]
   546→        if (!def) return
   547→        get()._pushHistory()
   548→        const modifier = {
   549→          id: `mod_${Math.random().toString(36).slice(2, 10)}`,
   550→          type,

   590→      // Aplica uma operação de malha e guarda o resultado em customGeometry
   591→      applyMeshOp: (id, op, params = {}) => {
   592→        const obj = get().objects.find((o) => o.id === id)
   593→        if (!obj) return
   594→        get()._pushHistory()
   595→
   596→        // Construir geometria atual
   597→        let geometry
   598→        if (obj.customGeometry) {
   599→          geometry = new THREE.BufferGeometry()
   600→          geometry.setAttribute('position', new THREE.Float32BufferAttribute(obj.customGeometry.positions, 3))
   601→          if (obj.customGeometry.normals) {
   602→            geometry.setAttribute('normal', new THREE.Float32BufferAttribute(obj.customGeometry.normals, 3))
   603→          }
   604→          if (obj.customGeometry.uvs) {
   605→            geometry.setAttribute('uv', new THREE.Float32BufferAttribute(obj.customGeometry.uvs, 2))
   606→          }
   607→        } else if (obj.imported && obj.bufferGeometry) {
   608→          geometry = obj.bufferGeometry
   609→        } else {
   610→          const def = PRIMITIVES[obj.type]
   611→          geometry = def ? def.build(THREE, obj.args) : new THREE.BoxGeometry(1, 1, 1)
   612→        }
   613→
   614→        let result = geometry
   615→        switch (op) {
   616→          case 'subdivide':
   617→            result = meshOps.subdivide(geometry, params.levels || 1)
   618→            break
   619→          case 'extrude':
   620→            result = meshOps.extrudeFaces(geometry, params.amount ?? 0.5)
   621→            break
   622→          case 'inset':
   623→            result = meshOps.insetFaces(geometry, params.amount ?? 0.1)
   624→            break
   625→          case 'bevel':
   626→            result = meshOps.bevelGeometry(geometry, params.radius ?? 0.05, params.segments ?? 2)
   627→            break
   628→          case 'loopCut':
   629→            result = meshOps.loopCut(geometry, params.axis || 'y', params.position || 0)
   630→            break
   631→          case 'merge':
   632→            result = meshOps.mergeVertices(geometry, params.threshold ?? 0.001)
   633→            break
   634→          case 'unwrap':
   635→            result = meshOps.unwrapUV(geometry, params.method || 'planar')
   636→            break
   637→          default:
   638→            get().toast(`Operação desconhecida: ${op}`, 'error')
   639→            return
   640→        }
   641→
   642→        // Serializar para o store
   643→        const positions = result.getAttribute('position').array
   644→        const normals = result.getAttribute('normal')?.array
   645→        const uvs = result.getAttribute('uv')?.array
   646→        set((s) => ({
   647→          objects: s.objects.map((o) =>
   648→            o.id === id
   649→              ? {
   650→                  ...o,
   651→                  customGeometry: {
   652→                    positions: Array.from(positions),
   653→                    normals: normals ? Array.from(normals) : null,
   654→                    uvs: uvs ? Array.from(uvs) : null,
   655→                  },
   656→                }
   657→              : o
   658→          ),
   659→        }))
