     1→/**
     2→ * taxonomy.js — definição central de todos os Conects disponíveis.
     3→ *
     4→ * Em FlirScript engine, um "Conect" é a unidade base de cena (similar a
     5→ * "node" na Godot ou "GameObject" na Unity). Cada Conect tem um tipo
     6→ * (RigidObject, VisualObject, etc.) com propriedades específicas.
     7→ *
     8→ * Esta taxonomy alimenta:
     9→ *  - A janela de Conects (lista + categorias + pesquisa)
    10→ *  - O painel de propriedades dinâmico (cada tipo tem campos diferentes)
    11→ *  - O runtime (cada tipo sabe como instanciar-se e atualizar-se)
    12→ *
    13→ * Estrutura de cada Conect:
    14→ *  {
    15→ *    type: 'RigidObject',         // identificador único
    16→ *    label: 'Rigid Object',       // nome visível
    17→ *    category: 'physics',         // categoria para agrupar
    18→ *    icon: '📦',                  // emoji ou ícone
    19→ *    description: '...',          // descrição curta
    20→ *    defaults: { ... },           // propriedades predefinidas
    21→ *    properties: [ ... ],         // esquema de propriedades para o painel
    22→ *    hasPhysics: true,            // participa no mundo de física?
    23→ *    hasVisual: true,             // tem malha visível?
    24→ *    flirScriptable: true,        // pode ter script FlirScript?
    25→ *  }
    26→ *
    27→ * Categorias: physics, visual, camera_audio, environment, ui, gameplay, organization
    28→ */

  1047→// Cria uma instância com valores predefinidos sensatos
  1048→export function createConectInstance(type, position = [0, 0.5, 0]) {
  1049→  const def = findConectDefinition(type)
  1050→  if (!def) throw new Error(`Conect desconhecido: ${type}`)
  1051→  // ViewObject tem posição default diferente — deve estar afastada para ver a cena
  1052→  if (type === 'ViewObject') {
  1053→    position = [5, 4, 6]
  1054→  }
  1055→  const defaults = {}
  1056→  for (const propDef of def.properties) {
  1057→    defaults[propDef.key] = JSON.parse(JSON.stringify(propDef.default))
  1058→  }
  1059→  return {
  1060→    instanceId: `conect_${Math.random().toString(36).slice(2, 10)}`,
  1061→    type,
  1062→    name: `${def.label}`,
  1063→    position: [...position],
  1064→    rotation: type === 'ViewObject' ? [-0.5, 0.7, 0] : [0, 0, 0],
  1065→    scale: [1, 1, 1],
  1066→    visible: def.hasVisual !== false,
  1067→    flirScript: null, // grafo FlirScript opcional
  1068→    ...defaults,
  1069→    ...def.defaults, // garantir que defaults da taxonomy também ficam
  1070→  }
  1071→}
