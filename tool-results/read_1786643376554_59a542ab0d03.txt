    65→// Componente interno que aplica o fundo da cena
    66→function SceneBackgroundSolid({ background }) {
    67→  const { scene } = useThree()
    68→  useEffect(() => {
    69→    if (background.type === 'gradient') {
    70→      const canvas = document.createElement('canvas')
    71→      canvas.width = 2; canvas.height = 256
    72→      const ctx = canvas.getContext('2d')
    73→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
    74→      grad.addColorStop(0, background.gradientTop)
    75→      grad.addColorStop(1, background.gradientBottom)
    76→      ctx.fillStyle = grad; ctx.fillRect(0, 0, 2, 256)
    77→      const tex = new THREE.CanvasTexture(canvas)
    78→      tex.colorSpace = THREE.SRGBColorSpace
    79→      scene.background = tex
    80→      return () => tex.dispose()
    81→    } else {
    82→      scene.background = new THREE.Color(background.color)
    83→    }
    84→  }, [background, scene])
    85→  return null
    86→}

    88→// ===== FogApplier — aplica FogObject à cena 3D =====
    89→function FogApplier({ conects }) {
    90→  const { scene } = useThree()
    91→  const fogConect = (conects || []).find(c => c.type === 'FogObject')
    92→  useEffect(() => {
    93→    if (!fogConect) {
    94→      scene.fog = null
    95→      return
    96→    }
    97→    if (fogConect.fogType === 'exponential') {
    98→      scene.fog = new THREE.FogExp2(fogConect.color || '#a0a0a0', fogConect.density || 0.02)
    99→    } else {
   100→      scene.fog = new THREE.Fog(fogConect.color || '#a0a0a0', fogConect.near || 5, fogConect.far || 50)
   101→    }
   102→  }, [fogConect?.fogType, fogConect?.color, fogConect?.near, fogConect?.far, fogConect?.density, scene])
   103→  return null
   104→}

   106→// ===== InstancingRenderer — adiciona HardwareInstancingSystems à cena e atualiza por frame =====
   107→function InstancingRenderer() {
   108→  const { scene, camera } = useThree()
   109→  const addedRef = useRef(new Set())
   110→
   111→  useFrame(() => {
   112→    const systems = window._flirInstancingSystems
   113→    if (!systems || systems.length === 0) return
   114→    for (const sys of systems) {
   115→      // Adicionar à cena se ainda não foi adicionado
   116→      if (!addedRef.current.has(sys)) {
   117→        try {
   118→          sys.addToScene(scene)
   119→          addedRef.current.add(sys)
   120→        } catch (e) {
   121→          // já foi adicionado ou sistema dispuesto
   122→        }
   123→      }
   124→      // Atualizar frustum culling + LOD + matrizes por instância
   125→      try { sys.update(camera) } catch {}
   126→    }
   127→  })
   128→
   129→  // Limpar ao desmontar
   130→  useEffect(() => {
   131→    return () => {
   132→      for (const sys of addedRef.current) {
   133→        try {
   134→          for (const mesh of sys.instanceMeshes || []) {
   135→            scene.remove(mesh)
   136→          }
   137→        } catch {}
   138→      }
   139→      addedRef.current.clear()
   140→    }
   141→  }, [scene])
   142→
   143→  return null
   144→}

   146→// ===== TerrainSculptBridge — ativa escultura 3D no viewport quando um TerrainObject está selecionado =====
   147→function TerrainSculptBridge({ activeScene, conectMeshRefs, isGameMode, onDragStateChange }) {
   148→  // Procurar se há um TerrainObject selecionado e um modo de escultura ativo
   149→  const terrainSculptActive = useStore((s) => s.terrainSculptActive)
   150→  const terrainBrush = useStore((s) => s.terrainBrush)
   151→  const selectedConectId = useStore((s) => s.selectedConectId)
   152→  const updateConect = useStore((s) => s.updateConect)
   153→
   154→  // Encontrar o TerrainObject atualmente selecionado
   155→  const terrainConect = useMemo(() => {
   156→    if (!activeScene?.conects || !selectedConectId) return null
   157→    const c = activeScene.conects.find(c => c.instanceId === selectedConectId && c.type === 'TerrainObject')
   158→    return c || null
   159→  }, [activeScene, selectedConectId])
   160→
   161→  const terrainMesh = useMemo(() => {
   162→    if (!terrainConect) return null
   163→    return conectMeshRefs.current.get(terrainConect.instanceId) || null
   164→  }, [terrainConect, conectMeshRefs, activeScene])
   165→
   166→  // Callback quando heightmap é alterado pela escultura 3D
   167→  const handleHeightmapChange = useCallback((newHm) => {
   168→    if (!terrainConect) return
   169→    // Persistir alteração no conect
   170→    updateConect(terrainConect.instanceId, {
   171→      heightmap: Array.from(newHm),
   172→    })
   173→  }, [terrainConect, updateConect])
   174→
   175→  if (!terrainSculptActive || !terrainConect || !terrainMesh || isGameMode) return null
   176→
   177→  return (
   178→    <TerrainSculpt3D
   179→      terrainMesh={terrainMesh}
   180→      heightmap={terrainConect.heightmap ? new Float32Array(terrainConect.heightmap) : null}
   181→      seg={terrainConect.segments || 64}
   182→      brushMode={terrainBrush?.mode || 'raise'}
   183→      brushSize={terrainBrush?.size || 8}
   184→      brushStrength={terrainBrush?.strength || 0.5}
   185→      falloffType={terrainBrush?.falloff || 'smooth'}
   186→      onHeightmapChange={handleHeightmapChange}
   187→      onDragStateChange={onDragStateChange}
   188→      isActive={terrainSculptActive}
   189→    />
   190→  )
   191→}

   193→// ===== ViewModelFPS — renderiza a arma como filho da câmara (first-person viewmodel) =====
   194→function ViewModelFPS({ activeScene, isGameMode }) {
   195→  const { camera } = useThree()
   196→  const weaponGroupRef = useRef()
   197→
   198→  // Procurar WeaponObject na cena ativa
   199→  const weaponConect = useMemo(() => {
   200→    if (!activeScene?.conects) return null
   201→    return (activeScene.conects || []).find(c => c.type === 'WeaponObject') || null
   202→  }, [activeScene])
   203→
   204→  // Procurar ViewObject para verificar followMode
   205→  const viewConect = useMemo(() => {
   206→    if (!activeScene?.conects) return null
   207→    const views = (activeScene.conects || []).filter(c => c.type === 'ViewObject')
   208→    return views.find(c => c.cameraRole === 'player') || views[0] || null
   209→  }, [activeScene])
   210→
   211→  // Só mostrar em modo jogo, com arma, e followMode first ou third
   212→  const shouldShow = isGameMode && weaponConect && viewConect &&
   213→    (viewConect.followMode === 'first' || viewConect.followMode === 'third')
   214→
   215→  // Parentar à câmara quando ativo
   216→  useEffect(() => {
   217→    if (!shouldShow || !weaponGroupRef.current) return
   218→    // Adicionar o grupo da arma como filho da câmara
   219→    camera.add(weaponGroupRef.current)
   220→    return () => {
   221→      // Remover ao desmontar
   222→      if (weaponGroupRef.current && weaponGroupRef.current.parent === camera) {
   223→        camera.remove(weaponGroupRef.current)
   224→      }
   225→    }
   226→  }, [shouldShow, camera])
   227→
   228→  if (!shouldShow) return null
   229→
   230→  // Posição relativa à câmara (view space)
   231→  // Em first-person: arma à direita, em baixo, à frente
   232→  const isFirst = viewConect.followMode === 'first'
   233→  const weaponPos = isFirst
   234→    ? [weaponConect.position?.[0] || 0.3, -(0.2), -(0.5)]  // direita, baixo, frente
   235→    : [0.5, -0.3, -0.8]  // third-person: mais afastada
   236→  const weaponScale = isFirst ? [0.5, 0.5, 0.5] : [0.6, 0.6, 0.6]
   237→
   238→  return (
   239→    <group ref={weaponGroupRef} position={weaponPos} scale={weaponScale}>
   240→      {/* Corpo da arma */}
   241→      <mesh>
   242→        <boxGeometry args={[0.15, 0.2, 0.6]} />
   243→        <meshStandardMaterial color="#1f2937" roughness={0.6} metalness={0.4} />
   244→      </mesh>
   245→      {/* Cano */}
   246→      <mesh position={[0, 0.05, -0.4]}>
   247→        <cylinderGeometry args={[0.04, 0.04, 0.4, 8]} rotation={[Math.PI / 2, 0, 0]} />
   248→        <meshStandardMaterial color="#374151" roughness={0.4} metalness={0.7} />
   249→      </mesh>
   250→      {/* Mira */}
   251→      <mesh position={[0, 0.18, -0.2]}>
   252→        <boxGeometry args={[0.02, 0.06, 0.02]} />
   253→        <meshStandardMaterial color="#fbbf24" emissive="#fbbf24" emissiveIntensity={0.8} />
   254→      </mesh>
   255→      {/* Mãos (simplificadas) */}
   256→      <mesh position={[0, -0.15, 0.15]}>
   257→        <boxGeometry args={[0.08, 0.1, 0.15]} />
   258→        <meshStandardMaterial color="#9ca3af" roughness={0.8} />
   259→      </mesh>
   260→      <mesh position={[0, -0.12, -0.2]}>
   261→        <boxGeometry args={[0.08, 0.08, 0.12]} />
   262→        <meshStandardMaterial color="#9ca3af" roughness={0.8} />
   263→      </mesh>
   264→    </group>
   265→  )
   266→}

   268→// ===== Componente que gere o modo jogo dentro do canvas =====
   269→function GameMode({ activeScene, objects, meshRefs, conectMeshRefs, isGameMode }) {
   270→  const { camera } = useThree()
   271→  const physicsRef = useRef(null)
   272→  const runtimesRef = useRef(new Map())
   273→  const animPlayersRef = useRef(new Map())
   274→  const npcAIsRef = useRef(new Map())
   275→  const timerStatesRef = useRef(new Map())
   276→  const joystickRef = useRef({ x: 0, z: 0, active: false })
   277→  const inventoryRef = useRef({})
   278→  const weaponStateRef = useRef({ equipped: false, ammo: 0, maxAmmo: 0, damage: 0, fireRate: 0.3, range: 50, reloadTime: 2, lastShot: 0 })
   279→  const collisionEventsRef = useRef(new Map()) // instanceId → Set de otherIds em contacto
   280→  const checkpointRef = useRef(null) // último checkpoint registado
   281→  const skyRef = useRef(null) // referência ao SkyObject ativo
   282→
   283→  // Expor meshRefs globalmente para TrailObject poder seguir objetos
   284→  useEffect(() => {
   285→    window._flirConectMeshRefs = conectMeshRefs
   286→    window._flirMeshRefs = meshRefs
   287→    return () => {
   288→      window._flirConectMeshRefs = null
   289→      window._flirMeshRefs = null
   290→    }
   291→  }, [conectMeshRefs, meshRefs])
   292→
   293→  // P2.5 fix: armazenar a cena ativa num ref para que modificações feitas pelo
   294→  // FlirCode (createObject, etc.) NÃO reiniciem o jogo (evita loop infinito).
   295→  // O useEffect só deve correr quando isGameMode muda (true→false ou false→true),
   296→  // não quando activeScene é modificado por spawnObject.
   297→  const activeSceneRef = useRef(activeScene)
   298→  const objectsRef = useRef(objects)
   299→  const gameStartedRef = useRef(false)
   300→
   301→  // Expor joystickRef globalmente para o GameUIOverlay poder atualizá-lo
   302→  useEffect(() => {
   303→    window._flirJoystick = joystickRef.current
   304→    return () => { window._flirJoystick = null }
   305→  }, [])
   306→
   307→  // Use o ref para o setup
   308→  const setupScene = activeSceneRef.current
   309→
   310→  // Fase 5: Expor multiplayer globalmente e configurar sync de estado do PersonalObject
   311→  useEffect(() => {
   312→    if (!isGameMode) return
   313→    let mp
   314→    import('../../utils/multiplayer/multiplayerManager').then(({ multiplayer }) => {
   315→      mp = multiplayer
   316→      window._multiplayer = mp
   317→      // Configurar getter de estado local (posição do PersonalObject)
   318→      mp.setLocalStateGetter(() => {
   319→        for (const conect of setupScene?.conects || []) {
   320→          if (conect.type === 'PersonalObject') {
   321→            const mesh = conectMeshRefs.current.get(conect.instanceId)
   322→            if (mesh) {
   323→              return {
   324→                position: [mesh.position.x, mesh.position.y, mesh.position.z],
   325→                rotation: [mesh.rotation.x, mesh.rotation.y, mesh.rotation.z],
   326→                playerId: mp.playerId,
   327→              }
   328→            }
   329→          }
   330→        }
   331→        return null
   332→      })
   333→      // Registar callbacks para eventos multiplayer → FlirCode
   334→      mp.on('playerJoin', (playerId) => {
   335→        for (const rt of runtimesRef.current.values()) {
   336→          rt.triggerEvent('onPlayerJoin', { playerId })
   337→        }
   338→      })
   339→      mp.on('playerLeave', (playerId) => {
   340→        for (const rt of runtimesRef.current.values()) {
   341→          rt.triggerEvent('onPlayerLeave', { playerId })
   342→        }
   343→      })
   344→      mp.on('message', (playerId, data) => {
   345→        for (const rt of runtimesRef.current.values()) {
   346→          rt.triggerEvent('onMessage', { playerId, data })
   347→        }
   348→      })
   349→    })
   350→    return () => {
   351→      window._multiplayer = null
   352→    }
   353→  }, [isGameMode, setupScene])
   354→
   355→  // Atualizar refs sem re-disparar o useEffect
   356→  useEffect(() => {
   357→    // Só atualizar o ref se o jogo ainda não começou OU se a cena mudou de verdade
   358→    // (não por spawnObject, mas por mudança de cena manual via changeScene)
   359→    if (!gameStartedRef.current || (activeScene && activeScene.id !== activeSceneRef.current?.id)) {
   360→      activeSceneRef.current = activeScene
   361→      objectsRef.current = objects
   362→    } else {
   363→      // Jogo em curso: manter o snapshot antigo, mas atualizar objects (para spawn)
   364→      objectsRef.current = objects
   365→    }
   366→  }, [activeScene, objects])
   367→
   368→  // Setup quando o modo jogo é activado
   369→  useEffect(() => {
   370→    if (!isGameMode || !setupScene) return
   371→    gameStartedRef.current = true
   372→
   373→    debugLog('Jogo iniciado', 'log', 'Game')
   374→
   375→    // Criar gameContext
   376→    const gameContext = {
   377→      globalVars: { _score: 0 },
   378→      // P2.5: setVar/getVar — expostos no gameContext para o FlirCode
   379→      setVar: (name, value) => { gameContext.globalVars[name] = value },
   380→      getVar: (name) => gameContext.globalVars[name] ?? 0,
   381→      moveObject: (instanceId, direction, speed) => {
   382→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   383→        if (mesh) {
   384→          mesh.position.x += direction[0] * speed * 0.016
   385→          mesh.position.y += direction[1] * speed * 0.016
   386→          mesh.position.z += direction[2] * speed * 0.016
   387→        }
   388→      },
   389→      rotateObject: (instanceId, rotation) => {
   390→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   391→        if (mesh) {
   392→          mesh.rotation.x += THREE.MathUtils.degToRad(rotation[0]) * 0.016
   393→          mesh.rotation.y += THREE.MathUtils.degToRad(rotation[1]) * 0.016
   394→          mesh.rotation.z += THREE.MathUtils.degToRad(rotation[2]) * 0.016
   395→        }
   396→      },
   397→      playAnimation: (instanceId, clip) => {
   398→        const player = animPlayersRef.current.get(instanceId)
   399→        if (player) {
   400→          // Usar blendTime do AnimationBoostObject se existir
   401→          const boost = (setupScene?.conects || []).find((c) => c.type === 'AnimationBoostObject')
   402→          player.play(clip, { loop: true, blendTime: boost?.blendTime ?? 0.3 })
   403→        }
   404→      },
   405→      playSound: (url) => { try { new Audio(url).play() } catch {} },
   406→      playSoundByName: (name) => {
   407→        const sc = (setupScene.conects || []).find((c) => c.type === 'SoundObject' && c.name === name)
   408→        if (sc && sc.url) { try { const a = new Audio(sc.url); a.volume = sc.volume ?? 1; a.loop = sc.loop || false; a.play() } catch {} }
   409→      },
   410→      destroyObject: (instanceId) => {
   411→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   412→        if (mesh) mesh.visible = false
   413→      },
   414→      spawnObject: (objectName, position) => {
   415→        const obj = objectsRef.current.find((o) => o.name === objectName)
   416→        if (obj) useStore.getState().addObjectToScene(obj.id, position)
   417→      },
   418→      changeScene: (sceneName) => {
   419→        const scenes = useStore.getState().scenes
   420→        const target = scenes.find((s) => s.name === sceneName)
   421→        if (target) useStore.getState().setActiveScene(target.id)
   422→      },
   423→      setVisible: (instanceId, visible) => {
   424→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   425→        if (mesh) mesh.visible = visible
   426→      },
   427→      applyForce: (instanceId, force) => physicsRef.current?.applyForce(instanceId, force),
   428→      jumpPlayer: (instanceId) => physicsRef.current?.jumpPersonal(instanceId),
   429→      showUIScreen: (name) => {
   430→        const s = useStore.getState().uiScreens.find((s) => s.name === name)
   431→        if (s) useStore.getState().setUIScreenVisible(s.id, true)
   432→      },
   433→      hideUIScreen: (name) => {
   434→        const s = useStore.getState().uiScreens.find((s) => s.name === name)
   435→        if (s) useStore.getState().setUIScreenVisible(s.id, false)
   436→      },
   437→      getUIValue: (name) => {
   438→        // Procurar em uiScreens
   439→        for (const s of useStore.getState().uiScreens) {
   440→          const el = s.elements.find((e) => e.name === name)
   441→          if (el) return el.value ?? el.text ?? ''
   442→        }
   443→        // Procurar em Conects de UI (TextObject, ButtonObject)
   444→        const scene = useStore.getState().scenes.find((s) => s.id === useStore.getState().activeSceneId)
   445→        const conect = (scene?.conects || []).find((c) => c.name === name && (c.type === 'TextObject' || c.type === 'ButtonObject'))
   446→        if (conect) return conect.text ?? conect.label ?? conect.value ?? ''
   447→        return ''
   448→      },
   449→      setUIValue: (name, val) => {
   450→        // Procurar em uiScreens
   451→        for (const s of useStore.getState().uiScreens) {
   452→          const el = s.elements.find((e) => e.name === name)
   453→          if (el) { useStore.getState().updateUIElement(el.id, { value: val, text: val, label: val }); return }
   454→        }
   455→        // Procurar em Conects de UI (TextObject, ButtonObject)
   456→        const state = useStore.getState()
   457→        const scene = state.scenes.find((s) => s.id === state.activeSceneId)
   458→        if (scene) {
   459→          const conect = (scene.conects || []).find((c) => c.name === name && (c.type === 'TextObject' || c.type === 'ButtonObject'))
   460→          if (conect) {
   461→            state.updateConect(conect.instanceId, { text: String(val), label: String(val) })
   462→            return
   463→          }
   464→        }
   465→      },
   466→      triggerUIEvent: (eventName, payload) => {
   467→        for (const rt of runtimesRef.current.values()) rt.triggerEvent(eventName, payload)
   468→      },
   469→      collidingWith: (instanceId, type) => {
   470→        if (!physicsRef.current) return false
   471→        // Usar contactos reais do cannon-es em vez de distância
   472→        const contacts = physicsRef.current.getContacts?.(instanceId)
   473→        if (contacts && contacts.length > 0) {
   474→          for (const otherId of contacts) {
   475→            const otherEntry = physicsRef.current.bodies.get(otherId)
   476→            if (!otherEntry) continue
   477→            if (otherEntry.conect.type === type || otherEntry.conect.name === type) {
   478→              return true
   479→            }
   480→          }

  1026→    return () => {

  1033→      animPlayersRef.current.clear()

  1121→            }

  1126→        if (!timerStatesRef.current.has(conect.instanceId + '_spawn')) {
  1127→          timerStatesRef.current.set(conect.instanceId + '_spawn', {
  1128→            remaining: conect.interval || 3,
  1129→            duration: conect.interval || 3,
  1130→            loop: true,

  1140→            : conect.position || [0, 1, 0]

  1237→            if (camRotation.enabled) {

  1239→              const offsetY = Math.sin(camRotation.pitch) * orbitDist
  1240→              const offsetX = Math.sin(camRotation.yaw) * Math.cos(camRotation.pitch) * orbitDist
  1241→              const offsetZ = Math.cos(camRotation.yaw) * Math.cos(camRotation.pitch) * orbitDist

  1246→              camera.position.lerp(_camTarget.current, 0.1)
  1247→              camera.lookAt(targetMesh.position)
  1248→            }
  1249→          } else if (mode === 'top') {
  1250→            _camTarget.current.set(targetMesh.position.x, targetMesh.position.y + dist, targetMesh.position.z)
  1251→            camera.position.lerp(_camTarget.current, 0.1)
  1252→            camera.lookAt(targetMesh.position)
  1253→          } else if (mode === 'side') {
  1254→            _camTarget.current.set(targetMesh.position.x + dist, targetMesh.position.y + height / 2, targetMesh.position.z)
  1255→            camera.position.lerp(_camTarget.current, 0.1)
  1256→            camera.lookAt(targetMesh.position)
  1257→          }
  1258→        }
  1259→      } else {
  1260→        camera.position.set(...(activeView.position || [5, 4, 6]))
  1261→        if (camRotation.enabled) {
  1262→          camera.rotation.set(camRotation.pitch, camRotation.yaw, 0, 'YXZ')
  1263→        } else if (activeView.rotation) {
  1264→          camera.rotation.set(...activeView.rotation)
  1265→        } else {
  1266→          camera.lookAt(0, 0, 0)
  1267→        }
  1268→      }
  1269→    } else if (activeScene.gameCamera) {
  1270→      const gc = activeScene.gameCamera
  1271→      const targetFov = gc.fov || 60
  1272→      const targetNear = gc.near || 0.1
  1273→      const targetFar = gc.far || 200
  1274→      if (camera.fov !== targetFov) {
  1275→        camera.fov = targetFov; camera.near = targetNear; camera.far = targetFar
  1276→        camera.updateProjectionMatrix()
  1277→      }
  1278→      camera.position.set(...(gc.position || [5, 4, 6]))
  1279→      if (camRotation.enabled) {
  1280→        camera.rotation.set(camRotation.pitch, camRotation.yaw, 0, 'YXZ')
  1281→      } else if (gc.rotation) {
  1282→        camera.rotation.set(...gc.rotation)
  1283→      } else {
  1284→        camera.lookAt(0, 0, 0)
  1285→      }
  1286→    }
  1287→  })
  1288→
  1289→  return null
  1290→}

  1380→        <div className="empty-state">
  1381→          <div style={{ fontSize: 32, opacity: 0.4 }}></div>
  1382→          <div>Nenhuma cena ativa.</div>
  1383→        </div>
  1384→      </div>
  1385→    )
  1386→  }
  1387→
  1388→  return (
  1389→    <div className="viewport" onDragOver={(e) => !isGameMode && e.preventDefault()} onDrop={handleDrop}>
  1390→      <Canvas
  1391→        shadows
  1392→        dpr={[1, 2]}
  1393→        camera={{ position: [8, 6, 10], fov: 50, near: 0.1, far: 200 }}
  1394→        gl={{ antialias: true, preserveDrawingBuffer: true, alpha: false }}
  1395→        onPointerMissed={() => {
  1396→          if (!isGameMode) {
  1397→            setSelectedInstanceId(null)
  1398→            setSelectedType(null)
  1399→            selectConect(null)
  1400→          }
  1401→        }}
  1402→      >

  1404→          <SceneBackgroundSolid background={background} />
  1405→          <FogApplier conects={activeScene?.conects} />

  1419→            shadow-camera-bottom={-20}

  1426→          )}

  1433→            const sceneObj = { ...obj, id: instance.instanceId, position: instance.position, rotation: instance.rotation, scale: instance.scale }

  1435→            const isPlayer = instance.instanceId === activeScene.playerObjectId

  1447→          ))}
  1448→

  1453→

  1466→              touches={{ ONE: THREE.TOUCH.ROTATE, TWO: THREE.TOUCH.DOLLY_PAN }}
  1467→              enabled={!isTerrainSculptDragging}

  1470→
  1471→          {/* Terrain Sculpt 3D — raycast + cursor + aplicação em tempo real */}

  1473→            activeScene={activeScene}
  1474→            conectMeshRefs={conectMeshRefs}

  1480→          <InstancingRenderer />
  1481→
  1482→          {/* ViewModel FPS — arma parented à câmara em first/third-person */}
  1483→          <ViewModelFPS activeScene={activeScene} isGameMode={isGameMode} />
  1484→
  1485→          {/* GameMode — activado quando scenePreviewOpen */}
  1486→          <GameMode activeScene={activeScene} objects={objects} meshRefs={meshRefs} conectMeshRefs={conectMeshRefs} isGameMode={isGameMode} />
  1487→        </Suspense>
  1488→      </Canvas>
  1489→    </div>
  1490→  )
