     1→/**
     2→ * ConectRenderer — renderiza um Conect na cena 3D.
     3→ *
     4→ * Dependendo do tipo de Conect:
     5→ *  - VisualObject: renderiza o modelo do catálogo (sourceObjectId)
     6→ *  - RigidObject/StaticObject/StopObject: renderiza um cubo placeholder + física
     7→ *  - PersonalObject: renderiza um capsule/cubo como jogador
     8→ *  - TriggerObject: renderiza um wireframe transparente (não visível no jogo)
     9→ *  - LuminousObject: adiciona uma luz à cena
    10→ *  - ParticleObject: sistema de partículas
    11→ *  - TerrainObject: terreno com heightmap
    12→ *  - WaterObject: plano de água
    13→ *  - PathObject: renderiza waypoints
    14→ *  - CheckpointObject: renderiza uma bandeira/marca
    15→ *  - Outros (UI, Sound, Sky, Fog, View, etc.): sem visual 3D (só lógica)
    16→ */
    17→import { forwardRef, useMemo, useEffect, useRef } from 'react'
    18→import { useThree, useFrame } from '@react-three/fiber'
    19→import * as THREE from 'three'
    20→import { useStore } from '../../store/useStore'
    21→import { findConectDefinition } from '../../utils/conects/taxonomy'
    22→import SceneObject from '../3d/SceneObject'
    23→
    24→const ConectRenderer = forwardRef(function ConectRenderer({ conect, objects, setMeshRef }, meshRef) {
    25→  const def = findConectDefinition(conect.type)
    26→
    27→  // Se não tem visual, não renderiza mesh — mas pode adicionar luzes, etc.
    28→  if (conect.type === 'LuminousObject') {
    29→    return <LuminousMesh conect={conect} setMeshRef={setMeshRef} />
    30→  }
    31→
    32→  if (conect.type === 'TerrainObject') {
    33→    return <TerrainMesh conect={conect} setMeshRef={setMeshRef} />
    34→  }
    35→
    36→  if (conect.type === 'WaterObject') {
    37→    return <WaterMesh conect={conect} setMeshRef={setMeshRef} />
    38→  }
    39→
    40→  if (conect.type === 'PathObject') {
    41→    return <PathMesh conect={conect} setMeshRef={setMeshRef} />
    42→  }
    43→
    44→  if (conect.type === 'CheckpointObject') {
    45→    return <CheckpointMesh conect={conect} setMeshRef={setMeshRef} />
    46→  }
    47→
    48→  if (conect.type === 'ParticleObject') {
    49→    return <ParticleMesh conect={conect} setMeshRef={setMeshRef} />
    50→  }
    51→
    52→  // ViewObject: renderiza um gizmo de câmara visível no editor (selecionável)
    53→  if (conect.type === 'ViewObject') {
    54→    return <ViewObjectMesh conect={conect} setMeshRef={setMeshRef} />
    55→  }
    56→
    57→  // TrailObject: rasto visual atrás de um objeto em movimento
    58→  if (conect.type === 'TrailObject') {
    59→    return <TrailMesh conect={conect} setMeshRef={setMeshRef} />
    60→  }
    61→
    62→  // ReflectObject: sonda de reflexo (renderiza uma esfera espelhada)
    63→  if (conect.type === 'ReflectObject') {
    64→    return <ReflectMesh conect={conect} setMeshRef={setMeshRef} />
    65→  }
    66→
    67→  // SkyObject: renderiza esfera de céu com shader procedural
    68→  if (conect.type === 'SkyObject') {
    69→    return <SkyMesh conect={conect} setMeshRef={setMeshRef} />
    70→  }
    71→
    72→  // SunObject: luz direcional (sol)
    73→  if (conect.type === 'SunObject') {
    74→    return <SunLightMesh conect={conect} setMeshRef={setMeshRef} />
    75→  }
    76→
    77→  // PointObject: luz pontual
    78→  if (conect.type === 'PointObject') {
    79→    return <PointLightMesh conect={conect} setMeshRef={setMeshRef} />
    80→  }
    81→
    82→  // SpotObject: luz spot
    83→  if (conect.type === 'SpotObject') {
    84→    return <SpotLightMesh conect={conect} setMeshRef={setMeshRef} />
    85→  }
    86→
    87→  // AmbientObject: luz ambiente (sem gizmo, só aplica à cena)
    88→  if (conect.type === 'AmbientObject') {
    89→    return <AmbientLightMesh conect={conect} setMeshRef={setMeshRef} />
    90→  }
    91→
    92→  // ReferenceObject: renderiza conteúdo de outra cena
    93→  if (conect.type === 'ReferenceObject') {
    94→    return <ReferenceMesh conect={conect} setMeshRef={setMeshRef} />
    95→  }
    96→
    97→  if (!def?.hasVisual && conect.type !== 'VisualObject') {
    98→    // Sem visual — ligar o meshRef a null para que physics não tente usar
    99→    useEffect(() => { setMeshRef?.(null) }, [])
   100→    return null
   101→  }
   102→
   103→  // VisualObject: usa modelo do catálogo
   104→  if (conect.type === 'VisualObject') {
   105→    const obj = objects.find((o) => o.id === conect.sourceObjectId)
   106→    if (!obj) return null
   107→    const sceneObj = {
   108→      ...obj,
   109→      id: conect.instanceId,
   110→      position: conect.position,
   111→      rotation: conect.rotation,
   112→      scale: conect.scale,
   113→    }
   114→    return (
   115→      <SceneObject
   116→        ref={(node) => setMeshRef?.(node)}
   117→        obj={sceneObj}
   118→        isSelected={false}
   119→        onSelect={() => {}}
   120→      />
   121→    )
   122→  }
   123→
   124→  // Rigid/Static/Stop/Personal: placeholder com geometria simples
   125→  return (
   126→    <PlaceholderMesh
   127→      ref={(node) => {
   128→        if (typeof meshRef === 'function') meshRef(node)
   129→        else if (meshRef) meshRef.current = node
   130→        setMeshRef?.(node)
   131→      }}
   132→      conect={conect}
   133→    />
   134→  )
   135→})
   136→
   137→export default ConectRenderer

   139→// ===== Placeholder para conects com física =====

   142→                : conect.type === 'StaticObject' ? '#6e7681'
   143→                : conect.type === 'StopObject' ? '#d29922'

   151→      position={conect.position}
   152→      rotation={conect.rotation}
   153→      scale={conect.scale}
   154→      visible={conect.visible !== false}
   155→      castShadow
   156→      receiveShadow
   157→      userData={{ conectInstanceId: conect.instanceId }}
   158→    >
   159→      {geometry}
   160→      <meshStandardMaterial color={color} roughness={0.6} metalness={0.1} />
   161→    </mesh>
   162→  )

   165→// ===== LuminousObject =====
   166→// Luz real + gizmo visual (esfera amarela) para seleção no editor
   167→function LuminousMesh({ conect, setMeshRef }) {
   168→  const lightProps = {
   169→    intensity: conect.intensity,
   170→    color: conect.color,
   171→    castShadow: conect.castShadow,
   172→  }
   173→  let lightElement
   174→  if (conect.lightType === 'directional') {
   175→    lightElement = <directionalLight {...lightProps} position={[0, 0, 0]} />
   176→  } else if (conect.lightType === 'spot') {
   177→    lightElement = <spotLight {...lightProps} distance={conect.distance} angle={Math.PI / 4} penumbra={0.3} position={[0, 0, 0]} />
   178→  } else {
   179→    lightElement = <pointLight {...lightProps} distance={conect.distance} position={[0, 0, 0]} />
   180→  }
   181→  return (
   182→    <group
   183→      ref={setMeshRef}
   184→      position={conect.position}
   185→      rotation={conect.rotation}
   186→      userData={{ conectInstanceId: conect.instanceId }}
   187→    >
   188→      {lightElement}
   189→      {/* Gizmo visual: esfera amarela para seleção */}
   190→      <mesh>
   191→        <sphereGeometry args={[0.2, 12, 12]} />
   192→        <meshBasicMaterial color="#ffd700" />
   193→      </mesh>
   194→      {/* Halo ao redor */}
   195→      <mesh>
   196→        <sphereGeometry args={[0.35, 12, 12]} />
   197→        <meshBasicMaterial color="#ffd700" transparent opacity={0.2} />
   198→      </mesh>
   199→    </group>
   200→  )
   201→}

   203→// ===== TerrainObject =====
   204→function TerrainMesh({ conect, setMeshRef }) {
   205→  const geometry = useMemo(() => {
   206→    const seg = conect.segments || 64
   207→    const g = new THREE.PlaneGeometry(
   208→      conect.width || 50, conect.depth || 50, seg, seg
   209→    )
   210→    g.rotateX(-Math.PI / 2)
   211→    const positions = g.attributes.position
   212→    const heightScale = conect.heightScale || 5
   213→
   214→    // Se há heightmap exportado do TerrainEditor, usá-lo
   215→    if (conect.heightmap && conect.heightmap.length > 0) {
   216→      const hm = conect.heightmap
   217→      for (let i = 0; i < positions.count && i < hm.length; i++) {
   218→        positions.setY(i, hm[i] * heightScale)
   219→      }
   220→    } else {
   221→      // Fallback: gerar procedural
   222→      const seed = conect.heightmapSeed || 1
   223→      for (let i = 0; i < positions.count; i++) {
   224→        const x = positions.getX(i)
   225→        const z = positions.getZ(i)
   226→        const h =
   227→          Math.sin(x * 0.3 + seed) * Math.cos(z * 0.3 + seed) * 0.5 +
   228→          Math.sin(x * 0.1 + z * 0.1) * 0.3
   229→        positions.setY(i, h * heightScale)
   230→      }
   231→    }
   232→    g.computeVertexNormals()
   233→
   234→    // Vertex colors from splatmap (multi-layer blending) — P7
   235→    if (conect.splatmap && conect.textureLayers && conect.splatmap.length > 0) {
   236→      const maxLayers = conect.maxLayers || 4
   237→      const layers = conect.textureLayers
   238→      const colors = new Float32Array(positions.count * 3)
   239→      const sm = conect.splatmap
   240→      const hexToRgb = (hex) => {
   241→        const r = parseInt(hex.slice(1, 3), 16) / 255
   242→        const g = parseInt(hex.slice(3, 5), 16) / 255
   243→        const b = parseInt(hex.slice(5, 7), 16) / 255
   244→        return [r, g, b]
   245→      }
   246→      const layerRgb = layers.map((l) => hexToRgb(l.color || '#5a7d3a'))
   247→      for (let i = 0; i < positions.count; i++) {
   248→        let r = 0, g = 0, b = 0
   249→        for (let l = 0; l < maxLayers && l < layers.length; l++) {
   250→          const w = sm[i * maxLayers + l] || 0
   251→          const c = layerRgb[l] || [0.35, 0.49, 0.23]
   252→          r += c[0] * w
   253→          g += c[1] * w
   254→          b += c[2] * w
   255→        }
   256→        colors[i * 3] = r
   257→        colors[i * 3 + 1] = g
   258→        colors[i * 3 + 2] = b
   259→      }
   260→      g.setAttribute('color', new THREE.BufferAttribute(colors, 3))
   261→    }
   262→
   263→    return g
   264→  }, [conect.width, conect.depth, conect.segments, conect.heightScale, conect.heightmapSeed, conect.heightmap, conect.splatmap, conect.textureLayers, conect.maxLayers])
   265→
   266→  const hasVertexColors = !!(conect.splatmap && conect.textureLayers && conect.splatmap.length > 0)
   267→
   268→  return (
   269→    <mesh
   270→      ref={setMeshRef}
   271→      geometry={geometry}
   272→      position={conect.position}
   273→      receiveShadow
   274→      castShadow
   275→    >
   276→      <meshStandardMaterial
   277→        color={hasVertexColors ? '#ffffff' : (conect.textureColor || '#5a7d3a')}
   278→        vertexColors={hasVertexColors}
   279→        roughness={0.9}
   280→        metalness={0}
   281→      />
   282→    </mesh>
   283→  )
   284→}

   286→// ===== WaterObject =====
   287→function WaterMesh({ conect, setMeshRef }) {
   288→  const meshRef = useRef()
   289→  const time = useRef(0)
   290→
   291→  const geometry = useMemo(() => {
   292→    const [w, h] = conect.size || [20, 20]
   293→    const g = new THREE.PlaneGeometry(w, h, 32, 32)
   294→    g.rotateX(-Math.PI / 2)
   295→    return g
   296→  }, [conect.size])
   297→
   298→  // Animar ondas no useFrame
   299→  useFrame((_, delta) => {
   300→    if (!meshRef.current) return
   301→    time.current += delta * (conect.waveSpeed || 0.5)
   302→    const pos = meshRef.current.geometry.attributes.position
   303→    const waveHeight = conect.waveHeight || 0.1
   304→    for (let i = 0; i < pos.count; i++) {
   305→      const x = pos.getX(i)
   306→      const z = pos.getZ(i)
   307→      const y = Math.sin(x * 0.5 + time.current) * waveHeight
   308→        + Math.cos(z * 0.5 + time.current * 0.7) * waveHeight * 0.5
   309→      pos.setY(i, y)
   310→    }
   311→    pos.needsUpdate = true
   312→    meshRef.current.geometry.computeVertexNormals()
   313→  })
   314→
   315→  return (
   316→    <mesh
   317→      ref={(node) => {
   318→        meshRef.current = node
   319→        if (typeof setMeshRef === 'function') setMeshRef(node)
   320→      }}
   321→      position={conect.position}
   322→      geometry={geometry}
   323→    >
   324→      <meshStandardMaterial
   325→        color={conect.color || '#2f81f7'}
   326→        transparent
   327→        opacity={conect.opacity ?? 0.6}
   328→        roughness={0.1}
   329→        metalness={0.3}
   330→      />
   331→    </mesh>
   332→  )
   333→}

   335→// ===== PathObject =====
   336→function PathMesh({ conect, setMeshRef }) {
   337→  if (!conect.points || conect.points.length === 0) return null
   338→  return (
   339→    <group ref={setMeshRef} position={conect.position}>
   340→      {conect.points.map((p, i) => (
   341→        <mesh key={i} position={p}>
   342→          <sphereGeometry args={[0.15, 12, 12]} />
   343→          <meshBasicMaterial color="#2f81f7" />
   344→        </mesh>
   345→      ))}
   346→      {/* Linhas entre pontos */}
   347→      {conect.points.length > 1 && (
   348→        <Line points={conect.points} loop={conect.loop} />
   349→      )}
   350→    </group>
   351→  )
   352→}
   353→
   354→function Line({ points, loop }) {
   355→  const geometry = useMemo(() => {
   356→    const pts = loop ? [...points, points[0]] : points
   357→    return new THREE.BufferGeometry().setFromPoints(pts.map((p) => new THREE.Vector3(...p)))
   358→  }, [points, loop])
   359→  return (
   360→    <line geometry={geometry}>
   361→      <lineBasicMaterial color="#2f81f7" transparent opacity={0.5} />
   362→    </line>
   363→  )
   364→}

   382→// ===== ParticleObject =====
   383→function ParticleMesh({ conect, setMeshRef }) {
   384→  const groupRef = useRef()
   385→  const timeRef = useRef(0)
   386→
   387→  // Usar BufferGeometry + Points para performance (1 draw call em vez de N)
   388→  const count = Math.min(conect.maxParticles || 100, 300)
   389→  const positionsRef = useRef(new Float32Array(count * 3))
   390→  const velocitiesRef = useRef(new Float32Array(count * 3))
   391→  const lifeRef = useRef(new Float32Array(count))
   392→
   393→  // Inicializar partículas
   394→  useMemo(() => {
   395→    const spread = conect.spread || 1
   396→    const speed = conect.particleSpeed || 0.5
   397→    for (let i = 0; i < count; i++) {
   398→      positionsRef.current[i * 3] = (Math.random() - 0.5) * spread
   399→      positionsRef.current[i * 3 + 1] = 0
   400→      positionsRef.current[i * 3 + 2] = (Math.random() - 0.5) * spread
   401→      velocitiesRef.current[i * 3] = (Math.random() - 0.5) * speed
   402→      velocitiesRef.current[i * 3 + 1] = Math.random() * speed + 0.2
   403→      velocitiesRef.current[i * 3 + 2] = (Math.random() - 0.5) * speed
   404→      lifeRef.current[i] = Math.random() * (conect.particleLife || 2)
   405→    }
   406→  }, [count, conect.spread, conect.particleSpeed, conect.particleLife])
   407→
   408→  const geometry = useMemo(() => {
   409→    const geo = new THREE.BufferGeometry()
   410→    geo.setAttribute('position', new THREE.BufferAttribute(positionsRef.current, 3))
   411→    return geo
   412→  }, [])
   413→
   414→  const material = useMemo(() => {
   415→    return new THREE.PointsMaterial({
   416→      color: conect.color || '#ffffff',
   417→      size: conect.particleSize || 0.1,
   418→      transparent: true,
   419→      opacity: 0.7,
   420→      sizeAttenuation: true,
   421→      depthWrite: false,
   422→    })
   423→  }, [conect.color, conect.particleSize])
   424→
   425→  // Animar partículas no useFrame
   426→  useFrame((_, delta) => {
   427→    timeRef.current += delta
   428→    const positions = positionsRef.current
   429→    const velocities = velocitiesRef.current
   430→    const life = lifeRef.current
   431→    const gravity = conect.gravity ?? -0.5
   432→    const maxLife = conect.particleLife || 2
   433→    const spread = conect.spread || 1
   434→
   435→    for (let i = 0; i < count; i++) {
   436→      // Atualizar vida
   437→      life[i] += delta
   438→      if (life[i] >= maxLife) {
   439→        // Reciclar partícula
   440→        positions[i * 3] = (Math.random() - 0.5) * spread
   441→        positions[i * 3 + 1] = 0
   442→        positions[i * 3 + 2] = (Math.random() - 0.5) * spread
   443→        velocities[i * 3] = (Math.random() - 0.5) * (conect.particleSpeed || 0.5)
   444→        velocities[i * 3 + 1] = Math.random() * (conect.particleSpeed || 0.5) + 0.2
   445→        velocities[i * 3 + 2] = (Math.random() - 0.5) * (conect.particleSpeed || 0.5)
   446→        life[i] = 0
   447→      } else {
   448→        // Mover partícula
   449→        positions[i * 3] += velocities[i * 3] * delta
   450→        positions[i * 3 + 1] += velocities[i * 3 + 1] * delta
   451→        positions[i * 3 + 2] += velocities[i * 3 + 2] * delta
   452→        // Aplicar gravidade
   453→        velocities[i * 3 + 1] += gravity * delta
   454→      }
   455→    }
   456→    geometry.attributes.position.needsUpdate = true
   457→  })
   458→
   459→  // Cleanup
   460→  useEffect(() => {
   461→    return () => {
   462→      geometry.dispose()
   463→      material.dispose()
   464→    }
   465→  }, [geometry, material])
   466→
   467→  return (
   468→    <points
   469→      ref={(node) => {
   470→        groupRef.current = node
   471→        if (typeof setMeshRef === 'function') setMeshRef(node)
   472→      }}
   473→      position={conect.position}
   474→      geometry={geometry}
   475→      material={material}
   476→    />
   477→  )
   478→}

   528→// ===== TrailObject (rasto visual) =====
   529→// Renderiza uma linha que segue a posição do objeto pai.
   530→// Em tempo de execução, o GameRunner atualiza as posições do trail.
   531→function TrailMesh({ conect, setMeshRef }) {
   532→  const groupRef = useRef()
   533→  const lineRef = useRef()
   534→  const maxPoints = conect.length || 30
   535→  const positionsRef = useRef(new Float32Array(maxPoints * 3))
   536→  const currentIndexRef = useRef(0)
   537→  const frameCountRef = useRef(0)
   538→
   539→  // Inicializar todas as posições na posição inicial do trail
   540→  useMemo(() => {
   541→    const px = conect.position?.[0] || 0
   542→    const py = conect.position?.[1] || 0
   543→    const pz = conect.position?.[2] || 0
   544→    for (let i = 0; i < maxPoints; i++) {
   545→      positionsRef.current[i * 3] = px
   546→      positionsRef.current[i * 3 + 1] = py
   547→      positionsRef.current[i * 3 + 2] = pz
   548→    }
   549→  }, [maxPoints, conect.position])
   550→
   551→  const geometry = useMemo(() => {
   552→    const geo = new THREE.BufferGeometry()
   553→    geo.setAttribute('position', new THREE.BufferAttribute(positionsRef.current, 3))
   554→    return geo
   555→  }, [])
   556→
   557→  // Seguir o followTarget e atualizar o trail
   558→  useFrame(() => {
   559→    if (!conect.followTarget) return
   560→    // Procurar o mesh do followTarget nos conectMeshRefs
   561→    const targetMesh = window._flirConectMeshRefs?.get(conect.followTarget)
   562→      || window._flirMeshRefs?.get(conect.followTarget)
   563→    if (!targetMesh) return
   564→
   565→    // Atualizar a cada N frames (não todos os frames — performance)
   566→    const updateInterval = Math.max(1, Math.floor(60 / (conect.updateRate || 30)))
   567→    frameCountRef.current++
   568→    if (frameCountRef.current < updateInterval) return
   569→    frameCountRef.current = 0
   570→
   571→    // Shift: mover todos os pontos uma posição para trás
   572→    const positions = positionsRef.current
   573→    for (let i = maxPoints - 1; i > 0; i--) {
   574→      positions[i * 3] = positions[(i - 1) * 3]
   575→      positions[i * 3 + 1] = positions[(i - 1) * 3 + 1]
   576→      positions[i * 3 + 2] = positions[(i - 1) * 3 + 2]
   577→    }
   578→    // Novo ponto na posição do target
   579→    positions[0] = targetMesh.position.x
   580→    positions[1] = targetMesh.position.y
   581→    positions[2] = targetMesh.position.z
   582→
   583→    geometry.attributes.position.needsUpdate = true
   584→    geometry.setDrawRange(0, maxPoints)
   585→  })
   586→
   587→  // Cleanup
   588→  useEffect(() => {
   589→    return () => { geometry.dispose() }
   590→  }, [geometry])
   591→
   592→  return (
   593→    <group ref={setMeshRef} position={[0, 0, 0]}>
   594→      <line ref={lineRef} geometry={geometry}>
   595→        <lineBasicMaterial
   596→          color={conect.color || '#2f81f7'}
   597→          transparent
   598→          opacity={conect.fade !== false ? 0.6 : 1}
   599→          linewidth={conect.width || 2}
   600→        />
   601→      </line>
   602→    </group>
   603→  )
   604→}

   623→// ===== SkyObject — esfera de céu com shader procedural =====
   624→function SkyMesh({ conect, setMeshRef }) {
   625→  const { scene } = useThree()
   626→
   627→  // Aplicar background à cena
   628→  useEffect(() => {
   629→    if (!scene) return
   630→    if (conect.skyType === 'procedural' || conect.skyType === 'gradient') {
   631→      // Gradiente via canvas
   632→      const canvas = document.createElement('canvas')
   633→      canvas.width = 2; canvas.height = 256
   634→      const ctx = canvas.getContext('2d')
   635→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
   636→      grad.addColorStop(0, conect.gradientTop || '#4a90d9')
   637→      grad.addColorStop(1, conect.gradientBottom || '#b0d4f1')
   638→      ctx.fillStyle = grad
   639→      ctx.fillRect(0, 0, 2, 256)
   640→      const tex = new THREE.CanvasTexture(canvas)
   641→      tex.colorSpace = THREE.SRGBColorSpace
   642→      scene.background = tex
   643→      return () => { tex.dispose() }
   644→    } else {
   645→      scene.background = new THREE.Color(conect.color || '#87CEEB')
   646→    }
   647→  }, [scene, conect.skyType, conect.color, conect.gradientTop, conect.gradientBottom])
   648→
   649→  // Esfera grande com material de céu (BackSide)
   650→  return (
   651→    <mesh ref={setMeshRef} scale={[100, 100, 100]}>
   652→      <sphereGeometry args={[1, 32, 16]} />
   653→      <meshBasicMaterial
   654→        color={conect.color || '#87CEEB'}
   655→        side={THREE.BackSide}
   656→        fog={false}
   657→      />
   658→    </mesh>
   659→  )
   660→}

   737→// ===== ReferenceObject — renderiza conteúdo de outra cena =====
   738→function ReferenceMesh({ conect, setMeshRef }) {
   739→  const scenes = useStore((s) => s.scenes)
   740→  const targetScene = scenes.find(s => s.id === conect.targetSceneId)
   741→
   742→  if (!targetScene) {
   743→    // Sem cena de destino — mostrar gizmo de placeholder
   744→    return (
   745→      <mesh ref={setMeshRef} position={conect.position} rotation={conect.rotation} scale={conect.scale}>
   746→        <boxGeometry args={[0.5, 0.5, 0.5]} />
   747→        <meshBasicMaterial color="#8957e5" wireframe />
   748→      </mesh>
   749→    )
   750→  }
   751→
   752→  // Renderizar objetos do catálogo da cena referenciada
   753→  const refObjects = (targetScene.objects || []).map((instance) => {
   754→    const obj = objects.find((o) => o.id === instance.objectId)
   755→    if (!obj) return null
   756→    const sceneObj = {
   757→      ...obj,
   758→      id: `${conect.instanceId}_ref_${instance.instanceId}`,
   759→      position: [
   760→        (conect.position?.[0] || 0) + (instance.position?.[0] || 0),
   761→        (conect.position?.[1] || 0) + (instance.position?.[1] || 0),
   762→        (conect.position?.[2] || 0) + (instance.position?.[2] || 0),
   763→      ],
   764→      rotation: instance.rotation || [0, 0, 0],
   765→      scale: instance.scale || [1, 1, 1],
   766→    }
   767→    return <SceneObject key={instance.instanceId} obj={sceneObj} isSelected={false} onSelect={() => {}} />
   768→  }).filter(Boolean)
   769→
   770→  return (
   771→    <group ref={setMeshRef} position={conect.position} rotation={conect.rotation} scale={conect.scale}>
   772→      {refObjects}
   773→    </group>
   774→  )
   775→}
