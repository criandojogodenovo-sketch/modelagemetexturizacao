    48→function snapshot(objects) {
    49→  // Clona profundamente mas descarta bufferGeometry (não serializável)
    50→  return JSON.parse(JSON.stringify(objects, (key, value) => {
    51→    if (key === 'bufferGeometry' && value && typeof value === 'object') return undefined
    52→    return value
    53→  }))
    54→}

   494→      // ---------- Operações de malha (edit mode) ----------
   495→      // Aplica uma operação de malha e guarda o resultado em customGeometry
   496→      applyMeshOp: (id, op, params = {}) => {
   497→        const obj = get().objects.find((o) => o.id === id)
   498→        if (!obj) return
   499→        get()._pushHistory()
   500→
   501→        // Construir geometria atual
   502→        let geometry
   503→        if (obj.customGeometry) {
   504→          geometry = new THREE.BufferGeometry()
   505→          geometry.setAttribute('position', new THREE.Float32BufferAttribute(obj.customGeometry.positions, 3))
   506→          if (obj.customGeometry.normals) {
   507→            geometry.setAttribute('normal', new THREE.Float32BufferAttribute(obj.customGeometry.normals, 3))
   508→          }
   509→          if (obj.customGeometry.uvs) {
   510→            geometry.setAttribute('uv', new THREE.Float32BufferAttribute(obj.customGeometry.uvs, 2))
   511→          }
   512→        } else if (obj.imported && obj.bufferGeometry) {
   513→          geometry = obj.bufferGeometry
   514→        } else {
   515→          const def = PRIMITIVES[obj.type]
   516→          geometry = def ? def.build(THREE, obj.args) : new THREE.BoxGeometry(1, 1, 1)
   517→        }
   518→
   519→        let result = geometry
   520→        switch (op) {
   521→          case 'subdivide':
   522→            result = meshOps.subdivide(geometry, params.levels || 1)
   523→            break
   524→          case 'extrude':
   525→            result = meshOps.extrudeFaces(geometry, params.amount ?? 0.5)
   526→            break
   527→          case 'inset':
   528→            result = meshOps.insetFaces(geometry, params.amount ?? 0.1)
   529→            break
   530→          case 'bevel':
   531→            result = meshOps.bevelGeometry(geometry, params.radius ?? 0.05, params.segments ?? 2)
   532→            break
   533→          case 'loopCut':
   534→            result = meshOps.loopCut(geometry, params.axis || 'y', params.position || 0)
   535→            break
   536→          case 'merge':
   537→            result = meshOps.mergeVertices(geometry, params.threshold ?? 0.001)
   538→            break
   539→          case 'unwrap':
   540→            result = meshOps.unwrapUV(geometry, params.method || 'planar')
   541→            break
   542→          default:
   543→            get().toast(`Operação desconhecida: ${op}`, 'error')
   544→            return
   545→        }
   546→
   547→        // Serializar para o store
   548→        const positions = result.getAttribute('position').array
   549→        const normals = result.getAttribute('normal')?.array
   550→        const uvs = result.getAttribute('uv')?.array
   551→        set((s) => ({
   552→          objects: s.objects.map((o) =>
   553→            o.id === id
   554→              ? {
   555→                  ...o,
   556→                  customGeometry: {
   557→                    positions: Array.from(positions),
   558→                    normals: normals ? Array.from(normals) : null,
   559→                    uvs: uvs ? Array.from(uvs) : null,
   560→                  },
   561→                }
   562→              : o
   563→          ),
   564→        }))
   565→        get().toast(`Operação "${op}" aplicada`, 'success')
   566→      },
   567→
   568→      // ---------- Booleanas entre objetos ----------
   569→      applyBooleanOp: (targetId, toolId, op) => {
   570→        const objects = get().objects
   571→        const target = objects.find((o) => o.id === targetId)
   572→        const tool = objects.find((o) => o.id === toolId)
   573→        if (!target || !tool) return
   574→        get()._pushHistory()
   575→
   576→        const buildGeo = (obj) => {
   577→          if (obj.customGeometry) {
   578→            const g = new THREE.BufferGeometry()
   579→            g.setAttribute('position', new THREE.Float32BufferAttribute(obj.customGeometry.positions, 3))
   580→            if (obj.customGeometry.normals) {
   581→              g.setAttribute('normal', new THREE.Float32BufferAttribute(obj.customGeometry.normals, 3))
   582→            }
   583→            return g
   584→          }
   585→          if (obj.imported && obj.bufferGeometry) return obj.bufferGeometry
   586→          const def = PRIMITIVES[obj.type]
   587→          return def ? def.build(THREE, obj.args) : new THREE.BoxGeometry(1, 1, 1)
   588→        }
   589→
   590→        const geoA = buildGeo(target)
   591→        const geoB = buildGeo(tool)
   592→        // Aplicar transform do tool ao geoB
   593→        geoB.applyMatrix4(new THREE.Matrix4().compose(
   594→          new THREE.Vector3(...tool.position),
   595→          new THREE.Quaternion().setFromEuler(new THREE.Euler(...tool.rotation)),
   596→          new THREE.Vector3(...tool.scale)
   597→        ))
   598→        const result = meshOps.booleanOp(geoA, geoB, op)
   599→
   600→        const positions = result.getAttribute('position').array
   601→        const normals = result.getAttribute('normal')?.array
   602→        set((s) => ({
   603→          objects: s.objects.map((o) =>
   604→            o.id === targetId
   605→              ? {
   606→                  ...o,
   607→                  customGeometry: {
   608→                    positions: Array.from(positions),
   609→                    normals: normals ? Array.from(normals) : null,
   610→                    uvs: null,
   611→                  },
   612→                }
   613→              : o
   614→          ),
   615→        }))
   616→        // Remover o objeto ferramenta
   617→        set((s) => ({
   618→          objects: s.objects.filter((o) => o.id !== toolId),
   619→          selectedId: targetId,
   620→        }))
   621→        get().toast(`Booleana "${op}" aplicada`, 'success')
   622→      },
