     2→ * ConectRenderer — renderiza um Conect na cena 3D.
     3→ *
     4→ * Dependendo do tipo de Conect:
     5→ *  - VisualObject: renderiza o modelo do catálogo (sourceObjectId)
     6→ *  - RigidObject/StaticObject/StopObject: renderiza um cubo placeholder + física
     7→ *  - PersonalObject: renderiza um capsule/cubo como jogador

    13→ *  - PathObject: renderiza waypoints
    14→ *  - CheckpointObject: renderiza uma bandeira/marca
    15→ *  - Outros (UI, Sound, Sky, Fog, View, etc.): sem visual 3D (só lógica)
    16→ */

    26→// ===== Helper: converter temperatura Kelvin para cor RGB =====
    27→// Baseado em http://www.tannerhelland.com/4435/convert-temperature-rgb-algorithm-code/
    28→function kelvinToRGB(kelvin) {
    29→  const temp = kelvin / 100
    30→  let r, g, b
    31→  if (temp <= 66) {
    32→    r = 255
    33→    g = 99.4708025861 * Math.log(temp) - 161.1195681661
    34→    b = temp <= 19 ? 0 : 138.5177312231 * Math.log(temp - 10) - 305.0447927307
    35→  } else {
    36→    r = 329.698727446 * Math.pow(temp - 60, -0.1332047592)
    37→    g = 288.1221695283 * Math.pow(temp - 60, -0.0755148492)
    38→    b = 255
    39→  }
    40→  r = Math.max(0, Math.min(255, r)) / 255
    41→  g = Math.max(0, Math.min(255, g)) / 255
    42→  b = Math.max(0, Math.min(255, b)) / 255
    43→  return new THREE.Color(r, g, b)
    44→}

    46→const ConectRenderer = forwardRef(function ConectRenderer({ conect, objects, setMeshRef }, meshRef) {
    47→  const def = findConectDefinition(conect.type)
    48→
    49→  // Se não tem visual, não renderiza mesh — mas pode adicionar luzes, etc.
    50→  if (conect.type === 'LuminousObject') {
    51→    return <LuminousMesh conect={conect} setMeshRef={setMeshRef} />
    52→  }
    53→
    54→  // Novos tipos de luz (expandir Conects de Luz)
    55→  if (conect.type === 'SunObject') {
    56→    return <SunMesh conect={conect} setMeshRef={setMeshRef} />
    57→  }
    58→  if (conect.type === 'PointObject') {
    59→    return <PointLightMesh conect={conect} setMeshRef={setMeshRef} />
    60→  }
    61→  if (conect.type === 'SpotObject') {
    62→    return <SpotLightMesh conect={conect} setMeshRef={setMeshRef} />
    63→  }
    64→  if (conect.type === 'AreaObject') {
    65→    return <AreaLightMesh conect={conect} setMeshRef={setMeshRef} />
    66→  }
    67→  if (conect.type === 'AmbientObject') {
    68→    return <AmbientLightMesh conect={conect} setMeshRef={setMeshRef} />
    69→  }
    70→
    71→  // SkyObject agora tem visual (céu procedural, HDRI, gradient, solid)
    72→  if (conect.type === 'SkyObject') {
    73→    return <SkyMesh conect={conect} setMeshRef={setMeshRef} />
    74→  }
    75→
    76→  if (conect.type === 'TerrainObject') {
    77→    return <TerrainMesh conect={conect} setMeshRef={setMeshRef} />
    78→  }
    79→
    80→  if (conect.type === 'WaterObject') {
    81→    return <WaterMesh conect={conect} setMeshRef={setMeshRef} />
    82→  }
    83→
    84→  if (conect.type === 'PathObject') {
    85→    return <PathMesh conect={conect} setMeshRef={setMeshRef} />
    86→  }
    87→
    88→  if (conect.type === 'CheckpointObject') {
    89→    return <CheckpointMesh conect={conect} setMeshRef={setMeshRef} />
    90→  }
    91→
    92→  if (conect.type === 'ParticleObject') {
    93→    return <ParticleMesh conect={conect} setMeshRef={setMeshRef} />
    94→  }
    95→
    96→  // ViewObject: renderiza um gizmo de câmara visível no editor (selecionável)
    97→  if (conect.type === 'ViewObject') {
    98→    return <ViewObjectMesh conect={conect} setMeshRef={setMeshRef} />
    99→  }
   100→
   101→  // TrailObject: rasto visual atrás de um objeto em movimento
   102→  if (conect.type === 'TrailObject') {
   103→    return <TrailMesh conect={conect} setMeshRef={setMeshRef} />
   104→  }
   105→
   106→  // ReflectObject: sonda de reflexo (renderiza uma esfera espelhada)
   107→  if (conect.type === 'ReflectObject') {
   108→    return <ReflectMesh conect={conect} setMeshRef={setMeshRef} />
   109→  }
   110→
   111→  if (!def?.hasVisual && conect.type !== 'VisualObject') {
   112→    // Sem visual — ligar o meshRef a null para que physics não tente usar
   113→    useEffect(() => { setMeshRef?.(null) }, [])
   114→    return null
   115→  }
   116→
   117→  // VisualObject, PersonalObject e NpcObject com sourceObjectId:
   118→  // usam o modelo do catálogo (incluindo esqueleto/animações) via SceneObject.
   119→  // Isto é CRÍTICO — sem isto, o runtime só mostra uma cápsula placeholder
   120→  // e o SkinnedMesh nunca é instanciado.
   121→  if (
   122→    conect.type === 'VisualObject' ||
   123→    ((conect.type === 'PersonalObject' || conect.type === 'NpcObject' || conect.type === 'RigidObject') && conect.sourceObjectId)
   124→  ) {
   125→    const obj = objects.find((o) => o.id === conect.sourceObjectId)
   126→    if (!obj) {
   127→      // sourceObjectId definido mas modelo não encontrado — fallback para placeholder
   128→      return (
   129→        <PlaceholderMesh
   130→          ref={(node) => {
   131→            if (typeof meshRef === 'function') meshRef(node)
   132→            else if (meshRef) meshRef.current = node
   133→            setMeshRef?.(node)
   134→          }}
   135→          conect={conect}
   136→        />
   137→      )
   138→    }
   139→    const sceneObj = {
   140→      ...obj,
   141→      id: conect.instanceId,
   142→      position: conect.position,
   143→      rotation: conect.rotation,
   144→      scale: conect.scale,
   145→    }
   146→    return (
   147→      <SceneObject
   148→        ref={(node) => setMeshRef?.(node)}
   149→        obj={sceneObj}
   150→        isSelected={false}
   151→        onSelect={() => {}}
   152→      />
   153→    )
   154→  }
   155→
   156→  // Rigid/Static/Stop/Personal sem modelo: placeholder com geometria simples
   157→  return (
   158→    <PlaceholderMesh
   159→      ref={(node) => {
   160→        if (typeof meshRef === 'function') meshRef(node)
   161→        else if (meshRef) meshRef.current = node
   162→        setMeshRef?.(node)
   163→      }}
   164→      conect={conect}
   165→    />
   166→  )
   167→})

   216→// ===== LuminousObject =====
   217→// Luz real + gizmo visual (esfera amarela) para seleção no editor
   218→function LuminousMesh({ conect, setMeshRef }) {
   219→  const lightProps = {
   220→    intensity: conect.intensity,
   221→    color: conect.color,
   222→    castShadow: conect.castShadow,
   223→  }
   224→  let lightElement
   225→  if (conect.lightType === 'directional') {
   226→    lightElement = <directionalLight {...lightProps} position={[0, 0, 0]} />
   227→  } else if (conect.lightType === 'spot') {
   228→    lightElement = <spotLight {...lightProps} distance={conect.distance} angle={Math.PI / 4} penumbra={0.3} position={[0, 0, 0]} />
   229→  } else {
   230→    lightElement = <pointLight {...lightProps} distance={conect.distance} position={[0, 0, 0]} />
   231→  }
   232→  return (
   233→    <group
   234→      ref={setMeshRef}
   235→      position={conect.position}
   236→      rotation={conect.rotation}
   237→      userData={{ conectInstanceId: conect.instanceId }}
   238→    >
   239→      {lightElement}
   240→      {/* Gizmo visual: esfera amarela para seleção */}
   241→      <mesh>
   242→        <sphereGeometry args={[0.2, 12, 12]} />
   243→        <meshBasicMaterial color="#ffd700" />
   244→      </mesh>
   245→      {/* Halo ao redor */}
   246→      <mesh>
   247→        <sphereGeometry args={[0.35, 12, 12]} />
   248→        <meshBasicMaterial color="#ffd700" transparent opacity={0.2} />
   249→      </mesh>
   250→    </group>
   251→  )
   252→}

   254→// ===== SunObject (luz direcional com temperatura de cor Kelvin) =====
   255→function SunMesh({ conect, setMeshRef }) {
   256→  // Converter temperatura Kelvin para cor RGB
   257→  const color = useMemo(() => kelvinToRGB(conect.temperature || 6500), [conect.temperature])
   258→  // Calcular direção do sol a partir de elevação/azimute
   259→  const sunDirection = useMemo(() => {
   260→    const elev = THREE.MathUtils.degToRad(conect.elevation ?? 45)
   261→    const azim = THREE.MathUtils.degToRad(conect.azimuth ?? 180)
   262→    return [
   263→      Math.cos(elev) * Math.sin(azim),
   264→      Math.sin(elev),
   265→      Math.cos(elev) * Math.cos(azim),
   266→    ]
   267→  }, [conect.elevation, conect.azimuth])
   268→
   269→  return (
   270→    <group
   271→      ref={setMeshRef}
   272→      position={conect.position}
   273→      userData={{ conectInstanceId: conect.instanceId, isLight: true, lightType: 'sun' }}
   274→    >
   275→      <directionalLight
   276→        intensity={conect.intensity}
   277→        color={color}
   278→        position={sunDirection}
   279→        castShadow={conect.castShadow}
   280→      />
   281→      {/* Gizmo: sol (esfera laranja + raios paralelos) */}
   282→      <mesh>
   283→        <sphereGeometry args={[0.3, 16, 16]} />
   284→        <meshBasicMaterial color="#ffa500" />
   285→      </mesh>
   286→      {/* Setas paralelas indicando direção da luz */}
   287→      <group position={[0, 0, 0]}>
   288→        {[0, 1, 2].map(i => (
   289→          <mesh key={i} position={[sunDirection[0] * (1 + i * 0.5), sunDirection[1] * (1 + i * 0.5), sunDirection[2] * (1 + i * 0.5)]}>
   290→            <coneGeometry args={[0.08, 0.2, 4]} />
   291→            <meshBasicMaterial color="#ffa500" transparent opacity={0.6} />
   292→          </mesh>
   293→        ))}
   294→      </group>
   295→    </group>
   296→  )
   297→}

   335→// ===== SpotObject (holofote com cone) =====
   336→function SpotLightMesh({ conect, setMeshRef }) {
   337→  const targetRef = useRef()
   338→  const angleRad = THREE.MathUtils.degToRad(conect.angle || 45)
   339→  // Calcular posição do target (aponta para -Y do grupo)
   340→  const targetPos = [0, -1, 0]
   341→
   342→  useEffect(() => {
   343→    if (targetRef.current) {
   344→      targetRef.current.position.set(...targetPos)
   345→    }
   346→  }, [])
   347→
   348→  return (
   349→    <group
   350→      ref={setMeshRef}
   351→      position={conect.position}
   352→      rotation={conect.rotation}
   353→      userData={{ conectInstanceId: conect.instanceId, isLight: true, lightType: 'spot' }}
   354→    >
   355→      <spotLight
   356→        intensity={conect.intensity}
   357→        color={conect.color}
   358→        distance={conect.distance}
   359→        angle={angleRad}
   360→        penumbra={conect.penumbra}
   361→        decay={conect.decay}
   362→        castShadow={conect.castShadow}
   363→        position={[0, 0, 0]}
   364→        target={targetRef.current}
   365→      />
   366→      <object3D ref={targetRef} position={targetPos} />
   367→      {/* Gizmo: cone a indicar direção do holofote */}
   368→      <mesh position={[0, -0.5, 0]} rotation={[Math.PI, 0, 0]}>
   369→        <coneGeometry args={[Math.tan(angleRad) * 1, 1, 16, 1, true]} />
   370→        <meshBasicMaterial color={conect.color || '#ffffff'} wireframe transparent opacity={0.4} side={THREE.DoubleSide} />
   371→      </mesh>
   372→      {/* Pequena esfera na fonte */}
   373→      <mesh>
   374→        <sphereGeometry args={[0.15, 12, 12]} />
   375→        <meshBasicMaterial color={conect.color || '#ffffff'} />
   376→      </mesh>
   377→    </group>
   378→  )
   379→}

   443→// ===== SkyObject (solid, gradient, hdri, procedural) =====
   444→// Céu procedural usando shader custom (flirSkyShader) — não depende do THREE.Sky
   445→// nem do tone mapping do renderer. Cores calculadas directamente em sRGB.
   446→function SkyMesh({ conect, setMeshRef }) {
   447→  const { scene, gl } = useThree()
   448→  const sunLightRef = useRef()
   449→  const skyMeshRef = useRef()
   450→  const timeRef = useRef(0)
   451→
   452→  // 1. Criar o mesh do céu com shader custom
   453→  const skyObj = useMemo(() => {
   454→    if (conect.skyType !== 'procedural') return null
   455→    const geometry = new THREE.SphereGeometry(100, 32, 16)
   456→    const material = new THREE.ShaderMaterial({
   457→      uniforms: {
   458→        sunPosition: { value: new THREE.Vector3(0, 1, 0) },
   459→        turbidity: { value: 10 },
   460→        rayleigh: { value: 1 },
   461→        time: { value: 0 },
   462→        starsEnabled: { value: 0 },
   463→      },
   464→      vertexShader: flirSkyVertexShader,
   465→      fragmentShader: flirSkyFragmentShader,
   466→      side: THREE.BackSide,
   467→      depthWrite: false,
   468→    })
   469→    const mesh = new THREE.Mesh(geometry, material)
   470→    mesh.frustumCulled = false
   471→    return mesh
   472→  }, [conect.skyType])
   473→
   474→  // 2. Atualizar uniforms quando parâmetros mudam
   475→  useEffect(() => {
   476→    if (!skyObj) return
   477→    // Converter elevação (graus do horizonte) e azimute para direção do sol
   478→    // elevation: 0=horizonte, 90=zénite
   479→    // azimuth: 0=norte, 90=este, 180=sul
   480→    const elevDeg = conect.sunElevation ?? 25
   481→    const azimDeg = conect.sunAzimuth ?? 180
   482→    const elevRad = THREE.MathUtils.degToRad(elevDeg)
   483→    const azimRad = THREE.MathUtils.degToRad(azimDeg)
   484→    // y = sin(elevation), x = cos(elevation) * sin(azimuth), z = cos(elevation) * cos(azimuth)
   485→    const sun = new THREE.Vector3(
   486→      Math.cos(elevRad) * Math.sin(azimRad),
   487→      Math.sin(elevRad),
   488→      Math.cos(elevRad) * Math.cos(azimRad)
   489→    )
   490→    skyObj.material.uniforms.sunPosition.value.copy(sun)
   491→    skyObj.material.uniforms.turbidity.value = conect.turbidity ?? 10
   492→    skyObj.material.uniforms.rayleigh.value = conect.rayleigh ?? 1
   493→    skyObj.material.uniforms.starsEnabled.value = conect.starsEnabled ? 1.0 : 0.0
   494→  }, [skyObj, conect.sunElevation, conect.sunAzimuth, conect.turbidity, conect.rayleigh, conect.starsEnabled])
   495→
   496→  // 3. Animar time (para nuvens e estrelas)
   497→  useFrame((_, delta) => {
   498→    if (!skyObj) return
   499→    timeRef.current += delta
   500→    skyObj.material.uniforms.time.value = timeRef.current
   501→  })
   502→
   503→  // 4. Adicionar/remover Sky à cena
   504→  useEffect(() => {
   505→    if (conect.skyType === 'procedural' && skyObj) {
   506→      scene.add(skyObj)
   507→      return () => { scene.remove(skyObj) }
   508→    }
   509→  }, [skyObj, scene, conect.skyType])
   510→
   511→  // 5. Calcular direção do sol para a luz direcional
   512→  const sunDirection = useMemo(() => {
   513→    const elevDeg = conect.sunElevation ?? 25
   514→    const azimDeg = conect.sunAzimuth ?? 180
   515→    const elevRad = THREE.MathUtils.degToRad(elevDeg)
   516→    const azimRad = THREE.MathUtils.degToRad(azimDeg)
   517→    return new THREE.Vector3(
   518→      Math.cos(elevRad) * Math.sin(azimRad),
   519→      Math.sin(elevRad),
   520→      Math.cos(elevRad) * Math.cos(azimRad)
   521→    )
   522→  }, [conect.sunElevation, conect.sunAzimuth])
   523→
   524→  // 6. Cor da luz do sol baseada na elevação
   525→  const sunLightColor = useMemo(() => {
   526→    const elev = conect.sunElevation ?? 25
   527→    if (elev < 5) return new THREE.Color(0xff6600)
   528→    if (elev < 15) return new THREE.Color(0xffaa44)
   529→    if (elev < 30) return new THREE.Color(0xffddaa)
   530→    return new THREE.Color(0xffffff)
   531→  }, [conect.sunElevation])
   532→
   533→  // 7. Intensidade da luz do sol
   534→  const sunLightIntensity = useMemo(() => {
   535→    const elev = conect.sunElevation ?? 25
   536→    return Math.max(0.3, (elev / 90) * 2.5)
   537→  }, [conect.sunElevation])
   538→
   539→  // 8. Aplicar background consoante o tipo
   540→  useEffect(() => {
   541→    if (conect.skyType === 'solid') {
   542→      scene.background = new THREE.Color(conect.solidColor || '#87ceeb')
   543→    } else if (conect.skyType === 'gradient') {
   544→      const canvas = document.createElement('canvas')
   545→      canvas.width = 2
   546→      canvas.height = 256
   547→      const ctx = canvas.getContext('2d')
   548→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
   549→      grad.addColorStop(0, conect.topColor || '#1a4d8f')
   550→      grad.addColorStop(1, conect.bottomColor || '#aac4e8')
   551→      ctx.fillStyle = grad
   552→      ctx.fillRect(0, 0, 2, 256)
   553→      const tex = new THREE.CanvasTexture(canvas)
   554→      tex.colorSpace = THREE.SRGBColorSpace
   555→      scene.background = tex
   556→    } else if (conect.skyType === 'hdri' && conect.hdriUrl) {
   557→      const pmrem = new THREE.PMREMGenerator(gl)
   558→      new RGBELoader().load(conect.hdriUrl, (texture) => {
   559→        const envMap = pmrem.fromEquirectangular(texture).texture
   560→        scene.background = envMap
   561→        scene.environment = envMap
   562→        texture.dispose()
   563→        pmrem.dispose()
   564→      })
   565→    } else if (conect.skyType === 'procedural') {
   566→      scene.background = null
   567→    }
   568→  }, [conect.skyType, conect.solidColor, conect.topColor, conect.bottomColor, conect.hdriUrl, scene, gl])
   569→
   570→  if (conect.skyType === 'procedural') {
   571→    return (
   572→      <group ref={setMeshRef} userData={{ conectInstanceId: conect.instanceId }}>
   573→        <directionalLight
   574→          ref={sunLightRef}
   575→          position={[sunDirection.x * 50, sunDirection.y * 50, sunDirection.z * 50]}
   576→          intensity={sunLightIntensity}
   577→          color={sunLightColor}
   578→          castShadow
   579→        />
   580→        {/* Sky mesh é adicionado à cena via useEffect acima */}
   581→        <mesh visible={false}>
   582→          <sphereGeometry args={[0.1, 8, 8]} />
   583→          <meshBasicMaterial />
   584→        </mesh>
   585→      </group>
   586→    )
   587→  }
   588→
   589→  return (
   590→    <group ref={setMeshRef} userData={{ conectInstanceId: conect.instanceId }}>
   591→      <mesh visible={false}>
   592→        <sphereGeometry args={[0.1, 8, 8]} />
   593→        <meshBasicMaterial />
   594→      </mesh>
   595→    </group>
   596→  )
   597→}

   601→// ===== TerrainObject =====
   602→function TerrainMesh({ conect, setMeshRef }) {
   603→  const geometry = useMemo(() => {
   604→    const seg = conect.segments || 64
   605→    const g = new THREE.PlaneGeometry(
   606→      conect.width || 50, conect.depth || 50, seg, seg
   607→    )
   608→    g.rotateX(-Math.PI / 2)
   609→    const positions = g.attributes.position
   610→    const heightScale = conect.heightScale || 5
   611→
   612→    // Se há heightmap exportado do TerrainEditor, usá-lo
   613→    if (conect.heightmap && conect.heightmap.length > 0) {
   614→      const hm = conect.heightmap
   615→      for (let i = 0; i < positions.count && i < hm.length; i++) {
   616→        positions.setY(i, hm[i] * heightScale)
   617→      }
   618→    } else {
   619→      // Fallback: gerar procedural
   620→      const seed = conect.heightmapSeed || 1
   621→      for (let i = 0; i < positions.count; i++) {
   622→        const x = positions.getX(i)
   623→        const z = positions.getZ(i)
   624→        const h =
   625→          Math.sin(x * 0.3 + seed) * Math.cos(z * 0.3 + seed) * 0.5 +
   626→          Math.sin(x * 0.1 + z * 0.1) * 0.3
   627→        positions.setY(i, h * heightScale)
   628→      }
   629→    }
   630→    g.computeVertexNormals()
   631→
   632→    // Vertex colors from splatmap (multi-layer blending) — P7
   633→    if (conect.splatmap && conect.textureLayers && conect.splatmap.length > 0) {
   634→      const maxLayers = conect.maxLayers || 4
   635→      const layers = conect.textureLayers
   636→      const colors = new Float32Array(positions.count * 3)
   637→      const sm = conect.splatmap
   638→      const hexToRgb = (hex) => {
   639→        const r = parseInt(hex.slice(1, 3), 16) / 255
   640→        const g = parseInt(hex.slice(3, 5), 16) / 255
   641→        const b = parseInt(hex.slice(5, 7), 16) / 255
   642→        return [r, g, b]
   643→      }
   644→      const layerRgb = layers.map((l) => hexToRgb(l.color || '#5a7d3a'))
   645→      for (let i = 0; i < positions.count; i++) {
   646→        let r = 0, g = 0, b = 0
   647→        for (let l = 0; l < maxLayers && l < layers.length; l++) {
   648→          const w = sm[i * maxLayers + l] || 0
   649→          const c = layerRgb[l] || [0.35, 0.49, 0.23]
   650→          r += c[0] * w
   651→          g += c[1] * w
   652→          b += c[2] * w
   653→        }
   654→        colors[i * 3] = r
   655→        colors[i * 3 + 1] = g
   656→        colors[i * 3 + 2] = b
   657→      }
   658→      g.setAttribute('color', new THREE.BufferAttribute(colors, 3))
   659→    }
   660→
   661→    return g
   662→  }, [conect.width, conect.depth, conect.segments, conect.heightScale, conect.heightmapSeed, conect.heightmap, conect.splatmap, conect.textureLayers, conect.maxLayers])
   663→
   664→  const hasVertexColors = !!(conect.splatmap && conect.textureLayers && conect.splatmap.length > 0)
   665→
   666→  return (
   667→    <mesh
   668→      ref={setMeshRef}
   669→      geometry={geometry}
   670→      position={conect.position}
   671→      receiveShadow
   672→      castShadow
   673→    >
   674→      <meshStandardMaterial
   675→        color={hasVertexColors ? '#ffffff' : (conect.textureColor || '#5a7d3a')}
   676→        vertexColors={hasVertexColors}
   677→        roughness={0.9}
   678→        metalness={0}
   679→      />
   680→    </mesh>
   681→  )
   682→}

   684→// ===== WaterObject (com ondas animadas via vertex displacement) =====
   685→function WaterMesh({ conect, setMeshRef }) {
   686→  const meshRef = useRef()
   687→  const time = useRef(0)
   688→
   689→  // Geometria com subdivisões para ondas
   690→  const geometry = useMemo(() => {
   691→    const [w, h] = conect.size || [20, 20]
   692→    const g = new THREE.PlaneGeometry(w, h, 32, 32)
   693→    g.rotateX(-Math.PI / 2)
   694→    return g
   695→  }, [conect.size])
   696→
   697→  // Animar ondas no useFrame
   698→  useFrame((_, delta) => {
   699→    time.current += delta * (conect.waveSpeed || 0.5)
   700→    if (!meshRef.current) return
   701→    const pos = meshRef.current.geometry.attributes.position
   702→    const waveHeight = conect.waveHeight || 0.1
   703→    for (let i = 0; i < pos.count; i++) {
   704→      const x = pos.getX(i)
   705→      const z = pos.getZ(i)
   706→      // Ondas senoidais combinadas
   707→      const y = Math.sin(x * 0.5 + time.current) * waveHeight
   708→        + Math.cos(z * 0.5 + time.current * 0.7) * waveHeight * 0.5
   709→      pos.setY(i, y)
   710→    }
   711→    pos.needsUpdate = true
   712→    meshRef.current.geometry.computeVertexNormals()
   713→  })
   714→
   715→  return (
   716→    <mesh
   717→      ref={(node) => {
   718→        meshRef.current = node
   719→        if (typeof setMeshRef === 'function') setMeshRef(node)
   720→      }}
   721→      position={conect.position}
   722→      geometry={geometry}
   723→    >
   724→      <meshStandardMaterial
   725→        color={conect.color}
   726→        transparent
   727→        opacity={conect.opacity}
   728→        roughness={0.1}
   729→        metalness={0.3}
   730→      />
   731→    </mesh>
   732→  )
   733→}

   754→function Line({ points, loop }) {

   758→  }, [points, loop])
   759→  return (
   760→    <line geometry={geometry}>
   761→      <lineBasicMaterial color="#2f81f7" transparent opacity={0.5} />
   762→    </line>

   802→      {particles.map((p, i) => (
   803→        <mesh key={i} position={p.position}>

   805→          <meshBasicMaterial color={conect.color} transparent opacity={0.7} />
