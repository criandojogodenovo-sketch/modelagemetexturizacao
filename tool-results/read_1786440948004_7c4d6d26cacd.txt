    14→// ===== FlirCode Parser (inline, sem dependências) =====
    15→function parseFlirCode(src) {
    16→  var errors = [], fns = {}, lines = src.split('\n'), cl = []
    17→  for (var i = 0; i < lines.length; i++) {
    18→    var t = lines[i].trim()
    19→    if (t && !t.startsWith('$$')) cl.push({ t: t, l: i + 1 })
    20→  }
    21→  var idx = 0
    22→  while (idx < cl.length) {
    23→    var m = cl[idx].t.match(/^fun\s+(\w+)\s*\(([^)]*)\)\s*begincode$/)
    24→    if (m) {
    25→      var body = parseBlock(cl, idx + 1, errors)
    26→      fns[m[1]] = { name: m[1], params: m[2].split(',').filter(function (p) { return p.trim() }), body: body.s, line: cl[idx].l }
    27→      idx = body.ni
    28→    } else { idx++ }
    29→  }
    30→  return { functions: fns, errors: errors }
    31→}
    32→
    33→function parseBlock(lines, si, errors) {
    34→  var s = [], idx = si, depth = 1
    35→  while (idx < lines.length && depth > 0) {
    36→    if (lines[idx].t === 'endcode') { depth--; if (depth === 0) return { s: s, ni: idx + 1 } }
    37→    else { s.push({ t: lines[idx].t, l: lines[idx].l }) }
    38→    idx++
    39→  }
    40→  return { s: s, ni: idx }
    41→}

    43→function evalCond(cond, vars, gc) {
    44→  var m = cond.match(/^(.+?)\s*(>=|<=|==|!=|>|<)\s*(.+)$/)
    45→  if (!m) return !!evalVal(cond, vars, gc)
    46→  var l = evalVal(m[1].trim(), vars, gc), r = evalVal(m[3].trim(), vars, gc)
    47→  switch (m[2]) {
    48→    case '>': return l > r; case '<': return l < r
    49→    case '>=': return l >= r; case '<=': return l <= r
    50→    case '==': return l == r; case '!=': return l != r
    51→  }
    52→  return false
    53→}
    54→
    55→function evalVal(v, vars, gc) {
    56→  v = v.trim()
    57→  if (v.startsWith('"') && v.endsWith('"')) return v.slice(1, -1)
    58→  var n = parseFloat(v); if (!isNaN(n)) return n
    59→  if (v === 'true') return true; if (v === 'false') return false
    60→  return vars[v] !== undefined ? vars[v] : (gc.globalVars && gc.globalVars[v]) || 0
    61→}

    63→function createFlirCodeRuntime(src, gc) {
    64→  var parsed = parseFlirCode(src)
    65→  for (var i = 0; i < parsed.errors.length; i++) dbg('FlirCode erro: ' + parsed.errors[i].message, 'error')
    66→  var vars = {}
    67→  var eventMap = {
    68→    onStart: 'beginPlay', onTick: 'tick', onCollide: 'onCollision',
    69→    onTouch: 'onTouch', onSeePlayer: 'onSeePlayer', onLoseSight: 'onLoseSight',
    70→    onTimer: 'onTimer', onClick: 'onClick', onChange: 'onChange', onSubmit: 'onSubmit',
    71→    onEnterZone: 'onEnterZone', onExitZone: 'onExitZone'
    72→  }

    74→  function execStmts(stmts, params) {
    75→    for (var i = 0; i < stmts.length; i++) {
    76→      try { execS(stmts[i], params) } catch (e) { dbg('Erro: ' + e.message, 'error') }
    77→    }
    78→  }
    79→
    80→  function execS(s, params) {
    81→    var m
    82→    if (m = s.t.match(/^var\s+(\w+)\s*=\s*(.+)$/)) { vars[m[1]] = evalVal(m[2], vars, gc); return }
    83→    if (m = s.t.match(/^(\w+)\s*=\s*(.+)$/)) { vars[m[1]] = evalVal(m[2], vars, gc); return }
    84→    if (m = s.t.match(/^if\s*\((.+)\)$/)) { if (evalCond(m[1], vars, gc)) { /* procurar begincode seguinte */ } return }
    85→    if (m = s.t.match(/^(\w+)\s*\(([^)]*)\)$/)) {
    86→      execBuiltin(m[1], m[2].split(',').map(function (a) { return evalVal(a.trim(), vars, gc) }), params)
    87→      return
    88→    }
    89→  }

    91→  function execBuiltin(name, args, params) {
    92→    switch (name) {
    93→      case 'print': dbg(args[0], 'log'); break
    94→      case 'warn': dbg(args[0], 'warn'); break
    95→      case 'error': dbg(args[0], 'error'); break
    96→      case 'move':
    97→        if (gc.mesh) { gc.mesh.position.x += args[0] * 0.016; gc.mesh.position.y += args[1] * 0.016; gc.mesh.position.z += args[2] * 0.016 }
    98→        break
    99→      case 'rotate':
   100→        if (gc.mesh) { gc.mesh.rotation.x += args[0] * 0.016; gc.mesh.rotation.y += args[1] * 0.016; gc.mesh.rotation.z += args[2] * 0.016 }
   101→        break
   102→      case 'scale':
   103→        if (gc.mesh) { gc.mesh.scale.set(args[0] || 1, args[1] || 1, args[2] || 1) }
   104→        break
   105→      case 'destroy': if (gc.mesh) gc.mesh.visible = false; break
   106→      case 'createObject': gc.spawnObject && gc.spawnObject(args[0], [args[1], args[2], args[3]]); break
   107→      case 'changeScene': dbg('changeScene: ' + args[0], 'log'); break
   108→      case 'wait': dbg('wait(' + args[0] + 's)', 'log'); break
   109→      case 'setVar': gc.globalVars = gc.globalVars || {}; gc.globalVars[args[0]] = args[1]; break
   110→      case 'getVar': return (gc.globalVars || {})[args[0]]
   111→      case 'setUIValue': gc.setUIValue && gc.setUIValue(args[0], args[1]); break
   112→      case 'getUIValue': return gc.getUIValue ? gc.getUIValue(args[0]) : ''
   113→      case 'showUIScreen': gc.showUIScreen && gc.showUIScreen(args[0]); break
   114→      case 'hideUIScreen': gc.hideUIScreen && gc.hideUIScreen(args[0]); break
   115→      case 'playSound': gc.playSound && gc.playSound(args[0]); break
   116→      case 'playAnim': dbg('playAnim: ' + args[0], 'log'); break
   117→      case 'collidingWith': return gc.collidingWith ? gc.collidingWith(gc._instanceId, args[0]) : false
   118→      case 'distanceTo': return gc.distanceTo ? gc.distanceTo(gc._instanceId, args[0]) : 0
   119→      case 'isTouching': return gc.isTouching ? gc.isTouching() : false
   120→      // Sistema 2: Armas
   121→      case 'shoot': gc.shoot && gc.shoot(); break
   122→      case 'reload': gc.reload && gc.reload(); break
   123→      case 'equipWeapon': gc.equipWeapon && gc.equipWeapon(args[0]); break
   124→      case 'getAmmo': return gc.getAmmo ? gc.getAmmo() : 0
   125→      case 'takeDamage': gc.takeDamage && gc.takeDamage(gc._instanceId, args[0]); break
   126→      case 'getHealth': return gc.getHealth ? gc.getHealth(gc._instanceId) : 100
   127→      // Sistema 3: Inventário
   128→      case 'addToInventory': gc.addToInventory && gc.addToInventory(args[0], args[1]); break
   129→      case 'removeFromInventory': gc.removeFromInventory && gc.removeFromInventory(args[0], args[1]); break
   130→      case 'getInventoryCount': return gc.getInventoryCount ? gc.getInventoryCount(args[0]) : 0
   131→      case 'hasItem': return gc.hasItem ? gc.hasItem(args[0]) : false
   132→      // Sistema 3: Sinais
   133→      case 'emitSignal': gc.emitSignal && gc.emitSignal(args[0], args[1]); break
   134→      // Aliases showUI/hideUI = showUIScreen/hideUIScreen
   135→      case 'showUI': gc.showUIScreen && gc.showUIScreen(args[0]); break
   136→      case 'hideUI': gc.hideUIScreen && gc.hideUIScreen(args[0]); break
   137→      // Multiplayer (básico no export)
   138→      case 'sendMessage': gc.sendMessage && gc.sendMessage(args[0]); break
   139→      case 'getPlayers': return gc.getPlayers ? gc.getPlayers() : 1
   140→      case 'getPlayerState': return gc.getPlayerState ? gc.getPlayerState(args[0]) : null
   141→      // Sistema: Links — navegar para cena ou tela
   142→      case 'linkTo': gc.linkTo && gc.linkTo(args[0], args[1]); break
   143→      // Sistema: Game State
   144→      case 'setGameState': gc.setGameState && gc.setGameState(args[0]); break
   145→      case 'getGameState': return gc.getGameState ? gc.getGameState() : 'menu'
   146→      // Sistema: Save/Load Progress
   147→      case 'saveProgress': gc.saveProgress && gc.saveProgress(args[0], args[1]); break
   148→      case 'loadProgress': return gc.loadProgress ? gc.loadProgress(args[0]) : null
   149→      // Sistema: Sequenciador
   150→      case 'playSequence': gc.playSequence && gc.playSequence(args[0]); break
   151→      // changeScene real
   152→      case 'changeScene':
   153→        if (gc.changeScene) { gc.changeScene(args[0]); break }
   154→        dbg('changeScene: ' + args[0], 'log'); break
   155→      default: dbg('Função desconhecida: ' + name, 'warn')
   156→    }
   157→  }
   158→
   159→  return {
   160→    functions: parsed.functions, hasErrors: parsed.errors.length > 0,
   161→    triggerEvent: function (en, payload) {
   162→      var fnName = null
   163→      for (var k in eventMap) { if (eventMap[k] === en) { fnName = k; break } }
   164→      if (!fnName) return
   165→      var fn = parsed.functions[fnName]; if (!fn) return
   166→      gc._instanceId = gc._instanceId
   167→      execStmts(fn.body, payload || {})
   168→    },
   169→    update: function () {}, dispose: function () {}
   170→  }
   171→}

   317→  // Setup meshes para objetos do catálogo
   318→  function setupMesh(obj, pos, rot, scl) {
   319→    var geo
   320→    if (obj.type === 'cube') geo = new THREE.BoxGeometry(1, 1, 1)
   321→    else if (obj.type === 'sphere') geo = new THREE.SphereGeometry(0.6, 32, 16)
   322→    else if (obj.type === 'cylinder') geo = new THREE.CylinderGeometry(0.5, 0.5, 1.2, 32)
   323→    else if (obj.type === 'cone') geo = new THREE.ConeGeometry(0.6, 1.2, 32)
   324→    else if (obj.type === 'plane') geo = new THREE.PlaneGeometry(1.5, 1.5)
   325→    else if (obj.type === 'torus') geo = new THREE.TorusGeometry(0.6, 0.2, 16, 64)
   326→    else geo = new THREE.BoxGeometry(1, 1, 1)
   327→    var m = obj.material || {}
   328→    var mat = new THREE.MeshStandardMaterial({ color: m.color || '#888', roughness: m.roughness || 0.7, metalness: m.metalness || 0 })
   329→    var mesh = new THREE.Mesh(geo, mat)
   330→    mesh.position.set.apply(mesh, pos || [0, 0.5, 0])
   331→    if (rot) mesh.rotation.set.apply(mesh, rot)
   332→    if (scl) mesh.scale.set.apply(mesh, scl)
   333→    mesh.castShadow = true; mesh.receiveShadow = true
   334→    scene3d.add(mesh)
   335→    return mesh
   336→  }

   360→      mesh.castShadow = true; mesh.receiveShadow = true
   361→      scene3d.add(mesh)
   362→      meshMap[conect.instanceId] = mesh
   363→      mesh._name = conect.name
   364→      mesh._conect = conect
   365→      // Physics
   366→      var shape = new CANNON.Box(new CANNON.Vec3(0.5, 0.5, 0.5))
   367→      var body = new CANNON.Body({ mass: conect.type === 'StaticObject' ? 0 : (conect.mass || 1), shape: shape, position: new CANNON.Vec3(conect.position[0], conect.position[1], conect.position[2]) })
   368→      if (conect.type === 'StaticObject') { body.type = CANNON.Body.STATIC; body.mass = 0 }
   369→      if (conect.type === 'StopObject') { body.type = CANNON.Body.KINEMATIC; body.mass = 0 }
   370→      body.fixedRotation = conect.fixedRotation || false
   371→      body._conect = conect
   372→      world.addBody(body)
   373→      bodies[conect.instanceId] = body
   374→      body.addEventListener('collide', function (e) {
   375→        var otherId = null
   376→        for (var k in bodies) { if (bodies[k] === e.body) { otherId = k; break } }
   377→        if (otherId && runtimes[conect.instanceId]) runtimes[conect.instanceId].triggerEvent('onCollision', { other: otherId })
   378→      })
   379→    } else if (conect.type === 'LuminousObject') {
   380→      var light
   381→      if (conect.lightType === 'directional') light = new THREE.DirectionalLight(conect.color || 0xffffff, conect.intensity || 1)
   382→      else if (conect.lightType === 'spot') light = new THREE.SpotLight(conect.color || 0xffffff, conect.intensity || 1)
   383→      else light = new THREE.PointLight(conect.color || 0xffffff, conect.intensity || 1, conect.distance || 10)
   384→      light.position.set.apply(light, conect.position || [0, 5, 0])
   385→      light.castShadow = conect.castShadow !== false
   386→      scene3d.add(light)
   387→    } else if (conect.type === 'SkyObject' && conect.skyType === 'gradient') {
   388→      var sc = document.createElement('canvas'); sc.width = 2; sc.height = 256
   389→      var sctx = sc.getContext('2d')
   390→      var sg = sctx.createLinearGradient(0, 0, 0, 256)
   391→      sg.addColorStop(0, conect.topColor || '#1a4d8f'); sg.addColorStop(1, conect.bottomColor || '#aac4e8')
   392→      sctx.fillStyle = sg; sctx.fillRect(0, 0, 2, 256)
   393→      scene3d.background = new THREE.CanvasTexture(sc)
   394→    } else if (conect.type === 'FogObject') {
   395→      if (conect.fogType === 'exponential') scene3d.fog = new THREE.FogExp2(conect.color || 0xa0a0a0, conect.density || 0.02)
   396→      else scene3d.fog = new THREE.Fog(conect.color || 0xa0a0a0, conect.near || 5, conect.far || 50)
   397→    } else if (conect.type === 'SoundObject' && conect.autoplay && conect.url) {
   398→      try { var audio = new Audio(conect.url); audio.volume = conect.volume || 1; audio.loop = conect.loop || false; audio.play() } catch (e) { }
   399→    }
   400→
   401→    // FlirCode para conects
   402→    if (conect.flirScript && typeof conect.flirScript === 'string' && conect.flirScript.startsWith('FLIRCODE:')) {
   403→      var rt2 = createFlirCodeRuntime(conect.flirScript.slice(9), Object.assign(gc, { _instanceId: conect.instanceId, mesh: mesh }))
   404→      if (!rt2.hasErrors) { runtimes[conect.instanceId] = rt2; rt2.triggerEvent('beginPlay') }
   405→    }

   445→  var lastTime = performance.now()
   446→  function animate() {
   447→    requestAnimationFrame(animate)
   448→    var now = performance.now()
   449→    var delta = (now - lastTime) / 1000
   450→    lastTime = now
   451→
   452→    // Physics
   453→    world.step(1 / 60, delta, 3)
   454→    for (var id in bodies) { var b = bodies[id]; var m = meshMap[id]; if (m) { m.position.copy(b.position); m.quaternion.copy(b.quaternion) } }
   455→
   456→    // FlirCode onTick
   457→    for (var rid in runtimes) { runtimes[rid].triggerEvent('tick', { deltaTime: delta }) }
   458→
   459→    // PersonalObject movement
   460→    var player = (scene.conects || []).find(function (c) { return c.type === 'PersonalObject' })
   461→    if (player && bodies[player.instanceId]) {
   462→      var speed = player.moveSpeed || 5
   463→      var mx = 0, mz = 0
   464→      if (joystick.active) { mx = joystick.x * speed; mz = joystick.z * speed }
   465→      if (keys['w'] || keys['arrowup']) mz = -speed
   466→      if (keys['s'] || keys['arrowdown']) mz = speed
   467→      if (keys['a'] || keys['arrowleft']) mx = -speed
   468→      if (keys['d'] || keys['arrowright']) mx = speed
   469→      bodies[player.instanceId].velocity.x = mx
   470→      bodies[player.instanceId].velocity.z = mz
   471→      if ((keys[' '] || keys['space']) && player.canJump) bodies[player.instanceId].velocity.y = player.jumpForce || 8
   472→    }
   473→
   474→    // Camera follow
   475→    if (activeView && activeView.cameraRole === 'player' && player && meshMap[player.instanceId]) {
   476→      var pm = meshMap[player.instanceId]
   477→      camera.position.lerp(new THREE.Vector3(pm.position.x, pm.position.y + (activeView.followHeight || 3), pm.position.z + (activeView.followDistance || 6)), 0.1)
   478→      camera.lookAt(pm.position)
   479→    }
   480→
   481→    renderer.render(scene3d, camera)
   482→  }
