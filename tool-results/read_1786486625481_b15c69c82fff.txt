    56→// Componente interno que aplica o fundo da cena
    57→function SceneBackgroundSolid({ background }) {
    58→  const { scene } = useThree()
    59→  useEffect(() => {
    60→    if (background.type === 'gradient') {
    61→      const canvas = document.createElement('canvas')
    62→      canvas.width = 2; canvas.height = 256
    63→      const ctx = canvas.getContext('2d')
    64→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
    65→      grad.addColorStop(0, background.gradientTop)
    66→      grad.addColorStop(1, background.gradientBottom)
    67→      ctx.fillStyle = grad; ctx.fillRect(0, 0, 2, 256)
    68→      const tex = new THREE.CanvasTexture(canvas)
    69→      tex.colorSpace = THREE.SRGBColorSpace
    70→      scene.background = tex
    71→      return () => tex.dispose()
    72→    } else {
    73→      scene.background = new THREE.Color(background.color)
    74→    }
    75→  }, [background, scene])
    76→  return null
    77→}

    79→// ===== FogApplier — aplica FogObject à cena 3D =====
    80→function FogApplier({ conects }) {
    81→  const { scene } = useThree()
    82→  const fogConect = (conects || []).find(c => c.type === 'FogObject')
    83→  useEffect(() => {
    84→    if (!fogConect) {
    85→      scene.fog = null
    86→      return
    87→    }
    88→    if (fogConect.fogType === 'exponential') {
    89→      scene.fog = new THREE.FogExp2(fogConect.color || '#a0a0a0', fogConect.density || 0.02)
    90→    } else {
    91→      scene.fog = new THREE.Fog(fogConect.color || '#a0a0a0', fogConect.near || 5, fogConect.far || 50)
    92→    }
    93→  }, [fogConect?.fogType, fogConect?.color, fogConect?.near, fogConect?.far, fogConect?.density, scene])
    94→  return null
    95→}

   162→      mp.on('playerJoin', (playerId) => {

   166→      })

   168→        for (const rt of runtimesRef.current.values()) {
   169→          rt.triggerEvent('onPlayerLeave', { playerId })

   291→          }
   292→        }
   293→      },
   294→      triggerUIEvent: (eventName, payload) => {

   296→      },
   297→      collidingWith: (instanceId, type) => {

   299→        // Usar contactos reais do cannon-es em vez de distância

   306→              return true
   307→            }
   308→          }
   309→          return false
   310→        }
   311→        // Fallback: usar eventos de colisão registados
   312→        const collisionSet = collisionEventsRef.current.get(instanceId)
   313→        if (collisionSet && collisionSet.size > 0) {
   314→          for (const otherId of collisionSet) {
   315→            const otherEntry = physicsRef.current.bodies.get(otherId)
   316→            if (!otherEntry) continue
   317→            if (otherEntry.conect.type === type || otherEntry.conect.name === type) {
   318→              return true
   319→            }
   320→          }
   321→        }

   935→    if (activeView) {
   936→      // Se cameraRole='player' e não tem followTarget, seguir PersonalObject
   937→      let targetId = activeView.followTarget
   938→      if (!targetId && activeView.cameraRole === 'player') {
   939→        const player = (activeScene.conects || []).find((c) => c.type === 'PersonalObject')
   940→        if (player) targetId = player.instanceId
   941→      }
   942→      if (targetId && activeView.followMode && activeView.followMode !== 'none') {
   943→        const targetMesh = meshRefs.current.get(targetId) || conectMeshRefs.current.get(targetId)
   944→        if (targetMesh) {
   945→          const mode = activeView.followMode
   946→          const dist = activeView.followDistance || 6
   947→          const height = activeView.followHeight || 3
   948→          if (mode === 'third') {
   949→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x, targetMesh.position.y + height, targetMesh.position.z + dist), 0.1)
   950→            camera.lookAt(targetMesh.position)
   951→          } else if (mode === 'top') {
   952→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x, targetMesh.position.y + dist, targetMesh.position.z), 0.1)
   953→            camera.lookAt(targetMesh.position)
   954→          } else if (mode === 'side') {
   955→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x + dist, targetMesh.position.y + height / 2, targetMesh.position.z), 0.1)
   956→            camera.lookAt(targetMesh.position)
   957→          }
   958→        }
   959→      } else {
   960→        // Câmara estática na posição da ViewObject
   961→        camera.position.set(...(activeView.position || [5, 4, 6]))
   962→        if (activeView.rotation) camera.rotation.set(...activeView.rotation)
   963→        else camera.lookAt(0, 0, 0)
   964→      }
   965→    }
   966→
   967→    // CheckpointObject — registar checkpoint quando jogador toca
   968→    const player3 = (setupScene?.conects || []).find((c) => c.type === 'PersonalObject')
   969→    if (player3) {
   970→      const playerMesh = conectMeshRefs.current.get(player3.instanceId)
   971→      if (playerMesh) {
   972→        for (const cp of setupScene?.conects || []) {
   973→          if (cp.type === 'CheckpointObject') {
   974→            const cpMesh = conectMeshRefs.current.get(cp.instanceId)
   975→            if (cpMesh && cpMesh.visible !== false) {
   976→              const dist = playerMesh.position.distanceTo(cpMesh.position)
   977→              if (dist <= (cp.triggerRadius || 2)) {
   978→                // Registar checkpoint
   979→                checkpointRef.current = {
   980→                  position: [cpMesh.position.x, cpMesh.position.y, cpMesh.position.z],
   981→                  sceneId: setupScene.id,
   982→                }
   983→                // Guardar no localStorage
   984→                try {
   985→                  localStorage.setItem('flir_checkpoint', JSON.stringify(checkpointRef.current))
   986→                } catch (e) {}
   987→                // Disparar evento
   988→                const rt = runtimesRef.current.get(cp.instanceId)
   989→                if (rt) rt.triggerEvent('onCheckpoint', { position: checkpointRef.current.position })
   990→                debugLog('Checkpoint atingido!', 'log', 'Checkpoint')
   991→              }
   992→            }
   993→          }
   994→        }
   995→      }
   996→    }
   997→    if (player3) {
   998→      const playerMesh = conectMeshRefs.current.get(player3.instanceId)
   999→      if (playerMesh) {
  1000→        for (const nav of setupScene?.conects || []) {
  1001→          if (nav.type === 'NavigatorObject' && nav.targetSceneId) {
  1002→            const navMesh = conectMeshRefs.current.get(nav.instanceId)
  1003→            if (navMesh && navMesh.visible !== false) {
  1004→              const dist = playerMesh.position.distanceTo(navMesh.position)
  1005→              if (dist <= (nav.triggerRadius || 2)) {
  1006→                // Transportar para a cena de destino
  1007→                debugLog(`Portal ativado! A mudar para cena ${nav.targetSceneId}`, 'log', 'Navigator')
  1008→                useStore.getState().closeScenePreview()
  1009→                setTimeout(() => {
  1010→                  useStore.getState().setActiveScene(nav.targetSceneId)
  1011→                  useStore.getState().openScenePreview()
  1012→                }, (nav.transitionDuration || 0.5) * 1000)
  1013→                break
  1014→              }
  1015→            }
  1016→          }
  1017→        }
  1018→      }
  1019→    }
  1020→
  1021→    // ItemObject — auto-pickup quando jogador próximo
  1022→    if (player3) {
  1023→      const playerMesh = conectMeshRefs.current.get(player3.instanceId)
  1024→      if (playerMesh) {
  1025→        for (const item of setupScene?.conects || []) {
  1026→          if (item.type === 'ItemObject' && item.autoPickup !== false) {
  1027→            const itemMesh = conectMeshRefs.current.get(item.instanceId)
  1028→            if (itemMesh && itemMesh.visible !== false) {
  1029→              const dist = playerMesh.position.distanceTo(itemMesh.position)
  1030→              if (dist <= (item.pickupRadius || 2)) {
  1031→                // Apanhar item
  1032→                if (gameContext.addToInventory) {
  1033→                  gameContext.addToInventory(item.itemName || 'Item', item.quantity || 1)
  1034→                }
  1035→                itemMesh.visible = false
  1036→                debugLog(`Item "${item.itemName}" apanhado!`, 'log', 'Inventory')
  1037→              }
  1038→            }
  1039→          }
  1040→        }
  1041→      }
  1042→    }
  1043→
  1044→    // SpawnObject — spawning temporal de objetos
  1045→    for (const spawn of setupScene?.conects || []) {
  1046→      if (spawn.type === 'SpawnObject' && spawn.autoStart !== false && spawn.objectToSpawn) {
  1047→        if (!timerStatesRef.current.has(spawn.instanceId + '_spawn')) {
  1048→          timerStatesRef.current.set(spawn.instanceId + '_spawn', {
  1049→            remaining: spawn.interval || 3,
  1050→            duration: spawn.interval || 3,
  1051→            loop: true,
  1052→          })
  1053→        }
  1054→        const st = timerStatesRef.current.get(spawn.instanceId + '_spawn')
  1055→        st.remaining -= delta
  1056→        if (st.remaining <= 0) {
  1057→          st.remaining = st.duration
  1058→          // Spawn do objeto
  1059→          const spawnMesh = conectMeshRefs.current.get(spawn.instanceId)
  1060→          const spawnPos = spawnMesh
  1061→            ? [spawnMesh.position.x, spawnMesh.position.y, spawnMesh.position.z]
  1062→            : spawn.position || [0, 1, 0]
  1063→          gameContext.spawnObject(spawn.objectToSpawn, spawnPos)
  1064→          debugLog(`Spawn: "${spawn.objectToSpawn}" em ${spawnPos}`, 'log', 'Spawn')
  1065→        }
  1066→      }
  1067→    }
  1068→
  1069→    // GroupObject — aplicar parenting
  1070→    for (const grp of setupScene?.conects || []) {
  1071→      if (grp.type === 'GroupObject' && grp.children && grp.children.length > 0) {
  1072→        const groupMesh = conectMeshRefs.current.get(grp.instanceId)
  1073→        if (groupMesh && !groupMesh.userData._grouped) {
  1074→          for (const childId of grp.children) {
  1075→            const childMesh = conectMeshRefs.current.get(childId) || meshRefs.current.get(childId)
  1076→            if (childMesh && childMesh.parent !== groupMesh) {
  1077→              groupMesh.attach(childMesh)
  1078→            }
  1079→          }
  1080→          groupMesh.userData._grouped = true
  1081→        }
  1082→      }
  1083→    }
  1084→  })
  1085→
  1086→  return null
  1087→}

  1090→export default function SceneLevel3D() {
  1091→  const scenes = useStore((s) => s.scenes)
  1092→  const activeSceneId = useStore((s) => s.activeSceneId)
  1093→  const objects = useStore((s) => s.objects)
  1094→  const background = useStore((s) => s.background)
  1095→  const grid = useStore((s) => s.grid)
  1096→  const lights = useStore((s) => s.lights)
  1097→  const transformMode = useStore((s) => s.transformMode)
  1098→  const updateSceneInstance = useStore((s) => s.updateSceneInstance)
  1099→  const updateConect = useStore((s) => s.updateConect)
  1100→  const addObjectToScene = useStore((s) => s.addObjectToScene)
  1101→  const addConectToScene = useStore((s) => s.addConectToScene)
  1102→  const selectConect = useStore((s) => s.selectConect)
  1103→  const scenePreviewOpen = useStore((s) => s.scenePreviewOpen)
  1104→
  1105→  const [selectedInstanceId, setSelectedInstanceId] = useState(null)
  1106→  const [selectedType, setSelectedType] = useState(null)
  1107→  const orbitRef = useRef(null)
  1108→  const meshRefs = useRef(new Map())
  1109→  const conectMeshRefs = useRef(new Map())
  1110→  const [selectedMesh, setSelectedMesh] = useState(null)
  1111→
  1112→  const activeScene = scenes.find((s) => s.id === activeSceneId)
  1113→  const isGameMode = scenePreviewOpen
  1114→
  1115→  useEffect(() => {
  1116→    if (selectedInstanceId && meshRefs.current.has(selectedInstanceId)) {
  1117→      setSelectedMesh(meshRefs.current.get(selectedInstanceId))
  1118→    } else if (selectedInstanceId && conectMeshRefs.current.has(selectedInstanceId)) {
  1119→      setSelectedMesh(conectMeshRefs.current.get(selectedInstanceId))
  1120→    } else {
  1121→      setSelectedMesh(null)
  1122→    }
  1123→  }, [selectedInstanceId, activeScene])
  1124→
  1125→  const setMeshRef = useCallback((id, node) => {
  1126→    if (node) meshRefs.current.set(id, node)
  1127→    else meshRefs.current.delete(id)
  1128→  }, [])
  1129→
  1130→  const setConectMeshRef = useCallback((id, node) => {
  1131→    if (node) conectMeshRefs.current.set(id, node)
  1132→    else conectMeshRefs.current.delete(id)
  1133→  }, [])
  1134→
  1135→  const handleDrop = useCallback((e) => {
  1136→    if (isGameMode) return
  1137→    e.preventDefault()
  1138→    const objectId = e.dataTransfer.getData('text/objectId')
  1139→    const conectType = e.dataTransfer.getData('text/conectType')
  1140→    const rect = e.currentTarget.getBoundingClientRect()
  1141→    const x = ((e.clientX - rect.left) / rect.width) * 2 - 1
  1142→    const y = -((e.clientY - rect.top) / rect.height) * 2 + 1
  1143→    const position = [x * 3, 0.5, -y * 3]
  1144→    if (objectId) addObjectToScene(objectId, position)
  1145→    else if (conectType) addConectToScene(conectType, position)
  1146→  }, [addObjectToScene, addConectToScene, isGameMode])
  1147→
  1148→  const selectObject = useCallback((instanceId) => {
  1149→    if (isGameMode) return
  1150→    setSelectedInstanceId(instanceId)
  1151→    setSelectedType('object')
  1152→    selectConect(null)
  1153→  }, [selectConect, isGameMode])
  1154→
  1155→  const selectConectInstance = useCallback((instanceId) => {
  1156→    if (isGameMode) return
  1157→    setSelectedInstanceId(instanceId)
  1158→    setSelectedType('conect')
  1159→    selectConect(instanceId)
  1160→  }, [selectConect, isGameMode])
  1161→
  1162→  const handleTransformUpdate = useCallback(() => {
  1163→    if (!selectedMesh || !selectedInstanceId) return
  1164→    const transform = {
  1165→      position: [selectedMesh.position.x, selectedMesh.position.y, selectedMesh.position.z],
  1166→      rotation: [selectedMesh.rotation.x, selectedMesh.rotation.y, selectedMesh.rotation.z],
  1167→      scale: [selectedMesh.scale.x, selectedMesh.scale.y, selectedMesh.scale.z],
  1168→    }
  1169→    if (selectedType === 'conect') updateConect(selectedInstanceId, transform)
  1170→    else updateSceneInstance(selectedInstanceId, transform)
  1171→  }, [selectedMesh, selectedInstanceId, selectedType, updateConect, updateSceneInstance])
  1172→
  1173→  if (!activeScene) {
  1174→    return (
  1175→      <div className="viewport" style={{ display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
  1176→        <div className="empty-state">
  1177→          <div style={{ fontSize: 32, opacity: 0.4 }}></div>
  1178→          <div>Nenhuma cena ativa.</div>
  1179→        </div>
  1180→      </div>
  1181→    )
  1182→  }
  1183→
  1184→  return (
  1185→    <div className="viewport" onDragOver={(e) => !isGameMode && e.preventDefault()} onDrop={handleDrop}>
  1186→      <Canvas
  1187→        shadows
  1188→        dpr={[1, 2]}
  1189→        camera={{ position: [8, 6, 10], fov: 50, near: 0.1, far: 200 }}
  1190→        gl={{ antialias: true, preserveDrawingBuffer: true, alpha: false }}
  1191→        onPointerMissed={() => {
  1192→          if (!isGameMode) {
  1193→            setSelectedInstanceId(null)
  1194→            setSelectedType(null)
  1195→            selectConect(null)
  1196→          }
  1197→        }}
  1198→      >
  1199→        <Suspense fallback={null}>
  1200→          <SceneBackgroundSolid background={background} />
  1201→          <FogApplier conects={activeScene?.conects} />
  1202→
  1203→          <ambientLight intensity={lights.ambient.intensity} color={lights.ambient.color} />
  1204→          <directionalLight
  1205→            intensity={lights.directional.intensity}
  1206→            color={lights.directional.color}
  1207→            position={lights.directional.position}
  1208→            castShadow
  1209→            shadow-mapSize-width={2048}
  1210→            shadow-mapSize-height={2048}
  1211→            shadow-camera-left={-20}
  1212→            shadow-camera-right={20}
  1213→            shadow-camera-top={20}
  1214→            shadow-camera-bottom={-20}
  1215→          />
  1216→          <hemisphereLight intensity={0.3} groundColor="#1a1a2e" color="#ffffff" />
  1217→
  1218→          {/* Grelha — só no editor */}
  1219→          {!isGameMode && grid.visible && (
  1220→            <Grid position={[0, 0, 0]} args={[grid.size, grid.divisions]} cellColor={grid.color} sectionColor={grid.color} sectionThickness={1.2} cellThickness={0.6} fadeDistance={30} fadeStrength={1} />
  1221→          )}
  1222→          {!isGameMode && <ContactShadows position={[0, 0.001, 0]} opacity={0.35} scale={40} blur={2.5} far={5} />}
  1223→
  1224→          {/* Objetos do catálogo */}
  1225→          {activeScene.objects.map((instance) => {
  1226→            const obj = objects.find((o) => o.id === instance.objectId)
  1227→            if (!obj) return null
  1228→            const sceneObj = { ...obj, id: instance.instanceId, position: instance.position, rotation: instance.rotation, scale: instance.scale }
  1229→            const isSelected = !isGameMode && instance.instanceId === selectedInstanceId
  1230→            const isPlayer = instance.instanceId === activeScene.playerObjectId
  1231→            return (
  1232→              <group key={instance.instanceId}>
  1233→                <SceneObject ref={(node) => setMeshRef(instance.instanceId, node)} obj={sceneObj} isSelected={isSelected} onSelect={() => selectObject(instance.instanceId)} />
  1234→                {isPlayer && !isGameMode && <PlayerMarker position={instance.position} />}
  1235→              </group>
  1236→            )
  1237→          })}
  1238→
  1239→          {/* Conects */}
  1240→          {(activeScene.conects || []).map((conect) => (
  1241→            <ConectSelectorWrapper key={conect.instanceId} conect={conect} objects={objects} isSelected={!isGameMode && conect.instanceId === selectedInstanceId} onSelect={() => selectConectInstance(conect.instanceId)} setMeshRef={(node) => setConectMeshRef(conect.instanceId, node)} isGameMode={isGameMode} />
  1242→          ))}
  1243→
  1244→          {/* GameCamera gizmo — só no editor */}
  1245→          {!isGameMode && !(activeScene.conects || []).some((c) => c.type === 'ViewObject') && (
  1246→            <GameCameraGizmo camera={activeScene.gameCamera} />
  1247→          )}
  1248→
  1249→          {/* TransformControls — só no editor */}
  1250→          {!isGameMode && selectedMesh && selectedInstanceId && (
  1251→            <TransformControls object={selectedMesh} mode={transformMode} size={1.2}
  1252→              onMouseDown={() => { if (orbitRef.current) orbitRef.current.enabled = false }}
  1253→              onMouseUp={() => { if (orbitRef.current) orbitRef.current.enabled = true; handleTransformUpdate() }}
  1254→              onObjectChange={handleTransformUpdate}
  1255→            />
  1256→          )}
  1257→
  1258→          {/* OrbitControls — só no editor */}
  1259→          {!isGameMode && (
  1260→            <OrbitControls ref={orbitRef} makeDefault enableDamping dampingFactor={0.08} minDistance={1} maxDistance={100} maxPolarAngle={Math.PI * 0.495}
  1261→              touches={{ ONE: THREE.TOUCH.ROTATE, TWO: THREE.TOUCH.DOLLY_PAN }}
  1262→            />
  1263→          )}
  1264→
  1265→          {/* GameMode — activado quando scenePreviewOpen */}
  1266→          <GameMode activeScene={activeScene} objects={objects} meshRefs={meshRefs} conectMeshRefs={conectMeshRefs} isGameMode={isGameMode} />
  1267→        </Suspense>
  1268→      </Canvas>
  1269→    </div>
  1270→  )
  1271→}
