    59→// Componente interno que aplica o fundo da cena
    60→function SceneBackgroundSolid({ background, hasSkyObject }) {
    61→  const { scene } = useThree()
    62→  useEffect(() => {
    63→    // Se há um SkyObject na cena, ele trata do background — não sobrescrever
    64→    if (hasSkyObject) return
    65→    if (background.type === 'gradient') {
    66→      const canvas = document.createElement('canvas')
    67→      canvas.width = 2; canvas.height = 256
    68→      const ctx = canvas.getContext('2d')
    69→      const grad = ctx.createLinearGradient(0, 0, 0, 256)
    70→      grad.addColorStop(0, background.gradientTop)
    71→      grad.addColorStop(1, background.gradientBottom)
    72→      ctx.fillStyle = grad; ctx.fillRect(0, 0, 2, 256)
    73→      const tex = new THREE.CanvasTexture(canvas)
    74→      tex.colorSpace = THREE.SRGBColorSpace
    75→      scene.background = tex
    76→      return () => tex.dispose()
    77→    } else {
    78→      scene.background = new THREE.Color(background.color)
    79→    }
    80→  }, [background, scene, hasSkyObject])
    81→  return null
    82→}

    84→// ===== Componente que aplica FogObject à cena =====
    85→function FogApplier({ conects }) {
    86→  const { scene } = useThree()
    87→  const fogConect = (conects || []).find(c => c.type === 'FogObject')
    88→  useEffect(() => {
    89→    if (!fogConect) {
    90→      scene.fog = null
    91→      return
    92→    }
    93→    if (fogConect.fogType === 'exponential') {
    94→      scene.fog = new THREE.FogExp2(fogConect.color || 0xa0a0a0, fogConect.density || 0.02)
    95→    } else {
    96→      scene.fog = new THREE.Fog(fogConect.color || 0xa0a0a0, fogConect.near || 5, fogConect.far || 50)
    97→    }
    98→  }, [fogConect?.fogType, fogConect?.color, fogConect?.near, fogConect?.far, fogConect?.density, scene])
    99→  return null
   100→}

   102→// ===== Helper: encontrar uma luz por nome ou instanceId =====
   103→// Procura em todos os conects de luz (LuminousObject, SunObject, PointObject, etc.)
   104→// e devolve o THREE.Light correspondente do meshRef.
   105→function findLight(lightId, setupScene, conectMeshRefs) {
   106→  const lightTypes = ['LuminousObject', 'SunObject', 'PointObject', 'SpotObject', 'AreaObject', 'AmbientObject']
   107→  const conects = (setupScene?.conects || []).filter(c => lightTypes.includes(c.type))
   108→  // Procurar por instanceId ou por name
   109→  const conect = conects.find(c => c.instanceId === lightId || c.name === lightId)
   110→  if (!conect) return null
   111→  const mesh = conectMeshRefs.current.get(conect.instanceId)
   112→  if (!mesh) return null
   113→  // Procurar a luz dentro do grupo
   114→  let light = null
   115→  mesh.traverse(obj => { if (obj.isLight && !light) light = obj })
   116→  return light
   117→}

   120→function GameMode({ activeScene, objects, meshRefs, conectMeshRefs, isGameMode }) {
   121→  const { camera } = useThree()
   122→  const physicsRef = useRef(null)
   123→  const runtimesRef = useRef(new Map())
   124→  const animPlayersRef = useRef(new Map())
   125→  const animControllersRef = useRef(new Map()) // instanceId → animationController
   126→  const playerSpeedRef = useRef(new Map()) // instanceId → speed (para blending idle/walk)
   127→  const npcAIsRef = useRef(new Map())
   128→  const timerStatesRef = useRef(new Map())

   131→  const inventoryRef = useRef({})
   132→
   133→  // P2.5 fix: armazenar a cena ativa num ref para que modificações feitas pelo
   134→  // FlirCode (createObject, etc.) NÃO reiniciem o jogo (evita loop infinito).
   135→  // O useEffect só deve correr quando isGameMode muda (true→false ou false→true),
   136→  // não quando activeScene é modificado por spawnObject.
   137→  const activeSceneRef = useRef(activeScene)
   138→  const objectsRef = useRef(objects)
   139→  const gameStartedRef = useRef(false)
   140→
   141→  // Expor joystickRef globalmente para o GameUIOverlay poder atualizá-lo
   142→  useEffect(() => {
   143→    window._flirJoystick = joystickRef.current
   144→    window._flirInventory = inventoryRef.current
   145→    return () => { window._flirJoystick = null; window._flirInventory = null }
   146→  }, [])
   147→
   148→  // Use o ref para o setup
   149→  const setupScene = activeSceneRef.current
   150→
   151→  // Fase 5: Expor multiplayer globalmente e configurar sync de estado do PersonalObject
   152→  useEffect(() => {
   153→    if (!isGameMode) return
   154→    let mp
   155→    import('../../utils/multiplayer/multiplayerManager').then(({ multiplayer }) => {
   156→      mp = multiplayer
   157→      window._multiplayer = mp
   158→      // Configurar getter de estado local (posição do PersonalObject)
   159→      mp.setLocalStateGetter(() => {
   160→        for (const conect of setupScene?.conects || []) {
   161→          if (conect.type === 'PersonalObject') {
   162→            const mesh = conectMeshRefs.current.get(conect.instanceId)
   163→            if (mesh) {
   164→              return {
   165→                position: [mesh.position.x, mesh.position.y, mesh.position.z],
   166→                rotation: [mesh.rotation.x, mesh.rotation.y, mesh.rotation.z],
   167→                playerId: mp.playerId,
   168→              }
   169→            }
   170→          }
   171→        }
   172→        return null
   173→      })
   174→      // Registar callbacks para eventos multiplayer → FlirCode
   175→      mp.on('playerJoin', (playerId) => {
   176→        for (const rt of runtimesRef.current.values()) {
   177→          rt.triggerEvent('onPlayerJoin', { playerId })
   178→        }
   179→      })
   180→      mp.on('playerLeave', (playerId) => {
   181→        for (const rt of runtimesRef.current.values()) {
   182→          rt.triggerEvent('onPlayerLeave', { playerId })
   183→        }
   184→      })
   185→      mp.on('message', (playerId, data) => {
   186→        for (const rt of runtimesRef.current.values()) {
   187→          rt.triggerEvent('onMessage', { playerId, data })
   188→        }
   189→      })
   190→    })
   191→    return () => {
   192→      window._multiplayer = null
   193→    }
   194→  }, [isGameMode, setupScene])
   195→
   196→  // Atualizar refs sem re-disparar o useEffect
   197→  useEffect(() => {
   198→    // Só atualizar o ref se o jogo ainda não começou OU se a cena mudou de verdade
   199→    // (não por spawnObject, mas por mudança de cena manual via changeScene)
   200→    if (!gameStartedRef.current || (activeScene && activeScene.id !== activeSceneRef.current?.id)) {
   201→      activeSceneRef.current = activeScene
   202→      objectsRef.current = objects
   203→    } else {
   204→      // Jogo em curso: manter o snapshot antigo, mas atualizar objects (para spawn)
   205→      objectsRef.current = objects
   206→    }
   207→  }, [activeScene, objects])
   208→
   209→  // Setup quando o modo jogo é activado
   210→  useEffect(() => {
   211→    if (!isGameMode || !setupScene) return
   212→    gameStartedRef.current = true
   213→
   214→    debugLog('Jogo iniciado', 'log', 'Game')
   215→
   216→    // Criar gameContext
   217→    const gameContext = {
   218→      globalVars: { _score: 0 },
   219→      // P2.5: setVar/getVar — expostos no gameContext para o FlirCode
   220→      setVar: (name, value) => { gameContext.globalVars[name] = value },
   221→      getVar: (name) => gameContext.globalVars[name] ?? 0,
   222→      moveObject: (instanceId, direction, speed) => {
   223→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   224→        if (mesh) {
   225→          mesh.position.x += direction[0] * speed * 0.016
   226→          mesh.position.y += direction[1] * speed * 0.016
   227→          mesh.position.z += direction[2] * speed * 0.016
   228→        }
   229→      },
   230→      rotateObject: (instanceId, rotation) => {
   231→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   232→        if (mesh) {
   233→          mesh.rotation.x += THREE.MathUtils.degToRad(rotation[0]) * 0.016
   234→          mesh.rotation.y += THREE.MathUtils.degToRad(rotation[1]) * 0.016
   235→          mesh.rotation.z += THREE.MathUtils.degToRad(rotation[2]) * 0.016
   236→        }
   237→      },
   238→      playAnimation: (instanceId, clip) => {
   239→        const player = animPlayersRef.current.get(instanceId)
   240→        if (player) player.play(clip, { loop: true })
   241→      },
   242→      playSound: (url) => { try { new Audio(url).play() } catch {} },
   243→      playSoundByName: (name) => {
   244→        const sc = (setupScene.conects || []).find((c) => c.type === 'SoundObject' && c.name === name)
   245→        if (sc && sc.url) { try { const a = new Audio(sc.url); a.volume = sc.volume ?? 1; a.loop = sc.loop || false; a.play() } catch {} }
   246→      },
   247→      destroyObject: (instanceId) => {
   248→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   249→        if (mesh) mesh.visible = false
   250→      },
   251→      spawnObject: (objectName, position) => {
   252→        const obj = objectsRef.current.find((o) => o.name === objectName)
   253→        if (obj) useStore.getState().addObjectToScene(obj.id, position)
   254→      },
   255→      changeScene: (sceneName) => {
   256→        const scenes = useStore.getState().scenes
   257→        const target = scenes.find((s) => s.name === sceneName)
   258→        if (target) useStore.getState().setActiveScene(target.id)
   259→      },
   260→      setVisible: (instanceId, visible) => {
   261→        const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   262→        if (mesh) mesh.visible = visible
   263→      },
   264→      applyForce: (instanceId, force) => physicsRef.current?.applyForce(instanceId, force),
   265→      jumpPlayer: (instanceId) => physicsRef.current?.jumpPersonal(instanceId),
   266→      showUIScreen: (name) => {
   267→        const s = useStore.getState().uiScreens.find((s) => s.name === name)
   268→        if (s) useStore.getState().setUIScreenVisible(s.id, true)
   269→      },
   270→      hideUIScreen: (name) => {
   271→        const s = useStore.getState().uiScreens.find((s) => s.name === name)
   272→        if (s) useStore.getState().setUIScreenVisible(s.id, false)
   273→      },
   274→      getUIValue: (name) => {
   275→        for (const s of useStore.getState().uiScreens) {
   276→          const el = s.elements.find((e) => e.name === name)
   277→          if (el) return el.value ?? el.text ?? ''
   278→        }
   279→        return ''
   280→      },
   281→      setUIValue: (name, val) => {
   282→        for (const s of useStore.getState().uiScreens) {
   283→          const el = s.elements.find((e) => e.name === name)
   284→          if (el) { useStore.getState().updateUIElement(el.id, { value: val, text: val, label: val }); return }
   285→        }
   286→      },
   287→      triggerUIEvent: (eventName, payload) => {
   288→        for (const rt of runtimesRef.current.values()) rt.triggerEvent(eventName, payload)
   289→      },
   290→      collidingWith: (instanceId, type) => {
   291→        if (!physicsRef.current) return false
   292→        const entry = physicsRef.current.bodies.get(instanceId)
   293→        if (!entry) return false
   294→        for (const [otherId, otherEntry] of physicsRef.current.bodies) {
   295→          if (otherId === instanceId) continue
   296→          if (otherEntry.conect.type === type || otherEntry.conect.name === type) {
   297→            if (entry.body.position.distanceTo(otherEntry.body.position) < 1.5) return true
   298→          }
   299→        }
   300→        return false
   301→      },
   302→      distanceTo: (instanceId, targetName) => {
   303→        // P2.5: procurar em objects E conects da cena ativa
   304→        for (const inst of setupScene.objects || []) {
   305→          const obj = objectsRef.current.find((o) => o.id === inst.objectId)
   306→          if (obj && (obj.name === targetName || inst.name === targetName)) {
   307→            const target = meshRefs.current.get(inst.instanceId)
   308→            const source = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   309→            if (source && target) return source.position.distanceTo(target.position)
   310→          }
   311→        }
   312→        for (const conect of setupScene.conects || []) {
   313→          if (conect.name === targetName) {
   314→            const target = conectMeshRefs.current.get(conect.instanceId)
   315→            const source = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   316→            if (source && target) return source.position.distanceTo(target.position)
   317→          }
   318→        }
   319→        return 0
   320→      },
   321→      isTouching: () => joystickRef.current.active,
   322→      // Fase 5: Multiplayer functions
   323→      sendMessage: (data) => {
   324→        try {
   325→          const mp = window._multiplayer
   326→          if (mp && mp.connected) mp.sendMessage(data)
   327→        } catch {}
   328→      },
   329→      getPlayers: () => {
   330→        const mp = window._multiplayer
   331→        return mp && mp.connected ? mp.getOtherPlayers().length + 1 : 1
   332→      },
   333→      getPlayerState: (playerId) => {
   334→        const mp = window._multiplayer
   335→        return mp && mp.connected ? mp.getPlayerState(playerId) : null
   336→      },
   337→      // Sistema 3: Sinais — emite para todos os runtimes da cena
   338→      emitSignal: (name, data) => {
   339→        for (const rt of runtimesRef.current.values()) {
   340→          rt.triggerEvent('onSignal', { name, data })
   341→        }
   342→        debugLog(`Signal emitido: ${name}`, 'log', 'Signals')
   343→      },
   344→      // Sistema 2: Armas e combate
   345→      shoot: () => {
   346→        const weaponState = weaponStateRef.current
   347→        if (!weaponState || !weaponState.equipped) {
   348→          debugLog('shoot() sem arma equipada', 'warning', 'Weapon')
   349→          return
   350→        }
   351→        const now = performance.now() / 1000
   352→        if (now - weaponState.lastShot < weaponState.fireRate) return
   353→        if (weaponState.ammo <= 0) {
   354→          debugLog('Sem munição — pressiona reload()', 'warning', 'Weapon')
   355→          return
   356→        }
   357→        weaponState.lastShot = now
   358→        weaponState.ammo--
   359→        debugLog(`Disparo! Munição restante: ${weaponState.ammo}`, 'log', 'Weapon')
   360→        // Raycast do centro do ecrã (câmara) ou da posição do jogador
   361→        const player = (setupScene?.conects || []).find((c) => c.type === 'PersonalObject')
   362→        if (player) {
   363→          const playerMesh = conectMeshRefs.current.get(player.instanceId)
   364→          if (playerMesh) {
   365→            const origin = new THREE.Vector3(playerMesh.position.x, playerMesh.position.y + 1, playerMesh.position.z)
   366→            const dir = new THREE.Vector3(0, 0, -1)
   367→            dir.applyQuaternion(playerMesh.quaternion)
   368→            const ray = new THREE.Raycaster(origin, dir, 0, weaponState.range)
   369→            // Verificar colisão com outros conects
   370→            let hit = false
   371→            for (const [id, entry] of physicsRef.current?.bodies || []) {
   372→              if (id === player.instanceId) continue
   373→              const targetMesh = meshRefs.current.get(id) || conectMeshRefs.current.get(id)
   374→              if (targetMesh) {
   375→                const intersects = ray.intersectObject(targetMesh, true)
   376→                if (intersects.length > 0) {
   377→                  // Aplicar dano
   378→                  gameContext.takeDamage(id, weaponState.damage)
   379→                  debugLog(`Alvo atingido: ${entry.conect.name} (-${weaponState.damage} HP)`, 'log', 'Weapon')
   380→                  hit = true
   381→                  break
   382→                }
   383→              }
   384→            }
   385→            if (!hit) debugLog('Disparo falhou (sem alvo)', 'log', 'Weapon')
   386→          }
   387→        }
   388→      },
   389→      reload: () => {
   390→        const ws = weaponStateRef.current
   391→        if (!ws || !ws.equipped) return
   392→        if (ws.ammo >= ws.maxAmmo) return
   393→        debugLog(`A recarregar... (${ws.reloadTime}s)`, 'log', 'Weapon')
   394→        setTimeout(() => {
   395→          ws.ammo = ws.maxAmmo
   396→          debugLog(`Recarregado! Munição: ${ws.ammo}`, 'log', 'Weapon')
   397→        }, ws.reloadTime * 1000)
   398→      },
   399→      equipWeapon: (weaponName) => {
   400→        const weapon = (setupScene?.conects || []).find((c) => c.type === 'WeaponObject' && (c.name === weaponName || c.instanceId === weaponName))
   401→        if (!weapon) {
   402→          debugLog(`Arma "${weaponName}" não encontrada`, 'warning', 'Weapon')
   403→          return
   404→        }
   405→        weaponStateRef.current = {
   406→          equipped: true,
   407→          damage: weapon.damage || 25,
   408→          fireRate: weapon.fireRate || 0.3,
   409→          range: weapon.range || 50,
   410→          maxAmmo: weapon.maxAmmo || 30,
   411→          ammo: weapon.maxAmmo || 30,
   412→          reloadTime: weapon.reloadTime || 2,
   413→          lastShot: 0,
   414→          weaponId: weapon.instanceId,
   415→        }
   416→        // Mostrar mira se necessário
   417→        if (weapon.showCrosshair) {
   418→          window._flirCrosshair = true
   419→        }
   420→        debugLog(`Arma "${weapon.name}" equipada (${weaponStateRef.current.ammo} balas)`, 'log', 'Weapon')
   421→      },
   422→      getAmmo: () => weaponStateRef.current?.ammo ?? 0,
   423→      takeDamage: (instanceId, amount) => {
   424→        // Reduzir vida do alvo
   425→        for (const conect of setupScene?.conects || []) {
   426→          if (conect.instanceId === instanceId) {
   427→            const currentHealth = conect.health ?? 100
   428→            const newHealth = Math.max(0, currentHealth - amount)
   429→            useStore.getState().updateConect(instanceId, { health: newHealth })
   430→            debugLog(`${conect.name} recebeu ${amount} de dano (vida: ${newHealth})`, 'log', 'Combat')
   431→            // Disparar onDamage no runtime do alvo
   432→            const rt = runtimesRef.current.get(instanceId)
   433→            if (rt) rt.triggerEvent('onDamage', { amount, source: 'weapon' })
   434→            if (newHealth <= 0) {
   435→              debugLog(`${conect.name} foi destruído`, 'log', 'Combat')
   436→              const mesh = meshRefs.current.get(instanceId) || conectMeshRefs.current.get(instanceId)
   437→              if (mesh) mesh.visible = false
   438→            }
   439→            break
   440→          }
   441→        }
   442→      },
   443→      getHealth: (instanceId) => {
   444→        for (const conect of setupScene?.conects || []) {
   445→          if (conect.instanceId === instanceId) return conect.health ?? 100
   446→        }
   447→        return 100
   448→      },
   449→      // Sistema 3: Inventário
   450→      addToInventory: (itemName, quantity = 1) => {
   451→        const inv = inventoryRef.current
   452→        inv[itemName] = (inv[itemName] || 0) + quantity
   453→        debugLog(`Item "${itemName}" adicionado (${quantity}). Total: ${inv[itemName]}`, 'log', 'Inventory')
   454→        // Trigger onPickup em todos os runtimes
   455→        for (const rt of runtimesRef.current.values()) {
   456→          rt.triggerEvent('onPickup', { itemName, quantity })
   457→        }
   458→      },
   459→      removeFromInventory: (itemName, quantity = 1) => {
   460→        const inv = inventoryRef.current
   461→        if (!inv[itemName]) return
   462→        inv[itemName] = Math.max(0, inv[itemName] - quantity)
   463→        if (inv[itemName] === 0) delete inv[itemName]
   464→        debugLog(`Item "${itemName}" removido (${quantity}). Restante: ${inv[itemName] || 0}`, 'log', 'Inventory')
   465→      },
   466→      getInventoryCount: (itemName) => inventoryRef.current[itemName] || 0,
   467→      hasItem: (itemName) => (inventoryRef.current[itemName] || 0) > 0,
   468→      // Sistema: Links — navegar para cena ou tela de UI
   469→      linkTo: (target, subTarget) => {
   470→        if (target === 'scene') {
   471→          // Mudar de cena
   472→          const scenes = useStore.getState().scenes
   473→          const targetScene = scenes.find((s) => s.name === subTarget || s.id === subTarget)
   474→          if (targetScene) {
   475→            useStore.getState().setActiveScene(targetScene.id)
   476→            debugLog(`Link: mudou para cena "${targetScene.name}"`, 'log', 'Links')
   477→          } else {
   478→            debugLog(`Link: cena "${subTarget}" não encontrada`, 'warning', 'Links')
   479→          }
   480→        } else if (target === 'screen') {
   481→          // Mostrar tela de UI e esconder as outras
   482→          const screens = useStore.getState().uiScreens
   483→          const targetScreen = screens.find((s) => s.name === subTarget || s.id === subTarget)
   484→          if (targetScreen) {
   485→            screens.forEach((s) => useStore.getState().setUIScreenVisible(s.id, s.id === targetScreen.id))
   486→            debugLog(`Link: mudou para tela "${targetScreen.name}"`, 'log', 'Links')
   487→          } else {
   488→            debugLog(`Link: tela "${subTarget}" não encontrada`, 'warning', 'Links')
   489→          }
   490→        } else if (target === 'url') {
   491→          // Abrir URL externa
   492→          window.open(subTarget, '_blank')
   493→          debugLog(`Link: abriu URL "${subTarget}"`, 'log', 'Links')
   494→        }
   495→      },
   496→
   497→      // ===== Funções de luz (FlirCode) =====
   498→      // setLightIntensity(nomeOuId, valor) — ajusta intensidade de uma luz por nome ou instanceId
   499→      setLightIntensity: (lightId, value) => {
   500→        const light = findLight(lightId, setupScene, conectMeshRefs)
   501→        if (light) {
   502→          light.intensity = value
   503→          debugLog(`Luz "${lightId}" intensidade = ${value}`, 'log', 'Light')
   504→        }
   505→      },
   506→      // setLightColor(nomeOuId, cor) — ajusta cor de uma luz (hex string ex: '#ff0000')
   507→      setLightColor: (lightId, color) => {
   508→        const light = findLight(lightId, setupScene, conectMeshRefs)
   509→        if (light) {
   510→          light.color = new THREE.Color(color)
   511→          debugLog(`Luz "${lightId}" cor = ${color}`, 'log', 'Light')
   512→        }
   513→      },
   514→      // setLightVisible(nomeOuId, bool) — liga/desliga uma luz
   515→      setLightVisible: (lightId, visible) => {
   516→        const mesh = conectMeshRefs.current.get(lightId)
   517→        if (mesh) {
   518→          mesh.visible = visible
   519→          // Também esconder luzes filho
   520→          mesh.traverse(obj => { if (obj.isLight) obj.visible = visible })
   521→        }
   522→      },
   523→    }
   524→    window._flirGameContext = gameContext
   525→
   526→    // Física
   527→    const gravity = setupScene.physics?.gravity || [0, -9.82, 0]
   528→    physicsRef.current = createPhysicsSystem({ gravity: gravity[1] })
   529→
   530→    // Registar conects com física
   531→    setTimeout(() => {
   532→      if (!physicsRef.current) return
   533→      for (const conect of setupScene.conects || []) {
   534→        const mesh = conectMeshRefs.current.get(conect.instanceId)
   535→        if (mesh) physicsRef.current.addConect(conect, mesh)
   536→      }
   537→      for (const conect of setupScene.conects || []) {
   538→        if (conect.type === 'JointObject' && conect.targetA && conect.targetB) {
   539→          physicsRef.current.addJoint(conect)
   540→        }
   541→      }
   542→    }, 50)
   543→
   544→    // Eventos de física
   545→    physicsRef.current.on('onCollision', ({ instanceId, otherInstanceId }) => {
   546→      const rt = runtimesRef.current.get(instanceId)
   547→      if (rt) rt.triggerEvent('onCollision', { other: otherInstanceId })
   548→    })
   549→    physicsRef.current.on('onTriggerEnter', ({ instanceId, otherInstanceId }) => {
   550→      const rt = runtimesRef.current.get(instanceId)
   551→      if (rt) rt.triggerEvent('onEnterZone', { other: otherInstanceId })
   552→    })
   553→    physicsRef.current.on('onTriggerExit', ({ instanceId, otherInstanceId }) => {
   554→      const rt = runtimesRef.current.get(instanceId)
   555→      if (rt) rt.triggerEvent('onExitZone', { other: otherInstanceId })
   556→    })
   557→
   558→    // FlirCode/FlirScript runtimes
   559→    // Sistema 2: obter classes FlirCode do store
   560→    const flirCodeClasses = useStore.getState().flirCodeClasses || []
   561→
   562→    const setupRuntime = (instance, scriptData) => {
   563→      try {
   564→        // Sistema 2: se o conect tem flirCodeClass definido, usar o código da classe
   565→        let source = null
   566→        let className = null
   567→
   568→        if (instance.flirCodeClass) {
   569→          const cls = flirCodeClasses.find((c) => c.name === instance.flirCodeClass)
   570→          if (cls) {
   571→            source = cls.source
   572→            className = cls.name
   573→          }
   574→        }
   575→
   576→        // Se há script próprio, concatenar com a classe (script próprio por cima)
   577→        if (scriptData && typeof scriptData === 'string' && scriptData.startsWith('FLIRCODE:')) {
   578→          const ownScript = scriptData.slice(9)
   579→          source = source ? source + '\n' + ownScript : ownScript
   580→        }
   581→
   582→        if (!source) return
   583→
   584→        const rt = createFlirCodeRuntime(source, {
   585→          ...gameContext,
   586→          _instanceId: instance.instanceId,
   587→          className, // Sistema 2: passar nome da classe para o runtime
   588→          mesh: conectMeshRefs.current.get(instance.instanceId) || meshRefs.current.get(instance.instanceId),
   589→        })
   590→        if (!rt.hasErrors) {
   591→          runtimesRef.current.set(instance.instanceId, rt)
   592→          rt.triggerEvent('beginPlay')
   593→        }
   594→      } catch (err) {
   595→        debugLog('Erro ao inicializar script: ' + err.message, 'error', 'Script')
   596→      }
   597→    }
   598→    for (const inst of setupScene.objects || []) setupRuntime(inst, inst.flirScript)
   599→    for (const conect of setupScene.conects || []) setupRuntime(conect, conect.flirScript)
   600→
   601→    // Animation players
   602→    // NOTA: as animações vivem no objeto do catálogo (em objects), não na instância.
   603→    // Para instâncias de objetos (setupScene.objects), precisamos de ir buscar o
   604→    // objeto original via inst.objectId. Para conects (PersonalObject/NpcObject com
   605→    // sourceObjectId), fazemos o mesmo via conect.sourceObjectId.
   606→    const setupAnimationPlayer = (inst, sourceObjectId) => {
   607→      if (!sourceObjectId) return
   608→      const catalogObj = objectsRef.current.find((o) => o.id === sourceObjectId)
   609→      const animations = catalogObj?.animations || inst.animations
   610→      if (!animations || Object.keys(animations).length === 0) return
   611→      const getMesh = () => meshRefs.current.get(inst.instanceId) || conectMeshRefs.current.get(inst.instanceId)
   612→      // getBones: extrai THREE.Bone[] do SkinnedMesh (se o objeto tem esqueleto)
   613→      // Isto é CRÍTICO — sem isto, applyPose nunca modifica os bones e o
   614→      // SkinnedMesh não deforma.
   615→      const getBones = () => {
   616→        const mesh = getMesh()
   617→        if (mesh && mesh.isSkinnedMesh && mesh.skeleton) return mesh.skeleton.bones
   618→        return null
   619→      }
   620→      const player = createAnimationPlayer(animations, getMesh, getBones)
   621→
   622→      // Procurar AnimationBoostObject na cena para ativar blending
   623→      const boostConect = (setupScene.conects || []).find(c => c.type === 'AnimationBoostObject')
   624→      if (boostConect) {
   625→        player.setBoost(true, boostConect.blendTime || 0.3)
   626→      }
   627→
   628→      animPlayersRef.current.set(inst.instanceId, player)
   629→
   630→      // Se é PersonalObject ou NpcObject com animationController, criar controller
   631→      if (inst.type === 'PersonalObject' || inst.type === 'NpcObject') {
   632→        const controllerGraph = inst.animationController || defaultAnimationController()
   633→        const getContext = () => {
   634→          const speed = playerSpeedRef.current.get(inst.instanceId) || 0
   635→          const mesh = getMesh()
   636→          let grounded = true
   637→          if (mesh && physicsRef.current) {
   638→            // Tentar obter grounded do physics system
   639→            try { grounded = physicsRef.current.isGrounded?.(inst.instanceId) ?? true } catch { grounded = true }
   640→          }
   641→          return { speed, grounded, attacking: false }
   642→        }
   643→        const controller = createAnimationController(controllerGraph, getContext)
   644→        animControllersRef.current.set(inst.instanceId, controller)
   645→
   646→        // Tocar o clip do estado inicial
   647→        const initialState = controller.getState()
   648→        if (initialState?.clip && animations[initialState.clip]) {
   649→          player.play(initialState.clip, { loop: true, blendTime: 0.3 })
   650→        } else if (animations.idle) {
   651→          player.play('idle', { loop: true })
   652→        }
   653→      } else {
   654→        // Objeto normal: tocar idle por defeito
   655→        if (animations.idle) player.play('idle', { loop: true })
   656→        else { const f = Object.keys(animations)[0]; if (f) player.play(f, { loop: true }) }
   657→      }
   658→    }
   659→    for (const inst of setupScene.objects || []) setupAnimationPlayer(inst, inst.objectId)
   660→    for (const conect of setupScene.conects || []) setupAnimationPlayer(conect, conect.sourceObjectId)
   661→
   662→    // NPC AI
   663→    for (const conect of setupScene.conects || []) {
   664→      if (conect.type === 'NpcObject') {
   665→        const ai = createNPCAI(conect, {
   666→          getPlayerPos: () => {
   667→            const player = (setupScene.conects || []).find((c) => c.type === 'PersonalObject')
   668→            if (!player) return null
   669→            const pm = conectMeshRefs.current.get(player.instanceId)
   670→            return pm ? [pm.position.x, pm.position.y, pm.position.z] : null
   671→          },
   672→          getPathPoints: (pathId) => {
   673→            const path = (setupScene.conects || []).find((c) => c.instanceId === pathId)
   674→            return path?.points || null
   675→          },
   676→          physicsMove: (id, dir, speed) => physicsRef.current?.movePersonal(id, dir, speed),
   677→          physicsJump: (id) => physicsRef.current?.jumpPersonal(id),
   678→          emitEvent: (en, payload) => {
   679→            const rt = runtimesRef.current.get(conect.instanceId)
   680→            if (rt) {
   681→              if (en === 'OnSeePlayer') rt.triggerEvent('onSeePlayer', payload)
   682→              else if (en === 'OnLoseSight') rt.triggerEvent('onLoseSight', payload)
   683→            }
   684→          },
   685→        })
   686→        npcAIsRef.current.set(conect.instanceId, ai)
   687→      }
   688→    }
   689→
   690→    // Timers
   691→    for (const conect of setupScene.conects || []) {
   692→      if (conect.type === 'TimerObject' && conect.autoStart) {
   693→        timerStatesRef.current.set(conect.instanceId, { remaining: conect.duration || 5, loop: conect.loop || false, duration: conect.duration || 5 })
   694→      }
   695→    }
   696→
   697→    // SkyObject / FogObject
   698→    const skyConect = (setupScene.conects || []).find((c) => c.type === 'SkyObject')
   699→    if (skyConect) {
   700→      // Aplicado via SceneBackgroundSolid se mudar o background
   701→    }
   702→    const fogConect = (setupScene.conects || []).find((c) => c.type === 'FogObject')
   703→    // Fog aplicado no useFrame
   704→
   705→    // Joystick touch
   706→    const onTouchStart = (e) => { if (e.touches.length === 1) { joystickRef.current.active = true } }
   707→    const onTouchEnd = () => { joystickRef.current.active = false; joystickRef.current.x = 0; joystickRef.current.z = 0 }
   708→    window.addEventListener('touchstart', onTouchStart)
   709→    window.addEventListener('touchend', onTouchEnd)
   710→
   711→    // Teclado
   712→    const keys = {}
   713→    const onKeyDown = (e) => { keys[e.key.toLowerCase()] = true }
   714→    const onKeyUp = (e) => { keys[e.key.toLowerCase()] = false }
   715→    window.addEventListener('keydown', onKeyDown)
   716→    window.addEventListener('keyup', onKeyUp)
   717→    window._flirKeys = keys
   718→
   719→    return () => {
   720→      for (const rt of runtimesRef.current.values()) rt.dispose()
   721→      runtimesRef.current.clear()
   722→      for (const ai of npcAIsRef.current.values()) ai.dispose()
   723→      npcAIsRef.current.clear()
   724→      for (const [, s] of timerStatesRef.current) { if (s.interval) clearInterval(s.interval); if (s.audio) s.audio.pause() }
   725→      timerStatesRef.current.clear()
   726→      animPlayersRef.current.clear()
   727→      animControllersRef.current.clear()
   728→      playerSpeedRef.current.clear()
   729→      physicsRef.current?.dispose()
   730→      physicsRef.current = null
   731→      window.removeEventListener('touchstart', onTouchStart)
   732→      window.removeEventListener('touchend', onTouchEnd)
   733→      window.removeEventListener('keydown', onKeyDown)
   734→      window.removeEventListener('keyup', onKeyUp)
   735→      window._flirGameContext = null
   736→      gameStartedRef.current = false
   737→    }
   738→  }, [isGameMode, setupScene])
   739→
   740→  // Loop do jogo
   741→  useFrame((_, delta) => {
   742→    if (!isGameMode) return
   743→
   744→    // OTIMIZAÇÃO: limpar cache de poses no início de cada frame
   745→    // (permite que 200 NPCs partilhem a mesma pose calculada)
   746→    clearPoseCache()
   747→
   748→    // Física
   749→    if (physicsRef.current) {
   750→      physicsRef.current.update(delta)
   751→      // Sincronizar meshes com bodies
   752→      for (const [id, entry] of physicsRef.current.bodies) {
   753→        const mesh = meshRefs.current.get(id) || conectMeshRefs.current.get(id)
   754→        if (mesh) {
   755→          mesh.position.copy(entry.body.position)
   756→          mesh.quaternion.copy(entry.body.quaternion)
   757→        }
   758→      }
   759→    }
   760→
   761→    // FlirCode onTick
   762→    for (const rt of runtimesRef.current.values()) {
   763→      rt.update(delta)
   764→      rt.triggerEvent('tick', { deltaTime: delta })
   765→    }
   766→
   767→    // Animation players + controllers
   768→    for (const [id, player] of animPlayersRef.current.entries()) {
   769→      const controller = animControllersRef.current.get(id)
   770→      if (controller) {
   771→        const prevState = controller.getState()
   772→        controller.update(delta)
   773→        const newState = controller.getState()
   774→        // Se o estado mudou, mudar de clip com blending
   775→        if (newState && newState.id !== prevState?.id && newState.clip) {
   776→          const boostConect = (setupScene?.conects || []).find(c => c.type === 'AnimationBoostObject')
   777→          player.play(newState.clip, { loop: true, blendTime: boostConect?.blendTime || 0.3 })
   778→          debugLog(`Anim: ${prevState?.clip || 'none'} → ${newState.clip} (speed=${playerSpeedRef.current.get(id)?.toFixed(1)})`, 'log', 'Anim')
   779→        }
   780→      }
   781→      player.update(delta)
   782→    }
   783→
   784→    // NPC AI
   785→    for (const ai of npcAIsRef.current.values()) ai.update(delta)
   786→
   787→    // Timers
   788→    for (const [id, state] of timerStatesRef.current) {
   789→      if (id.startsWith('_')) continue
   790→      state.remaining -= delta
   791→      if (state.remaining <= 0) {
   792→        const rt = runtimesRef.current.get(id)
   793→        if (rt) rt.triggerEvent('onTimer')
   794→        if (state.loop) state.remaining = state.duration
   795→        else timerStatesRef.current.delete(id)
   796→      }
   797→    }
   798→
   799→    // Joystick → PersonalObject — usar setupScene (ref)
   800→    const keys = window._flirKeys || {}
   801→    // Reset de speeds no início de cada frame
   802→    for (const id of playerSpeedRef.current.keys()) playerSpeedRef.current.set(id, 0)
   803→    if (joystickRef.current.active || keys['w'] || keys['a'] || keys['s'] || keys['d']) {
   804→      for (const conect of setupScene?.conects || []) {
   805→        if (conect.type === 'PersonalObject') {
   806→          const speed = conect.moveSpeed || 5
   807→          let mx = 0, mz = 0
   808→          if (joystickRef.current.active) { mx = joystickRef.current.x * speed; mz = joystickRef.current.z * speed }
   809→          if (keys['w']) mz = -speed
   810→          if (keys['s']) mz = speed
   811→          if (keys['a']) mx = -speed
   812→          if (keys['d']) mx = speed
   813→          physicsRef.current?.movePersonal(conect.instanceId, [mx, 0, mz], 1)
   814→          if ((keys[' '] || keys['space']) && conect.canJump) {
   815→            physicsRef.current?.jumpPersonal(conect.instanceId)
   816→          }
   817→          // Guardar speed para o animation controller (blending idle/walk)
   818→          const currentSpeed = Math.hypot(mx, mz)
   819→          playerSpeedRef.current.set(conect.instanceId, currentSpeed)
   820→        }
   821→      }
   822→    }
   823→
   824→    // Sistema 3: Auto-pickup de ItemObjects
   825→    const player2 = (setupScene?.conects || []).find((c) => c.type === 'PersonalObject')
   826→    if (player2) {
   827→      const playerMesh = conectMeshRefs.current.get(player2.instanceId)
   828→      if (playerMesh) {
   829→        for (const item of setupScene?.conects || []) {
   830→          if (item.type === 'ItemObject' && item.autoPickup !== false) {
   831→            const itemMesh = conectMeshRefs.current.get(item.instanceId)
   832→            if (itemMesh && itemMesh.visible !== false) {
   833→              const dist = playerMesh.position.distanceTo(itemMesh.position)
   834→              if (dist <= (item.pickupRadius || 2)) {
   835→                // Apanhar item
   836→                gameContext.addToInventory(item.itemName || 'Item', item.quantity || 1)
   837→                itemMesh.visible = false
   838→                debugLog(`Item "${item.itemName}" apanhado!`, 'log', 'Inventory')
   839→              }
   840→            }
   841→          }
   842→        }
   843→      }
   844→    }
   845→
   846→    // Câmara: ViewObject ativa — usar setupScene (ref) para consistência
   847→    const viewConects = (setupScene?.conects || []).filter((c) => c.type === 'ViewObject')
   848→    const activeView = viewConects.find((c) => c.cameraRole === 'player') || viewConects.find((c) => c.cameraRole === 'primary') || viewConects[0]
   849→    if (activeView) {
   850→      // Se cameraRole='player' e não tem followTarget, seguir PersonalObject
   851→      let targetId = activeView.followTarget
   852→      if (!targetId && activeView.cameraRole === 'player') {
   853→        const player = (setupScene?.conects || []).find((c) => c.type === 'PersonalObject')
   854→        if (player) targetId = player.instanceId
   855→      }
   856→      if (targetId && activeView.followMode && activeView.followMode !== 'none') {
   857→        const targetMesh = meshRefs.current.get(targetId) || conectMeshRefs.current.get(targetId)
   858→        if (targetMesh) {
   859→          const mode = activeView.followMode
   860→          const dist = activeView.followDistance || 6
   861→          const height = activeView.followHeight || 3
   862→          if (mode === 'third') {
   863→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x, targetMesh.position.y + height, targetMesh.position.z + dist), 0.1)
   864→            camera.lookAt(targetMesh.position)
   865→          } else if (mode === 'top') {
   866→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x, targetMesh.position.y + dist, targetMesh.position.z), 0.1)
   867→            camera.lookAt(targetMesh.position)
   868→          } else if (mode === 'side') {
   869→            camera.position.lerp(new THREE.Vector3(targetMesh.position.x + dist, targetMesh.position.y + height / 2, targetMesh.position.z), 0.1)
   870→            camera.lookAt(targetMesh.position)
   871→          }
   872→        }
   873→      } else {
   874→        // Câmara estática na posição da ViewObject
   875→        camera.position.set(...(activeView.position || [5, 4, 6]))
   876→        // Sempre olhar para a origem (ou para o centro dos objetos da cena)
   877→        // a menos que followMode esteja definido
   878→        camera.lookAt(0, 0, 0)
   879→      }
   880→    }
   881→  })
   882→
   883→  return null
   884→}

   911→
   912→  useEffect(() => {

   915→    } else if (selectedInstanceId && conectMeshRefs.current.has(selectedInstanceId)) {

   917→    } else {
   918→      setSelectedMesh(null)
   919→    }
   920→  }, [selectedInstanceId, activeScene])

   934→    e.preventDefault()
   935→    const objectId = e.dataTransfer.getData('text/objectId')
   936→    const conectType = e.dataTransfer.getData('text/conectType')
   937→    const rect = e.currentTarget.getBoundingClientRect()
   938→    const x = ((e.clientX - rect.left) / rect.width) * 2 - 1
   939→    const y = -((e.clientY - rect.top) / rect.height) * 2 + 1
   940→    const position = [x * 3, 0.5, -y * 3]
   941→    if (objectId) addObjectToScene(objectId, position)
   942→    else if (conectType) addConectToScene(conectType, position)
   943→  }, [addObjectToScene, addConectToScene, isGameMode])
   944→
   945→  const selectObject = useCallback((instanceId) => {
   946→    if (isGameMode) return
   947→    setSelectedInstanceId(instanceId)
   948→    setSelectedType('object')
   949→    selectConect(null)
   950→  }, [selectConect, isGameMode])
   951→
   952→  const selectConectInstance = useCallback((instanceId) => {
   953→    if (isGameMode) return
   954→    setSelectedInstanceId(instanceId)
   955→    setSelectedType('conect')
   956→    selectConect(instanceId)
   957→  }, [selectConect, isGameMode])
   958→
   959→  const handleTransformUpdate = useCallback(() => {
   960→    if (!selectedMesh || !selectedInstanceId) return
   961→    const transform = {
   962→      position: [selectedMesh.position.x, selectedMesh.position.y, selectedMesh.position.z],
   963→      rotation: [selectedMesh.rotation.x, selectedMesh.rotation.y, selectedMesh.rotation.z],
   964→      scale: [selectedMesh.scale.x, selectedMesh.scale.y, selectedMesh.scale.z],
   965→    }
   966→    if (selectedType === 'conect') updateConect(selectedInstanceId, transform)
   967→    else updateSceneInstance(selectedInstanceId, transform)
   968→  }, [selectedMesh, selectedInstanceId, selectedType, updateConect, updateSceneInstance])
   969→
   970→  if (!activeScene) {
   971→    return (
   972→      <div className="viewport" style={{ display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
   973→        <div className="empty-state">
   974→          <div style={{ fontSize: 32, opacity: 0.4 }}>🎬</div>
   975→          <div>Nenhuma cena ativa.</div>
   976→        </div>
   977→      </div>
   978→    )
   979→  }
   980→
   981→  return (
   982→    <div className="viewport" onDragOver={(e) => !isGameMode && e.preventDefault()} onDrop={handleDrop}>
   983→      <Canvas
   984→        shadows
   985→        dpr={[1, 2]}
   986→        camera={{ position: [8, 6, 10], fov: 50, near: 0.1, far: 200 }}
   987→        gl={{ antialias: true, preserveDrawingBuffer: true, alpha: false }}
   988→        onPointerMissed={() => {
   989→          if (!isGameMode) {
   990→            setSelectedInstanceId(null)
   991→            setSelectedType(null)
   992→            selectConect(null)
   993→          }
   994→        }}
   995→      >
   996→        <Suspense fallback={null}>
   997→          <SceneBackgroundSolid background={background} hasSkyObject={!!(activeScene?.conects || []).find(c => c.type === 'SkyObject')} />
   998→          <FogApplier conects={activeScene?.conects} />
   999→
  1000→          <ambientLight intensity={lights.ambient.intensity} color={lights.ambient.color} />
  1001→          <directionalLight
  1002→            intensity={lights.directional.intensity}
  1003→            color={lights.directional.color}
  1004→            position={lights.directional.position}
  1005→            castShadow
  1006→            shadow-mapSize-width={2048}
  1007→            shadow-mapSize-height={2048}
  1008→            shadow-camera-left={-20}
  1009→            shadow-camera-right={20}
  1010→            shadow-camera-top={20}
  1011→            shadow-camera-bottom={-20}
  1012→          />
  1013→          <hemisphereLight intensity={0.3} groundColor="#1a1a2e" color="#ffffff" />
  1014→
  1015→          {/* Grelha — só no editor */}
  1016→          {!isGameMode && grid.visible && (
  1017→            <Grid position={[0, 0, 0]} args={[grid.size, grid.divisions]} cellColor={grid.color} sectionColor={grid.color} sectionThickness={1.2} cellThickness={0.6} fadeDistance={30} fadeStrength={1} />
  1018→          )}
  1019→          {!isGameMode && <ContactShadows position={[0, 0.001, 0]} opacity={0.35} scale={40} blur={2.5} far={5} />}
  1020→
  1021→          {/* Objetos do catálogo */}
  1022→          {activeScene.objects.map((instance) => {
  1023→            const obj = objects.find((o) => o.id === instance.objectId)
  1024→            if (!obj) return null
  1025→            const sceneObj = { ...obj, id: instance.instanceId, position: instance.position, rotation: instance.rotation, scale: instance.scale }
  1026→            const isSelected = !isGameMode && instance.instanceId === selectedInstanceId
  1027→            const isPlayer = instance.instanceId === activeScene.playerObjectId
  1028→            return (
  1029→              <group key={instance.instanceId}>
  1030→                <SceneObject ref={(node) => setMeshRef(instance.instanceId, node)} obj={sceneObj} isSelected={isSelected} onSelect={() => selectObject(instance.instanceId)} />
  1031→                {isPlayer && !isGameMode && <PlayerMarker position={instance.position} />}
  1032→              </group>
  1033→            )
  1034→          })}
  1035→
  1036→          {/* Conects */}
  1037→          {(activeScene.conects || []).map((conect) => (
  1038→            <ConectSelectorWrapper key={conect.instanceId} conect={conect} objects={objects} isSelected={!isGameMode && conect.instanceId === selectedInstanceId} onSelect={() => selectConectInstance(conect.instanceId)} setMeshRef={(node) => setConectMeshRef(conect.instanceId, node)} isGameMode={isGameMode} />
  1039→          ))}
  1040→
  1041→          {/* GameCamera gizmo — só no editor */}
  1042→          {!isGameMode && !(activeScene.conects || []).some((c) => c.type === 'ViewObject') && (
  1043→            <GameCameraGizmo camera={activeScene.gameCamera} />
  1044→          )}
  1045→
  1046→          {/* TransformControls — só no editor */}
  1047→          {!isGameMode && selectedMesh && selectedInstanceId && (
  1048→            <TransformControls object={selectedMesh} mode={transformMode} size={0.8}
  1049→              onMouseDown={() => { if (orbitRef.current) orbitRef.current.enabled = false }}
  1050→              onMouseUp={() => { if (orbitRef.current) orbitRef.current.enabled = true; handleTransformUpdate() }}
  1051→              onObjectChange={handleTransformUpdate}
  1052→            />
  1053→          )}
  1054→
  1055→          {/* OrbitControls — só no editor */}
  1056→          {!isGameMode && (
  1057→            <OrbitControls ref={orbitRef} makeDefault enableDamping dampingFactor={0.08} minDistance={1} maxDistance={100} maxPolarAngle={Math.PI * 0.495}
  1058→              touches={{ ONE: THREE.TOUCH.ROTATE, TWO: THREE.TOUCH.DOLLY_PAN }}
  1059→            />
  1060→          )}
  1061→
  1062→          {/* GameMode — activado quando scenePreviewOpen */}
  1063→          <GameMode activeScene={activeScene} objects={objects} meshRefs={meshRefs} conectMeshRefs={conectMeshRefs} isGameMode={isGameMode} />
  1064→        </Suspense>
  1065→      </Canvas>
  1066→    </div>
  1067→  )
  1068→}
  1069→
  1070→function ConectSelectorWrapper({ conect, objects, isSelected, onSelect, setMeshRef, isGameMode }) {
  1071→  const handlePointerDown = (e) => {
  1072→    if (isGameMode) return
  1073→    e.stopPropagation()
  1074→    onSelect()
  1075→  }
  1076→  // Sistema 4: ViewObject deve ser invisível no modo jogo (só é gizmo do editor)
  1077→  if (isGameMode && conect.type === 'ViewObject') {
  1078→    return null
  1079→  }
  1080→  return (
  1081→    <group onPointerDown={handlePointerDown}>
  1082→      <ConectRenderer conect={conect} objects={objects} setMeshRef={setMeshRef} />
  1083→      {/* Sistema 1: gizmo do colisor independente (só no editor) */}
  1084→      {!isGameMode && <ColliderGizmo conect={conect} isSelected={isSelected} />}
  1085→    </group>
  1086→  )
  1087→}
