     1→/**
     2→ * taxonomy.js — definição central de todos os Conects disponíveis.
     3→ *
     4→ * Em FlirScript engine, um "Conect" é a unidade base de cena (similar a
     5→ * "node" na Godot ou "GameObject" na Unity). Cada Conect tem um tipo
     6→ * (RigidObject, VisualObject, etc.) com propriedades específicas.
     7→ *
     8→ * Esta taxonomy alimenta:
     9→ *  - A janela de Conects (lista + categorias + pesquisa)
    10→ *  - O painel de propriedades dinâmico (cada tipo tem campos diferentes)
    11→ *  - O runtime (cada tipo sabe como instanciar-se e atualizar-se)
    12→ *
    13→ * Estrutura de cada Conect:
    14→ *  {
    15→ *    type: 'RigidObject',         // identificador único
    16→ *    label: 'Rigid Object',       // nome visível
    17→ *    category: 'physics',         // categoria para agrupar
    18→ *    icon: '📦',                  // emoji ou ícone
    19→ *    description: '...',          // descrição curta
    20→ *    defaults: { ... },           // propriedades predefinidas
    21→ *    properties: [ ... ],         // esquema de propriedades para o painel
    22→ *    hasPhysics: true,            // participa no mundo de física?
    23→ *    hasVisual: true,             // tem malha visível?
    24→ *    flirScriptable: true,        // pode ter script FlirScript?
    25→ *  }
    26→ *
    27→ * Categorias: physics, visual, camera_audio, environment, ui, gameplay, organization
    28→ */

   849→// Sistema 1: injetar propriedades de colisor independente em todos os Conects
   850→// com hasPhysics: true. Garante que qualquer Conect de física tem as
   851→// propriedades colliderShape, colliderSize, colliderOffset, etc.
   852→CONECT_TAXONOMY.forEach((conect) => {
   853→  if (conect.hasPhysics) {
   854→    conect.properties = [...conect.properties, ...colliderProps()]
   855→    conect.defaults = {
   856→      ...conect.defaults,
   857→      colliderShape: 'model',
   858→      colliderSize: [1, 1, 1],
   859→      colliderOffset: [0, 0, 0],
   860→      colliderRadius: 0.5,
   861→      colliderHeight: 1.5,
   862→    }
   863→  }

   866→// Cria uma instância com valores predefinidos sensatos
   867→export function createConectInstance(type, position = [0, 0.5, 0]) {
   868→  const def = findConectDefinition(type)
   869→  if (!def) throw new Error(`Conect desconhecido: ${type}`)
   870→  // ViewObject tem posição default diferente — deve estar afastada para ver a cena
   871→  if (type === 'ViewObject') {
   872→    position = [5, 4, 6]
   873→  }
   874→  const defaults = {}
   875→  for (const propDef of def.properties) {
   876→    defaults[propDef.key] = JSON.parse(JSON.stringify(propDef.default))
   877→  }
   878→  return {
   879→    instanceId: `conect_${Math.random().toString(36).slice(2, 10)}`,
   880→    type,
   881→    name: `${def.label}`,
   882→    position: [...position],
   883→    rotation: type === 'ViewObject' ? [-0.5, 0.7, 0] : [0, 0, 0],
   884→    scale: [1, 1, 1],
   885→    visible: def.hasVisual !== false,
   886→    flirScript: null, // grafo FlirScript opcional
   887→    ...defaults,
   888→    ...def.defaults, // garantir que defaults da taxonomy também ficam
   889→  }
   890→}
